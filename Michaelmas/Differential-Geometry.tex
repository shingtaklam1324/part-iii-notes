\section{Manifolds}

The concept of a manifold is intended to generalise curves and surfaces in \(\R^3\).

% Motivation
% Definition of a topological space
% Definition of Huasdorff and second countable

\begin{definition}
    [smooth structure]
    A \emph{smooth structure}\index{smooth structure} on a topological space \(M\) is a collection of \emph{charts}\index{charts} (an \emph{atlas}\index{atlas})
    \[\varphi_\alpha : U_\alpha \to V_\alpha\]
    where \(U_\alpha \sse M\), \(V_\alpha \sse \R^d\) are open, \(\varphi_\alpha\) is a homeomorphism, such that
    \begin{enumerate}
        \item \(M = \bigcup_\alpha U_\alpha\). That is, the charts cover \(M\).
        \item Given any charts \(\varphi_\alpha, \varphi_\beta\), the map
        \[\varphi_\beta \circ \varphi_\alpha^{-1} : \varphi_\alpha(U_\alpha \cap U_\beta) \to \varphi_\beta(U_\alpha \cap U_\beta)\]
        is a smooth map between open subsets of \(\R^d\).
        \item If \(\varphi\) is \emph{compatible} (as in the previous point) with every chart in the atlas, then \(\varphi\) is a chart in the atlas.
    \end{enumerate}
\end{definition}

In practice (i.e. using Zorn's lemma), given any collection of charts satisfying the first two conditions, we can find a maximal atlas containing it.

\begin{remark}
    \begin{enumerate}
        \item The second condition implies that \(\varphi_\beta\circ \varphi_\alpha^{-1}\) is a diffeomorphism.
        \item \(d\) is fixed, assuming that \(M\) is connected.
    \end{enumerate}
\end{remark}

\begin{definition}
    [manifold] A \emph{(smooth) manifold}\index{manifold} \(M\) is a second countable Huasdorff space topological space, equipped with a smooth structure.

    \(d = \dim(M)\) is the \emph{dimension}\index{dimension} of \(M\).
\end{definition}

In practice, we may induce a topology on \(M\) from a smooth structure (without the mention of open or homeomorphism), then \(D \sse M\) is open if for any chart \(\varphi_\alpha\), \(\varphi_\alpha(D \cap U_\alpha)\) is an open subseteq of \(\R^d\).

\begin{remark}
    \begin{enumerate}
        \item We can also define a \(C^k\) manifold by replacing smooth with \(C^k\) as above. If \(k = 0\), then we have a \emph{topological manifold}\index{topological manifold}.
        \item On the other hand, we can replace \(\R^d\) with \(\C^n\), and obtain a \emph{complex manifold}\index{complex manifold}.
    \end{enumerate}
\end{remark}

\(\R^n\) is (trivially) a manifold.

\begin{example}
    [unit sphere]
    Define
    \[S^n = \left\{x = (x_0, \dots, x_n) \in \R^n \mid \norm{x} = 1\right\} \sse \R^{n+1}\]
    For the charts, we can consider the stereographic projections
    \[\varphi(x) = \frac{1}{1-x_0}(x_1, \dots, x_n) \qquad \text{for }x_0 \ne 1\]
    and
    \[\psi(x) = \frac{1}{1+x_0}(x_1, \dots, x_n) \qquad \text{for }x_0 \ne -1\]
    In particular,
    \[\psi \circ \varphi^{-1}(u) = \frac{u}{\norm{u}^2}\]
\end{example}

\begin{example}
    If \(M, N\) are manifolds of dimensions \(d,e\) respectively, then \(M \times N\) is a manifold of dimension \(d + e\).
\end{example}

As a corollary, \(T^n = (S^1)^n\) is a manifold.

\begin{example}
    If \(M\) is a manifold, \(X \subseteq M\) is open, then \(X\) is a manifold.
\end{example}

A convention, we will will assume for any manifold \(M\), we will assume all components are of the same dimension. In particular, \(\dim(M)\) is well defined.

\begin{notation}
    We will write \(M^d\) to say \(M\) is a manifold of dimension \(d\).
\end{notation}

\begin{example}
    [real projective space] The real projective space\index{real projective space@real projective space \(\R\P^n\)} is
    \[\R\P^n = \left\{\text{straight lines in }\R^{n+1}\text{ through }0\right\}\]
    Points in \(\R\P^n\) can be written using homogeneous coordinates \((x_0 : \dots : x_n)\), where the \(x_i\) are not all zero, and
    \[(x_0 : \dots : x_n) = (\lambda x_0 : \dots : \lambda x_n)\]
    for all \(\lambda \ne 0\). We will induce a topology from a smooth structure. The charts are given by \((\varphi_i, U_i)\) where
    \[U_i = \left\{x_i \ne 0\right\}\]
    and we define
    \[\varphi_i(x_0:\dots:x_n) = \left(\frac{x_0}{x_i}, \dots, \frac{x_{i-1}}{x_i},\frac{x_{i+1}}{x_i}, \dots, \frac{x_n}{x_i}\right) \in \R^n\]
    For \(i < j\), 
    \[\varphi_j \circ \varphi_i^{-1}(y_0, \dots, y_n) = \left(\frac{y_1}{y_j}, \dots, \frac{1}{y_j}, \dots, \frac{y_n}{y_j}\right)\]
    where the \(1\) is in the \(i\)-th position, and we omit the \(j\)-th position. Thus, \(\R\P^n\) is an \(n\)-dimensional manifold. Similarly, we can define the complex projective space \(\C\P^n\)\index{complex projective space@complex projective space \(\C\P^n\)}, which is a \(2n\)-manifold (and a \(n\)-dimensional complex manifold).
\end{example}

\begin{example}
    [Grassmannian]
    \[\Gr(k, n) = \left\{k\text{-dimensional subspaces of }\R^n\right\}\]\index{Grassmannian@Grassmannian \(\Gr(k, n)\)}
    Then \(\Gr(k, n)\) is a manifold of dimension \(k(n - k)\). Note we can also define the Grassmannian over \(\C\).
\end{example}

\begin{proof}
    [Sketch proof] An example of a coordinate neighbourhood is
    \[U = \left\{k\text{-dimensional subspaces obtainable as the span of rows of }k \times n\text{ matrices of the form} \begin{pmatrix}
        I_k & *
    \end{pmatrix}\right\}\]
    where \(I_k\) is the \(k \times k\) identity matrix. In this case, the \(k \times (n-k)\) block \(*\) defines local coordinates. Choosing different columns, we have \({n \choose k}\) neighbourhoods, \(U_{i_1, \dots, i_k}\), where \(i_1 < i_2 < \dots < i_k\).
\end{proof}

In particular, note that \(\R\P^n = \Gr(1, n+1)\).

\begin{example}
    [Non-example] Consider an equivalence relation on \(\R^2\) given by
    \[(x, y) \sim \left(\lambda x, \frac{1}{\lambda}y\right)\]
    for any \(\lambda \ne 0\). Let \(X = \R^2/\sim\) for the quotient, with the quotient topology. We can consider
    \[\left\{(x, y) \mid xy = c\right\}\]
    If \(c \ne 0\), then this is one equivalence class. If \(c = 0\), this splits into three, which are
    \begin{itemize}
        \item \(\{(0, 0)\}\),
        \item \(\{(x, 0) \mid x \ne 0\}\),
        \item \(\{(0, y) \mid y \ne 0\}\)
    \end{itemize}
    We can write
    \[X = (-\infty, 0) \cup (0, \infty) \cup \left\{0', 0'', 0'''\right\}\]
    We have three charts \(\varphi_i\) corresponding to each choice of zero. For example,
    \[U_1 = (-\infty, 0) \cup \{0'\} \cup (0, \infty)\]
    with the ``natural'' map \(\varphi_1\). On the other hand, the induced topology is not Hausdorff. Note the induced topology is \emph{not} the quotient topology.
\end{example}

Refer to Examples Sheet 1 Question 12 for an example of a non-second countable (non-)example.

\begin{definition}
    [smooth map] Let \(M, N\) be manifolds, a continuous map \(f : M \to N\) is \emph{smooth}\index{smooth map} if for any \(p \in M\), we can find charts \((\varphi, U)\) near \(p\), \((\psi, V)\) near \(f(p)\), and the composition
    \[\psi \circ f \circ \varphi^{-1}\]
    is smooth. as a map on
    \[\varphi(U \cap f^{-1}(V))\]
\end{definition}

\begin{definition}
    [diffeomorphism] A smooth map \(f : M \to N\) is a \emph{diffeomorphism}\index{diffeomorphism} if \(f\) is a bijection, and \(f^{-1}\) is smooth. In this case, we say that \(M, N\) are \emph{diffeomorphic}\index{diffeomorphic}.
\end{definition}

\begin{proposition}
    \begin{enumerate}
        \item If \(U \sse \R^n\), then \(f : U \to \R^m\) is smooth in the above definition if and only if it is smooth in the sense of multi-variate calculus.
        \item Every chart \(\varphi : U \to \varphi(U) \sse \R^d\) is a smooth map of manifolds. Moreover, it is a diffeomorphism.
        \item If \(f : M \to N\), \(g : N \to P\) are smooth, then \(g \circ f : M \to P\) is smooth.
    \end{enumerate}
\end{proposition}

\begin{proof}
    Obvious, and so omitted.
\end{proof}

\subsection{Matrix Lie groups}

Consider the group \(\GL(n, \R) \sse \R^{n^2}\). In fact, this is an open subset, and so it is a manifold. Moreover, matrix multiplication and inversion are smooth.

\begin{definition}
    [Lie group] A group \(G\) is called a \emph{Lie group}\index{Lie group} if
    \begin{itemize}
        \item \(G\) is a manifold,
        \item the multiplication map \(G \times G \to G\) is smooth,
        \item inversion \(G \to G\) is smooth.
    \end{itemize}
\end{definition}

\begin{remark}
    Suffices to show that the map
    \[G \times G \ni (\sigma, \tau) \mapsto \sigma\tau^{-1} \in G\]
    is smooth.
\end{remark}

In particular, \(\GL(n, \R\) is a Lie group. Similarly, \(\GL(n, \C)\) is also a Lie group. More generally, for \(A = (a_{ij}) \in \Mat_n(\C)\), we define the norm \(\norm{A} = n\max_{i, j}\abs{i, j}\). (Note in finite dimensional, all norms are equivalent). In particular,
\[\norm{AB} \le \norm{A}\norm{B}\]
We define the \emph{exponential of \(A\)}\index{exponential map!of matrices} as
\[\exp(A) = \sum_k \frac{A^k}{k!}\]
This series converges for all \(A\), by the same proof as for \(\exp : \C \to \C\) (or more generally, any Banach algebra). Moreover, it converges uniformly on balls
\[\left\{\norm{A} \le R\right\}\]
by the Weierstrass \(M\)-test. In particular, \(\exp : \Mat_n(\C) \to \Mat_n(\C)\) is continuous.

\lec

First of all, note that
\[f(A) = A^n\]
is a differentiable function, with
\[\norm{(\dd f)_A} \le n\abs{A}^{n-1}\]
Therefore, the series which we get by differentiating term by term, converges uniformly on any bounded set, again by the Weierstrass \(M\)-test.

Therefore, \(\exp(A)\) is (at least) \(C^1\) on \(\Mat_n(\C)\). We can similarly consider the higher derivatives, and we find that \(\exp(A)\) is \(C^\infty\).

\begin{remark}
    Some basic properties of the matrix exponential:

    \begin{enumerate}
        \item \(\exp(A^\T) = \exp(A)^\T\),
        \item \(\exp(A^*) = \exp(A^*)\),
        \item \(\exp(CAC^{-1}) = C\exp(A)C^{-1}\),
    \end{enumerate}

    In general,
    \[\exp(A + B) = \exp(A)\exp(B)\]
    does not hold. However, it does hold if \(AB = BA\). In particular,

    \begin{enumerate}
        \setcounter{enumi}{3}
        \item \(\exp(A)\exp(-A) = \exp(-A)\exp(A) = I\)
    \end{enumerate}
\end{remark}

We can also define the \emph{matrix logarithm}\index{logarithm map!for matrices} via the series
\[\log(I + A) = \sum_m(-1)^{m+1}\frac{A^m}{m}\]
We can show that this converges absolutely and uniformly on \(\{\abs{A} < \epsilon\}\) for any \(0 < \epsilon < 1\). The same result holds for the term by term differentiation, and again we get that \(\log(I + A)\) is \(C^\infty\) on \(\{\abs{A} < 1\}\)

Suppose \(\log(A)\) is defined, that is, \(\abs{A - I} < 1\), then
\[\exp(\log(A)) = A\]
since the same proof as in \(\C\) works for any Banach algebra.

On the other hand, if we would like
\[\log(\exp(A)) = A\]
we clearly need \(\abs{\exp(A) - I} < 1\). Suppose
\[A_\theta = \begin{pmatrix}
    0 & -\theta \\
    \theta & 0
\end{pmatrix}\]
for some \(\theta \in \R\). In particular,
\[A_\theta^2 = -\theta^2I\]
and we find that
\[\exp(A_\theta) = \begin{pmatrix}
    \cos(\theta) & -\sin(\theta) \\
    \sin(\theta) & \cos(\theta)
\end{pmatrix}\]
Setting \(\theta = 2\pi\), then we find that \(\exp(A_{2\pi}) = I\), but
\[\log(\exp(A_{2\pi})) = \log(I) = 0 \ne A_{2\pi}\]
Hence \(\abs{\exp(A) - I} < 1\) is a necessary, but not sufficient. On the other hand, \(\abs{A} < \log(2)\) is sufficient, since in this case,
\[\abs{e^\abs{A} - 1} < 1\]
which is required for absolute convergence.

\begin{example}
    [orthogonal group]
    Consider
    \[O(n) = \left\{A \in \GL(n, \R) \mid AA^\T = A^\T A = I\right\}\]
    We will show that \(O(n)\)\index{orthogonal group, \(O(n)\)} is a Lie group. Let \(A\) be an orthogonal matrix, \(\abs{A - I} < \epsilon < 1\), then \(B = \log(A)\) is defined. Moreover, \(\exp(B) = A\). By continuity, we can take \(\epsilon\) small enough such that \(\abs{B} < \log(2)\). Now
    \[\exp(B)\exp(B^\T) = AA^\T = \id\]
    and so
    \[\exp(B) = A = (A^\T)^{-1} = \exp(B^\T)^{-1} = \exp(-B^\T)\]
    Taking \(\log\), since \(\abs{B} = \abs{B^\T} < \log(2)\), we see that \(B = -B^\T\).

    Conversely, if \(B\) is skew-symmetric, with \(\abs{B} < \log(2)\), then
    \[\exp(B)\exp(B^\T) = I\]

    Define
    \[V_0 = \left\{B \in \Mat_n(\R) \mid  \abs{B} < \log(2), B^\T + B = 0\right\}\]
    and define
    \[U = \exp(V_0)\]
    which is an open neighbourhood of \(I \in O(n)\).

    \begin{proposition}
        \(O(n)\) has a \(C^\infty\) structure, making it into a manifold and a Lie group, with dimension \(n(n-1)/2\).
    \end{proposition}

    \begin{proof}
        We will use the subspace topology induced from \(O(n) \sse \R^{n^2}\). Define
        \begin{align*}
            h : U &\to V_0 \\
            h(A) &= \log(A)
        \end{align*}
        This is a homeomorphism onto its image, which is an open neighbourhood of \(0 \in \so(n) \cong \R^{n(n-1)/2}\).

        Given \(C \in O(n)\), define
        \[U_C = \left\{CA \mid A \in U\right\}\]
        and
        \begin{align*}
            h_C : U_C &\to V_0 \\
            h_C(A) &= \log(C^{-1}A)
        \end{align*}
        Again, this is a homeomorphism onto its image. Since \(C \in U_C\), we have an open cover. Moreover,
        \[h_{C_2} \circ h_{C_1}^{-1}(B) = h_{C_2}(C_1\exp(B)) = \log(C_2^{-1}C_2\exp(B))\]
        which is a smooth map between open neighbourhoods of \(\so(n)\). Therefore, \(O(n)\) is a smooth manifold. Similarly, for consider the map
        \[F(A_1, A_2) = A_1A_2^{-1}\]
        In local coordinates,
        \begin{align*}
            F_{\text{loc}}(B_1, B_2) &= h_{C_1C_2^{-1}}(F(h_{C_1}^{-1}(B_1), h_{C_2}^{-1}(B_2))) \\
            &= \log((C_1C_2^{-1})^{-1}C_1\exp(B_1)(C_2\exp(B_2))^{-1}) \\
            &= \log(C_2\exp(B_1)\exp(-B_2)C_2^{-1})
        \end{align*}
        and we can see that this is a smooth map as a function of \(B_1, B_2\).
    \end{proof}
\end{example}

This method works for more groups, and is sometimes called the Cayley construction (for matrix Lie groups).

\section{Tangent spaces to manifolds}

Consider a local curve in a ball in \(\R^n\), defined by a smooth parametrisation \(x(t) = (x_1(t), \dots, x_n(t))\). Suppose we have that \(x(0) = p \in \R^n\). In this case, its tangent vector at \(p\) is
\[\dot x(0)\in \R^n\]
If \(y(x)\) is a smooth change of coordinates near \(p\), then by the chain rule, we have
\[\dv{t}\bigg\vert_{t=0}y(x(t)) = J(y)_p \dot x(0)\]
where \(J(y)\) is the Jacobian matrix. More explicitly,
\[\dot y_i(0) = \sum_j \pdv{y_j}{x_j}(p)\dot x_j(0)\]

\begin{definition}
    [tangent vector]
    A \emph{tangent vector}\index{tangent vector} \(a\) to a manifold \(M^n\) at a point \(p \in M\), is an assignment to each coordinate chart \((U, \varphi)\) with \(p \in U\), an \(n\)-tuple of coordinates \((a_1, \dots, a_n) \in \R^n\), such that if \((U', \varphi')\) is another chart near \(p\), \((x_i), (x_i')\) the respective local coordinates,
    \[a_i' = \sum_j \pdv{x_i'}{x_j}(p)a_j\] 
\end{definition}

This definition is called the \emph{tensorial definition}, where we focus on the transformation law. There are several other definitions, for example, derivations, equivalence of curves to first order and so on.

\begin{definition}
    [tangent space]
    The \emph{tangent space}\index{tangent space} to \(M\) at a point \(p\), denoted \(\T_pM\) is the space of all tangent vectors to \(M\) at \(p\). It is naturally a vector space (of dimension \(n\) over \(\R\)).
\end{definition}

\lec

A choice of chart (with local coordinates \((x_i)\) say), determines a linear isomorphism
\[\T_p M \to \R^n\]
where \(n = \dim(M)\). We denote by \(\pdv{x_i}\), the basis corrsponding to the usual standard basis of \(\R^n\), under this linear isomorphism.

If we want to emphasise the point \(p \in M\), we may write
\[\left(\pdv{x_i}\right)_p \text{ or }\pdv{x_i}\bigg\vert_p\]
Then by the usual linear algebra, we have that
\[\left(\pdv{x_i}\right)_p = \sum_j \left(\pdv{x_j'}{x_i}\right)_p \left(\pdv{x_j'}\right)_p\]

An alternative view on tangent vectors is as \emph{derivations}. Given
\[a = \sum a_i \pdv{x_i} \in \T_pM\]
we can define a \emph{first-order derivation}\index{first order derivation} at \(p\),
\[a : C^\infty(M) \to \R\]
where
\[a(f) = \sum_i a_i \pdv{f}{x_i}\]
in local coordinates. This definition is independent of the choice of local coordinates. We can define this in a coordinate independent way, by defining
\[a(f) = \dv{t}\bigg\vert_{t=0}f(x(t))\]
where \(x\) is a local curve on \(M\), with \(\dot x(0) = a\). Moreover, every derivation satisfies the \emph{Leibniz rule}\index{Leibniz rule}
\[a(fg) = f \cdot a(g) + a(f) \cdot g\]
Conversely, every linear map \(a : C^\infty(M) \to \R\) satisfying the Leibniz rule above arises as a derivation corresponding to a tangent vector.

\begin{example}
    Let \(r : \R^2_{u, v}\supseteq D \to S \sse \R^3\) be a regular parametrisation for a surface \(S\). Then \(\varphi = r^{-1}\) defines a chart. The partial derivatives \(r_u, r_v\) correspond to
    \[\pdv{u}, \pdv{v}\]
    respectively.
\end{example}

For a Lie group \(G\), the tangent spaces has an ``infinitessimal counterpart'' of the group structure.

\begin{definition}
    [Lie algebra] A \emph{Lie algebra}\index{Lie algebra} is a vector space \(\mfg\), wiht a bilinear map
    \[[\cdot, \cdot] : \mfg \times \mfg \to \mfg\]
    called the \emph{Lie bracket}\index{Lie bracket}, with
    \begin{itemize}
        \item anticommutativity \([x, y] = [-y, x]\),
        \item the Jacobi idenity
        \[[x,[y,z]] + [y, [z,x]] + [z, [x, y]] = 0\]
    \end{itemize}
\end{definition}

\begin{theorem}
    Let \(G \sse \GL(n, F)\) be a matrix Lie group (where \(F = \R\) or \(\C\)), such that \(\log\) defines a chart near \(I \in G\). Let
    \[\mfg = \T_IG\]
    is identified with a (real) subspace of \(\Mat_n(F)\). Then \(\mfg\) is a Lie algebra, with
    \[[x,y] = xy-yx\]
    We call \(\mfg\) the \emph{Lie algebra of \(G\)}\index{Lie algebra!of a Lie group}. We may also write
    \[\mfg = \Lie(G)\]
\end{theorem}

\begin{proof}
    Clearly \(\mfg\) is a vector space, and \([\cdot, \cdot]\) is anticommutative and satisfies the Jacobi identity. Therefore, all we need to show is that if \(x, y \in \mfg\), then so is \([x,y]\).

    For \(B_1, B_2 \in \mfg\), consider
    \[A(t) = \exp(B_1t)\exp(B_2t)\exp(-B_1t)\exp(-B_2t)\]
    This defines a smooth path in \(G\), with \(A(0) = I\). Recall
    \[\exp(B t) = I + Bt + \frac{1}{2}B^2t^2 + o(t^2)\]
    as \(t \to 0\). We obtain that
    \[A(t) = I + [B_1, B_2]t^2 + o(t^2)\]
    Letting \(B(t) = \log(A(t))\), for small \(t\), we have that \(B(t) = [B_1, B_2]t^2 + o(t^2)\). But we also have that
    \[\exp(A(t)) = B(t)\]
    Moreover, by definition, \(B(t) \in \mfg\) for \(t\) small, as \(B(t)\) is in the image of the logarithm chart. So
    \[\frac{B(t)}{t^2} \in \mfg\]
    as \(\mfg\) is a vector space. So
    \[\lim_{t \to 0}\frac{B(t)}{t^2} = [B_1, B_2] \in \mfg\]
    since \(\mfg \le \Mat_n(F)\) is closed.
\end{proof}

If we take \(G = O(n)\), then we have that
\[\mfg = \mfo(n) = \left\{A \in \Mat_n(\R) \mid A^\T + A = 0\right\}\]
is the space of skew-symmetric matrices.

\begin{definition}
    Let \(M\) be a manifold. Then the \emph{tangent bundle of \(M\)}\index{tangent bundle} is
    \[\T M = \bigsqcup_{p \in M}\T_p M\]
\end{definition}

\begin{theorem}
    \(\T M\) has a smooth structure, making it into a manifold of dimension \(2 \dim(M)\).
\end{theorem}

\begin{proof}
    The topology on \(\T M\) will be induced from the smooth structure. Let \((\varphi, U)\) be a chart on \(M\). Say the local coordinates corresponding to \(\varphi\) are \(x_1, \dots, x_n\). So we can write \(a \in \T_pM\) as
    \[\sum a_i \pdv{x_i}\]
    Define
    \begin{align*}
        U_T &= \bigsqcup_{p \in U}\T_p M \\
        \varphi_T(p, a) &= (\varphi(p), (a_i))
    \end{align*}
    It is clear that the sets of the form \(U_T\) cover, so all we need to check is that the change of coordinate maps are smooth. Let \((\psi, V)\) be another chart on \(M\), with local coordinates \(y_i\). Then
    \[\psi \circ \varphi^{-1}(x, a) = (y, b)\]
    and it is easy to see that \(y = \psi \circ \varphi^{-1}(x)\), and
    \[b_i = \sum_j\pdv{y_i}{x_j} a_j\]
    Hausdorff and second countable follow from the fact that \(M\) and \(\R^n\) are.
\end{proof}

\lec

We will use the notation \(\pi : \T M \to M\) for the canonical projection map
\[\pi(p, a) = p\]

\begin{proposition}
    The map \(\pi : \T M \to M\) is smooth.
\end{proposition}

\begin{proof}
    In local coordinates given by \(\varphi\) and \(\varphi_T\), we have
    \[\pi(x, a) = x\]
\end{proof}

\begin{remark}
    In general, the tangent bundle \(\T M\) need not be diffeomorphic to \(M \times \R^n\).
\end{remark}

\begin{definition}
    [smooth vector field] A \emph{smooth vector field}\index{smooth vector field} \(X\) on a manifold \(M\) is a smooth map \(X : M \to \T M\), such that \(\pi(X(p)) = p\). That is, \(X(p) \in \T_pM\).
\end{definition}

\begin{example}
    \(X = 0\) is a vector field. Similarly, \(X\) supported on a coordinate neighbourhood is a well defined vector field.
\end{example}

\(X\) is \(C^\infty\) if and only if for each coordinate neighbourhood \(U\), with local coordinates \(x_i\), we have
\[X = \sum_i a_i(p) \pdv{x_i}\]
Smoothness of \(X\) becomes the requirement that each \(a_i : U \to \R\) is smooth.

\begin{theorem}
    \label{thm:parallelisable-iff-exists-lin-indep-vf}
    Suppose \(M^n\) is a manifold, and \(X^{(1)}, \dots, X^{(n)}\) are smooth vector fields on \(M\), such that for all \(p \in M\),
    \[X^{(1)}(p), \dots, X^{(n)}(p)\]
    is a basis for \(\T_pM\). Then \(\T M\) is isomorphic to the product \(M \times \R^d\). That is, there exists \(\Phi : \T M \to M \times \R^n\) is a diffeomorphism, such that
    \[\Phi(p, m) = (p, \phi_p(m))\]
    and for each fixed \(p\), \(\phi_p : \T_p M \to \R^n\) is a linear isomorphism.
\end{theorem}

Manifolds which satisfies the requirements of the theorem are called \emph{parallelisable}\index{parallelisable}.

\begin{proof}
    For \((p, a) \in \T M\), we can write it as
    \[a = \sum a_i X^{(i)}(p)\]
    for some unique \(a_1, \dots, a_n \in \R\). Using this, we can define a map
    \[\Phi(p, a) = (p,  (a_i)) \in M \times \R^n\]
    It is easy to see that this is a bijection. Moreover, \(\phi_p\) defines a linear isomorphism. Therefore, all we need to check is the smoothness. Let \(\psi : U \to \R^n\) be a chart on \(M\), and \(\psi_T : U_T \to \R^n \times \R^n\) be the corresponding chart on \(\T M\). In this case,
    \[(\psi \times \id) \circ \Phi \circ \psi_T^{-1}(x, b) = (x, a)\]
    such that
    \[a = \sum_i a_i X^{(i)}(p) = \sum_j b_j \left(\pdv{x_j}\right)_p\]
    We can write \(X^{(i)}\) in the \(\pdv{x_j}\) basis, to get
    \[X^{(i)} = \sum_j X^{(i)}_j \pdv{x_j}\]
    Thus,
    \[b_j = \sum_i a_i X^{(i)}_j(x)\]
    Since the \(X^{(i)}_j\) are smooth, we are done. Moreover, the inverse transformation is also smooth, and thus \(\Phi\) defines a local diffeomorphism, which must then be a diffeomorphism.
\end{proof}

\begin{remark}
    \begin{itemize}
        \item The converse of the theorem is also true, and it is easy, since all we need to choose is (the image of) \((x, e_1), \dots, (x, e_n)\).
        \item The condition that \(M\) is parallelisable is a restriction. We know that all Lie groups are parallelisable, but general manifolds are not. For example, there are no non-vanishing vector fields on \(S^2\). More generally, \(S^n\) is parallelisable if and only if \(n = 1, 3\) or \(7\).
    \end{itemize}
\end{remark}

\begin{definition}
    [differential]
    Let \(F : M \to N\) be a smooth map, Define the \emph{differential of \(F\) at \(p \in M\)}\index{differential!of a smooth map between manifolds} as
    \begin{align*}
        \dd F_p : \T_p M &\to \T_{F(p)} N \\
        \pdv{x_i}\bigg\vert_p &\mapsto \sum_j {\pdv{y_i}{x_j}}(x(p))\pdv{y_j}\bigg\vert_{F(p)}
    \end{align*}
    where \(x_i\) are local coordinates around \(p\), given by the chart \(\varphi\), and \(y_j\) are local coordinates aroung \(F(p)\), given by \(\psi\). In local coordinates, \(y = y(x) = \psi \circ F \circ \psi^{-1}(x)\).

    Moreover, it is easy to see this is independent of choice of local coordinate, which follows from the chain rule in multivariave calculus.
\end{definition}

\begin{remark}
    For smooth maps \(F : M \to N\), \(G : N \to P\),
    \[\dd(G \circ F) = \dd G \circ \dd F\]
\end{remark}

Now suppose \(F : M \to N\) is a diffeomorphism, \(X\) is a smooth vector field on \(M\). Then the \emph{pushforward of \(X\) by \(F\)}\index{pushforward!of a vector field by a diffeomorphism} is
\[\left((\dd F)X\right)_{F(p)} = (\dd F)_pX(p)\]
which defines a vector field on \(N\). We want \(F^{-1}\) to be smooth, since otherwise we don't know whether the image will be smooth.

Every vector field \(X\) on \(M\) defines a linear map
\[X : \C^\infty(M) \to \C^\infty(M)\]
where if
\[X = \sum_i X_i \pdv{x_i}\]
we define locally
\[Xh = \sum_i X_i \pdv{h}{x_i}\]
again by the chain rule, this is independent of choice of coordinates.

Let \(f \in C^\infty(N)\), \(F : M \to N\) be a diffeomorphism. Then \(f \circ F \in C^\infty(M)\). In coordinates \(x_i\) on \(M\), \(y_j\) on \(N\) given by \(F\),
\[\pdv{x_i}(f \circ F) = \sum_j \pdv{f}{y_j}\pdv{y_j}{x_i}\]
Therefore,
\[X(f \circ F) = \sum_{i, j}X_i\pdv{f}{y_j}\pdv{y_j}{x_i} = \left(\left((\dd F)X\right)f\right) \circ F\]
That is, the diagram

Let \(X, Y\) be vector fields on \(M\). \(X\) and \(Y\) can be consider as a first order linear differential operator. Then the composition \(XY\) is not, but the \emph{Lie bracket}\index{Lie bracket!of vector fields}
\[[X, Y] = XY - YX\]
is a well defined first order linear differential operator. That is, it defines an \(\R\)-linear map \(C^\infty(M) \to C^\infty(M)\), satisfying
\[[X, Y](fg) = f \cdot [X,Y]g + g \cdot [X, Y]f\]

One way to see this is by the symmetry of mixed partial derivatives. Thus, the space of all vector fields on \(M\) forms a Lie algebra. Note that it is infinite dimensional, and so it can't come from a Lie group.

\begin{notation}
    We write \(V(M)\) for the vector space of all \(C^\infty\) vector fields on \(M\).
\end{notation}

\subsection{Left invariant vector fields}

Let \(G\) be a Lie group, \(e \in G\) is the identity element of \(G\), and \(\mfg = \T_eG\) for the tangent space of \(G\) at \(e\), i.e. its Lie algebra.

Given \(g \in G\), we can define the \emph{left translation} \(L_g : G \to G\),
\[L_g(h) = gh\]
\(L_g\) is a diffeomorphism, as it is smooth with smooth inverse \(L_g^{-1} = L_{g^{-1}}\). Consider \(\xi \in \mfg\), and consider \index{Xxi@\(X_\xi\), vector field generated by \(\xi\)}
\[X_\xi(g) = (\dd L_g)_e\xi \in \T_gG\]
Since \(L_g\) is a diffeomorphism, \((\dd L_g)_e\) is a linear isomorphism \(\mfg \to \T_gG\), with inverse
\[(\dd L_g)_e^{-1} = (\dd L_{g^{-1}})_g\]

\begin{lemma}
    \(X_\xi(g)\) as defined above, is smooth. Thus, \(X_\xi\) is a smooth vector field.
\end{lemma}

\begin{proof}
    Consider the map \(L : G \times G \to G\), with \(L(g, h) = gh\). Then \(L\) is smooth. Fix \(g_0 \in G\), the local coordinate expression of \(L\) in a neighbourhood of \((g_0, e)\) is
    \[\widehat{L} : U_{g_0} \times U_e \to V_{g_0}\]
    Explicitly, \(\widehat L = \varphi_{g_0} \circ L \circ (\varphi_{g_0}^{-1} \times \varphi_e^{-1})\). With this,
    \[\widehat{L_g} = \widehat L(\varphi_{g_0}(g), \cdot) : U_e \to V_{g_0}\]
    where \(g\) is in a neighbourhood of \(g_0\). With this, consider the derivative \(D_2 \widehat{L}\) of \(\widehat{L}\) in the \(V_e\) coordinates. Since \(D_2\widehat{L}\) depends smoothly on the \(V_{g_0}\) coordinates on \(L\), \(X_\xi\) has smooth coefficients in a neighbourhood of \(g_0\).
\end{proof}

\begin{proposition}
    If \(\xi_1, \dots, \xi_n \in \mfg\) are linearly independent. That for all \(g \in G\),
    \[X_{\xi_1}(g), \dots, X_{\xi_n}(g)\]
    are linearly independent.
\end{proposition}

\begin{proof}
    Clear from definition as \((\dd L_g)_e\) is invertible.
\end{proof}

\begin{theorem}
    Every Lie group is parallelisable.
\end{theorem}

\begin{proof}
    From the proposition and \cref{thm:parallelisable-iff-exists-lin-indep-vf}.
\end{proof}

Also, consider \(g, h \in G\), and \(\xi \in \mfg\). In this case,
\[(\dd L_g)_hX_\xi(h) = (\dd L_g)_h(\dd L_h)_e(\xi) = (\dd L_g \circ \dd L_h)_e(\xi) = (\dd L_{gh})_e(\xi) = X_\xi(gh)\]
That is,
\[(\dd L_g)X_\xi = X_\xi \circ L_g\]

\begin{definition}
    [left-invariant vector field]
    A vector field \(X\) on a Lie group \(G\), with
    \begin{equation}
        \label{eq:left-inv-vf-def}
        (\dd L_g)X = X\circ L_g
    \end{equation}
    is called a \emph{left-invariant vector field}\index{left invariant vector field}. We write
    \[\ell(G) \sse V(G)\]
    for the subspace of all left-invariant vector fields on \(G\). 
\end{definition}

It is easy to see that \(\ell(G)\) is finite dimensional, since any \(X \in \ell(G)\) is \(X_\xi\) for \(\xi = X(e) \in \mfg\). This also shows that the map \(\mfg \to \ell(G)\) given by \(\xi \mapsto X_\xi\) is a linear isomorphism.

Thus,
\[\dim(\ell(G)) = \dim(\mfg) = \dim(G)\]

\begin{theorem}
    \(\ell(G)\) is a Lie subalgebra of \(V(G)\). More precisely, \([X_\xi, X_\eta]\) is left invariant for any \(\xi, \eta \in \mfg\).
\end{theorem}

\begin{proof}
    Recall that
    \[X(f \circ F) = \left(\left((\dd F)X\right)f\right) \circ F\]
    We will take \(F = L_g\), \(f \in C^\infty(G)\), \(X = [X_\xi, X_\eta]\). We will show that \(X\) satisfies \cref{eq:left-inv-vf-def}.
    \begin{align*}
        \left((\dd L_g)[X_\xi, X_\eta](f)\right)\circ L_g &= [X_\xi, X_\eta](f \circ L_g) \\
        &= X_\xi X_\eta (f \circ L_g) - X_\eta X_\xi(f \circ L_g) \\
        &= X_\xi \cdot (\dd L_g)X_\eta(f) \circ L_g -  X_\eta \cdot (\dd L_g)X_\xi(f) \circ L_g \\
        &= ((\dd L_g)X_\xi)((\dd L_g)X_\eta)(f) \circ L_g - ((\dd L_g)X_\eta)((\dd L_g)X_\xi)(f) \circ L_g \\
        &= ([(\dd L_g)X_\xi, (\dd L_g)X_\eta]f) \circ L_g \\
        &= [X_\xi \circ L_g, X_\eta \circ L_g](f) \circ L_g \\
        &= ([X_\xi, X_\eta] \circ L_g)(f) \circ L_g
    \end{align*}
    But \(f, g\) are arbitrary, and as such, we have
    \[(\dd L_g)[X_\xi, X_\eta] = [X_\xi, X_\eta] \circ L_g\]
    which means that \([X_\xi, X_\eta] \in \ell(G)\).
\end{proof}

With this, for all \(\xi, \eta \in \mfg\), there exists \(\zeta \in \mfg\) such that
\[[X_\xi, X_\eta] = X_\eta\]
This defines a Lie algebra structure on \(\mfg\).

\begin{theorem}
    \label{thm:matrix-lie-group-lie-alg-iso}
    Let \(G\) be a matrix Lie group and \(\log\) defines charts on \(G\). Then the map
    \[\mfg \ni \xi \mapsto X_\xi \in \ell(G)\]
    is an isomorphism of Lie algebras, where on \(\mfg\) we use the matrix commutator.
\end{theorem}

We will prove this later.

\section{Submanifolds}

Let \(M\) be a manifold, \(N \sse M\) and \(N\) is a manifold. Consider the inclusion map \(\iota : N \inj M\).

\begin{definition}
    [embedded submanifold]
    Suppose
    \begin{enumerate}[(i)]
        \item \(\iota\) is smooth,
        \item \((\dd \iota)_p : \T_p N \to \T_pM\) is injective for all \(p\),
        \item \(\iota\) is a homeomorphism onto its image.
    \end{enumerate}
    Then we say that \(N\) is an \emph{embedded submanifold}\index{embedded submanifold} of \(M\).
\end{definition}

\begin{remark}
    For (iii), \(D \sse N\) is open if and only if there exists \(U \sse M\) open, with \(D = D \cap N\).
\end{remark}
