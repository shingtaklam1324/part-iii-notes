\section{Manifolds}

The concept of a manifold is intended to generalise curves and surfaces in \(\R^3\).

% Motivation
% Definition of a topological space
% Definition of Huasdorff and second countable

\begin{definition}
    [smooth structure]
    A \emph{smooth structure}\index{smooth structure} on a topological space \(M\) is a collection of \emph{charts}\index{charts} (an \emph{atlas}\index{atlas})
    \[\varphi_\alpha : U_\alpha \to V_\alpha\]
    where \(U_\alpha \sse M\), \(V_\alpha \sse \R^d\) are open, \(\varphi_\alpha\) is a homeomorphism, such that
    \begin{enumerate}
        \item \(M = \bigcup_\alpha U_\alpha\). That is, the charts cover \(M\).
        \item Given any charts \(\varphi_\alpha, \varphi_\beta\), the map
        \[\varphi_\beta \circ \varphi_\alpha^{-1} : \varphi_\alpha(U_\alpha \cap U_\beta) \to \varphi_\beta(U_\alpha \cap U_\beta)\]
        is a smooth map between open subsets of \(\R^d\).
        \item If \(\varphi\) is \emph{compatible} (as in the previous point) with every chart in the atlas, then \(\varphi\) is a chart in the atlas.
    \end{enumerate}
\end{definition}

In practice (i.e. using Zorn's lemma), given any collection of charts satisfying the first two conditions, we can find a maximal atlas containing it.

\begin{remark}
    \begin{enumerate}
        \item The second condition implies that \(\varphi_\beta\circ \varphi_\alpha^{-1}\) is a diffeomorphism.
        \item \(d\) is fixed, assuming that \(M\) is connected.
    \end{enumerate}
\end{remark}

\begin{definition}
    [manifold] A \emph{(smooth) manifold}\index{manifold} \(M\) is a second countable Huasdorff space topological space, equipped with a smooth structure.

    \(d = \dim(M)\) is the \emph{dimension}\index{dimension} of \(M\).
\end{definition}

In practice, we may induce a topology on \(M\) from a smooth structure (without the mention of open or homeomorphism), then \(D \sse M\) is open if for any chart \(\varphi_\alpha\), \(\varphi_\alpha(D \cap U_\alpha)\) is an open subseteq of \(\R^d\).

\begin{remark}
    \begin{enumerate}
        \item We can also define a \(C^k\) manifold by replacing smooth with \(C^k\) as above. If \(k = 0\), then we have a \emph{topological manifold}\index{topological manifold}.
        \item On the other hand, we can replace \(\R^d\) with \(\C^n\), and obtain a \emph{complex manifold}\index{complex manifold}.
    \end{enumerate}
\end{remark}

\(\R^n\) is (trivially) a manifold.

\begin{example}
    [unit sphere]
    Define
    \[S^n = \left\{x = (x_0, \dots, x_n) \in \R^n \mid \norm{x} = 1\right\} \sse \R^{n+1}\]
    For the charts, we can consider the stereographic projections
    \[\varphi(x) = \frac{1}{1-x_0}(x_1, \dots, x_n) \qquad \text{for }x_0 \ne 1\]
    and
    \[\psi(x) = \frac{1}{1+x_0}(x_1, \dots, x_n) \qquad \text{for }x_0 \ne -1\]
    In particular,
    \[\psi \circ \varphi^{-1}(u) = \frac{u}{\norm{u}^2}\]
\end{example}

\begin{example}
    If \(M, N\) are manifolds of dimensions \(d,e\) respectively, then \(M \times N\) is a manifold of dimension \(d + e\).
\end{example}

As a corollary, \(T^n = (S^1)^n\) is a manifold.

\begin{example}
    If \(M\) is a manifold, \(X \subseteq M\) is open, then \(X\) is a manifold.
\end{example}

A convention, we will will assume for any manifold \(M\), we will assume all components are of the same dimension. In particular, \(\dim(M)\) is well defined.

\begin{notation}
    We will write \(M^d\) to say \(M\) is a manifold of dimension \(d\).
\end{notation}

\begin{example}
    [real projective space] The real projective space\index{real projective space@real projective space \(\R\P^n\)} is
    \[\R\P^n = \left\{\text{straight lines in }\R^{n+1}\text{ through }0\right\}\]
    Points in \(\R\P^n\) can be written using homogeneous coordinates \((x_0 : \dots : x_n)\), where the \(x_i\) are not all zero, and
    \[(x_0 : \dots : x_n) = (\lambda x_0 : \dots : \lambda x_n)\]
    for all \(\lambda \ne 0\). We will induce a topology from a smooth structure. The charts are given by \((\varphi_i, U_i)\) where
    \[U_i = \left\{x_i \ne 0\right\}\]
    and we define
    \[\varphi_i(x_0:\dots:x_n) = \left(\frac{x_0}{x_i}, \dots, \frac{x_{i-1}}{x_i},\frac{x_{i+1}}{x_i}, \dots, \frac{x_n}{x_i}\right) \in \R^n\]
    For \(i < j\), 
    \[\varphi_j \circ \varphi_i^{-1}(y_0, \dots, y_n) = \left(\frac{y_1}{y_j}, \dots, \frac{1}{y_j}, \dots, \frac{y_n}{y_j}\right)\]
    where the \(1\) is in the \(i\)-th position, and we omit the \(j\)-th position. Thus, \(\R\P^n\) is an \(n\)-dimensional manifold. Similarly, we can define the complex projective space \(\C\P^n\)\index{complex projective space@complex projective space \(\C\P^n\)}, which is a \(2n\)-manifold (and a \(n\)-dimensional complex manifold).
\end{example}

\begin{example}
    [Grassmannian]
    \[\Gr(k, n) = \left\{k\text{-dimensional subspaces of }\R^n\right\}\]\index{Grassmannian@Grassmannian \(\Gr(k, n)\)}
    Then \(\Gr(k, n)\) is a manifold of dimension \(k(n - k)\). Note we can also define the Grassmannian over \(\C\).
\end{example}

\begin{proof}
    [Sketch proof] An example of a coordinate neighbourhood is
    \[U = \left\{k\text{-dimensional subspaces obtainable as the span of rows of }k \times n\text{ matrices of the form} \begin{pmatrix}
        I_k & *
    \end{pmatrix}\right\}\]
    where \(I_k\) is the \(k \times k\) identity matrix. In this case, the \(k \times (n-k)\) block \(*\) defines local coordinates. Choosing different columns, we have \({n \choose k}\) neighbourhoods, \(U_{i_1, \dots, i_k}\), where \(i_1 < i_2 < \dots < i_k\).
\end{proof}

In particular, note that \(\R\P^n = \Gr(1, n+1)\).

\begin{example}
    [Non-example] Consider an equivalence relation on \(\R^2\) given by
    \[(x, y) \sim \left(\lambda x, \frac{1}{\lambda}y\right)\]
    for any \(\lambda \ne 0\). Let \(X = \R^2/\sim\) for the quotient, with the quotient topology. We can consider
    \[\left\{(x, y) \mid xy = c\right\}\]
    If \(c \ne 0\), then this is one equivalence class. If \(c = 0\), this splits into three, which are
    \begin{itemize}
        \item \(\{(0, 0)\}\),
        \item \(\{(x, 0) \mid x \ne 0\}\),
        \item \(\{(0, y) \mid y \ne 0\}\)
    \end{itemize}
    We can write
    \[X = (-\infty, 0) \cup (0, \infty) \cup \left\{0', 0'', 0'''\right\}\]
    We have three charts \(\varphi_i\) corresponding to each choice of zero. For example,
    \[U_1 = (-\infty, 0) \cup \{0'\} \cup (0, \infty)\]
    with the ``natural'' map \(\varphi_1\). On the other hand, the induced topology is not Hausdorff. Note the induced topology is \emph{not} the quotient topology.
\end{example}

Refer to Examples Sheet 1 Question 12 for an example of a non-second countable (non-)example.

\begin{definition}
    [smooth map] Let \(M, N\) be manifolds, a continuous map \(f : M \to N\) is \emph{smooth}\index{smooth map} if for any \(p \in M\), we can find charts \((\varphi, U)\) near \(p\), \((\psi, V)\) near \(f(p)\), and the composition
    \[\psi \circ f \circ \varphi^{-1}\]
    is smooth. as a map on
    \[\varphi(U \cap f^{-1}(V))\]
\end{definition}

\begin{definition}
    [diffeomorphism] A smooth map \(f : M \to N\) is a \emph{diffeomorphism}\index{diffeomorphism} if \(f\) is a bijection, and \(f^{-1}\) is smooth. In this case, we say that \(M, N\) are \emph{diffeomorphic}\index{diffeomorphic}.
\end{definition}

\begin{proposition}
    \begin{enumerate}
        \item If \(U \sse \R^n\), then \(f : U \to \R^m\) is smooth in the above definition if and only if it is smooth in the sense of multi-variate calculus.
        \item Every chart \(\varphi : U \to \varphi(U) \sse \R^d\) is a smooth map of manifolds. Moreover, it is a diffeomorphism.
        \item If \(f : M \to N\), \(g : N \to P\) are smooth, then \(g \circ f : M \to P\) is smooth.
    \end{enumerate}
\end{proposition}

\begin{proof}
    Obvious, and so omitted.
\end{proof}

\subsection{Matrix Lie groups}

Consider the group \(\GL(n, \R) \sse \R^{n^2}\). In fact, this is an open subset, and so it is a manifold. Moreover, matrix multiplication and inversion are smooth.

\begin{definition}
    [Lie group] A group \(G\) is called a \emph{Lie group}\index{Lie group} if
    \begin{itemize}
        \item \(G\) is a manifold,
        \item the multiplication map \(G \times G \to G\) is smooth,
        \item inversion \(G \to G\) is smooth.
    \end{itemize}
\end{definition}

\begin{remark}
    Suffices to show that the map
    \[G \times G \ni (\sigma, \tau) \mapsto \sigma\tau^{-1} \in G\]
    is smooth.
\end{remark}

In particular, \(\GL(n, \R\) is a Lie group. Similarly, \(\GL(n, \C)\) is also a Lie group. More generally, for \(A = (a_{ij}) \in \Mat_n(\C)\), we define the norm \(\norm{A} = n\max_{i, j}\abs{i, j}\). (Note in finite dimensional, all norms are equivalent). In particular,
\[\norm{AB} \le \norm{A}\norm{B}\]
We define the \emph{exponential of \(A\)}\index{exponential map!of matrices} as
\[\exp(A) = \sum_k \frac{A^k}{k!}\]
This series converges for all \(A\), by the same proof as for \(\exp : \C \to \C\) (or more generally, any Banach algebra). Moreover, it converges uniformly on balls
\[\left\{\norm{A} \le R\right\}\]
by the Weierstrass \(M\)-test. In particular, \(\exp : \Mat_n(\C) \to \Mat_n(\C)\) is continuous.

\lec

First of all, note that
\[f(A) = A^n\]
is a differentiable function, with
\[\norm{(\dd f)_A} \le n\abs{A}^{n-1}\]
Therefore, the series which we get by differentiating term by term, converges uniformly on any bounded set, again by the Weierstrass \(M\)-test.

Therefore, \(\exp(A)\) is (at least) \(C^1\) on \(\Mat_n(\C)\). We can similarly consider the higher derivatives, and we find that \(\exp(A)\) is \(C^\infty\).

\begin{remark}
    Some basic properties of the matrix exponential:

    \begin{enumerate}
        \item \(\exp(A^\T) = \exp(A)^\T\),
        \item \(\exp(A^*) = \exp(A^*)\),
        \item \(\exp(CAC^{-1}) = C\exp(A)C^{-1}\),
    \end{enumerate}

    In general,
    \[\exp(A + B) = \exp(A)\exp(B)\]
    does not hold. However, it does hold if \(AB = BA\). In particular,

    \begin{enumerate}
        \setcounter{enumi}{3}
        \item \(\exp(A)\exp(-A) = \exp(-A)\exp(A) = I\)
    \end{enumerate}
\end{remark}

We can also define the \emph{matrix logarithm}\index{logarithm map!for matrices} via the series
\[\log(I + A) = \sum_m(-1)^{m+1}\frac{A^m}{m}\]
We can show that this converges absolutely and uniformly on \(\{\abs{A} < \epsilon\}\) for any \(0 < \epsilon < 1\). The same result holds for the term by term differentiation, and again we get that \(\log(I + A)\) is \(C^\infty\) on \(\{\abs{A} < 1\}\)

Suppose \(\log(A)\) is defined, that is, \(\abs{A - I} < 1\), then
\[\exp(\log(A)) = A\]
since the same proof as in \(\C\) works for any Banach algebra.

On the other hand, if we would like
\[\log(\exp(A)) = A\]
we clearly need \(\abs{\exp(A) - I} < 1\). Suppose
\[A_\theta = \begin{pmatrix}
    0 & -\theta \\
    \theta & 0
\end{pmatrix}\]
for some \(\theta \in \R\). In particular,
\[A_\theta^2 = -\theta^2I\]
and we find that
\[\exp(A_\theta) = \begin{pmatrix}
    \cos(\theta) & -\sin(\theta) \\
    \sin(\theta) & \cos(\theta)
\end{pmatrix}\]
Setting \(\theta = 2\pi\), then we find that \(\exp(A_{2\pi}) = I\), but
\[\log(\exp(A_{2\pi})) = \log(I) = 0 \ne A_{2\pi}\]
Hence \(\abs{\exp(A) - I} < 1\) is a necessary, but not sufficient. On the other hand, \(\abs{A} < \log(2)\) is sufficient, since in this case,
\[\abs{e^\abs{A} - 1} < 1\]
which is required for absolute convergence.

\begin{example}
    [orthogonal group]
    Consider
    \[O(n) = \left\{A \in \GL(n, \R) \mid AA^\T = A^\T A = I\right\}\]
    We will show that \(O(n)\)\index{orthogonal group, \(O(n)\)} is a Lie group. Let \(A\) be an orthogonal matrix, \(\abs{A - I} < \epsilon < 1\), then \(B = \log(A)\) is defined. Moreover, \(\exp(B) = A\). By continuity, we can take \(\epsilon\) small enough such that \(\abs{B} < \log(2)\). Now
    \[\exp(B)\exp(B^\T) = AA^\T = \id\]
    and so
    \[\exp(B) = A = (A^\T)^{-1} = \exp(B^\T)^{-1} = \exp(-B^\T)\]
    Taking \(\log\), since \(\abs{B} = \abs{B^\T} < \log(2)\), we see that \(B = -B^\T\).

    Conversely, if \(B\) is skew-symmetric, with \(\abs{B} < \log(2)\), then
    \[\exp(B)\exp(B^\T) = I\]

    Define
    \[V_0 = \left\{B \in \Mat_n(\R) \mid  \abs{B} < \log(2), B^\T + B = 0\right\}\]
    and define
    \[U = \exp(V_0)\]
    which is an open neighbourhood of \(I \in O(n)\).

    \begin{proposition}
        \(O(n)\) has a \(C^\infty\) structure, making it into a manifold and a Lie group, with dimension \(n(n-1)/2\).
    \end{proposition}

    \begin{proof}
        We will use the subspace topology induced from \(O(n) \sse \R^{n^2}\). Define
        \begin{align*}
            h : U &\to V_0 \\
            h(A) &= \log(A)
        \end{align*}
        This is a homeomorphism onto its image, which is an open neighbourhood of \(0 \in \so(n) \cong \R^{n(n-1)/2}\).

        Given \(C \in O(n)\), define
        \[U_C = \left\{CA \mid A \in U\right\}\]
        and
        \begin{align*}
            h_C : U_C &\to V_0 \\
            h_C(A) &= \log(C^{-1}A)
        \end{align*}
        Again, this is a homeomorphism onto its image. Since \(C \in U_C\), we have an open cover. Moreover,
        \[h_{C_2} \circ h_{C_1}^{-1}(B) = h_{C_2}(C_1\exp(B)) = \log(C_2^{-1}C_2\exp(B))\]
        which is a smooth map between open neighbourhoods of \(\so(n)\). Therefore, \(O(n)\) is a smooth manifold. Similarly, for consider the map
        \[F(A_1, A_2) = A_1A_2^{-1}\]
        In local coordinates,
        \begin{align*}
            F_{\text{loc}}(B_1, B_2) &= h_{C_1C_2^{-1}}(F(h_{C_1}^{-1}(B_1), h_{C_2}^{-1}(B_2))) \\
            &= \log((C_1C_2^{-1})^{-1}C_1\exp(B_1)(C_2\exp(B_2))^{-1}) \\
            &= \log(C_2\exp(B_1)\exp(-B_2)C_2^{-1})
        \end{align*}
        and we can see that this is a smooth map as a function of \(B_1, B_2\).
    \end{proof}
\end{example}

This method works for more groups, and is sometimes called the Cayley construction (for matrix Lie groups).

\section{Tangent spaces to manifolds}

Consider a local curve in a ball in \(\R^n\), defined by a smooth parametrisation \(x(t) = (x_1(t), \dots, x_n(t))\). Suppose we have that \(x(0) = p \in \R^n\). In this case, its tangent vector at \(p\) is
\[\dot x(0)\in \R^n\]
If \(y(x)\) is a smooth change of coordinates near \(p\), then by the chain rule, we have
\[\dv{t}\bigg\vert_{t=0}y(x(t)) = J(y)_p \dot x(0)\]
where \(J(y)\) is the Jacobian matrix. More explicitly,
\[\dot y_i(0) = \sum_j \pdv{y_j}{x_j}(p)\dot x_j(0)\]

\begin{definition}
    [tangent vector]
    A \emph{tangent vector}\index{tangent vector} \(a\) to a manifold \(M^n\) at a point \(p \in M\), is an assignment to each coordinate chart \((U, \varphi)\) with \(p \in U\), an \(n\)-tuple of coordinates \((a_1, \dots, a_n) \in \R^n\), such that if \((U', \varphi')\) is another chart near \(p\), \((x_i), (x_i')\) the respective local coordinates,
    \[a_i' = \sum_j \pdv{x_i'}{x_j}(p)a_j\] 
\end{definition}

This definition is called the \emph{tensorial definition}, where we focus on the transformation law. There are several other definitions, for example, derivations, equivalence of curves to first order and so on.

\begin{definition}
    [tangent space]
    The \emph{tangent space}\index{tangent space} to \(M\) at a point \(p\), denoted \(\T_pM\) is the space of all tangent vectors to \(M\) at \(p\). It is naturally a vector space (of dimension \(n\) over \(\R\)).
\end{definition}

\lec

A choice of chart (with local coordinates \((x_i)\) say), determines a linear isomorphism
\[\T_p M \to \R^n\]
where \(n = \dim(M)\). We denote by \(\pdv{x_i}\), the basis corrsponding to the usual standard basis of \(\R^n\), under this linear isomorphism.

If we want to emphasise the point \(p \in M\), we may write
\[\left(\pdv{x_i}\right)_p \text{ or }\pdv{x_i}\bigg\vert_p\]
Then by the usual linear algebra, we have that
\[\left(\pdv{x_i}\right)_p = \sum_j \left(\pdv{x_j'}{x_i}\right)_p \left(\pdv{x_j'}\right)_p\]

An alternative view on tangent vectors is as \emph{derivations}. Given
\[a = \sum a_i \pdv{x_i} \in \T_pM\]
we can define a \emph{first-order derivation}\index{first order derivation} at \(p\),
\[a : C^\infty(M) \to \R\]
where
\[a(f) = \sum_i a_i \pdv{f}{x_i}\]
in local coordinates. This definition is independent of the choice of local coordinates. We can define this in a coordinate independent way, by defining
\[a(f) = \dv{t}\bigg\vert_{t=0}f(x(t))\]
where \(x\) is a local curve on \(M\), with \(\dot x(0) = a\). Moreover, every derivation satisfies the \emph{Leibniz rule}\index{Leibniz rule}
\[a(fg) = f \cdot a(g) + a(f) \cdot g\]
Conversely, every linear map \(a : C^\infty(M) \to \R\) satisfying the Leibniz rule above arises as a derivation corresponding to a tangent vector.

\begin{example}
    Let \(r : \R^2_{u, v}\supseteq D \to S \sse \R^3\) be a regular parametrisation for a surface \(S\). Then \(\varphi = r^{-1}\) defines a chart. The partial derivatives \(r_u, r_v\) correspond to
    \[\pdv{u}, \pdv{v}\]
    respectively.
\end{example}

For a Lie group \(G\), the tangent spaces has an ``infinitessimal counterpart'' of the group structure.

\begin{definition}
    [Lie algebra] A \emph{Lie algebra}\index{Lie algebra} is a vector space \(\mfg\), wiht a bilinear map
    \[[\cdot, \cdot] : \mfg \times \mfg \to \mfg\]
    called the \emph{Lie bracket}\index{Lie bracket}, with
    \begin{itemize}
        \item anticommutativity \([x, y] = [-y, x]\),
        \item the Jacobi idenity
        \[[x,[y,z]] + [y, [z,x]] + [z, [x, y]] = 0\]
    \end{itemize}
\end{definition}

\begin{theorem}
    Let \(G \sse \GL(n, F)\) be a matrix Lie group (where \(F = \R\) or \(\C\)), such that \(\log\) defines a chart near \(I \in G\). Let
    \[\mfg = \T_IG\]
    is identified with a (real) subspace of \(\Mat_n(F)\). Then \(\mfg\) is a Lie algebra, with
    \[[x,y] = xy-yx\]
    We call \(\mfg\) the \emph{Lie algebra of \(G\)}\index{Lie algebra!of a Lie group}. We may also write
    \[\mfg = \Lie(G)\]
\end{theorem}

\begin{proof}
    Clearly \(\mfg\) is a vector space, and \([\cdot, \cdot]\) is anticommutative and satisfies the Jacobi identity. Therefore, all we need to show is that if \(x, y \in \mfg\), then so is \([x,y]\).

    For \(B_1, B_2 \in \mfg\), consider
    \[A(t) = \exp(B_1t)\exp(B_2t)\exp(-B_1t)\exp(-B_2t)\]
    This defines a smooth path in \(G\), with \(A(0) = I\). Recall
    \[\exp(B t) = I + Bt + \frac{1}{2}B^2t^2 + o(t^2)\]
    as \(t \to 0\). We obtain that
    \[A(t) = I + [B_1, B_2]t^2 + o(t^2)\]
    Letting \(B(t) = \log(A(t))\), for small \(t\), we have that \(B(t) = [B_1, B_2]t^2 + o(t^2)\). But we also have that
    \[\exp(A(t)) = B(t)\]
    Moreover, by definition, \(B(t) \in \mfg\) for \(t\) small, as \(B(t)\) is in the image of the logarithm chart. So
    \[\frac{B(t)}{t^2} \in \mfg\]
    as \(\mfg\) is a vector space. So
    \[\lim_{t \to 0}\frac{B(t)}{t^2} = [B_1, B_2] \in \mfg\]
    since \(\mfg \le \Mat_n(F)\) is closed.
\end{proof}

If we take \(G = O(n)\), then we have that
\[\mfg = \mfo(n) = \left\{A \in \Mat_n(\R) \mid A^\T + A = 0\right\}\]
is the space of skew-symmetric matrices.

\begin{definition}
    Let \(M\) be a manifold. Then the \emph{tangent bundle of \(M\)}\index{tangent bundle} is
    \[\T M = \bigsqcup_{p \in M}\T_p M\]
\end{definition}

\begin{theorem}
    \(\T M\) has a smooth structure, making it into a manifold of dimension \(2 \dim(M)\).
\end{theorem}

\begin{proof}
    The topology on \(\T M\) will be induced from the smooth structure. Let \((\varphi, U)\) be a chart on \(M\). Say the local coordinates corresponding to \(\varphi\) are \(x_1, \dots, x_n\). So we can write \(a \in \T_pM\) as
    \[\sum a_i \pdv{x_i}\]
    Define
    \begin{align*}
        U_T &= \bigsqcup_{p \in U}\T_p M \\
        \varphi_T(p, a) &= (\varphi(p), (a_i))
    \end{align*}
    It is clear that the sets of the form \(U_T\) cover, so all we need to check is that the change of coordinate maps are smooth. Let \((\psi, V)\) be another chart on \(M\), with local coordinates \(y_i\). Then
    \[\psi \circ \varphi^{-1}(x, a) = (y, b)\]
    and it is easy to see that \(y = \psi \circ \varphi^{-1}(x)\), and
    \[b_i = \sum_j\pdv{y_i}{x_j} a_j\]
    Hausdorff and second countable follow from the fact that \(M\) and \(\R^n\) are.
\end{proof}

\lec

We will use the notation \(\pi : \T M \to M\) for the canonical projection map
\[\pi(p, a) = p\]

\begin{proposition}
    The map \(\pi : \T M \to M\) is smooth.
\end{proposition}

\begin{proof}
    In local coordinates given by \(\varphi\) and \(\varphi_T\), we have
    \[\pi(x, a) = x\]
\end{proof}

\begin{remark}
    In general, the tangent bundle \(\T M\) need not be diffeomorphic to \(M \times \R^n\).
\end{remark}

\begin{definition}
    [(smooth) vector field] A \emph{(smooth) vector field}\index{vector field} \(X\) on a manifold \(M\) is a smooth map \(X : M \to \T M\), such that \(\pi(X(p)) = p\). That is, \(X(p) \in \T_pM\).
\end{definition}

\begin{example}
    \(X = 0\) is a vector field. Similarly, \(X\) supported on a coordinate neighbourhood is a well defined vector field.
\end{example}

\(X\) is \(C^\infty\) if and only if for each coordinate neighbourhood \(U\), with local coordinates \(x_i\), we have
\[X = \sum_i a_i(p) \pdv{x_i}\]
Smoothness of \(X\) becomes the requirement that each \(a_i : U \to \R\) is smooth.

\begin{theorem}
    \label{thm:parallelisable-iff-exists-lin-indep-vf}
    Suppose \(M^n\) is a manifold, and \(X^{(1)}, \dots, X^{(n)}\) are smooth vector fields on \(M\), such that for all \(p \in M\),
    \[X^{(1)}(p), \dots, X^{(n)}(p)\]
    is a basis for \(\T_pM\). Then \(\T M\) is isomorphic to the product \(M \times \R^d\). That is, there exists \(\Phi : \T M \to M \times \R^n\) is a diffeomorphism, such that
    \[\Phi(p, m) = (p, \phi_p(m))\]
    and for each fixed \(p\), \(\phi_p : \T_p M \to \R^n\) is a linear isomorphism.
\end{theorem}

Manifolds which satisfies the requirements of the theorem are called \emph{parallelisable}\index{parallelisable}.

\begin{proof}
    For \((p, a) \in \T M\), we can write it as
    \[a = \sum a_i X^{(i)}(p)\]
    for some unique \(a_1, \dots, a_n \in \R\). Using this, we can define a map
    \[\Phi(p, a) = (p,  (a_i)) \in M \times \R^n\]
    It is easy to see that this is a bijection. Moreover, \(\phi_p\) defines a linear isomorphism. Therefore, all we need to check is the smoothness. Let \(\psi : U \to \R^n\) be a chart on \(M\), and \(\psi_T : U_T \to \R^n \times \R^n\) be the corresponding chart on \(\T M\). In this case,
    \[(\psi \times \id) \circ \Phi \circ \psi_T^{-1}(x, b) = (x, a)\]
    such that
    \[a = \sum_i a_i X^{(i)}(p) = \sum_j b_j \left(\pdv{x_j}\right)_p\]
    We can write \(X^{(i)}\) in the \(\pdv{x_j}\) basis, to get
    \[X^{(i)} = \sum_j X^{(i)}_j \pdv{x_j}\]
    Thus,
    \[b_j = \sum_i a_i X^{(i)}_j(x)\]
    Since the \(X^{(i)}_j\) are smooth, we are done. Moreover, the inverse transformation is also smooth, and thus \(\Phi\) defines a local diffeomorphism, which must then be a diffeomorphism.
\end{proof}

\begin{remark}
    \begin{itemize}
        \item The converse of the theorem is also true, and it is easy, since all we need to choose is (the image of) \((x, e_1), \dots, (x, e_n)\).
        \item The condition that \(M\) is parallelisable is a restriction. We know that all Lie groups are parallelisable, but general manifolds are not. For example, there are no non-vanishing vector fields on \(S^2\). More generally, \(S^n\) is parallelisable if and only if \(n = 1, 3\) or \(7\).
    \end{itemize}
\end{remark}

\begin{definition}
    [differential]
    Let \(F : M \to N\) be a smooth map, Define the \emph{differential of \(F\) at \(p \in M\)}\index{differential!of a smooth map between manifolds} as
    \begin{align*}
        \dd F_p : \T_p M &\to \T_{F(p)} N \\
        \pdv{x_i}\bigg\vert_p &\mapsto \sum_j {\pdv{y_i}{x_j}}(x(p))\pdv{y_j}\bigg\vert_{F(p)}
    \end{align*}
    where \(x_i\) are local coordinates around \(p\), given by the chart \(\varphi\), and \(y_j\) are local coordinates aroung \(F(p)\), given by \(\psi\). In local coordinates, \(y = y(x) = \psi \circ F \circ \psi^{-1}(x)\).

    Moreover, it is easy to see this is independent of choice of local coordinate, which follows from the chain rule in multivariave calculus.
\end{definition}

\begin{remark}
    For smooth maps \(F : M \to N\), \(G : N \to P\),
    \[\dd(G \circ F) = \dd G \circ \dd F\]
\end{remark}

Now suppose \(F : M \to N\) is a diffeomorphism, \(X\) is a smooth vector field on \(M\). Then the \emph{pushforward of \(X\) by \(F\)}\index{pushforward!of a vector field by a diffeomorphism} is
\[\left((\dd F)X\right)_{F(p)} = (\dd F)_pX(p)\]
which defines a vector field on \(N\). We want \(F^{-1}\) to be smooth, since otherwise we don't know whether the image will be smooth.

Every vector field \(X\) on \(M\) defines a linear map
\[X : \C^\infty(M) \to \C^\infty(M)\]
where if
\[X = \sum_i X_i \pdv{x_i}\]
we define locally
\[Xh = \sum_i X_i \pdv{h}{x_i}\]
again by the chain rule, this is independent of choice of coordinates.

Let \(f \in C^\infty(N)\), \(F : M \to N\) be a diffeomorphism. Then \(f \circ F \in C^\infty(M)\). In coordinates \(x_i\) on \(M\), \(y_j\) on \(N\) given by \(F\),
\[\pdv{x_i}(f \circ F) = \sum_j \pdv{f}{y_j}\pdv{y_j}{x_i}\]
Therefore,
\[X(f \circ F) = \sum_{i, j}X_i\pdv{f}{y_j}\pdv{y_j}{x_i} = \left(\left((\dd F)X\right)f\right) \circ F\]
That is, the diagram

commutes.

Let \(X, Y\) be vector fields on \(M\). \(X\) and \(Y\) can be consider as a first order linear differential operator. Then the composition \(XY\) is not, but the \emph{Lie bracket}\index{Lie bracket!of vector fields}
\[[X, Y] = XY - YX\]
is a well defined first order linear differential operator. That is, it defines an \(\R\)-linear map \(C^\infty(M) \to C^\infty(M)\), satisfying
\[[X, Y](fg) = f \cdot [X,Y]g + g \cdot [X, Y]f\]

One way to see this is by the symmetry of mixed partial derivatives. Thus, the space of all vector fields on \(M\) forms a Lie algebra. Note that it is infinite dimensional, and so it can't come from a Lie group.

\lec

Suppose in local coordinates \(X = \sum_i X_i \pdv{x_i}\) and \(Y = \sum_i Y_i \pdv{x_i}\), then
\[[X, Y] = \sum_{i, j}\left(X_i \pdv{Y_j}{x_i} - Y_i \pdv{X_j}{x_i}\right)\pdv{x_k}\]

\begin{notation}
    We write \(V(M)\) for the vector space of all \(C^\infty\) vector fields on \(M\).
\end{notation}

\subsection{Left invariant vector fields}

Let \(G\) be a Lie group, \(e \in G\) is the identity element of \(G\), and \(\mfg = \T_eG\) for the tangent space of \(G\) at \(e\), i.e. its Lie algebra.

Given \(g \in G\), we can define the \emph{left translation} \(L_g : G \to G\),
\[L_g(h) = gh\]
\(L_g\) is a diffeomorphism, as it is smooth with smooth inverse \(L_g^{-1} = L_{g^{-1}}\). Consider \(\xi \in \mfg\), and consider \index{Xxi@\(X_\xi\), vector field generated by \(\xi\)}
\[X_\xi(g) = (\dd L_g)_e\xi \in \T_gG\]
Since \(L_g\) is a diffeomorphism, \((\dd L_g)_e\) is a linear isomorphism \(\mfg \to \T_gG\), with inverse
\[(\dd L_g)_e^{-1} = (\dd L_{g^{-1}})_g\]

\begin{lemma}
    \(X_\xi(g)\) as defined above, is smooth. Thus, \(X_\xi\) is a smooth vector field.
\end{lemma}

\begin{proof}
    Consider the map \(L : G \times G \to G\), with \(L(g, h) = gh\). Then \(L\) is smooth. Fix \(g_0 \in G\), the local coordinate expression of \(L\) in a neighbourhood of \((g_0, e)\) is
    \[\widehat{L} : U_{g_0} \times U_e \to V_{g_0}\]
    Explicitly, \(\widehat L = \varphi_{g_0} \circ L \circ (\varphi_{g_0}^{-1} \times \varphi_e^{-1})\). With this,
    \[\widehat{L_g} = \widehat L(\varphi_{g_0}(g), \cdot) : U_e \to V_{g_0}\]
    where \(g\) is in a neighbourhood of \(g_0\). With this, consider the derivative \(D_2 \widehat{L}\) of \(\widehat{L}\) in the \(V_e\) coordinates. Since \(D_2\widehat{L}\) depends smoothly on the \(V_{g_0}\) coordinates on \(L\), \(X_\xi\) has smooth coefficients in a neighbourhood of \(g_0\).
\end{proof}

\begin{proposition}
    If \(\xi_1, \dots, \xi_n \in \mfg\) are linearly independent. That for all \(g \in G\),
    \[X_{\xi_1}(g), \dots, X_{\xi_n}(g)\]
    are linearly independent.
\end{proposition}

\begin{proof}
    Clear from definition as \((\dd L_g)_e\) is invertible.
\end{proof}

\begin{theorem}
    Every Lie group is parallelisable.
\end{theorem}

\begin{proof}
    From the proposition and \cref{thm:parallelisable-iff-exists-lin-indep-vf}.
\end{proof}

Also, consider \(g, h \in G\), and \(\xi \in \mfg\). In this case,
\[(\dd L_g)_hX_\xi(h) = (\dd L_g)_h(\dd L_h)_e(\xi) = (\dd L_g \circ \dd L_h)_e(\xi) = (\dd L_{gh})_e(\xi) = X_\xi(gh)\]
That is,
\[(\dd L_g)X_\xi = X_\xi \circ L_g\]

\begin{definition}
    [left-invariant vector field]
    A vector field \(X\) on a Lie group \(G\), with
    \begin{equation}
        \label{eq:left-inv-vf-def}
        (\dd L_g)X = X\circ L_g
    \end{equation}
    is called a \emph{left-invariant vector field}\index{left invariant vector field}. We write
    \[\ell(G) \sse V(G)\]
    for the subspace of all left-invariant vector fields on \(G\). 
\end{definition}

It is easy to see that \(\ell(G)\) is finite dimensional, since any \(X \in \ell(G)\) is \(X_\xi\) for \(\xi = X(e) \in \mfg\). This also shows that the map \(\mfg \to \ell(G)\) given by \(\xi \mapsto X_\xi\) is a linear isomorphism.

Thus,
\[\dim(\ell(G)) = \dim(\mfg) = \dim(G)\]

\begin{theorem}
    \(\ell(G)\) is a Lie subalgebra of \(V(G)\). More precisely, \([X_\xi, X_\eta]\) is left invariant for any \(\xi, \eta \in \mfg\).
\end{theorem}

\begin{proof}
    Recall that
    \[X(f \circ F) = \left(\left((\dd F)X\right)f\right) \circ F\]
    We will take \(F = L_g\), \(f \in C^\infty(G)\), \(X = [X_\xi, X_\eta]\). We will show that \(X\) satisfies \cref{eq:left-inv-vf-def}.
    \begin{align*}
        \left((\dd L_g)[X_\xi, X_\eta](f)\right)\circ L_g &= [X_\xi, X_\eta](f \circ L_g) \\
        &= X_\xi X_\eta (f \circ L_g) - X_\eta X_\xi(f \circ L_g) \\
        &= X_\xi \cdot (\dd L_g)X_\eta(f) \circ L_g -  X_\eta \cdot (\dd L_g)X_\xi(f) \circ L_g \\
        &= ((\dd L_g)X_\xi)((\dd L_g)X_\eta)(f) \circ L_g - ((\dd L_g)X_\eta)((\dd L_g)X_\xi)(f) \circ L_g \\
        &= ([(\dd L_g)X_\xi, (\dd L_g)X_\eta]f) \circ L_g \\
        &= [X_\xi \circ L_g, X_\eta \circ L_g](f) \circ L_g \\
        &= ([X_\xi, X_\eta] \circ L_g)(f) \circ L_g
    \end{align*}
    But \(f, g\) are arbitrary, and as such, we have
    \[(\dd L_g)[X_\xi, X_\eta] = [X_\xi, X_\eta] \circ L_g\]
    which means that \([X_\xi, X_\eta] \in \ell(G)\).
\end{proof}

With this, for all \(\xi, \eta \in \mfg\), there exists \(\zeta \in \mfg\) such that
\[[X_\xi, X_\eta] = X_\eta\]
This defines a Lie algebra structure on \(\mfg\).

\begin{theorem}
    \label{thm:matrix-lie-group-lie-alg-iso}
    Let \(G\) be a matrix Lie group and \(\log\) defines charts on \(G\). Then the map
    \[\mfg \ni \xi \mapsto X_\xi \in \ell(G)\]
    is an isomorphism of Lie algebras, where on \(\mfg\) we use the matrix commutator.
\end{theorem}

We will prove this later.

\section{Submanifolds}

Let \(M\) be a manifold, \(N \sse M\) and \(N\) is a manifold. Consider the inclusion map \(\iota : N \inj M\).

\begin{definition}
    [embedded submanifold]
    Suppose
    \begin{enumerate}[(i)]
        \item \(\iota\) is smooth,
        \item \((\dd \iota)_p : \T_p N \to \T_pM\) is injective for all \(p\),
        \item \(\iota\) is a homeomorphism onto its image.
    \end{enumerate}
    Then we say that \(N\) is an \emph{embedded submanifold}\index{embedded submanifold} of \(M\).
\end{definition}

\begin{remark}
    For (iii), \(D \sse N\) is open if and only if there exists \(U \sse M\) open, with \(D = D \cap N\).
\end{remark}

\lec

The condition (iii) is required, since we don't want an example which looks like a figure-8, or other `weird' topological spaces, such as

    \includegraphics[scale=0.125]{diagrams/2023_10_19.jpg}
\end{center}

\begin{remark}
    If \(\iota\) only satisfies (i) and (ii), we call \(N\) an \emph{immersed submanifold}\index{immersed submanifold}.
\end{remark}

\begin{remark}
    A slight generalisation of this is that: if \(N, M\) are manifolds, a map \(\psi : N \to M\) is called an \emph{embedding}\index{embedding}, denoted \(\psi : N \inj M\), if \(\psi\)  is injective, and satisfies conditions (i), (ii) and (iii). That is, \(\psi(N) \sse M\) is an immersed submanifold.

    If \(\dd\Psi\) is injective for all \(p\), then we call \(\psi\) an \emph{immersion}\index{immersion}.
\end{remark}

\underline{Convention:} Submanifold will mean embedded submanifold unless otherwise stated.

\begin{example}
    [parametrised urves and surfaces in \(\R^3\)] In this case, condition (ii) simply means what we called a `regular' parametrisation.
\end{example}

\begin{example}
    [irrational twist flow]

    Define the map
    \begin{align*}
        \R &\to T^2 \\
        t &\mapsto (e^{it}, e^{i\alpha t})
    \end{align*}
    In this case, we see that the map is injective and an immersion, but not an embedding as the image is dense in \(T^2\).
\end{example}

Is a submanifold the same as \(f^{-1}(0)\) for some smooth map \(f : \R^n \to \R^m\), or more generally, \(f^{-1}(p)\) for some \(f : M \to N\).

In general, no. Since \(f\) is smooth, it is continuous, and so \(f^{-1}(p)\) is a closed set. On the other hand,

\begin{proposition}    
    For every closed subset \(E \sse \R^2\), we can find a smooth map \(f : \R^2 \to \R\), such that \(f^{-1}(0) = E\).
\end{proposition}

\begin{proof}
    Omitted, optional exercise.
\end{proof}

\begin{definition}
    [regular value]
    Let \(f : M \to Y\) be a smooth map between manifolds. Then \(q \in Y\) is a \emph{regular value}\index{regular value} for \(f\) if for every \(p \in f^{-1}(q)\), \((\dd f)_p : \T_p M \to \T_q Y\) is surjective.
\end{definition}

\begin{theorem}
    [preimage theorem] \index{preimage theorem}Suppose \(f : M \to Y\) is a smooth map, \(y \in Y\) is a regular value, with \(f^{-1}(y)\) nonempty. Then \(N = f^{-1}(y)\) is an embedded submanifold of \(M\), with
    \[\dim(N) = \dim(M) - \dim(Y)\]
\end{theorem}

\begin{proof}
    Omitted.
\end{proof}

A fact from differential topology: Suppose \(M\) is a manifold, \(N \sse M\) a subspace (with the subspace topology). Then there is at most one smooth structure on \(N\) making it into an embedded submanifold. Therefore, it makes sense for us to say \(N\) is or is not a submanifold.

The converse of the preimage theorem is only true locally.

\begin{proposition}
    Let \(N \inj M\) be an embedded submanifold, \(p \in N\). Then there exists an open neighbourhood \(p \in U \sse M\), and \(f : U \to \R^d\), \(d = \dim(M) - \dim(N)\), with \(0 \in \R^d\) is a regular value of \(f\), and \(f^{-1}(0) = U \cap f^{-1}(0)\).
\end{proposition}

\begin{proof}
    Let \(\varphi : U_0 \to \R^n\) be a chart on \(M\) near \(p\), with \(\varphi(p) = 0\), with local coordinates \(x_1, \dots, x_n\). Let \(\psi : V_0 \to \R^\ell\) be a chart on \(N\), \(\psi(p) = 0\) and with local coordinates \(u_1, \dots, u_\ell\).

    We may assume without loss of generality that \(V_0 = U_0 \cap N\). In this case, \(\iota : N \inj M\) is expressed locally as
    \[x_i = x_i(u_1, \dots, u_\ell)\]
    Thus, the rank of the Jacobian matrix
    \[\begin{pmatrix}
        \pdv{x_i}{u_j}(0)
    \end{pmatrix}\]
    is \(\ell\), by assumption. Without loss of generality,
    \[\det\begin{pmatrix}
        \pdv{x_i}{u_j}(0)
    \end{pmatrix}_{i, j = 1}^\ell \ne 0\]
    From the inverse function theorem from multivariate calculus, we have a local inverse
    \[u_i = u_i(x_1, \dots, x_\ell)\]
    near \(0\). Moreover, for \(i > \ell\), then \(x_i = x_i(u) = x_i(u(x_1, \dots, x_\ell)) = h_i(x_1, \dots, x_\ell)\).

    Define \(f_i(x) = x_i - h_i(x_1, \dots, x_\ell)\) for \(\ell < i < n\). This gives the required map, with Jacobian at \(0\) given in block form as
    \(\begin{pmatrix}
        * & I_d
    \end{pmatrix}\)
    Thus, \(0\) is a regular value.
\end{proof}

\begin{example}
    Consider
    \[N = \{(x_0:x_1:x_2) \in \R\P^2 \mid x_2 = 0\}\]
    We can see that \(N \inj \R\P^2\), and \(N \cong S^1\). However, it is impossible to write
    \[N = f^{-1}(q)\]
    for all \(f : \R\P^2 \to M\) where \(M\) is a \(1\)-manifold, \(q \in M\) a regular value.

    To see this, suppose we found such an \(f\). Then consider a chart \(\Psi\) around \(q\) on \(M\), \(\psi(q) = 0\). Then we have that
    \[\psi \circ f : \R\P^2 \supseteq U \to (-1, 1)\]
    with \(N \sse U \sse \R\P^2\), \(U\) is open and connected. In this case, we have
    \[N = \left\{p \mid (\psi \circ f)(p) = 0\right\}\]
    As \(q\) is a regular value, there exists \(p_+, p_- \in U\) such that
    \[\psi(f(p_+)) > 0 \qqtext{and} \psi(f(p_-)) < 0\]
    But \(U \setminus N \sse \R\P^2 \setminus N\) is connected, since it is either an annulus or a disc.
\end{example}

\begin{theorem}
    [Whitney embedding theorem]
    \index{Whitney embedding theorem}
    For any manifold \(M^n\), there exists an embedding \(\psi : M \inj \R^{2n}\).
\end{theorem}

On the examples sheet, we will show the result for \(\R^N\), and with the additional assumption that \(M\) is compact. It's not too difficult to show that an embedding into \(\R^{2n+1}\) exists, Reducing to \(2n\) is the main difficulty in this theorem. Moreover, this result is optimal, for example by considering \(\R\P^2\) which embeds into \(\R^4\) but not \(\R^3\).

\lec

Using what we have done so far, we can now prove \cref{thm:matrix-lie-group-lie-alg-iso}. That is, for matrix Lie groups, the map
\begin{align*}
    \mfg &\to \ell(G) \\
    \xi &\mapsto X_\xi
\end{align*}
is a Lie algebra isomorphism.

\begin{proof}
    We have shown before that \([X_\xi, X_\eta] = X_\zeta\) for some \(\zeta \in \mfg\). We would like to show that \(\zeta = [\xi, \eta] = \xi\eta - \eta\xi\).

    First consider the case when \(G = \GL(n) \sse \Mat(n)\), which is an open subset. In this case, the Lie algebra is \(\mfg = \Mat(n)\). For \(g = (g^i_{\phantom{i}j}) \in G\), the left translation map is a linear function in the coordinates \(g^i_{\phantom{i}j}\). Thus, for \(g, h \in \GL(n)\), we have that
    \begin{align*}
        (\dd L_g)_h : \Mat(n) &\to \Mat(n) \\
        A &\mapsto gA
    \end{align*}
    For \(A = (A^i_{\phantom{i}j}) \in \mfg\), we have that
    \[X_A(\mfg) = \sum X^i_{\phantom i j}(g)\pdv{g^i_{\phantom i j}}\]
    with
    \[X^i_{\phantom i j}(g) = \sum_k g^i_{\phantom i k}A^k_{\phantom k j}\]
    With this, the claim follows from a computation. That is, (using the summation convention),
    \begin{align*}
        g^i_{\phantom i k}\left(A^k_{\phantom k j}\pdv{g^i_{\phantom i j}}\left(g^\ell_{\phantom \ell p} B^p_{\phantom p q}\right) - B^k_{\phantom k j}\pdv{g^i_{\phantom i j}}\left(g^\ell_{\phantom \ell p} A^p_{\phantom p q}\right)\right) \pdv{g^\ell_{\phantom \ell q}}
        &= g^i_{\phantom i k}\left(A^k_{\phantom kj}B^j_{\phantom jq}\delta^i_{\phantom i\ell} - B^k_{\phantom kj}A^j_{\phantom jq}\delta^i_{\phantom i\ell}\right)\pdv{g^\ell_{\phantom \ell q}} \\
        &= g^i_{\phantom ik}(AB - BA)^k_{\phantom k q}\pdv{g^i_{\phantom j q}}
    \end{align*}
    Using this, we see that
    \[[X_A, X_B] = X_{[A, B]}\]

    Now note that the change of basis of \(\mfg\) from the identity chart as above, to the \(\log\) chart is given by
    \[(\dd \log)_I = \id\]
    Thus, the basis are the same, and so the above computation is valid in the \(\log\) chart as well.

    Now consider a more general group satisfying our assumptions. The \(\log\) chart assumption shows that \(\iota : G \to \GL(n)\) is an immersion. Moreover, \(L_g : G \to G\) is a restriction of \(L_g : \GL(n) \to \GL(n)\), and its derivative the restriction of \(\dd L_g : \Mat(n) \to \Mat(n)\).

    Using this, for \(\xi \in \mfg\), the left invariant vector field \(X_\xi \in \ell(G)\) is a restriction of \(X_\xi \in \ell(\GL(n))\). As such,
    \[[X_\xi\vert_G, X_\eta\vert_G] = [X_\xi, X_\eta]\vert_G\]
    We can use test functions which is constant in the variables which are transverse to \(G\) to see this. But we know that \([X_\xi, X_\eta] = X_{\xi, \eta}\) on \(\GL(n)\), and so we have that
    \[X_{[\xi, \eta]}\vert_G = [X_\xi\vert_G, X_\eta\vert_G] \in \ell(G)\]
\end{proof}

\section{Differential forms}

Let \(M^n\) be a manifold, \(p \in M\). We have defined the tangent space \(\T_pM\). Define the \emph{cotangent space of \(M\) at \(p\)}\index{cotangent space} to be
\[\T_p^*M = (\T_pM)^* = \left\{\text{linear maps } \T_pM \to \R\right\}\]
Choosing local coordinates \(x_i\) around \(p\), we have a basis
\[\pdv{x_i} \in \T_pM\]
for \(\T_pM\). This gives us a dual basis, which we will denote as
\[\dd x_i \in \T_p^*M\]
That is,
\[\inner{\dd x_i, \pdv{x_j}} := \dd x_i\left(\pdv{x_j}\right) = \delta_{ij}\]
For any \(a \in \T_p^*M\), we have that
\[a = \sum_i a_i \dd x_i\]
for \(a_I \in \R\). Recall if \(y_i\) is a different choice of local coordinates, then
\[\pdv{y_j} = \sum_i \pdv{x_i}{y_j}\pdv{x_i}\]
Then by standard linear algebra arguments,
\[\dd x_i = \sum_j \pdv{x_i}{y_j}\dd y_j\]
Thus, for
\[a = \sum a_i \dd x_i = \sum b_i \dd y_i \in \T_p^*M\]
we have that
\[b_j = \sum_i a_i \pdv{x_i}{y_j}\]

We can define the \emph{cotangent bundle of \(M\)}\index{cotangent bundle}, as
\[\T M = \bigsqcup_{p \in M}\T_p^* M\]

\begin{theorem}
    \(\T^* M\) is a smooth manifold, with \(\dim(\T^* M) = 2\dim(M)\). Again, we have a projection map \(\pi : \T^* M \to M\), with \(\pi(p, a) = p\), which is a smooth map.
\end{theorem}

\begin{proof}
    As for the tangent bundle. The only difference is the transformation law.
\end{proof}

\begin{definition}
    [differential \(1\)-form] A \emph{((smooth) differential) \(1\)-form}\index{differential \(1\)-form} \(\alpha\) on \(M\), is by definition a smooth map \(\alpha : M \to \T^* M\), such that \(\pi \circ \alpha = \id\). That is, for all \(p \in M\), \(\alpha(p) \in \T_p^*M\).
\end{definition}

As for vector fields, in local coordinates \(x_i\), we have that
\[\alpha\vert_U = \sum_i \alpha_i(p)\dd x_i\]
where \(\alpha_i \in C^\infty(U)\). This is equivalent to saying that
\[\inner{\alpha, X} \in C^\infty(M)\]
for all \(X \in V(M)\). To see this, note that
\[\inner{\alpha, \pdv{x_i}} = \alpha_i\]

We will now recall/define some multilinear algebra. For \(r = 0,1, 2, \dots\), and a vector space \(V\), define the \emph{\(r\)-th exterior power of \(V^*\)} as
\[\Lambda^rV^* = \left\{\text{alternating multilinear maps }\underbrace{V \times \cdots \times V}_{r \text{ copies}} \to \R\right\}\]
In particular, if \(r = 0\), by convention \(\Lambda^0V^* = \R\), and if \(r = 1\), \(\Lambda^1V^* = V^*\). If \(r > \dim(V)\), then \(\Lambda^rV^* = 0\). 

From this, we obtain the vector spaces \(\Lambda^r\T_p^* M\), and we can define the \emph{\(r\)-th exterior power of \(\T^*M\)} as
\[\Lambda^r\T^* M = \bigsqcup_{p \in M}\Lambda^r\T_p^* M\]
Consider a smooth map \(\alpha : M \to \Lambda^r\T^*M\), with \(\pi \circ \alpha = \id\). We call \(\alpha\) a \emph{differential \(r\)-form}\index{differential \(r\)-form}.

\lec

This is a vector space, with dimension \(\dim(\Lambda^r\T_p^* M) = {n \choose r}\). To see this, we can consider the basis
\[\dd x_{i_1} \wedge \cdots \wedge \dd x_{i_r} \text{ with }1 \le i_1 < \cdots < i_r \le n\]
By definition,
\[(\dd x_{i_1} \wedge \cdots \wedge \dd x_{i_r})(v_1, \dots, v_r) := \det((\dd x_{i_k}(v_\ell))_{k, \ell})\]

In particular, we see that \(\wedge\) is antisymmetric, and this extends by linearity and induction on \(p, q\), to extend it to an assosiative bilinear product
\[\Lambda^p \T_x^*M \times \Lambda^q \T_x^* M \to \Lambda^{p+q}\T_x^* M\]

\begin{notation}
    [multi-index notation] If \(I = (i_1, \dots, i_r)\), then we write \(\abs{I} = r\), and
    \[\dd x_I = \dd x_{i_1} \wedge \cdots \wedge \dd x_{i_r}\]
\end{notation}

The transformation law from \(\dd x_I\) to \(\dd y_J\) is given by
\[\dd y_J = \sum_{\abs{I} = r}\left(\prod_{k=1}^r\pdv{y_{j_k}}{x_{i_k}}\right)\dd x_I\]
Using this, we can show that \(\Lambda^r \T^* M\) is a smooth manifold, with dimension \(n + {n \choose r}\). The projection map \(\pi : \Lambda^r \T^*M \to M\) is smooth, and we call \(\Lambda^r\T^*M\) the \emph{bundle of \(r\)-forms}\index{bundle of \(r\)-forms, \(\Lambda^r\T^*M\)}.

\begin{definition}
    [smooth differential \(r\)-form]
    A \emph{((smooth) differential) \(r\)-form}\index{differential \(r\)-form} is a section \(\alpha\) of \(\Lambda^r\T^*M\). The \emph{degree} of \(\alpha\) is \(r\).
\end{definition}

Note that \(\alpha\) as above is smooth if and only if for any \(X_1, \dots, X_r\) vector fields on \(M\), \(\alpha(X_1, \dots, X_n) = 0\). In local coordinates, we can write
\[\alpha = \sum_{\abs{I} = r}\alpha_I \dd x_I\]
where the \(\alpha_I\) are smooth functions.

\begin{definition}
    [space of \(r\)-forms] For a manifold \(M\), we write \(\Omega^r(M)\)\index{space of \(r\)-forms, \(\Omega^r(M)\)} for the space of all \(r\)-forms on \(M\). By convention, \(\Omega^0(M) = C^\infty(M)\).
\end{definition}

\subsection{Orientability}

\begin{theorem}
    For any \(n\)-dimensional manifold, the following are equivalent:
    \begin{enumerate}[(a)]
        \item There exists a nowhere vanishing \(\omega \in \Omega^n(M)\).
        \item There exists a collection of coordinate charts \((\varphi_\alpha, U_\alpha)\) such that
        \begin{itemize}
            \item \(\bigcup_\alpha U_\alpha = M\),
            \item for \(\alpha, \beta\) with corresponding coordinates \(x_i, y_j\), we have
            \[\det(\pdv{y_j}{x_i}) > 0\]
        \end{itemize}
        \item \(\Lambda^n \T^* M\) is isomorphic to \(M \times \R\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    \textbf{(a) \(\implies\) (b):} From the transformation law and taking \(r = n\), we see that
    \begin{align*}
        \dd x_1 \wedge \cdots \wedge \dd x_n &=\left(\sum_{i_1}\pdv{x_1}{y_{i_1}}\dd y_{i_1}\right) \wedge \cdots \wedge \left(\sum_{i_n}\pdv{x_n}{y_{i_n}}\dd y_{i_n}\right) \\
        &= \left(\sum_{\sigma \in S_n}\sign(\sigma)\pdv{x_1}{y_{\sigma(1)}}\cdots \pdv{x_n}{y_{\sigma(n)}}\right) \\
        &= \det(\pdv{x_i}{y_j})\dd y_1 \wedge \cdots \wedge \dd y_n
    \end{align*}
    
    Given \(\omega\) as in (a), consider an open cover by coordinate neighbourhoods \(U_\alpha\), with all \(U_\alpha\) connected. In this case,
    \[\omega\vert_{U_\alpha} = f_\alpha(x) \dd x^\alpha_1 \wedge \cdots \wedge \dd x^\alpha_n\]
    We can assume without loss of generality that \(f_\alpha > 0\) on each coordinate neighbourhood, but then
    \[f_\alpha = \det(\pdv{x^\alpha_i}{x^\beta_j})f_\beta\]
    and so the determinant must be positive.

    \textbf{(b) \(\implies\) (a)}: We will assume the following theorem without proof:
    \begin{theorem}
        For every open cover \(M = \bigcup_\alpha U_\alpha\), we can find a countable collection of smooth functions \(\rho_i \in C^\infty(M)\), such that
        \begin{enumerate}[(i)]
            \item \(\Supp(\rho_i)\) is compact, and contained in some \(U_\alpha\),
            \item the \(\rho_i\) are locally finite, that is, given \(x \in M\), there exists a neighbourhood \(W_x\) of \(x\), such that all but finitely many \(\rho_i\) are identically zero on \(W_x\).
            \item \(\rho_i(x) \in [0, 1]\), and for any \(x \in M\),
            \[\sum_i \rho_i(x) = 1\]
        \end{enumerate}
        We call \(\{\rho_i\}\) a \emph{partition of unity subordinate to \(U_\alpha\)}\index{partition of unity}.
    \end{theorem}

    For any \(\alpha\), define \(\omega_\alpha = \dd x^\alpha_1 \wedge \cdots \wedge \dd x^\alpha_n \in \Omega^n(U_\alpha)\). By passing to a subcollection, we may assume we only have countably many \(\alpha\), and say we have a parition of unity \(\rho_\alpha\) subordinate to \(\{U_\alpha\}\). Then
    \[\rho_\alpha\omega_\alpha \in \Omega^n(M)\]
    and using this, we can define
    \[\omega = \sum \rho_\alpha \omega_\alpha\]
    Since the sum is locally finite, \(\omega\) is well defined. Moreover, since the \(\rho_\alpha\) are nonnegative, and
    \[\det(\pdv{x_i^\alpha}{x_j^\beta}) > 0\]
    \(\omega\) never vanishes.

    For \textbf{(c) \(\implies\) (a)}, suppose we have an isomorphism \(\Phi : \Lambda^n\T^* M \to M \times \R\) of vector bundles, consider \(f : M \to M \times \R\), given by \(f(x) = (x, 1)\). Then \(\Phi^{-1} \circ f \in \Omega^n(M)\) never vanishes.

    Conversely, given a non-zero \(\omega\), notice for any \((p, \xi) \in \Lambda^n \T^* M\), there exists \(a_\xi\) such that \(\xi = a_\xi\omega(p)\). Thus, we can define
    \[\Phi(p, \xi) = (\pi(\xi), a_\xi)\]
    We leave as an exercise to check that \(\Phi\) is a diffeomorphism.
\end{proof}

\begin{definition}
    [orientable] A manifold \(M\) satisfying any of the conditions in the above theorem is called \emph{orientable}\index{orientable!manifold}.
\end{definition}

Suppose \(M\) is connected. then there are \emph{two} choices of \emph{orientation}\index{orientation}, given by respectively:
\begin{enumerate}[(a)]
    \item \(\omega\) up to rescaling by a positive function,
    \item a choice of an open up to `positive compatibility'.
    \item a choice of \(\Phi\) up to compositions \((x, a) \mapsto (x, h(x)a)\), where \(h \in C^\infty(M)\) is positive.
\end{enumerate}

\begin{definition}
    [oriented] A manifold \(M\) with a choice of orientation is called \emph{oriented}\index{oriented!manifold}.
\end{definition}

\subsection{Exterior derivative}

Consider \(f \in C^\infty(M)\). Formally, it is a smooth map \(f : M \to \R\) between manifolds, and so we have a differential
\[\dd f_p : \T_p M \to \T_{f(p)}\R = \R\]
which depends on \(p\) in a smooth way. Then
\[\dd f \in \Omega^1(M)\]

\begin{remark}
    A coordinate \(x_i : U \to \R\) is a smooth function, and so \(\dd x_i\) is a \(1\)-form.    
\end{remark}

% ES2 is out, questions 1 and 2

\begin{theorem}
    There exists a unique \(\R\)-linear map \(\dd : \Omega^k(M) \to \Omega^{k+1}(M)\), determined by
    \begin{enumerate}[(i)]
        \item If \(f \in \Omega^0(M) = C^\infty(M)\), then \(\dd f\) is just the differential. That is,
        \[\dd f = \sum_i \pdv{f}{x_i}\dd x_i\]
        \item \(\dd(\omega \wedge \eta) = (\dd\omega)\wedge \eta + (-1)^{\deg(\omega)}\omega \wedge \dd\eta\)
        \item \(\dd^2 = 0\).
    \end{enumerate}
    The map \(\dd\) is called the \emph{exterior derivative}\index{exterior derivative}.
\end{theorem}

\begin{proof}
    Consider a coordinate neighbourhood \(U \sse M\), with coordinates \(x_i\). Consider \(\omega \in \Omega^r(M)\), with
    \[\omega\vert_U = f(x)\dd x_I\]
    We can assume this since \(\dd\) is linear. By (ii), (iii) and then (i),
    \begin{align*}
        \dd(\omega\vert_U) &= \dd(f(x)\dd x_I) = \dd f \wedge \dd x_I + f(x)\sum_{k=1}^r\left(\cdots \wedge (-1)^{k-1}\dd\dd x_{i_k} \wedge \cdots\right) \\
        &= \sum_{i=1}^m \pdv{f}{x_i}\dd x_i \wedge \dd x_I \tag{\(*\)}
    \end{align*}

    Using this definition, (i) is clear, (ii) can be verified by a computation, and (iii) follows from the symmetry of mixed partial derivatives, since
    \[\pdv[2]{f}{x_i}{x_j} = \pdv[2]{f}{x_j}{x_i}\]
    we have that
    \[\sum_{i, j}\pdv[2]{f}{x_i}{x_j}\dd x_i \wedge \dd x_j = 0\]

    Using this, we have shown uniqueness since it is well defined and unique with respect to anny choice of local coordinates. To show existence, we need to show that it is well defined with respect to a change of coordinates. But by \((*)\), \((\dd\omega)_p\) is determined by \(\omega\) restricted to any neighbourhood of \(p\).

    Thus, to show that \(\dd\) is well defined, it suffices to consider the intersection of coordinate neighbourhoods. Let \(y_i\) be the other coordinates. The same computation as above gives an operator \(\delta\). We want to show that \(\delta = \dd\). With this,
    \[\delta(f(x)\dd x_I) = \delta f \wedge \dd x_I + \sum_k\left(\cdots \wedge (-1)^{k-1}\delta\dd x_{i_k}\wedge \cdots\right)\]
    Now \(\delta f = \dd f\) as the differential is well defined, and \(\delta(x_{i_k}) = \dd x_{i_k}\) as well. Thus, \(\delta\dd x_{i_k} = \delta^2 x_{i_k} = 0\). With this, we find that
    \[\delta(f(x)\dd x_I) = \dd f \wedge \dd x_I\]
    as required.
\end{proof}

\begin{definition}
    [pullback] Let \(f : M \to N\) be a smooth map between manifolds. Then we have a map
    \[f^* : \T^*_{f(p)}N \to \T^*_pM\]
    and more generally, we have
    \[f^* : \Omega^r(N) \to \Omega^r(M)\]
    which we call the \emph{pullback map of \(f\)}\index{pullback!of differential forms}. By definition,
    \[(f^*\alpha)_p(v_1, \dots, v_r) = \alpha_{f(p)}((\dd f)_pv_1, \dots, (\dd f)_p v_r)\]
    for every \(r\)-form \(\alpha\) on \(N\), \(p \in M, v_1, \dots, v_r \in \T_pM\).
\end{definition}

\begin{remark}
    \begin{itemize}
        \item \(f^*\) is defined for any smooth map. Compare this to the pushforward of vector fields, which is defined only when \(f\) is a diffeomorphism.
        \item \((f \circ g)^* = g^* \circ f^*\).
        \item \(f^*(\alpha \wedge \beta) = f^*(\alpha) \wedge f^*(\beta)\). In particular, for any \(h \in C^\infty(N)\), \(f^*(h\alpha) = (h \circ f)f^*\alpha\). That is, \(f^*\) is \(C^\infty(\cdot)\)-linear.
        \item \(\dd(f^*\alpha) = f^*(\dd\alpha)\). To see this, it suffices to do this in local coordinates, i.e. we can take \(M, N\) to be open subsets of Euclidean space. Moreover, we can assume \(\alpha = \dd y_j\), \(v = \pdv{x_k}\), since we know both operators are \(\R\)-linear, and we can use the Leibniz rule to extend to general \(r\)-forms. In this case, \(f\) becomes \(y_j = y_j(x)\), and we have that
        \[(f^*\dd y_j)\left(\pdv{x_k}\right) = \dd y_j\left((\dd f)\pdv{x_k}\right) = \dd y_j\left(\sum_\ell\pdv{y_\ell}{x_k}\pdv{y_\ell}\right) = \pdv{y_j}{x_k}\]
        We see that \(f^*\) is determined by
        \[f^*\dd y_j = \sum_{k=1}^n \pdv{y_j}{x_k}\dd x_k = \dd(y_j \circ f) = \dd(f^*y_j)\]
    \end{itemize}
\end{remark}

If \(\alpha = \dd\beta\), then \(\dd\alpha = 0\). The converse may not hold. We can use this:

\begin{definition}
    [exact form] If \(\alpha = \dd\beta\) for some \(\beta\), we say that \(\alpha\) is \emph{exact}. Equivalently, \(\alpha \in \im(\dd_r)\) for some \(r\).
\end{definition}

\begin{definition}
    [closed form] If \(\alpha \in \ker(\dd_r)\), then we say that \(\alpha\) is \emph{exact}.
\end{definition}

Since \(\dd^2 = 0\), \(\im(\dd_{r-1}) \sse \ker(\dd_r)\). Moreover, it is a vector subspace.

\begin{definition}
    [de Rham cohomology]
    The vector space
    \[\HH^r(M) = \frac{\ker(\dd_r)}{\im(\dd_{r-1})}\]
    is called the \emph{\(r\)-th de Rham cohomology group of \(M\)}\index{de Rham cohomology}. We write
    \[\HH^* = \bigoplus_r \HH^r(M)\]
\end{definition}

\begin{proposition}
    If \(f : M \to N\) is smooth, the pullback \(f^* : \Omega^r(N) \to \Omega^r(M)\) descends to the quotient, i.e. we have a homomorphism
    \[f^* : \HH^r(N) \to \HH^r(M)\]
    Moreover, as for pullback of forms,
    \[(g \circ f^*) = f^* \circ g^*\]
\end{proposition}

\begin{proof}
    This follows from \(f^*\dd = \dd f^*\) (i.e. \(f^*\) is a chain map).
\end{proof}

In particular, if \(f\) is a diffeomorphism, then \(f^* : \HH^*(N) \to \HH^*(M)\) is an isomorphism. The converse is false, for example
\[\HH^*(\R\P^3) \cong \HH^*(S^3)\]
but they are diffeomorphic (not even homeomorphic, for example consider \(\pi_1\)).

\begin{theorem}
    [de Rham]
    \index{de Rham's theorem}
    \[\HH^r_{\text{dR}}(M) \cong \HH^r(M; \R)\]
    where the right hand side is the singular cohomology of \(M\) as a topological space, with real coefficients.
\end{theorem}

If \(M\) is connected, then \(\HH^0(M) = \R\), as it is just the space of constant functions (i.e. functions with zero differential).

\begin{theorem}
    [Poincar\'e lemma] 
    \label{thm:poincare-lemma}
    \index{Poincar\'e lemma}
    Suppose \(U \sse \R^n\) is an open ball, \(\alpha \in \Omega^k(U)\), for some \(k > 0\), with \(\dd\alpha = 0\). Then \(\alpha\) is exact, that is, we can find \(\beta \in \Omega^{k-1}(U)\), with \(\alpha = \dd\beta\).
\end{theorem}

\begin{corollary}
    \[\HH^k(U) = \begin{cases}
        \R & k = 0 \\
        0 & \text{otherwise}
    \end{cases}\]
\end{corollary}

\begin{proof}
    [Sketch proof of \cref{thm:poincare-lemma}] The main idea is to invert the exterior derivative \(\dd\), by constructing appropriate integral operators,
    \[h_k : \Omega^k(U) \to \Omega^{k-1}(U)\]
    with
    \[h_{k+1} \circ \dd_k + \dd_{k-1} \circ h_k = \id_{\Omega^k(U)}\]
    That is, \(h\) defines a homotopy between \(\id\) and the zero map on the de Rham cochain complex.
\end{proof}

\subsection{Integration}

Throughout, assume \(M\) is a oriented \(n\)-manifold, with orientation given by positively oriented charts \((U_\alpha, \varphi_\alpha)\).

Let \(\omega \in \Omega^n(M)\), and suppose \(\Supp(\omega)\) is compact\footnote{This is different to lectures, but I'll use the definition that
\[\Supp(\omega) = \overline{\{p \in M \mid \omega(p) \ne 0\}}\]}. If \(\Supp(\omega) \sse U_\alpha\) for some \(\alpha\), with local coordinates \(x_i\), then we can write
\[\omega\vert_{U_\alpha} = f(x)\dd x_1 \wedge \cdots \wedge \dd x_n\]
In this case, we define
\[\int_M \omega = \int_{\varphi_\alpha(U_\alpha)}f(x)\dd x_1 \cdots \dd x_n\]
where the right hand side is the integral over a subset of \(\R^n\), i.e. Riemann/Lebesgue integration.

To see that this is well defined, suppose \(\Supp(\omega) \sse U_\beta\), and \(U_\beta\) has local coordinates \(y_j\), then
\[\omega\vert_{U_\beta} = h(y)\dd y_1 \wedge \cdots \wedge \dd y_n\]
In this case,
\begin{align*}
    \int_{\varphi_\beta(U_\beta)} h(y)\dd y_1\cdots \dd y_n &= \int_{\varphi_\alpha(U_\alpha)}h(y(x))\abs{\det\left(\pdv{y_i}{x_j}\right)}\dd x_1 \cdots \dd x_n \\
    &= \int_{\varphi_\alpha(U_\alpha)}h(y(x))\det\left(\pdv{y_i}{x_j}\right)\dd x_1 \cdots \dd x_n \\
    &= \int_{\varphi_\alpha(U_\alpha)}f(x)\dd x_1 \wedge \cdots \wedge \dd x_n
\end{align*}

For more general \(n\)-forms with compact support, use a parition of unity \(\{\rho_i\}\) subordinate to a finite cover \(\{U_i\}\) of \(\Supp(\omega)\) by coordinate patches \(U_1, \dots, U_N\). Let \(U_0 = M \setminus \Supp(\omega)\).
\begin{definition}
    [integral of a differential form]
    The \emph{integral of \(\omega\)}\index{integral of a differential form} is
    \[\int_M \omega = \sum_{i=1}^N \int_{U_i}\rho_i\omega\]
\end{definition}
Basic properties:
\begin{itemize}
    \item it is linear in \(\omega\),
    \item it is additive when we integrate over disjoint coordinate neighbourhoods,
    \item it is independent of the choice of a parition of unity. If \(\{\tilde\rho_j\}\) is a different partition of unity, then
    \[\sum_i \int_{U_i} \rho_i\omega = \sum_{i, j}\int_{U_i} \rho_i\tilde\rho_j\omega = \sum_{i, j}\int_{U_i \cap \tilde U_j}\rho_i\tilde\rho_j \omega\]
    Swapping \(\rho_i\) and \(\tilde\rho_j\) gives the same result, and so we are done.
\end{itemize}

\begin{theorem}
    [Stoke's for manifolds without boundary] \index{Stoke's theorem}If \(\eta \in \Omega^{n-1}(M)\) is compactly supported, then
    \[\int_M \dd \eta = 0\]
\end{theorem}

Note \(\eta\) being compactly supported implies \(\dd\eta\) is compactly supported, but the converse is false.

\begin{proof}
    As above, let \(U_1, \dots, U_N\) be a positively oriented coordinate neighbourhoods covering \(\Supp(\eta)\), and let \(U_0 = M \setminus \Supp(\eta)\), which is positively oriented with respect to \(U_1, \dots, U_N\). Choose a partition of unity subordinate to \(\{U_0, \dots, U_N\}\). Then
    \[\dd \eta = \dd\left(\sum_{i=1}^N \rho_i\eta\right) = \sum_{i=1}^N \dd(\rho_i\eta)\]
    By additivity, suffices to show
    \[\int_M \dd(\rho_i\eta) = 0\]
    Fix \(i\), and choose local coordinates \(x_k\) on \(U_i\). Without loss of generality,
    \[\rho_i\eta = h(x) \dd x_2 \wedge \cdots \dd x_n\]
    We can do this as the integral is linear. Then
    \[\dd(\rho_i\eta) = \pdv{h}{x_i}\dd x_1 \wedge \cdots \dd x_n\]
    Integrating, choose \(R\) large enough such that
    \begin{align*}
        \int_{\R^n}\dd(\rho_i\eta) &= \int_{\R^{n-1}}\left(\int_{x_1=-R}^R\pdv{h}{x_1}\dd x_1\right)\dd x_2 \cdots \dd x_n \\
        &= \int_{\R^{n-1}}h(R, x_2, \dots, x_n) - h(-R, x_2, \dots, x_n)\dd x_2\cdots \dd x_n \\
        &= 0
    \end{align*}
    as we can assume \(h(R, x_2, \dots, x_n) = h(-R, x_2, \dots, x_n) = 0\).
\end{proof}

\begin{corollary}
    [integration by parts] \index{integration by parts}Assume \(\alpha, \beta\) are differential forms on \(M\), at least one is compactly supported, with \(\deg(\alpha) + \deg(\beta) = \dim(M) - 1\). Then
    \[\int_M \alpha \wedge \dd \beta = (-1)^{\deg(\alpha) + 1}\int_M (\dd\alpha) \wedge \beta\]
\end{corollary}

\begin{proof}
    Apply Stokes to \(\eta = \alpha \wedge\beta\), which is compactly supported.
\end{proof}

\section{Vector bundles}

\begin{definition}
    [vector bundle] A \emph{vector bundle \(E\) over a manifold \(B\)}\index{vector bundle} is
    \begin{enumerate}
        \item a smooth manifold \(E\),
        \item a surjective submersion \(\pi : E \to B\), i.e. \(\pi\) is smooth, and \((\dd\pi)_p\) is surjective,
    \end{enumerate}
    Moreover,
    \begin{enumerate}[(i)]
        \item there exists a (finite dimensional) vector space \(V\), such that for all \(p \in B\), \(E_p = \pi^{-1}(p)\) is a vector space isomorphic to \(V\),
        \item for any \(p \in B\), there exists an open neighbourhood \(U\) of \(p\), and a diffeomorphism \(\Phi_U\) making the diagram
        % https://q.uiver.app/#q=WzAsMyxbMCwwLCJcXHBpXnstMX0oVSkiXSxbMiwwLCJVIFxcdGltZXMgViJdLFsxLDIsIlUiXSxbMCwxLCJcXFBoaV9VIl0sWzAsMiwiXFxwaSIsMl0sWzEsMiwiXFxwcl8xIl1d
        \[\begin{tikzcd}
            {\pi^{-1}(U)} && {U \times V} \\
            \\
            & U
            \arrow["{\Phi_U}", from=1-1, to=1-3]
            \arrow["\pi"', from=1-1, to=3-2]
            \arrow["{\pr_1}", from=1-3, to=3-2]
        \end{tikzcd}\]
        commute.
        \item for each \(p \in U\), the restriction \(\Phi_U : E_p \to \{p\} \times V \cong V\) is a linear isomorphism.
    \end{enumerate}

    We call 
    \begin{itemize}
        \item \(B\) the \emph{base}\index{base space}, 
        \item \(E\) the \emph{total space}\index{total space}, 
        \item \(\pi\) the \emph{bundle projection}\index{bundle projection},
        \item \(V\) the \emph{typical fibre}\index{typical fibre}, if \(V\) is a real vector space, we call \(E\to B\) a \emph{real vector bundle}\index{real vector bundle}, and if \(V\) is a complex vector space, then we call \(E \to B\) a \emph{complex vector bundle}\index{complex vector bundle}.
        \item \(\dim(V)\) is the \emph{rank of the vector bundle}\index{rank!of a vector bundle}\footnote{Note in the case of a complex vector bundle this depends on the real/complex dimension.},
        \item \(\Phi_U\) a \emph{local trivialisation over \(U\)}\index{local trivialisation},
        \item \(U\) is a \emph{trivialising neighbourhood}\index{trivialising neighbourhood}
    \end{itemize}
\end{definition}

\begin{remark}
    Note surjectivity of \(\pi\) follows from (i), and the fact that it is a submersion follows from (ii).
\end{remark}

\begin{definition}
    [(local) section] A \emph{section}\index{section} of a vector bundle \(E \to B\) is a smooth map \(s : B \to E\) such that \(\pi \circ s = \id_B\). A \emph{local section}\index{local section} is a smooth map \(s ; U \to E\), with \(\pi \circ s = \id_U\).
\end{definition}

\begin{example}
    [product] If \(E = B \times V\), \(E \to B\) is a vector bundle, and the space of sections is just \(C^\infty(B, V)\).
\end{example}

\begin{example}
    [(co)tangent space] \(\T M\) and \(\T^*M\) are real vector bundles of rank \(n = \dim(M)\). The sections are vector fields and \(1\)-forms respectively. In general, these are non-trivial, i.e. they are not products.

    In general, \(\Lambda^r\T^*M\) is also a real vector bundle, and sections are \(r\)-forms. If \(\T M\) is trivial, then so are the bundles of differential forms.
\end{example}

\begin{example}
    [tautological vector bundle] Over \(\R\P^n, \C\P^n\) (or more generally, Grassmannians), we have the \emph{tautological vector bundle}\index{tautological vector bundle}.

    Say \(B = \C\P^n\), take
    \[E = \bigsqcup_{\ell \text{ line in }\C^{n+1}\text{ through 0}}\ell\]
    The bundle projection sends
    \[\pi : E\ni \ell \mapsto \ell \in \C\P^n\]
    We can check that this is a rank \(1\) complex vector bundle (i.e. a complex line bundle).
\end{example}

\subsection{Structure group and transition functions}

Let \((U_\alpha, \Phi_\alpha), (U_\beta, \Phi_\beta)\) be local trivialisations of \(E\). In this case, we have that for \(b \in U_\alpha \cap U_\beta, v \in V\),
\[\Phi_\beta \circ \Phi_\alpha^{-1}(b, v) = (b, \psi_{\beta\alpha}(b)v)\]
where \(\psi_{\beta\alpha} : U_\alpha \cap U_\beta \to \GL(V)\) is smooth.

\begin{definition}
    [transition functions] The \(\psi_{\beta\alpha}\) are called the \emph{transition functions}\index{transition functions!of a vector bundle}.
\end{definition}

The transition functions satisfy the following:
\begin{itemize}
    \item \(\psi_{\alpha\alpha}(b) = \id\),
    \item \(\psi_{\alpha\beta}(b)\psi_{\beta\alpha}(b) = \id\),
    \item \(\psi_{\alpha\beta}(b)\psi_{\beta\gamma}(b)\psi_{\gamma\alpha}(b) = \id\)
\end{itemize}
which we call the \emph{cocycle conditions}\index{cocycle conditions}.

\begin{example}
    [(co)tangent bundle]
    In the case of the (co)tangent bundles, with the charts defined earlier, the transition functions are given by the derivatives (or the dual matrix)
    \[\left(
        \pdv{x_i}{y_j}
    \right)_{i, j}\]
\end{example}

\begin{proposition}
    The following data:
    \begin{itemize}
        \item base manifold \(B\),
        \item trivialising neighbourhoods \(\{U_\alpha\}_{\alpha \in A}\) covering \(B\),
        \item maps \(\psi_{\beta\alpha} : U_\alpha \cap U_\beta \to \GL(V)\) smooth, satisfying the cocycle condition.
    \end{itemize}
    determines a vector bundle \(E\) over \(B\), with typical fibre \(V\).
\end{proposition}

The proof is called the Steenrod construction.

\begin{proof}
    First we define the total space
    \[E := \frac{\bigsqcup_{\alpha \in A}(U_\alpha \times V)}{\sim}\]
    where \(U_\alpha \times V \ni (b, v) \sim (b, \psi_{\beta\alpha}(b)v) \in U_\beta \times V\).

    Without loss of generality, we may assume that the \(U_\alpha \sse B\) are coordinate neighbourhoods, with charts \(\varphi_\alpha\) Then
    \[\widetilde\varphi_\alpha = (\varphi_\alpha, \id_V)\]
    for each \(\alpha\) is a chart on \(U_\alpha \times V\). This implies that \(E\) is a manifold, as these form a smooth structure. The definitions of \(\pi\) and the existence of local trivialisations are clear.
\end{proof}

\begin{definition}
    [\(G\)-structure, structure group]
    Let \(E \to B\) be a real vector bundle with typical fibre \(V\). Let \(G \le \GL(V)\) be a subgroup. Suppose there exists local trivialisations \((U_\alpha, \Phi_\alpha)\) covering \(B\), where the transition maps \(\psi_{\alpha\beta}(b) \in G\) for all \(b \in U_\alpha \cap U_\beta\). We call this a \emph{\(G\)-structure on \(E \to B\)}\index{G-structure@\(G\)-structure}, and that \(E\) has \(G\) as a \emph{structure group}\index{structure group}.
\end{definition}

\begin{example}
    If \(G = \{\id\}\), then \(E\) has a global trivialisation \(\Phi : E \to B \times V\).
    
    If \(G = \GL_+(V)\) is the subgroup of matrices with positive determinant. Then a \(G\)-structure on \(E\) is an orientation of \(E\). If \(E = \T M\), then this determines an orientation of \(M\).
\end{example}

\begin{definition}
    [orientable, oriented] A vector bundle \(E \to B\) is \emph{orientable}\index{orientable!vector bundle} if it admits a \(\GL_+(V)\) structure. \(E \to B\) is \emph{oriented}\index{oriented!vector bundle} if we made a choice of orientation.
\end{definition}

\lec

\begin{example}
    [orthogonal trivialisations] Let \(E\) be a real vector bundle, if \(G = O(V)\), then there exists an invariantly defined \emph{inner product} on each fibre \(E_p\). More precisely, this is obtained via a linear isometry \(E_p \to (\R^k, \inner{\cdot, \cdot})\). The respective \(\Phi_{\alpha\beta}\) giving the \(O(V)\) structure are then called \emph{orthogonal (local) trivialisations}\index{orthogonal trivialisation}.

    In particular, if \(G = \SO(V) = \GL_+(V) \cap O(V)\), then this is equivalent to a choice of an inner product as well as an orientation on the fibres. 
\end{example}

\begin{example}
    Say now \(E\) is a real vector bundle with rank \(2m\). Then we can consider \(G = \GL(m, \C) \le \GL(2m, \R)\). For all \(p \in B\), there exists \(J_p \in \GL(E_p)\), with \(J_p^2 = -\id_{E_p}\). Moreover, \(J_p\) depends smoothly on \(p\). This corresponds to \(J_0\) on \(\R^{2n} = \C^n\), defined by multiplication by \(i\). This naturally makes \(E\) into a complex vector bundle.

    If \(E = \T M\), a \(\GL(n, \C)\)-structure on \(M\) is called an \emph{almost complex structure}\index{almost complex structure} on \(M\).
\end{example}

\begin{example}
    If \(G = U(k) \sse \GL(k, \C)\), we get a \emph{unitary structure}, analogous to the orthogonal case. The trivialisations are called \emph{unitary trivialisations}\index{unitary trivialisation}.
\end{example}

More generally, if \(G\) preserves a tensor \(T\) on \(V\) under the inclusion \(G \to \GL(V)\) (or the appropriate action on the dual), then a \(G\)-structure induces a family \(T_p\) of tensors on \(E_p\), which are equivalent to \(T\) under the trivialisations.

In general, existence of a \(G\)-structure for a given \(G\) can be a non-trivial problem. But for \(G = O(V)\) and \(G = U(V)\), the answer is yes.

\subsection{Principal bundles}

Let \(G\) be a Lie group, with identity element \(1_G\).

\begin{definition}
    [smooth free right action] A \emph{smooth free right action}\index{smooth free right action} of \(G\) on a manifold \(P\) is a smooth map
    \begin{align*}
        P \times G &\to P \\
        (p, h) &\mapsto p\cdot h
    \end{align*}
    where
    \begin{itemize}[(i)]
        \item free: If \(p\cdot h = p\), then \(h = 1_G\),
        \item \((p\cdot h_1)\cdot h_2 = p\cdot (h_1h_2)\). 
    \end{itemize}
\end{definition}

\begin{remark}
    The fact that it is a right action implies that the map \(p \mapsto ph\) is a diffeomorphism for all \(h \in G\).
\end{remark}

\begin{definition}
    [principal bundle] Let \(G\) be a Lie group. A \emph{principal \(G\)-bundle}\index{principal bundle} \(P\) over a manifold \(B\) is
    \begin{enumerate}
        \item a smooth manifold \(P\), with a smooth free right action of \(G\) on \(P\),
        \item a surjective submersion \(\pi : P \to B\),
    \end{enumerate}
    Moreover, 
    \begin{enumerate}[(i)]
        \item 
        for any \(b \in B\), there exists an open neighbourhood \(U\) of \(b\) and a diffeomorphism \(\Phi_U\) making the diagram
        % https://q.uiver.app/#q=WzAsMyxbMCwwLCJcXHBpXnstMX0oVSkiXSxbMiwwLCJVIFxcdGltZXMgRyJdLFsxLDIsIlUiXSxbMCwyLCJcXHBpIiwyXSxbMCwxLCJcXHBoaV9VIl0sWzEsMiwiXFxwcl8xIl1d
        \[\begin{tikzcd}
            {\pi^{-1}(U)} && {U \times G} \\
            \\
            & U
            \arrow["\pi"', from=1-1, to=3-2]
            \arrow["{\phi_U}", from=1-1, to=1-3]
            \arrow["{\pr_1}", from=1-3, to=3-2]
        \end{tikzcd}\]
        commute.
        \item \(\Phi_U\) commutes with the action of \(G\). That is,
        \[\Phi_U(ph) = (b, gh) = (\pi(p), gh)\]
        where \(\Phi_U(p) = (b, g)\). 
    \end{enumerate}
    We call
\end{definition}

\begin{remark}
    From the definitions, it follows that the fibres \(\pi^{-1}(p)\) are embedded submanifolds, diffeomorphic to \(G\). By analogy, we will call \(P\) the total space, \(B\) the base and so on.
\end{remark}

\begin{remark}
    [Warning] If \(P \to B\) is a fibre bundle, where each fibre is a Lie group, \(P\) does not have to be a principal bundle, since we don't need to have a right action on each fibre.
\end{remark}

Similar to the case of a vector bundle, we have: For every \(b \in U_\alpha \cap U_\beta\), consider
\[\Phi_\beta \circ \Phi_\alpha^{-1}(b, g) = (b, \psi_{\beta\alpha}(b, g))\]
where
\[\psi_{\beta\alpha}(b, \cdot) : G \to G\]
The requirement that \(\Phi_U\) commutes with the \(G\)-action gives that
\[\psi_{\beta\alpha}(b, gh) = \psi_{\beta\alpha}(b, g)h\]
for any \(g, h \in G\). We will abuse notation, and write
\[\psi_{\beta\alpha}(b) := \psi_{\beta\alpha}(b, 1_G)\]
Using this, we get that
\[\psi_{\beta\alpha}(b, g) = \psi_{\beta\alpha}(b)g = L_{\psi_{\beta\alpha}(b)}g\]
With this, we have maps
\[\psi_{\beta\alpha} : U_\alpha \cap U_\beta \to G\]
which are called the \emph{transition functions}\index{transition functions!of a principal bundle}, and these satisfy the same cocycle conditions as for vector bundles.

\begin{theorem}
    Given the data:
    \begin{itemize}
        \item base manifold \(B\),
        \item trivialising neighbourhoods \(\{U_\alpha\}\) covering \(B\),
        \item maps \(\psi_{\beta\alpha} : U_\alpha \cap U_\beta \to G\) satisfying the cocycle conditions,
    \end{itemize}
    Let
    \[P = \frac{\bigsqcup(U_\alpha \times G)}{\sim}\]
    where \(U_\alpha \times G \ni (b, h) \sim (b, \psi_{\beta\alpha}(b)h) \in U_\beta \times G\). Then \(P\) is a principal \(G\)-bundle, with proiection map induced by the projections on each element of the disjoint union, with transition functions \(\{\psi_{\beta\alpha}\}\). The right multiplication action is given by right multiplication on each element in the disjoint union.
\end{theorem}

\begin{proof}
    Basically the same as the vector bundle case.
\end{proof}

\begin{remark}
    If the \(G\)-action on a manifold \(P\) is smooth, free and proper, that is the map \(P \times G \to P\) is proper, i.e. the preimage of a compact set is compact. Then we can make the orbit space \(P/G\) into a manifold, and we can define \(B = P/G\) and \(\pi\) is the quotient map. This then defines a principal bundle \(P \to B\).
\end{remark}

\begin{remark}
    Let \(G\) be a matrix Lie group. Suppose we have a vector bundle \(E \to B\) with a \(G\)-structure, with transition maps \(\psi_{\beta\alpha}\). From this we can obtain a principal \(G\)-bundle.

    Conversely, if we have a principal \(G\)-bundle, and a smooth faithful representation \(G \to \GL(V)\), we can make a vector bundle \(E \to B\) using the transition functions.

    We then say that \(E\) is \emph{associated to the principal \(G\)-bundle \(P \to B\)} (and vice versa).
\end{remark}

\lec

\begin{example}
    If \(E = \T M\), \(\dim(M) = n\), taking \(G = \GL(n, \R)\), then the principal bundle \(P\) is called a \emph{frame bundle}\index{frame bundle}, and each fibre \(P_p\) is the set of a bases of the vector space \(\T_p M\), by considering the columnes of an invertible matrix.

    If \(G = \mathrm O(n)\)< then \(P\) is called an \emph{orthonormal frame bundle}\index{orthonormal frame bundle}, and each fibre is the set of orthonormal bases of \(\T_pM\).
\end{example}

\subsection{Hopf bundle}

\index{Hopf bundle}

A Hopf bundle is an example of a tautological complex line bundle over \(\C\P^1\). Recall that the fibre over \((z_1 : z_2) \in \C\P^1\) is the complex line spanned by \((z_1, z_2)\) in \(\C^2\). We will work out the transition functions, and appeal to the Steenrod construction.

Write \(U_i = \left\{(z_1 : z_2) \in \C\P^1 \mid z_i \ne 0\right\}\). \(U_1 \cup U_2\) is an open cover of \(\C\P^1\). Write \(z = z_2 / z_1\) for the local coordinate on \(U_1\), and \(\zeta = z_1/z_2 = 1/z\) for the local coordinate on \(U_2\).

Set
\begin{align*}
    \Phi_1 : \pi^{-1}(U_1) &\to U_1 \times \C \\
    (w, wz) &\mapsto ((1:z), w\sqrt{1 + \abs{z}^2})
\end{align*}
and
\begin{align*}
    \Phi_2 : \pi^{-1}(U_2) &\to U_2 \times \C \\
    (\zeta w, w) &\mapsto ((\zeta:1), w\sqrt{\abs{\zeta}^2 + 1})
\end{align*}
We can compute
\[\Phi_1^{-1}((1 : z), \widetilde w) = \left(\frac{\widetilde w}{\sqrt{1 + \abs{z}^2}}, \frac{\widetilde w}{\sqrt{1 + \abs{z}^2}}z\right)\]
If \(z \ne 0\), then
\begin{align*}
    \Phi_2 \circ \Phi_1^{-1}((1 : z), \widetilde w) &= \Phi_2\left(\frac{\widetilde w}{\zeta\sqrt{1 + \abs{\zeta}^{-2}}}\zeta,\frac{\widetilde w}{\zeta\sqrt{1 + \abs{\zeta}^{-2}}}\right) \\
    &= \left((\zeta: 1), \frac{\abs{\zeta}}{\zeta}\widetilde w\right) \\
    &= \left((1 : z), \frac{z}{\abs{z}}\widetilde w\right)
\end{align*}
Hence the transition function is
\[\psi_{21}((1 : z)) = \frac{z}{\abs{z}}\]
and
\[\psi_{12}((\zeta:1)) = \frac{\abs{z}}{z} = \frac{\zeta}{\abs{\zeta}}\]
Note that the cocycle conditions are trivial as we only have two transition functions. Note
\[\psi_{12}, \psi_{21} : U_1 \cap U_2 \to \mathrm U(1) = S^1 \sse \C^\times = \GL(1, \C)\]

With this, the Hopf bundle admits a \(\mathrm U(1)\) structure. Explicitly,
\[\norm{(w, wz)} = \abs{w}\sqrt{1 + \abs{z}^2}\]
and we have a similar computation for \(\pi^{-1}(U_2)\). If we now remove the zero section, then we have an isomorphism
\[E \setminus \{\text{image of zero section}\} \cong \C^2 \setminus \{(0, 0)\})\]
which is the total space of an associated \(\GL(1, \C)\)-bundle.

The associated principal \(\mathrm U(1) = S^1\) bundle has total space
\[P = \{(w_1, w_2) \in \C^2 \mid \abs{w_1}^2 + \abs{w_2}^2 = 1\} = S^3\]
and the bundle projection will send \((w_1, w_2) \to (w_1:w_2) \in \C\P^1 \cong S^2\).

We can see that the Hopf bundle is not the trivial bundle, since \(S^3\) and \(S^2 \times S^1\) are not homeomorphic.

In particular, if \(E\) is associated to \(P\), then \(f^*E\) is associated to \(f^*P\).
