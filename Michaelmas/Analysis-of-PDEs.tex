\section{Basics}

Let \(\mcU \subseteq \R^n\) be an open set. A \emph{PDE of order \(k\)}\index{PDE of order \(k\)} is an expression of the following form
\begin{equation}
    \label{eq:pde}
    F(x, u, Du, \dots, D^ku) = 0
\end{equation}
where \(u : \mcU \to \R\) is the unknown, \(F : \mcU \times \R \times \R^n \times \cdots \times \R^{n^k}\) is a general function. We say that \(u\) is a \emph{classical solution of}\index{classical solution} \cref{eq:pde} if \(u\) satisfies \cref{eq:pde} in \(\mcU\).

We may also consider the case where \(u(x) \in \R^p\) and \(F \in \R^q\), and we call this a \emph{system of PDEs}\index{system of PDEs}.

\subsection{Examples of PDEs}

\begin{example}
    [ODE system]
    One example of a model used in mathematical biology is the ODE system
    \begin{align*}
        \dv{u}{t} &= f(u, v) \\
        \dv{v}{t} &= g(u, v)
    \end{align*}
\end{example}

\begin{example}
    [Laplace's equation]
    \[\Delta u = \sum_{i=1}^n \partial_i^2 u = 0\]
    The Laplacian is an \emph{averaging} operator.
\end{example}

\begin{example}
    [Heat equation]
    \[u_t = D\Delta u\]
    This is also called the diffusion equation, \(D\) is called the diffusion constant.
\end{example}

\begin{example}
    [Navier-Stokes]
    The Navier-Stokes equations in fluid dynamics is
    \begin{align*}
        u_t &= \nu\Delta u - u \cdot \grad u - \grad \rho + f \\
        \div(u) &= 0
    \end{align*}
\end{example}

\begin{example}
    [Transport equation]

    The transport equation is
    \[u_t + vu_x = 0\]
    where \(v\) is a constant, corresponding to the velocity. A modification is the \emph{advection-diffusion} equation,
    \[u_t + v \vdot \grad u = D\Delta u + f\]
\end{example}

\begin{example}
    [Poisson equation]
    \[\Delta u = f\]
    Describes electric field due to some charge, or Newtonian gravity.
\end{example}

\begin{example}
    [Wave equation]
    \[\Box u = -u_tt + c^2 \Delta u = 0\]
    This models sound waves, seismic waves, ...
\end{example}

\begin{example}
    [KdV equation]
    This equation admits \emph{soliton} solution.
    \[u_t + \partial_x^3 u - 6u\partial_x u = 0\]
\end{example}

\begin{example}
    [Maxwell equations]
    \begin{align*}
        \div(E) &= \rho \\
        \div(B) &= 0 \\
        \partial_t E &= \curl B = J \\
        \partial_t B &= -\curl F
    \end{align*}
\end{example}

\begin{example}
    [Einstein's equations]
    \[\Ric(g) - \frac{1}{2}gR(g) = 0\]
\end{example}

\subsection{Data and Well-Posedness}

All of the examples from above need additional information to solve, which we call the \emph{data}\index{data}. For example, we might need \(u\vert_{\partial \mcU}\) and so on. A guiding principle to this process is called \emph{well-posedness (in the sense of Hadamard)}.

We say that a PDE problem (equation and the data) is well-posed\index{well-posed} if we have

\begin{enumerate}
    \item A solution exists (in some function space).
    \item Given some data, the solution should be unique (depends on the function space of choice).
    \item The solution depends continuously on the data.
\end{enumerate}

The aim is to find the largest space for which a solution exists, but small enough so that it is unique. (For example, strong cosmic censorship in GR?)

\begin{notation}
    [Multi-index notation] We will use multi-index notation\index{multi-index notation},
    \[\N = \{0, 1, \dots\}\]
    \(\alpha = (\alpha_1, \dots, \alpha_n) \in \N^n\) is called a \emph{multi-index}, and we define the \emph{order} of \(\alpha\)
    \[\abs{\alpha} = \alpha_1 + \alpha_2 + \dots + \alpha_n\]
    and the \(\alpha\)-th derivative is
    \[D^\alpha f(x) = \partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}f\]
    If \(x = (x_1, \dots, x_n)\), then \(x^\alpha = x_1^{\alpha_1}\cdots x_n^{\alpha_n}\), and
    \[\alpha! = \alpha_1!\cdots \alpha_n!\]
\end{notation}

\subsection{Classifying PDEs (of order \(k\))}

We say \cref{eq:pde} is \emph{linear}\index{linear PDE} if \(F\) is a linear function of \(x, u\) and its derivative. That is, we can write it as

\[\sum_{\abs{\alpha} \le k}a_\alpha(x) D^\alpha u = f(x)\]

Moreover, we say that a linear PDE is \emph{homogeneous}\index{linear PDE!homogeneous} if \(f = 0\). We say \cref{eq:pde} is \emph{semilinear}\index{semilinear PDE} if the highest order derivatives appear linearly with coefficients depending only on \(x\). That is, we have

\[\sum_{\abs{\alpha} = k}a_\alpha(x)D^\alpha u(x) + F(x, u, Du, \dots, D^{k-1}u) = 0\]

One example would be \(\Delta u = u_x^2\). Finally, we say \cref{eq:pde} is \emph{quasilinear}\index{quasilinear PDE} if the highest order derivatives appear linearly, but the coefficients depending on lower order derivatives. That is, we have

\[\sum_{\abs{\alpha} = k}a_\alpha(x, u, \dots, D^{k-1}u)D^k u + F(x, u, \dots, D^{k-1}u) = 0\]

For example, we can have \(u u_{xx} + u_{yy} - u_x^2 = 0\). Finally, we say \cref{eq:pde} is \emph{fully non-linear}\index{fully non-linear PDE} if it is non of the above.

\section{Cauchy-Kovalevskaya theorem}

\subsection{ODE theory}

Fix \(\mcU \sse \R^n\) open, and suppose \(f : \mcU \to \R^n\) is given. We would like to consider the ODE

\begin{equation}
    \label{eq:ode}
    \begin{cases}
        \dot u(t) = f(u(t)) \\
        u(0) = u_0
    \end{cases}
\end{equation}

\begin{theorem}
    [Picard-Lindel\"of]\index{Picard-Lindel\"of} Suppose we have \(r, K > 0\) such that \(B_r(u_0) \sse U\) and
    \[\norm{f(x) - f(y)} \le K\norm{x - y}\]
    for all \(x, y \in B_r(u_0)\). Then there exists \(\epsilon > 0\), depending on \(K, r\) and a unique \(C^1\) solution \(u : (-\epsilon, \epsilon) \to \mcU\) solving \cref{eq:ode}.
\end{theorem}

\lec

\begin{proof}
    [Sketch proof, see examples sheet 1] If \(u \in \C^1\) solves \cref{eq:ode}, then by the fundamental theorem of calculus, \(u\) satisfies the \emph{weak formulation}
    \begin{equation}
        \label{eq:weak-sol-ode}
        u(t) = u_0 + \int_0^t f(u(s))\dd s
    \end{equation}
    Moreover, if \(u \in C^0\) is a solution to \cref{eq:weak-sol-ode}, then it is a \(C^1\) solution to \cref{eq:ode}. Moreover, if \(u\) exists, then it is a fixed point of
    \[G(w) = u_0 + \int_0^t f(w(s))\dd s\]
    Let \(\mcS = \left\{w : (-\epsilon, \epsilon) \to \overline{B_{r/2}(u_0)}\right\} \text{ continuous}\). We want to show that \(\mcS\) is a complete metric space, \(G : \mcS \to \mcS\) is a contraction for \(\epsilon\) sufficiently small, and by the contraction mapping theorem \(G\) has a fixed point.
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item The solution (in general) can't be global. Consider for example
        \[\dot u(t) = u(t)^2 \qqtext{with} u(0) = u_0 > 0\]
        Solutions to this equation blow up in finite time.
        \item This does not apply to
        \[\dot u(t) = \sqrt{u(t)} \qqtext{with} u(0) = 0\]
        There are two solutions. Note we \emph{can} apply the Peano existence theorem.
    \end{enumerate}
\end{remark}

Now suppose \(f\) is smooth, and we have \(\dot u(t) = f(u(t))\) is \(C^1\). By the chain rule,
\[\ddot u(t) = Df(u(t))\vdot \dot u(t) = f_2(u(t), \dot u(t))\]
which is continuous. Hence \(\ddot u\) is continuous, and so \(u \in C^2\). Repeating this, we get that \(u \in C^k\) for all \(k\). That is, \(u\) is smooth.

In principle, given \(u_0 = u(0)\), we can determine
\[u^{(k)}(0) = F_k\left(u, u', \dots, u^{(k-1)}\right)\big\vert_{t=0}\]
and so we can write
\[\sum_{k\ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
We call this a \emph{formal power series solution}. Does our solution \(u(t)\) agree with this? That is, do we have
\[u(t) = \sum_{k \ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
in a neighbourhood of \(0\)?

\begin{theorem}
    [Cauchy-Kovalevskaya for simple ODEs] 
    \label{thm:c-k-simple-ode}
    \index{Cauchy-Kovalevskaya Theorem!for simple ODEs}If \(f(u)\) is real analytic in a neighbourhood of \(u_0\), then the series
    \[\sum_{k \ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
    converges in a neighbourhood of \(0\) to the unique solution of \cref{eq:ode} given by Picard-Lindel\"of.
\end{theorem}

\subsection{Real analyticity and majorants}

Suppose \(f : (-\epsilon, \epsilon) \to \R\) is a smooth function. Therefore, \(f^{(n)}(0)\) exists for all \(n \ge 0\). Does the partial sums
\[\sum_{n \ge 0}\abs{f^{(n)}(0)}{n!}x^n\]
for some \(\abs{x} \le \delta\)? No, consider the function
\[f(x) = \begin{cases}
    \exp(-\frac{1}{x}) & x > 0 \\
    0 & x \le 0
\end{cases}\]
This is a smooth function, with \(f^{(n)}(0) = 0\) for all \(n\).

\begin{definition}
    [real analytic]
    Let \(\mcU \sse \R^n\) be open and \(f : \mcU \to \R\) is \emph{real analytic}\index{real analytic} (at \(x_0\)) if there exists \(r > 0\), \(f_\alpha \in \R\) such that
    \[f(x) = \sum_{\alpha}f_\alpha(x-x_0)^\alpha\]
    when \(\abs{x - x_0} < r\). 
\end{definition}

\begin{remark}
    \begin{enumerate}
        \item That is, \(f\) can be written as a convergent power series and
        \[f_\alpha = \frac{D^\alpha f(x_0)}{n!}\]
        \item Real analyticity is a local property.
        \item \(f\) is real analytic on an open set \(U\) if it is real analytic at each \(x_0 \in U\).
        \item We will denote the set of real analytic functions on \(\mcU\) by \(C^\omega(\mcU)\).
        \item If \(f\) is \(C^\omega\), then \(f\) is smooth (e.g. Weierstrass \(M\)-test).
        \item If \(f\) is real analytic, and \(\mcU\) is connected, then \(f\) is uniquely determined in \(\mcU\) by its derivatives \(D^\alpha f(x)\) at some point \(x \in \mcU\).
        \item In particular, \(f\) is real analytic if and only if for any compact \(K \sse U\), there exists \(C, r\) such that
        \[\sup_{x \in K}\abs{D^\alpha f(x)} \le C\abs{\abs{\alpha}!}{r^{\abs{\alpha}}}\]
    \end{enumerate}
\end{remark}

\underline{Exercise:} Show \(f(x) = 1/x\) and \(f(x) =\sqrt{x}\) are real analytic for \(x > 0\).

\begin{example}
    Recall
    \[\frac{1}{1-x} = \sum_{k \ge 0}x^k\]
    for \(\abs{x} < 1\). Let \(r > 0\), and consider
    \[f(x) = \frac{r}{r - (x_1 + \dots + x_n)} = \frac{1}{1- \frac{x_1 + \dots + x_n}{r}} = \sum_{k \ge 0}\left(\frac{x_1 + \dots + x_n}{r}\right)^k\]
    provided \(\abs{x_1 + \dots + x_n} \le \sqrt n\left(\sum_j \abs{x_j}^2\right)^{1/2} = \sqrt n \norm{x} < r\).
    By the multinomial theorem (sheet 1),
    \[f(x) = \sum_{k \ge 0}\frac{1}{r^k}\sum_{\abs{\alpha} = k}{\abs{\alpha}\choose \alpha}x^\alpha = \sum_\alpha \frac{\abs{\alpha}!}{\alpha!r^{\abs{\alpha}}}x^\alpha\]
    where
    \[{\abs{\alpha}\choose\alpha} = \frac{\abs{\alpha}!}{\alpha!}\]
    and so,
    \[f(x) = \sum_\alpha f_\alpha x^\alpha\]
    where
    \[f_\alpha = \frac{\abs{\alpha}!}{\alpha!}\frac{1}{r^{\abs{\alpha}}}\]

    This series is absolutely convergent near zero, since
    \[\sum_\alpha \frac{\abs{\alpha!}}{\alpha!}\frac{x^\alpha}{r^{\abs{\alpha}}} = \sum_{k \ge 0}\left(\frac{\abs{x_1} + \dots + \abs{x_n}}{r}\right)^k < \infty\]
\end{example}

\begin{definition}
    [majorise]
    Let \(f = \sum_\alpha f_\alpha x^\alpha\) and \(g = \sum_\alpha g_\alpha x^\alpha\), we say that \emph{\(g\) majorises \(f\)}\index{majorise}, or \emph{\(g\) is a mojorant of \(f\)}, written \(g \gg f\), if \(g_\alpha \ge \abs{f_\alpha}\) for all \(\alpha\).
\end{definition}

For vector valued functions, we require each component to be a majorant.

\begin{lemma}
    [properties of majorants]
    \label{lem:majorant-props}
    \begin{enumerate}
        \item If \(g \gg f\), and \(g\) converges for \(\norm{x} < r\), then \(f\) converges for \(\norm{x} < r\).
        \item If \(f = \sum f_\alpha x^\alpha\) converges for \(\norm{x} < r\), then for any \(s \in (0, r/\sqrt n)\), there exists a majorant of \(f\) which converges for \(\norm{x} < s/\sqrt n\).
    \end{enumerate}
\end{lemma}

\begin{proof}
    1. Looking at the partial sums
    \[\sum_{\abs{\alpha} \le k}\abs{f_\alpha x^\alpha} = \sum_{\abs{\alpha} \le k}\abs{f_\alpha}\abs{x_1}^{\alpha_1} \cdots \abs{x_n}^{\alpha_n} \le \sum_{\abs{\alpha}\le k}\abs{g_\alpha}\abs{x_1}^{\alpha_1}\cdots\abs{x_n}^{\alpha_n} \le \sum_\alpha\abs{x_1}^{\alpha_1} \cdots \abs{x_n}^{\alpha_n} = g(\tilde x)\]
    where \(\tilde(x) = (\abs{x_1}, \dots, \abs{x_n})\). So \(\norm{\tilde x} = \norm{x}\), and so if \(\norm{x} < r\), then \(g\) converges at \(\tilde x\). That is, \(g(\tilde x) < \infty\). Therefore, we have a uniform bound on the partial sum.

    2. Let \(s \in (0, r/\sqrt n)\), and set \(y = (s, \dots, s)\). Then \(\norm{y} = s\sqrt n\), and by assumption,
    \[f(y) = \sum_\alpha f_\alpha y^\alpha\]
    converges, as \(\norm{y} = s\sqrt n < r\). So there exists a constant \(c\) such that \(\abs{f_\alpha y^\alpha} \le c\). Hence \[\abs{f_\alpha} \le \frac{C}{\abs{y^\alpha}} = \frac{C}{\abs{y_1}^{\alpha_1}\cdots \abs{y_n}^{\alpha_n}} = \frac{C}{\abs{s}^{\abs{\alpha}}} \le \frac{C}{\abs{s}^{\abs{\alpha}}}\frac{\abs{\alpha}!}{\alpha!}\]
    
    So we define
    \[g(x) = \frac{Cs}{s - (x_1 + \dots + x_n)}\]
    From the above, \(g\) majorises \(f\).
\end{proof}

\subsection{Proof of Cauchy-Kovalevskaya for ODEs}

% Meet and greet 2:30 - 3:15 this afternoon.
\begin{proof}
    [Proof of \cref{thm:c-k-simple-ode}] We will use the method of majorants. Without loss of generality \(u_0 = 0\), and for simplicity, we can assume \(n = 1\). We need to find the series coefficients. So
    \[\dot u = f(u)\]
    and so \(\dot u(0) = f(u(0)) = f(0)\), that is, \(u_1 = f(0)\). Next,
    \[\ddot u(t) = f'(u(t))\dot u(t)\]
    and so \(\ddot u(0) = f'(0)f(0)\), that is, \(u_2 = f'(0)f(0) = \frac{1}{2!}f'(0)f(0)\). Repeating,
    \[u^{(3)}(0) = f''(0)f(0)^2 + f'(0)^2 f(0)\]
    and so
    \[u_3 = \frac{1}{3!}(f''(0)f(0)^2 + f'(0)^2 f(0))\]
    Iterating this procedure,
    \[u_k = P_k\left(f(0), \dots, f^{(k-1)}(0)\right)\]
    where \(P_k\) is a polynomial in \(k\)-variables, with nonnegative coefficients. For example,
    \begin{align*}
        P_1(x) &= x \\
        P_2(x,y) &= \frac{1}{2!}xy \\
        P_3(x,y,z) &= \frac{1}{3!}(x^2z + xy^2)
    \end{align*} 

    Since \(f\) is real analytic, we have that
    \[f(v) = \sum_{k \ge 0} f_k v^k\]
    where \(f_k = \frac{1}{k!}f^{(k)}(0)\). Hence we have that
    \[f^{(k)}(0) = k!\cdot f_k\]
    Substituting, we have that
    \[u_k = Q_k(f_0, \dots, f_{k-1})\]
    which again is a polynomial in \(k\)-variables and nonnegative coefficients. This polynomial is ``universal''.

    \underline{Aim:} We would like to show that the power series
    \[\sum_k u_k t^k\]
    converges in a neighbourhood of \(t = 0\), and solves the ODE, \cref{eq:ode}. Since \(f\) is analytic, we know that
    \[f(u) = \sum_k f_k u^k\]
    on \(\abs{u} < k\). Fixing some \(s < r\), there exists a majorant
    \[g(u) = \sum_k g_k u^k\]
    of \(f\), from \cref{lem:majorant-props} (ii). Consider the auxilliary differential equation
    \[\dot w(t) = g(w(t))\]
    and \(w(0) = 0\). Using the definition of \(g\), we that
    \[\dv{w}{t} = \frac{Cs}{s - w(t)}\]
    We get that
    \[w = s\pm\sqrt{s^2 - 2Cst}\]
    Due to the initial data, we take the \(-\) solution. That is,
    \[w = s - \sqrt{s^2 - 2Cst}\]
    This is real analytic, for \(\abs{t} < s/2C\). This tells us that
    \[w(t)  = \sum_k w_kt^k\]
    converges for \(\abs{t} < s/2C\). Moreover,
    \[w_k = Q_k(g_0, \dots, g_{k-1})\]
    since \(Q_k\) is ``universal''.

    \begin{claim}
        \(w\) majorises \(u\).
    \end{claim}

    By construction, \(g\) majorises \(f\), i.e. \(g_k \ge \abs{f_k}\) for all \(k\). Moreover, since \(Q_k\) has nonnegative coefficients,
    \[w_k = Q_k(g_0,\dots, g_{k-1}) \ge Q_k(\abs{f_0}, \dots, \abs{f_{k-1}}) \ge \abs{Q_k(f_0, \dots, f_{k-1})} = \abs{u_k}\]
    Hence by \cref{lem:majorant-props} (i), we know that the series
    \[\sum_k u_kt^k\]
    converges for \(\abs{t} < s/2C\).

    To conclude, set
    \[u(t) := \sum_{k\ge 0}u_kt^k\]
    and we need to check that it solves \cref{eq:ode}. Both sides are analytic, so suffices to check the derivatives on each side agree to all orders at \(t = 0\).
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item We can extend to systems, where we replace \(u_k\) with
        \[u_k^j = Q_k^j(D^\abs{\alpha} \mid \abs{\alpha} \le k)\]
    
        For \(w\), we can replace \(w^j = w^1\) as before.
        \item For the non-autonomous case,
        \begin{align*}
            u(t) &= f(u, t) \\
            u(0) &= 0
        \end{align*}
        Consider \(v(t) = (u(t), t)\), then \(\dot v(t) = (\dot u(t), 1) = (f(u, 1), 1) = (f(v), 1) = F(v)\)
        with \(v(0) = 0\), and we can apply the system version.
    \end{enumerate}
\end{remark}

\subsection{Cauchy-Kovalevskaya for PDEs}

Let \(u : \R^n \to R^m\), and choose \(r > 0\). Consider the following problem
\[u_t = \sum_{j=1}^n B_j(u, x)u_{x_j} + C(u, x)\]
on \(\norm{x}^2 + t^2 < r^2\), with
\[u = 0\]
on \(\norm{x}^2 + t^2 < r^2\) and \(t = 0\). The \(B_j\) are matrices, \(B_j\) and \(C\) are real analytic.

\begin{theorem}
    [Cauchy-Kovalevskaya for first order systems]
    \index{Cauchy-Kovalevskaya Theorem!for first order systems}
    Suppose \(B_j, C\) are real analytic, for small \(r > 0\). Then there exists a unique real analytic function
    \[u = \sum_\alpha u_\alpha x^\alpha\]
    solving the above PDE.
\end{theorem}

\begin{proof}
    [Idea] Compute
    \[u_\alpha = \frac{D^\alpha u}{\alpha!}\]
    in terms of \(B_j, C\), and show that the power series converges for small \(r\). We use the PDE to find all derivatives.
\end{proof}

\begin{example}
    Consider the system
    \begin{align*}
        u_t &= v_x - f \\
        v_t &= -u_x
    \end{align*}
    with \(u = v = 0\) on \(t = 0\). The boundary conditions give us that
    \[u(0, 0) = v(0, 0) = 0\]
    We would like to determine \(u_\alpha\) for all \(\alpha\). By differentiating the boudary conditions,
    \[\partial_x^nu(x, 0) = \partial_x^nv(x, 0) = 0\]
    for all \(n\). That is, for the case \(\alpha = (n, 0)\). From the PDE,
    \[u_t(x, 0) = 0 - f = -f \qquad v_t(x, 0) = 0\]
    This then means that
    \[\partial_x^n \partial_t u(x, 0) = -\partial_x^n f(x, 0)\]
    and
    \[\partial_x^n \partial_tv(x, 0) = 0\]
    for all \(n \ge 1\).
    
    Next, if \(\alpha = (n, 2)\) use the PDE and we get
    \[u_{tt}(x, 0) = f_t(x, 0)\]
    and
    \[v_{tt}(x, 0) = f_x(x, 0)\]
    The same method as above gives us that
    \begin{align*}
        \partial_x^n\partial_t^2 u(x, 0) &= -(\partial x)^n \partial_t f(x, 0) \\
        \partial_x^n\partial_t^2 v(x, 0) &= (\partial_x)^{n+1}f(x, 0)
    \end{align*}

    Repeating this, we can compute all of the derivatives.
\end{example}

\subsection{Reduction to first order systems}

\begin{example}
    Consider \(u : \R^3 \to \R\) satisfying
    \[u_{tt} = u u_{xy} - u_{xx} + u_t\]
    with conditions
    \[u\vert_{t=0} = u_0(x, y) \qqtext{and} u_t\vert_{t=0}u_1(x, y)\]

    where \(u_0, u_1\) are real analytic near \(0 \in \R^3\). Note
    \[f(t, x, y) = u_0 + tu_1\]
    is real analytic near \(0 \in \R^3\). Note
    \[f\vert_{t=0} = u_0 \qqtext{and} \partial_tf\vert_{t=0} = u_1\]
    Set \(w(t,x,y) = u - f\). Then we find that
    \[w_tt = ww_{xy} - w_{xx} + w_t + fw_{xy} + f_{xy}w + F\]
    where
    \[R = ff_{xy} -  f_{xx} + f_t\]
    where
    \[w\vert_{t=0} = \partial_tw\vert_{t=0} = 0\]
    Observe \(F\) is real analytic, and independent of \(w\) and its derivatives. Let \(x = (x, y, t) = (x^1, x^2, x^3)\), and set
    \[v = (w, w_x, w_y, w_z) = (v^1, v^2, v^3, v^4)\]
    Then
    \begin{align*}
        v^1_t &= w_t = v^4 \\
        v_t^2 &= w_{xt} = v^4_{x_1} \\
        v_t^3 &= w_{yt} = v^4_{x_2} \\
        v_t^4 &= v^1v_{x_2}^2 - v_{x_1}^2 + v^4 + fv_{x_2}^2 + f_{xy}v^1 + F
    \end{align*}
    Define
    \[B_1 = \begin{pmatrix}
        & & & 0\\
        & & & 1 \\
        & & & 0 \\
        0 & -1 & 0 & 0
    \end{pmatrix} \qquad B_2 = \begin{pmatrix}
        & & & \\
        & & & \\
        & & & 1 \\
        v^1 + f & 0 & 0 & 0
    \end{pmatrix}\]
    and
    \[c = \begin{pmatrix}
        v^4 \\
        0 \\
        0 \\
        v^4 + f_{xy}v^1 + F
    \end{pmatrix}\]
    We can see then
    \[\pdv{x_3}v = B_1v_{x_1} + B_2v_{x_2} + c\]
    In this case, \(B_1, B_2, c\) are real analytic functions of \(x, v\), and so we can apply Cauchy-Kovalevskaya.
\end{example}

More generally, consider the scalar quasilinear problem

\[\sum_{\abs{\alpha} = k}a_\alpha(D^{k-1}u, \dots, u, x)D^\alpha u + a_0(D^{k-1}u, \dots, u, x) = 0\]
where
\[u : B_r(0) \to \R \qqtext{and} \pdv{u}{x_n} = \dots = \left(\pdv{x_n}\right)^{k-1}u = 0\]
for \(\norm{x'} < r, x_n = 0\).

Define
\[v = \left(u, \pdv{u}{x_1}, \dots, \pdv{u}{x_n}, \pdv[2]{u}{x_1}{x_1}, \dots, \left(\pdv{x_n}\right)^{k-1}u\right) = (v^1, \dots, v^m) \in \R^m\]
We would like to get a first order system in \(v\). That is, express
\[\pdv{v^j}{x_n}\]
in terms of \(v^j\) and \(\pdv{v}{x_j}\) for \(j = 1, \dots, m-1\). If \(j = 1\), then
\[\pdv{v^1}{x_n} = \pdv{u}{x_n} = v^\ell\]
for some \(\ell\). If \(2 \le j \le m-1\), then
\[v^j = D^\alpha u\]
for some \(\abs{\alpha} \le k-1\), such that \(\alpha_n < k-1\). So
\[\pdv{v^j}{x_n} = D^\alpha \pdv{u}{x_n} = \frac{\partial^{\abs{\alpha} + 1}}{\partial_1^{\alpha_1}\cdots \partial_n^{\alpha_n + 1}}u\]
If \(\abs{\alpha} \le k-2\), then \(\abs{\alpha} + 1 \le k-1\),a nd so \(\pdv{v^j}{x_n} = v^\ell\) for some \(\ell\).

If \(\abs{\alpha} = k-1\), and \(\alpha_n < k-1\), then there exists \(p \ne n\) such that \(\alpha_p \ge 1\). So we have that
\[\pdv{v^j}{x_n} = \pdv{x_n}(\frac{\partial^{\abs{\alpha}}u}{\partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}}) = \pdv{x_p}(\frac{\partial^{\abs{\alpha}}u}{\partial_1^{\alpha_1}\cdots \partial_p^{\alpha_p - 1}\cdots \partial_n^{\alpha_n - 1}}) = \pdv{v^\ell}{x_p}\]
for some \(\ell\). Finally, to compute
\[\pdv{v^m}{x_n}\]
we will use the PDE. Recall the coefficients are \(a_\alpha(v, x)\) for \(v \in \R^m, x \in \R^n\). We assume \(a_\alpha : B_\rho(0) \to \R\) is real analytic, and suppose
\[a_c = a_{(0, \dots, 0, k)}(0) \ne 0\]
Since \(a_\alpha\) are real analytic near zero, they are continuous. Therefore, \(a_c(z, w) \ne 0\) for all \(\norm{z}^2 + \norm{w}^2 \le \delta^2\), where \(\delta < \rho\). Then
\[a_c \frac{\partial^k u}{\partial x_n^k} = -\left(\sum_{\abs{\alpha} = k, \alpha_n < k} a_\alpha D^\alpha u + a_0\right)\]
Dividing by \(a_c\), we get
\[\frac{\partial^k u}{\partial x_n^k} = -\frac{1}{a_c}\left(\sum_{\abs{\alpha} = k, \alpha_n < k} a_\alpha D^\alpha u + a_0\right)\]

The right hand side can be written in terms of
\[\pdv{v^\ell}{x_p}, v\]
for \(p < n\). Therefore, if \(a_c \ne 0\), we have turned the scalar quasilinear PDE into a first order PDE system, on which we can apply Cauchy-Kovalevskaya.

\begin{definition}
    If \(a_c = a_{(0, \dots, 0, k)}(0)\) is non-zero, then we say the plane
    \[\{x^n = 0\}\]
    is \emph{non-characteristic}\index{non-characteristic!equation}. Otherwise, it is \emph{characteristic}.
\end{definition}

\subsection{Exotic boundary conditions}

\begin{definition}
    [real analytic hypersurface]
    We say \(\Sigma \sse \R^n\) is a \emph{real analytic hypersurface near \(x_0 \in \Sigma\)}\index{real analytic hypersurface} if there exists \(\epsilon > 0\), and a real analytic function
    \[\Phi : B_\epsilon(x_0) \to U \sse \R^n\]
    where \(U\) is an open neighbourhood of \(0 \in \R^n\), and defining \(y = \Phi(x)\), with \(\Phi(x_0) = 0\). Moreover, we require
    \begin{enumerate}[(i)]
        \item \(\Phi\) is a bijection,
        \item \(\Phi^{-1} : U \to B_\epsilon(x_0)\) is real analytic,
        \item \(\Phi(\Sigma \cap B_\epsilon(x_0)) = \{y_n = 0\} \cap U\). 
    \end{enumerate}

    We can think of \(\Phi\) as ``straightening out'' \(\Sigma\).
\end{definition}

\begin{example}
    Spheres, planes, tori, etc. are real analytic (hyper)surfaces.
\end{example}

Let \(\gamma\) be unit normal to \(\Sigma\), and consider
\[\sum_{\abs{\alpha} = k} a_\alpha(D^{k-1}u, \dots, u, x)D^\alpha u + a_0(D^{k-1}u, \dots, u, x)\]
where
\begin{equation}
    \label{eq:surf-pde}
    u = \sum_i\gamma^i\partial_i u = (\gamma^i\partial_i)^{k-1}u = 0 \text{ on }\Sigma
\end{equation}

Define
\[v(y) = u(\Phi^{-1}(y))\]
for \(y \in U\). That is, \(u(x) = v(\Phi(x))\) for \(x \in B_\epsilon(x_0)\). Using the chain rule,
\[\pdv{u}{x_i} = \sum \pdv{u}{y_j} \pdv{\Phi^j}{x_i}\]
where \(\Phi = (\Phi^1, \dots, \Phi^n) \in \R^n\). So the PDE becomes

\[\sum_{\abs{\alpha} = k} b_\alpha D^\alpha v + b_0 = 0\]
where \(b_0, b_\alpha\) dependis on \(u\) and \(D^\alpha u\) for \(\abs{\alpha} \le k-1\), and also \(\Phi\) (which is given). The boundary conditions becomes
\[v = \pdv{y_n}v = \dots = \left(\pdv{y_n}\right)^{k-1}v = 0\]
on \(\{y_n = 0\}\). Since \(\Phi\) is real analytic, so are \(b_0, b_\alpha\).

\lec

We would like to apply Cauchy-Kovalevskaya, therefore we need to check that whether the hypersurface \(\{y_n = 0\}\) is non-characteristic. That is,
\[b_{(0, \dots, 0, k)}(D^{k-1}v = 0, \dots, Dv = 0, y = 0) \ne 0\]

Note if \(\abs{\alpha} = 2\), we can compute
\[D^\alpha u = \frac{\partial^kv}{\partial y_n^k}(D\Phi^n)^\alpha + \text{terms not involving} \frac{\partial^kv}{\partial y_n^k}\]

For example, if \(k = 2, n = 2, \alpha = (0, 2)\), then
\[D^\alpha u = u_{x_2x_2} = v_{y_2y_2}\underbrace{(\Phi^2_{x_2})(\Phi^2_{x_2})}_{=(D\Phi^2)^\alpha} + \text{terms not involving} v_{y_2y_2}\]
Thus,
\[b_{(0, \dots, 0, k)} = \sum_{\abs{\alpha} = k}a_\alpha \left(D \Phi^n\right)^\alpha\]

\begin{definition}
    We say that \(\Sigma\) is \emph{non-characteristic at \(x_0 \in \Sigma\)}\index{non-characteristic!hypersurface} if
    \[b_{(0, \dots, 0, k)}(0) = \sum_{\abs{\alpha} = k}a_\alpha(0, \dots, 0, x_0)\left(D\Phi^n(x_0)\right)^\alpha \ne 0\]

    Otherwise, \(\Sigma\) is characteristic at \(x_0\).
\end{definition}

\begin{remark}
    Note that \(\Sigma = \left\{x \in \R^n \mid \Phi^{(n)}(x) = y_n = 0\right\}\). This tells us that
    \[D\Phi^n(x) = c(x)\gamma(x)\]
    where \(\gamma\) is the unit normal of \(\Sigma\). In particular,

    \[D\Phi^n(x_0) = c(x_0)\gamma(x_0)\]
    and so the non-characteristic condition is equivalent to
    \[\sum_{\abs{\alpha} = k}a_\alpha\gamma^\alpha(x) \ne 0\]
\end{remark}

\begin{theorem}
    [Cauchy-Kovalevskaya for non-characteristic hypersurfaces] Suppose \(\Sigma \sse \R^n\) is a hypersurface, with normal \(\gamma\), and consider the PDE \cref{eq:surf-pde} as above. Suppose \(a_\alpha, a_0\) are real analytic near \(x_0 \in \Sigma\), and \(\Sigma\) is non-characteristic near \(x_0\). Then there exists a unique real analytic solution in a neighbourhood of \(x_0\).
\end{theorem}

\subsection{Characteristic surfaces}

Consider the linear operator
\[L = \sum_{i, j=  1}^n a_{ij}\pdv[2]{}{x_i}{x_j}\]
where \(a_{ij} \in \R\). Without loss of generality, we can assume \(a_{ij} = a_{ji}\). Consider the PDE problem
\begin{equation}
    \label{eq:plane-pde}
    \begin{cases}
        Lu = f & \\
        u = \gamma^i\partial_i u = 0 & \text{ on }\Pi_\gamma = \left\{x \mid x \vdot \gamma = 0\right\}
    \end{cases}
\end{equation}
That is, the boundary conditions are on the plane with unit normal \(\gamma\). In particular, \(\Pi_\gamma\) is non-characteristic for \cref{eq:plane-pde} if
\[\sum_{i, j= 1}^n a_{ij}\gamma^i\gamma^j \ne 0\]

We would like to find non-characteristic \(\Pi_\gamma\). Note that the left hand side is just \(\inner{A\gamma, \gamma}\), where \(A = (a_{ij})\) is a symmetric matrix, with the usual Euclidean inner product. In particular, \(A\) is diagonalisable, say \(A = P^\T\Lambda P\), where \(P\) is orthogonal and \(\Lambda\) is diagonal. Then
\[\inner{A\gamma, \gamma} = \inner{P^\T\Lambda P\gamma, \gamma} = \inner{\Lambda v, v}\]
where \(v = P\gamma\). If \(\{\lambda_i\}\) are the eigenvalues for \(A\), then the non-characteristic condition becomes
\[\sum_{i=1}^n \lambda_i(v_i)^2 \ne 0\]

\begin{example}
    [Laplacian]
    \[L = \Delta = \sum_{i=1}^n \partial^2_{1,1}\]
    gives
    \[A = \begin{pmatrix}
        1 \\
        &\ddots \\
        & & 1
    \end{pmatrix}\]
    The Laplacian is an elliptic operator.
\end{example}

\begin{example}
    [Wave equation]
    \[L = \partial_t^2 + \Delta\]
    gives
    \[A = \begin{pmatrix}
        1 \\
        & \ddots \\
        && 1 \\
        &&&-1
    \end{pmatrix}\]
\end{example}

\textbf{Case 1: All eigenvalues have the same sign (and are all non-zero)}. Since \(v\) is a unit vector, the characteristic condition is impossible. That is, there are no characteristic hyperplanes \(\Pi_\gamma\). In this case, we call \(L\) an \emph{elliptic operator}\index{elliptic operator}.

\textbf{Case 2: Say \(\lambda_n < 0\) and \(\lambda_j > 0\) for \(j \ne n\) (or vice versa).} In this case, we call \(L\) an \emph{hyperbolic operator}\index{hyperbolic operator}. In particular, the characteristic condition becomes
\[\sum \lambda_i v_i^2 = 0 \iff \lambda_1v_1^2 + \sum_{j=2}^n\lambda_jv_j^2 = 0\]
Considering the wave equation again, we have the condition that
\[v_n^2 = \sum_{j=1}^{n-1}v_j^2\]
subject to the condition
\[\norm{v} = 1\]

\begin{center}
    \includegraphics[scale=0.1]{diagrams/2023_10_14 (1).jpg}
\end{center}

\textbf{Note that these cases are \emph{not} exhaustive.}

Now we want to different features of elliptic and hyperbolic operators. We will forget about boundary conditions, and look for solutions of the form
\[u(x) = e^{ik \vdot x}\]
for \(k \in \R^n\). We are looking for \emph{wave-like}\index{wave-like solutions} solutions. Substituting,
\[L(e^{ik\vdot x}) = e^{ik\vdot x} = \sum_{j, \ell}a_{j\ell}k_jk_\ell\]
We would like to consider \(Lu = 0\). Taking \(k = c\gamma\), \(\norm{\gamma} = 1\), then the condition is equivalent to
\[c^2\sum a_{j\ell}\gamma_j\gamma_\ell = 0\]

If \(L\) is elliptic, then the only solution is when \(k = 0\). That is, there are no wave-like solutions. On the other hand, if \(L\) is hyperbolic, then we can have wave like solutions, that is,
\[\sum a_{ij}\gamma_i\gamma_j = 0\]
has a solution with \(\norm{\gamma} = 1\). In this case, we have
\[u(x) = e^{i\lambda \gamma \vdot x}\]
gives an infinite family of solutions, indexed by \(\lambda \in \R\).

As we take \(\abs{\lambda} \to \infty\), we see that \(u'(x)\) can grow large. In particular, solutions can be rough.

\begin{center}
    \includegraphics[scale=0.1]{diagrams/2023_10_14 (2).jpg}
\end{center}

By contrast, we will see that solutions to elliptic equations are smooth.

\lec

\begin{example}
    Consider the IVP for Laplace's equation
    \begin{align*}
        \begin{cases}
            u_{xx} + u_{yy} = 0 \\
            u(x, 0) = \varphi(x) \\
             \partial_y u(x, 0) = 0
        \end{cases}
    \end{align*}
    Is this problem well-posed? If \(\varphi(x) = 0\), then \(0\) is a solution. On the other hand, we don't have Cauchy stability. Consider
    \[u_k(x, y) = e^{-\sqrt k}\cos(kx)\cosh(ky)\]
    See typed notes for more details.
\end{example}

\section{Sobolev spaces}

\subsection{H\"older spaces \(C^{k, \gamma}\)}

Let \(U \sse \R^n\) be open, \(k \in \N\). 

\begin{definition}
    [\(C^k\) spaces]
    Define\index{\(C^k(U)\)}
    \[C^k(U) = \left\{f : U \to \R \mid u\text{ is }k\text{ times continuously differentiable}\right\}\]
    
    and define\index{\(C^k(\overline U)\)}
    \[C^k(\overline U) = \left\{u \in C^k(U) \mid u \text{ and its derivatives are bounded and uniformly continuous on }U\right\}\]

    We will define the norm
    \[\norm{u}_{C^k(\overline U)} = \sum_{\abs{\alpha} \le k}\sup_{x \in U}\abs{D^\alpha u(x)}\]
\end{definition}

The idea is that \(C^k(\overline U)\) is the space of functions which can be extended continuously to \(\partial U\). Note that this is is contained in, but not equal to
\[u : \overline U \to \R \text{ such that }u \text{ and its derivatives are continuous}\]

On examples sheet 2, we will show that \(C^k(\overline U)\) is a Banach space.

\begin{definition}
    [H\"older continuous]
    We say \(u : U \to \R\) is \emph{H\"older continuous of index \(\gamma\)}\index{H\"older continuous} with \(0 < \gamma \le 1\) if there exists a constant \(C > 0\), such that
    \[\abs{u(x) - u(y)} \le C\abs{x-y}^\gamma\]
    for all \(x, y \in U\).

    If \(\gamma = 1\), then we say that \(u\) is \emph{Lipschitz continuous}\index{Lipschitz continuous}.
\end{definition}

\begin{remark}
    If \(\gamma > 1\), and \(u\) is H\"older continuous of index \(\gamma\), then \(u\) is constant.
\end{remark}

\begin{definition}
    [\(0\)-H\"older space]
    For \(\gamma \in \Ioc{0, 1}\), we define the \emph{\(0\)-H\"older space}\index{H\"older space!\(C^{0, \gamma}\)}:
    \[C^{0, \gamma}(\overline U) = \left\{u \in C^0(\overline U) \ \big\vert\ u \text{ is }\gamma\text{-H\"older continuous}\right\}\]
    Define the \(\gamma\)-H\"older seminorm by
    \[[u]_{C^{0, \gamma}(\overline U)} = \sup_{x, y \in U}\frac{\abs{u(x) -  u(y)}}{\abs{x - y}}\]
    That is, the smallest \(C\) such that \(u\) is \(\gamma\)-H\"older continuous. Since constant functions vanish, we add the \(C^0\) norm, and define
    \[\norm{u}_{C^{0, \gamma}(\overline U)} = \norm{u}_{C^0(\overline U)} + [u]_{C^{0, \gamma}(\overline U)}\]
\end{definition}

The space \(C^{0, \gamma}\), with the H\"older norm \(\norm{\cdot}_{C^{0, \gamma}(\overline U)}\) is a Banach space. We can extend this to higher order, that is,

\begin{definition}
    [\(k\)-th H\"older space]
    Define the \emph{\(k\)-th H\"older space}\index{H\"older space!\(C^{k, \gamma}\)}
    \[C^{k, \gamma}(\overline U) = \left\{u \in C^k(\overline U) \mid D^\alpha u \in C^{0, \gamma}(\overline U) \text{ for all }\abs{\alpha} = k\right\}\]
    with norm
    \[\norm{u}_{C^{k, \gamma}(\overline U)} = \norm{u}_{C^k(\overline U)} + \sum_{\abs{\alpha} = k}[D^\alpha u]_{C^{0, \gamma}(\overline U)}\]
\end{definition}

As above, \(C^{k, \gamma}\) is a Banach space with the H\"older norm.

\subsection{The Lebesgue spaces}


\begin{definition}
    [\(L^p\) space]
    Let \(U \sse \R^n\) be open, and suppose \(1 \le p \le \infty\), define \index{Lebsgue space!\(L^p\)}
    \[L^p(U) = \frac{\left\{f : U \to \R \mid f \text{ measurable, and with} \norm{u}_{L^p(U)} < \infty\right\}}{\sim}\]
    where
    \[\norm{u}_{L^p} = \begin{cases}
        \left(\int_U \abs{u(x)}^p\dd x\right)^{1/p} & 1 \le p < \infty \\
        \esssup_{x \in U}\abs{u(x)} = \inf\left\{C > 0 \mid \abs{u(x)} \le C \text{ a.e.}\right\} & p = \infty
    \end{cases}\]
    and \(u \sim v\) if \(u = v\) almost everywhere.
\end{definition}

\(L^p(U)\) with the \(\norm{\cdot}_{L^p}\) norm is a Banach space. We also define local versions,

\begin{definition}
    [\(L^p_\loc\) space]
    We say that \(u \in L^p_\loc(U)\)\index{Lebesgue space!\(L^p_\loc\)} if \(f \in L^p(V)\) for every \(V \Subset U\) (\(V \sse K \sse U\), where \(K\) is compact). Equivalently,
    \[L^p_\loc(U) = \bigcap_{V \Subset U}L^p(V)\]
\end{definition}

Note that \(L^p_\loc(U)\) is not Banach, on the other hand, it is a Fr\'echet space.

\begin{remark}
    If \(K \sse U\) is compact, \(U\) is open, then
    \[d(K, \partial U) = \inf\{\abs{x - y} \mid x \in K, y \in \R^n \setminus U\} > 0\]
    We will use the space outside \(K\) as a ``buffer zone''.
\end{remark}

\subsection{Weak derivatives}

That is, a notion of derivative for \(L^p\).

\begin{definition}
    [weak derivative]
    Suppose \(u, v \in L^1_\loc(U)\), \(\alpha\) a multi-index. We say \(v\) is the \emph{\(\alpha\)-th weak derivative of \(U\)}\index{weak derivative} if
    \[\int_U u D^\alpha \phi \dd x = (-1)^{\abs{\alpha}}\int_Uv\phi\dd x\]
    for all \(\phi \in C_c^\infty(U)\). We will also call the space \(C_c^\infty(U)\) the space of \emph{test functions}\index{test function}.
\end{definition}

\begin{remark}
    \begin{enumerate}
        \item since \(\Supp(D^\alpha\phi)\) and \(\Supp(\phi)\) are compact, the integrals are finite,
        \item \(u, v\) obey the correct integration by parts formula
    \end{enumerate}
\end{remark}

\begin{example}
    \(u(x) = \abs{x}\) is not differentiable at \(x = 0\), but it is weakly differentiable with \(v(x) = \sign(x)\).
\end{example}

\begin{lemma}
    [uniqueness of weak derivative] Suppose \(v, \tilde v \in L^1_\loc(U)\) are both the \(\alpha\)-th weak derivative of \(u \in L^1_\loc(U)\), then \(v = \tilde v\) almost everywhere.
\end{lemma}

\begin{proof}
    For all \(\varphi \in C_c^\infty(U)\),
    \[\int_U v\phi \dd x = (-1)^\abs{\alpha}\int_U u D^\alpha \phi \dd x = \int_U\overline v \phi \dd x\]
    Thus,
    \[\int_U (v - \overline v)\phi \dd x = 0\]
    for all \(\phi \in C_c^\infty(U)\). Taking \(\phi\) to be a smooth approximation to an indicator, we get the required result.
\end{proof}

Suppose \(u\) is smooth, then the weak derivative agrees with the usual derivative almost everywhere.

\begin{notation}
    We will write \(v = D^\alpha u\).
\end{notation}

\begin{definition}
    [Sobolev space] Define the \emph{Sobolev space}\index{Sobolev space!\(W^{k, p}\)}
    \[W^{k, p}(U) = \left\{u \in L^1_\loc(U) \mid u \in L^p(U) \text{, the weak derivatives }D^\alpha u\text{ exists for }\abs{\alpha} \le k\text{, }D^\alpha u \in L^p(U)\right\}\]
    with the \emph{Sobolev norm}
    \[\norm{u}_{W^{k,p}(U)} = \begin{cases}
        \left(\sum_{\abs{\alpha} = k}\int_U\abs{D^\alpha u}^p \dd x\right)^{1/p} & p < \infty \\
        \sum_{\abs{\alpha} \le k}\esssup_U \abs{D^\alpha u} & p = \infty
    \end{cases}\]
    When \(p = 2\), we write \(H^k = W^{k, 2}\)\index{Sobolev space!\(H^k\)}.
\end{definition}

\begin{definition}
    [\(W_0^{k, p}\)]
    We denote by \(W^{k, p}_0(U)\)\index{Sobolev space!\(W^{k, p}_0\)} the completion of \(C_c^\infty(U)\) with respect to the \(W^{k,p}\) norm. Analogously, we define \(H^k_0 = W^{k, 2}_0(U)\)\index{Sobolev space!\(H^k_0\)}.
\end{definition}

The \(_0\) denotes that the function vanishes on the boundary.

\lec

\begin{example}
    Let \(n > 2, \lambda > 0\), and take \(U = B_1(0) \sse \R^n\) the open ball. Consider
    \[u(x) = \begin{cases}
        \abs{x}^{-\lambda} & \text{ for }x \ne 0 \\
        \text{anything} & x = 0
    \end{cases}\]
    When is \(u \in W^{p, 1}(U)\)?

    First of all, we compute
    \[\int_U\frac{1}{\abs{x}^\lambda}\dd x = C\int_0^1 r^{n-1-\lambda}\dd r\]
    which is finite if and only if \(\lambda < n\). Moreover, \(u \in L^p(U)\) if and only if \(\lambda p < n\).

    Let \(\phi \in C_c^\infty(B_1(0) \setminus \{0\})\), if \(u\) has a weak derivative \(v\), then
    \[v_i = D_iu = -\frac{\lambda x_i}{\abs{x}^{\lambda + 2}}\]
    on \(B_1(0)\setminus 0\). Thus,
    \[\abs{D u} = \frac{\abs{\lambda}}{\abs{x}^{\lambda + 1}}\]
    Hence \(v_i \in L^1\loc(U)\) if and only if \(\lambda + 1 < n\). Suppose \(\lambda + 1 < n\), then we claim that
    \begin{align*}
        v_i = \begin{cases}
            -\frac{\lambda x_i}{\abs{x}^{\lambda + 1}} & x \ne 0 \\
            \text{anything} & x = 0
        \end{cases}
    \end{align*}
    is a weak derivative of \(u\) on \(U\).

    For \(\phi \in C_c^\infty(U)\), by Stokes' theorem,
    \[(-1)\int_{U \setminus B_\epsilon(0)}u \phi_{x_i}\dd x = \int_{U\setminus B_\epsilon(0)}D_iu \phi \dd x - \int_{\partial B_\epsilon(0)} u \phi n \vdot \dd S\]
    Therefore, we can estimate
    \[\abs{\int_{\partial B_\epsilon}u \phi n \vdot \dd S} = \abs{\phi}_{L^\infty}\abs{\int_{\partial B_\epsilon}\epsilon^{-\lambda}n \vdot \dd S} \le C \epsilon^{n-1-\lambda} \to 0\]
    as \(\lambda \to 0\). Thus, by the dominated convergence theorem,
    \[-\int_U u \phi_{x_i}\dd x = \int_U v_i\phi \dd x\]
\end{example}

\begin{remark}
    \begin{enumerate}
        \item Weak derivatives can exist even when the function is not continuous.
        \item Since \(D_iu \in L^p(U)\) if and only if \(p(\lambda + 1) < n\), we see that
        \[u \in W^{1, p}(U) \iff \lambda < \frac{n}{p} - 1\]
        and if \(p > n\), we see that \(\lambda\) must be negative, and so it is continuous.

        Heuristically, larger \(p\) gives us nicer functions.
    \end{enumerate}
\end{remark}

\begin{theorem}
    \(W^{k, p}(U)\) is a Banach space for \(k \in \N, 1 \le p \le \infty\).
\end{theorem}

\begin{proof}
    First, we need to show that it is a normed vector space. This is straightforward, and for the triangle inequality we will need to use Minkowski's inequality
    \[\left(\sum_{i=1}^\ell (a_i + b_i)^p\right)^{1/p} \le \left(\sum_{i=1}^n a_i^p\right)^{1/p} + \left(\sum_{i=1}^n b_i^p\right)^{1/p}\]

    For completeness, we will use the completeness of \(L^p\). Let \((u_j)\) be a Cauchy sequence on \(W^{k, p}(U)\).

    Note that \(\norm{D^\alpha v}_{L^p(U)} \le \norm{v}_{W^{k, p}(U)}\) for \(\abs{\alpha} \le k\). Setting \(v = u_j\), we see that \((D^\alpha u_j)_j\) is a Cauchy sequence in \(L^p\). But we know that \(L^p\) is complete, and so there exists a function \(u^\alpha \in L^p(U)\), with \(D^\alpha u_j \to u^\alpha\) in \(L^p\) for all \(\abs{\alpha}\le k\). We will set \(u = u^{(0, \dots, 0)}\).

    \begin{claim}
        \(u^\alpha\) is the \(\alpha\)-th weak derivative of \(u\). That is, \(D^\alpha u\) exists and \(D^\alpha u = u^\alpha\).
    \end{claim}

    \begin{proof}
        Choose a test function \(\phi \in C_c^\infty(U)\). Since \(u_j \in W^{k, p}\), we know that \(D^\alpha u_j\) exists, and
        \[(-1)^{\abs{\alpha}}\int_U u_j D^\alpha \phi \dd x = \int_U(D^\alpha u_j)\phi \dd x\]
        for all \(j\). Taking \(j \to \infty\), using the fact that \(D^\alpha u_j \to u^\alpha\) (using H\"older, or the dominated convergence theorem), we get that
        \[(-1)^{\abs{\alpha}}\int_U uD^\alpha \phi \dd x = \int_U u^\alpha \phi \dd x\]
        and so \(D^\alpha u = u^\alpha \in L^p(U)\).
    \end{proof}

    Thus, \(u \in W^{k,p}(U)\).
\end{proof}

\subsection{Approximations of Sobolev spaces}

\subsubsection*{Convolution and mollifiers}

\begin{definition}
    [standard mollifier]
    Let
    \[\eta(x) = \begin{cases}
        C \exp(\frac{-1}{1-\abs{x}^2}) & \abs{x} < 1 \\
        0 & \abs{x} \ge 1
    \end{cases}\]
    where \(C\) is chosen such that
    \[\int_{\R^n}\eta(x)\dd x = 1\]
    For \(\epsilon > 0\), we denote
    \[\eta_\epsilon(x) = \frac{1}{\epsilon^n}\eta(x/\epsilon)\]
    We call \(\eta_\epsilon\) to be the \emph{standard mollifer}\index{standard mollifier}.
\end{definition}

\begin{remark}
    \begin{itemize}
        \item \(\eta_\epsilon \in C_c^\infty(\R^n)\),
        \item \(\Supp(\eta_\epsilon) = \overline{B_\epsilon(0)}\),
        \item \(\int_{\R^n}\eta_\epsilon(x)\dd x = 1\)
    \end{itemize}
\end{remark}

\begin{definition}
    Given \(U \sse \R^n\) open, define
    \[U_\epsilon = \left\{x \in U \mid d(x, \partial U) > \epsilon\right\}\]
\end{definition}

\begin{definition}
    [mollification]
    Given \(f \in L^1_\loc(U)\), the \emph{mollification of \(f\)}\index{mollification} is
    \begin{align*}
        f_\epsilon : U_\epsilon &\to \R \\
        f_\epsilon(x) &= \eta_\epsilon * f(x) = \int_{U}\eta_\epsilon(x-y)f(y)\dd y = \int_{B_\epsilon(0)}\eta_\epsilon(y)f(x-y) \dd y
    \end{align*}
    where \(*\) denotes the convolution.
\end{definition}

We can think of \(f_\epsilon(x)\) as the average of \(f\) in an \(\epsilon\)-ball, weighted by \(\eta\).

\begin{theorem}
    [properties of mollification]
    \index{mollification!properties of}
    Let \(f \in L^1_\loc(U)\), then
    \begin{enumerate}
        \item \(f_\epsilon \in C^\infty(U_\epsilon)\),
        \item \(f_\epsilon \to f\) a.e. on \(U\) as \(\epsilon\to 0\),
        \item if \(f \in C^0(U)\), then \(f_\epsilon \to f\) locally uniformly (i.e. uniformly on \(K \sse U\) compact).
        \item if \(1 \le p < \infty\), and \(f \in L^p_\loc(U)\), then \(f_\epsilon \to f\) in \(L^p_\loc(U)\). That is,
        \[\norm{f_\epsilon - f}_{L^p(V)} \to 0\]
        for all \(V \Subset U\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    See handout on moodle.
\end{proof}

In particular, there is a big improvement going from \(f \in L^1_\loc\) to \(f_\epsilon \in C^\infty\).

\begin{lemma}
    [local smooth approximation of Sobolev functions away from \(\partial U\)] \index{Sobolev approximation!local smooth approximation away from \(\partial U\)}\label{lem:local-smooth-approx-away-from-boundary}Let \(u \in W^{k, p}(U)\) for some \(1 \le p < \infty\). Set \(u_\epsilon = \eta_\epsilon * u\) in \(U_\epsilon\). Then
    \begin{enumerate}
        \item \(u_\epsilon \in C^\infty(U_\epsilon)\) for all \(\epsilon > 0\),
        \item \(u_\epsilon \to u\) in \(W^{k, p}_\loc(U)\). Note that for \(V \Subset U\), \(V \subseteq U_\epsilon\) for \(\epsilon\) sufficiently small.
    \end{enumerate}
\end{lemma}

\begin{proof}
    (i) follows from the theorem. For (ii),
    \begin{claim}
        \[D^\alpha u_\epsilon = D^\alpha(\eta_\epsilon * u) = \eta_\epsilon * D^\alpha u\]
    \end{claim}
    \begin{proof}
        Since \(u_\epsilon \in C^\infty\), we can compute the classical derivative as follows:
        \begin{align*}
            D^\alpha_x u_\epsilon(x) &= D^\alpha_x\int_U \eta_\epsilon(x - y)u(y) \dd y \\
            &= \int_U (D^\alpha_x\eta_\epsilon(x - y))u(y) \dd y \\
            &= (-1)^\abs{\alpha}\int_U(D^\alpha_u\eta_\epsilon(x - y))u(y) \dd y \\
            &= \int_U \eta_\epsilon(x - y)D^\alpha u(y) \dd y \\
            &= (\eta_\epsilon * u)(x)
        \end{align*}
        See handout for justification of swapping the integral and derivative.
    \end{proof}
    Note for \(V \Subset U\), by (iv) of the theorem, since \(D^\alpha u \in L^p(U)\), then
    \[D^\alpha u_\epsilon = \eta_\epsilon * D^\alpha u \to D^\alpha u\]
    in \(L^p(V)\) as \(\epsilon \to 0\). Thus, for all \(V \Subset U\), and \(\delta > 0\), there exists \(\epsilon_0 = \epsilon_0(\delta, V)\) such that
    \begin{align*}
        \norm{u_\epsilon - u}_{W^{k, p}(V)}^p = \sum_{\abs{\alpha} \le k}\norm{D^\alpha u_\epsilon - D^\alpha u}_{L^p(V)}^p \le \delta
    \end{align*}
    for \(0 < \epsilon \le \epsilon_0\).
\end{proof}

In conclusion, \(u \in W^{k, p}(U)\) can be approximated by \(C^\infty\) functions away from \(\partial U\).

\lec

\begin{theorem}
    [global Sobolev approximation globally away from \(\partial U\)]
    \index{Sobolev approximation!global Sobolev approximation globally away from \(\partial U\)}\label{thm:global-sobolev-approx-away-from-boundary}
    Suppose \(U \sse \R^n\) open bounded, and suppose \(u \in W^{k, p}(U)\), for \(1 \le p < \infty\). Then there exists a sequence \((u_j) \in C^\infty \cap W^{k, p}(U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\).
\end{theorem}

\underline{Exercise:} Drop the assumption that \(U\) is bounded.

\begin{remark}
    Note that we don't assume \(u \in C^\infty(\overline U)\).
\end{remark}

\begin{proof}
    \textbf{Step 1:} We have
    \[U = \bigcup_{j=1}^\infty\]
    where
    \[U_j = \left\{x \in U \mid d(x, \partial U) > 1/j\right\}\]
    and define \(V_j = U_{j+3} \setminus \overline{U_{j+1}} \Subset U\). Choose \(V_0 \Subset U\) such that \(U = \bigcup_{j=0}^\infty V_j\). Note in particular only the consecutive \(V_j\) intersect.

    Let \(\xi_j\) be a \emph{partition of unity subordinate to \(V_j\)}. That is,
    \begin{itemize}
        \item \(0 \le \xi_j \le 1\),
        \item \(\xi_j \in C_c^\infty(V_j)\),
        \item \(\sum_{j=0}^\infty \xi_j(x) = 1\) for all \(x \in U\). Note at any point at most two \(\xi_j\) are non-zero.
    \end{itemize}

    Given \(u \in W^{k, p}(U)\), then we see that \(\xi_j u \in W^{k, p}(U)\), and \(\Supp(\xi_j u)\in V_j\).

    \textbf{Step 2:} We would like to smooth out the split up function. Let \(W_j = U_{j+4} \setminus \overline{U_j} \supseteq V_j\). Let
    \[u_j = \eta_{\epsilon_j} * (\xi_ju)\]
    Fix \(\delta > 0\), for each \(j \ge 1\), we can choose \(\epsilon_j\) sufficiently small such that \(\Supp(u_j) \sse W_j\).

    By \cref{lem:local-smooth-approx-away-from-boundary}, we have that \(u_j \to \xi_j u\) in \(W^{k, p}(W_j)\). With this, we can make
    \[\norm{u_j -  \xi_j u}_{W^{k, p}(U)} = \norm{u_j - \xi_j u}_{W^{k, p}(U_j)} \le \frac{\delta}{2^{j+1}}\]

    Summing everything together, let \(v = \sum_{j=0}^\infty u_j\). Note that each \(u_j\) is non-zero on finitely many \(W_j\), and so at each point it is a finite sum. With this, \(v\) is smooth. Also note that \(u = \sum_j \xi_j u\) on \(U\), and so for any \(V \Subset U\),
    \[\norm{v - u}_{W^{k,p}(V)} \le \sum_{j=1}^\infty \norm{u_j - \xi_j u}_{W^{k, p}(V)} \le \delta \sum_{j=1}^\infty \frac{1}{2^{j+1}} = \delta\]
    where we applied the triangle inequality. Since \(\delta\) is independent of \(V\),  taking the \(\sup\) over all \(V \Subset U\), we get that
    \[\norm{v - u}_{W^{k, p}(U)} \le \delta\]
\end{proof}

\underline{Question:} Can we approximate \(u \in W^{k, p}(U)\) by \(u \in C^\infty(\overline U)\)?

The issue here is that \(\partial U\) could be a problem. For example, we can consider \(\partial U\) to be the Cantor set.

\begin{definition}
    [\(C^{k, \delta}\)-domain]
    Suppose \(U \sse \R^n\) is bounded and open. Then we say that \(\partial U\) is a \emph{\(\C^{k, \delta}\)-domain}\index{\(C^{k, \delta}\) domain} if for every \(p \in \partial U\), there exists \(r > 0\), and a function \(\gamma : \R^{n-1} \to \R\), with \(\gamma \in C^{k,\delta}(\R^{n-1})\), such that (after relabelling axes),
    \[U \cap B_r(p) = \left\{(x', x_n) \in B_(p) \mid x_n > \gamma(x')\right\}\]
\end{definition}

\begin{theorem}
    [smooth approximation of Sobolev functions up to \(\partial U\)]
    \index{Sobolev approximation!smooth approximation up to \(\partial U\)}\label{thm:smooth-approx-up-to-boundary}
    Let \(U \sse \R^n\) be open, bounded and \(\partial U\) a \(C^{0, 1}\) domain. Let \(u \in W^{k, p}(U)\), for some \(1 \le p < \infty\). Then there exists a sequence \((u_j) \in C^\infty(\overline U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\).
\end{theorem}

\begin{proof}
    \textbf{Step 1:} Fix \(x_0 \in \partial U\). Since \(\partial U\) is Lipschitz, there exists \(r > 0\) and a Lipschitz function \(\gamma : \R^{n-1} \to \R\), such that
    \[U \cap B_r(x_0) = \left\{x \in B_r(x_0) \mid x_n > \gamma(x')\right\}\]
    Let \(V = U \cap B_{r/2}(x_0)\).

    \textbf{Step 2:} Define the shifted point \(x^\epsilon = x + \lambda \epsilon e_n\), for \(x \in V, \epsilon > 0\).

    \begin{claim}
        For \(\lambda > 0\) large enough, \(B_\epsilon(x^\epsilon) \sse U \cap B_r(x_0)\) for all \(\epsilon > 0\).
    \end{claim}

    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_21.jpg}
    \end{center}

    That is, we need to show that for \(y \in B_\epsilon(x^\epsilon)\), \(y_n > \gamma(y')\). As \(\gamma\) is Lipschitz, there exists a constant \(L > 0\) such that
    \[\abs{\gamma(x') - \gamma(y')} \le L\abs{x' - y'}\]
    So we have that
    \[\abs{y' - (x^\epsilon)'} = \abs{y' - x'} < \epsilon\]
    and so,
    \[\gamma(y') \le \gamma(x') + L_\epsilon < x_n + L\epsilon\]
    by rearranging \(y_n > x_n^\epsilon - \epsilon = x_n + \lambda\epsilon - \epsilon = x_n + (\lambda - 1)\epsilon\), we see that
    \[y_n > \gamma(y')\]
    if \(\lambda \ge L + 1\).

    Define \(u_\epsilon(x) = u(x^\epsilon)\) for \(x \in V\). Set
    \[v_{\delta, \epsilon} = \eta_\delta * u_\epsilon\]
    for \(0 < \delta < \epsilon\). Then \(v_{\delta, \epsilon} \in C^\infty(\overline V)\). Fix \(\mu \ge 0\), then we note
    \[\norm{v_{\delta, \epsilon} - u}_{W^{k, p}(V)} \le \norm{v_{\delta, \epsilon} - u_\epsilon}_{W^{k, p}(V)} + \underbrace{\norm{u_\epsilon - u}_{W^{k, p}(V)}}_{\text{ translation is continuous on \(W^{k, p}\)}}\]
    We can choose \(\epsilon > 0\) such that the second term is at most \(\mu\). Fix \(\epsilon > 0\), we can choose \(\delta < \epsilon\) such that the first term is at most \(\mu\), using the same proof as in \cref{lem:local-smooth-approx-away-from-boundary}.

    \textbf{Step 3:} Let \(x_0\) vary over the boundary, then the \(V\)s which we get will cover the boundary, which is compact, and so we have a finite subcover. That is, finitely many points \(x_1, \dots, x_N \in \partial U\) and radii \(r_i\), where
    \[V_i = B_{r_i/2}(x_i) \cap U\]
    Choose \(V_0 \Subset U\) such that
    \[U = V_0 \cup V_0 \cup \cdots \cup V_N\]
    By step 2, we have \(v_i \in C^\infty(\overline{V_i})\), such that \(\norm{v_i - u}_{W^{k, p}(V_i)} \le \mu\). By \cref{lem:local-smooth-approx-away-from-boundary} there exists \(v_0 \in C^\infty(\overline{V_0})\) such that \(\norm{v_0 - u}_{W^{k, p}(V_0)} \le \mu\).

    \textbf{Step 4:} Summing everything together, using a partition of unity \(\xi_0, \dots, \xi_N\) subordinate to the open cover \(V_0, \dots, V_N\). Define
    \[v = \sum_{i=0}^n\xi_i v_i\]
    This sum is finite, and so \(v \in C^infty(\overline{U})\), and for \(\abs{\alpha} \le k\),
    \begin{align*}
        \norm{D^\alpha v - D^\alpha u}_{L^p(U)} &\le \sum_{i=0}^N\norm{D^\alpha(\xi_i(v_i - u))}_{L^p(V_i)} \\
        &\le C_k\sum_{i=0}^N\norm{v_i - u}_{W^{k, p}(V_i)} \\
        &= C_k(1+ N)\mu
    \end{align*}

    But \(\mu\) was arbitrary, and so we are done.
\end{proof}

\lec

To conclude, we consider some examples of functions:

\begin{itemize}
    \item \(\abs{x} \notin C^\infty(-1, 1)\), but it is in \(W^{1,1}(-1, 1)\).
    \item \(1/x\) is \(C^\infty\) and \(L^1_\loc\) on \((0, 1)\), but not \(C^\infty(\overline{(-1,1)})\) or \(W^{1, 1}\).
\end{itemize}
and so, \(C^\infty(U), C^\infty(\overline U) \not\subseteq W^{k,p}(U)\).

\subsection{Extensions and traces}

Suppose \(U \sse \R^n\) is open and bounded, \(u \in W^{k,p}(U)\). We would like to extend \(u\) to \(\overline{u} : \R^n \to \R\). What happens if we set
\begin{align*}
    \overline u = \begin{cases}
        u & \text{ on }U \\
        0 & \text{ on }U^c
    \end{cases}
\end{align*}
This is okay for \(L^p\), but not for \(W^{k, p}\) as the derivatives become an issue. Moreover, we can expect at most \(\overline u \in W^{k, p}(\R^n)\).


\begin{theorem}
    [Calderon, Stein] Suppose \(U \sse \R^n\) is open and bounded, and \(\partial U\) is \(C^1\). Choose \(V\) bounded in \(\R^n\), with \(U \Subset V\). Let \(1 \le p < \infty\). Then there exists a bounded linear operator
    \begin{align*}
        E : W^{1, p}(U) &\to W^{1, p}(\R^n) \\
        u &\mapsto \overline u
    \end{align*}
    such that for all \(u \in W^{1, p}(U)\),
    \begin{itemize}[(i)]
        \item \(\overline{u}\vert_U = u\) a.e.
        \item \(\Supp(E(u)) \sse V\),
        \item There exists a constant \(C\) depending only on \(U, V, p\), such that \[\norm{E(u)}_{W^{1,p}(\R^n)} \le C\norm{u}_{W^{1, p}(U)}\]
    \end{itemize}

    We call \(Eu\) the \emph{extension of \(U\) to \(R^n\)}\index{extension}.
\end{theorem}

\begin{proof}
    \textbf{Step 1:} Fix \(p \in \partial U\), and suppose \(\partial U\) is flat near \(p\). We may assume there exists \(r > 0\) such that
    \begin{align*}
        B^+ &= B_r(p) \cap \{x_n \ge 0\} \sse \overline{U} \\
        B^- &= B_r(p) \cap \{x_n < 0\} \sse \R^n \setminus \overline{U}
    \end{align*}
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_24 (1).jpg}
    \end{center}
    Suppose also that \(u \in C^1(\overline U)\).Denote \(x' = (x_1, \dots, x_{n-1})\). We define
    \[\overline{u}(x) = \begin{cases}
        u(x) & \text{ if }x \in B^+ \\
        -3u(x', -x_n) + 4u(x', -x_n/2) & \text{ if }x \in B^-
    \end{cases}\]
    which is called a \emph{higher order reflection of \(u\) from \(B^+\) to \(B^-\)}\index{higher order reflection}
    \begin{claim}
        \(\overline{u} \in C^1(B_r(p))\)
    \end{claim}
    \begin{proof}
        Clearly \(\overline{u}\) is continuous. Computing the derivatives:
        \[\partial_{x_n}\overline{u} = \begin{cases}
            \partial_{x_n}u(x) & x \in B^+ \\
            3\partial_{x_n}u(x', -x_n) - 2\partial_{x_n}u(x', -x_n/2) & x \in B^-
        \end{cases}\]
        Similarly,
        \[\partial_{x_i}\overline{u} = \begin{cases}
            \partial_{x_i}\overline(u) & x \in B^+ \\
            -3\partial_{x_i}u(x', -x_n) + 4\partial_{x_i}u(x', -x_n/2) & x \in B^-
        \end{cases}\]
        and so the derivative is continuous. 
    \end{proof}
    We can also check that the inequality holds in this case.

    \textbf{Step 2:} Suppose \(\partial U\) is not flat near \(p\). Since \(\partial U\) is \(C^1\), there exists \(r > 0\) and \(\gamma : \R^{n-1} \to \R\), such that
    \[U \cap B_r(p) = \left\{x \mid x_n > \gamma(x')\right\}\]
    
    Define
    \begin{align*}
        \Phi : \R^n &\to \R^n \\
        \Phi(x) &= (x_1, \dots, x_{n-1}, x_n - \gamma(x'))
    \end{align*}
    We can see that \(\Phi\) maps \(\partial U\) to \(\{y_n=0\}\), and it is invertible with \(C^1\) inverse
    \[\Psi(y) = (y_1, \dots, y_{n-1}, y_n + \gamma(y'))\]
    with
    \begin{itemize}
        \item \(\Phi(U \cap B_r(p)) \sse \{y_n > 0\}\)
        \item \(\det(D\Phi) = \det(D\Psi) = 1\)
    \end{itemize}
    \begin{center}
        \includegraphics[scale=0.15]{diagrams/2023_10_24 (2).jpg}
    \end{center}
    Moreover, there exists a neighbourhood \(W\) of \(p\), with \(\Phi(W) = B_s(\tilde p)\) for some \(s > 0\). In this case,
    \[\Phi(U \cap W) = B_s(\tilde p) \cap \{y_n > 0\} = B^+\]
    Define \(v(y) = u(\Psi(y))\) for \(y \in B_+\). Then \(v\) is \(C^1\), and so by step 1 there exists an extension \(\overline{v} \in C^1(B_s(\tilde p))\) with \(\overline v\vert_{B^+} = v\) and
    \[\norm{\overline v}_{W^{1, p}(B_s(\tilde p))} \le C\norm{v}_{W^{1, p}(B^+)}\]
    Define \(\overline u(x) = \overline{v}(\Phi(x))\), then \(\overline{u} \in C^1(W)\), and
    \[\norm{\overline u}_{W^{1, p}(W)} \le C\norm{u}_{W^{1, p}(U)}\]
    which we will see on examples sheet 1.

    \textbf{Step 3:} Now we have local extensions near all \(p \in \partial U\). We assume \(U\) is bounded, and so we have an open cover \(\{W_0, \dots, W_N\}\), with
    \[U \sse \bigcup_{i=0}^N W_i\]
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_24 (3).jpg}
    \end{center}
    and we have extensions \(\overline u_i \in C^1(W_i)\). Let \((\xi_i)_{i=0}^N\) be a partition of unity subordinate to \(\{W_i\}\). Let
    \[\overline{u} = \sum_{i=0}^N \xi_i \overline{u_i}\]
    where \(\overline{u_0} = u\). Then \(\overline u\vert_U = u\) ae., and
    \[\overline{u} \in C_c^1(\R^n)\]
    with
    \[\norm{\overline u}_{W^{1, p}(\R^n)} \le C\norm{u}_{W^{1, p}(U)}\]

    We may assume \(\Supp(\overline u) \sse V\), since \(U \Subset V\), for example by using a cutoff function.

    \textbf{Step 4:} Given \(u \in W^{1, p}(U)\), by \cref{thm:smooth-approx-up-to-boundary}, there exists a sequence \((u_j) \in C^\infty(\overline U)\) with \(u_j \to u\) in \(W^{1,p}(U)\).

    \begin{claim}
        \((E(u_j))_j\) is a Cauchy sequence in \(W^{1, p}(\R^n)\).
    \end{claim}

    \begin{proof}
        By the previous steps, we have that \(E(u_j) \in W^{1, p}(\R^n)\). By linearity,
        \[\norm{E(u_j) - E(u_k)}_{W^{1, p}(\R^n)} = \norm{E(u_j - u_k)}_{W^{1, p}(\R^n)} \le C\norm{u_j - u_k}_{W^{1, p}(U)}\]
        But we know that \((u_j)\) is convergent, and thus Cauchy in \(W^{1, p}(U)\).
    \end{proof}
    Since \(W^{1, p}(\R^n)\) is complete, the sequence converges and we define
    \[E(u) = \lim_j E(u_j)\]
\end{proof}

\begin{remark}
    If \(\partial U\) is \(C^k\), then we have the extension operators
    \[E : W^{k, p}(U) \to W^{k, p}(\R^n)\]
    Given \(u \in C^k(\overline U)\), we set
    \[\overline{u} = \begin{cases}
        u(x) & x\in B_+ \\
        \sum_{j=1}^k c_ju(x', -x_n/j) & x \in B_-
    \end{cases}\] 
    To match at the boundary, we need
    \[\sum_{j=1}^kc_j\left(\frac{-1}{j}\right)^m = 1\]
    for all \(m = 0, \dots, k-1\).
\end{remark}

\lec

\subsubsection*{Traces}

If we have \(u \in C^0(\overline U)\), then \(u \vert_{\partial U}\) makes sense. But for \(u \in W^{k, p}(U)\), then \(u\vert_{\partial U}\) does not make sense, as \(\partial U\) has measure zero.

\begin{theorem}
    Let \(U\) be open bounded and \(\partial U\) is \(C^1\). Then there exists a bounded linear operator
    \[T : W^{1, p}(U) \to L^p(\partial U)\]
    called the \emph{trace of \(u\) on \(\partial U\)}\index{trace}, such that
    \begin{enumerate}[(i)]
        \item \(T(u) = u\vert_{\partial U}\) if \(u \in W^{1, p}(U) \cap C^0(\overline U)\),
        \item \(\norm{T(u)}_{L^p(\partial U)} \le C \norm{u}_{W^{1, p}(U)}\) for all \(u \in W^{k, p}(U)\), where \(C\) depends only on \(U, p\).
    \end{enumerate}
\end{theorem}

\begin{remark}
    We have \(u, Du \in L^p\), which is giving us the control of \(u\) on the boundary.
\end{remark}

\begin{proof}
    [Sketch proof] See examples sheet 2. Suppose \(u \in C^1(\overline U)\), and \(\partial U\) is flat near \(p\). Let
    \begin{align*}
        B^+ &= B_r(p) \cap \{x_n \ge 0\} \sse \overline U \\
        B^- &= B_r(p) \cap \{x_n < 0\} \sse \R^n \setminus U
    \end{align*}
    as before, and let \(\Gamma\) be the portion of \(\partial U\) within \(B_r(p)\). Choose \(\xi \in C_c^\infty(B_r(p))\) such that \(0 \le \xi \le 1\) on \(B_r(p)\), and \(\xi = 1\) on \(B_{r/2}(p)\). Then
    \begin{align*}
        \int_\Gamma \abs{u(x', 0)}^p \dd x' &\le \int_{B_r(p) \cap \{x_n = 0\}} \xi \abs{u(x', 0)}^p \dd x' \\
        &\underbrace{=}_{\text{FTC}} (-1)\int_{B^+}\partial x_n(\xi\abs{u}^p)\dd x_n \dd x' \\
        &= (-1)\int_{B^+}\abs{u^p}\partial_{x_n}\xi + p \abs{u}^{p-1}\sign(u)\partial_{x_n}u \xi \dd x \\
        &\underbrace{\le}_{\text{Young's inequality}} C_p\int_{B^+}\abs{u}^p + \abs{Du}^p \dd x \\
        &= C_p\norm{u}^p_{W^{1, p}(U)}
    \end{align*}

    In Sheet 2, we will extend to general \(\partial U\) using a partition unity, and the fact that it is compact. Then define
    \[T(u) = u\vert_{\partial U}\]
    for \(u \in C^1(\overline U)\), and we have that
    \[\norm{T(u)}_{L^p(\partial U)} \le C\norm{u}_{W^{1, p}(U)}\]
    Using density of \(C^\infty(\overline U)\) in \(W^{1, p}(U)\), we are done.
\end{proof}

\begin{remark}
    \begin{itemize}
        \item The map \(T\) above is not surjective, however in the case of \(T : \HH^s \to \HH^{s-1/2}\), it is surjective.
        \item Recall \(W^{k,p}_0(U)\) is the closure of \(C_c^\infty(U)\) in \(W^{k, p}(U)\). If \(u \in W^{k,p}_0(U)\), then there exists \((u_j) \sse C_c^\infty(U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\). In particular,
        \[T(u) = T(\lim_j u_j) = \lim_j T(u_j) = 0\]
        In fact, the converse is true also. If \(T(u) = 0\), then \(u \in W^{1, p}_0(U)\).
        \item if \(u \in W^{k, p}(U)\), then we can define trace operators for \(Du, \dots, D^{k-1}(U)\).
    \end{itemize}
\end{remark}

\subsection{Sobolev inqualities}

In this case, the basic idea is that we can trade differentiability (measured by \(k\)) for integrability (measured by \(p\)). Note it does not work the other way. For example, if \(f' \in L^1(\R)\) then \(f \in L^\infty(\R)\), but the converse is not true.

The idea is that we will prove estimates of the form
\[\norm{u}_{L_q(\R^n)} \le C\norm{D u}_{L^p(\R^n)} \left(+ \norm{u}_{L^p(\R^n)}\right)\]

We have three cases:
\begin{enumerate}
    \item \(1 \le p < n\),
    \item \(p = n\),
    \item \(n < p \le \infty\).
\end{enumerate}

\subsubsection*{Case 1: \(1 \le p < n\)}

\begin{lemma}
    Let \(n \ge 2\), and \(f_1, \dots, f_n \in L^{n-1}(\R^{n-1})\). Then for \(1 \le i \le n\), define 
    \[\widetilde x_i = (x_1, \dots, x_{i-1}, x_{i+1}, \dots, x_n) \in \R^{n-1}\]
    and
    \[f(x) = \prod_{i=1}^n f_i(\widetilde x_i) : \R^n \to \R\]
    Then \(f \in L^1(\R^n)\), with
    \[\norm{f}_{L^1(\R^n)} \le \prod_{i=1}^n \norm{f_i}_{L^{n-1}(\R^{n-1})}\]
\end{lemma}

\begin{proof}
    We induct on \(n\). The case \(n = 2\) gives
    \[f(x_1, x_2) = f_1(x_1)f_2(x_2)\]
    But
    \begin{align*}
        \norm{f}_{L^1(\R^2)} &= \int_{\R^2}\abs{f_1(x_1)}\abs{f_2(x_2)}\dd x_1 \dd x_2 \\
        &= \int_\R \abs{f_1(x_1)}\dd x_1 \int \abs{f_2(x_2)}\dd x_2 \\
        &= \norm{f_1}_{L^1(\R)}\norm{f_2}_{L^1(\R)}
    \end{align*}
    Suppose the result is true for \(n\). Write
    \[F(x) = f_1(\widetilde x_1) \cdots f_n (\widetilde x_n)\]
    and so \(f(x) = F(x)f_{n+1}(\tilde x_{n+1})\). Fix \(x_{n+1}\) and integrate over \(x_1, \dots, x_n\):
    \begin{align*}
        \int_{\R^n}\abs{f(\xi_1, \dots, \xi_n, x_{n+1})} \dd \xi_1 \cdots \dd x_n &= \int_{\R^n}\abs{F(\xi, x_{n+1})} \abs{f_{n+1}(\xi)}\dd \xi \\
        &\underbrace{=}_{\text{H\"older}} \norm{F(\cdot, x_{n+1})}_{L^{n/(n-1)}(\R^n)}\norm{f_{n+1}}_{L^n(\R^n)}
    \end{align*}
    By the induction hypothesis, if \(q = n/(n-1)\), then
    \begin{align*}
        \norm{F(\cdot, x_{n+1})}_{L^{n/(n-1)}(\R^n)} &= \norm{F(\cdot, x_{n+1}^q)}^{1/q}_{L^1(\R^n)} \\
        &\le \prod_{i=1}^n \norm{f_i(\cdot, x_{n+1})^{n/(n-1)}}^{(n-1)/n}_{L^{n-1}(\R^n)} \\
        &= \prod_{i=1}^n \norm{f(\cdot, x_{n+1})}_{L^n(\R^{n-1})}
    \end{align*}
    Integrating over \(x_{n+1}\),
    \begin{align*}
        \norm{f}_{L1(\R^{n+1})} &\le \norm{f_{n+1}}_{L^n(\R^n)}\int_\R \prod_{i=1}^n \norm{f_i(\cdot, x_{n+1})}_{L^n(\R^{n-1})}\dd x_{n+1} \\
        &\le \norm{f_{n+1}}_{L^n(\R^n)}\prod_{i=1}^n\left(\int_\R \norm{f_i(\cdot, x_{n+1})}_{L^n(\R^{n-1})}^n \dd x_{n+1}\right)^{1/n} \\
        &= \prod_{i=1}^{n+1}\norm{f_i}_{L^n(\R^n)}
    \end{align*}
    Where we used the generalised H\"older inequality with \(p = n\), that is,
    \[\norm{\prod_i f_i}_{L^1} \le \prod_{i=1}^n \norm{f_i}_{L^{p_i}}\]
    where \(\sum \frac{1}{p_i} = 1\).
\end{proof}

\begin{theorem}
    [Gagliado-Nirenberg-Sobolev inequality]
    \label{thm:Gagliado-Nirenberg-Sobolev}
    \index{Gagliado-Nirenberg-Sobolev inequality}
    Suppose \(1\le p < n\), set
    \[p^* = \frac{np}{n-p}\]
    for the \emph{Sobolev conjugate of \(p\)}\index{Sobolev conjugate}. Then we have a continuous embedding
    \[W^{1, p}(\R^n) \sse L^{p^*}(\R^n)\]
    That is, there exists a constant \(C\) depending only on \(n, p\), such that
    \begin{equation}
        \label{eq:gns}
        \norm{f}_{L^{p^*}(\R^n)} \le C\norm{D u}_{L^q(\R^n)}
    \end{equation}
\end{theorem}

\begin{remark}
    \begin{enumerate}
        \item \(p^* > p\).
        \item Nothing is said about \(\norm{Du}_{L^{p^*}}\).
    \end{enumerate}
\end{remark}

\subsubsection*{Intuition}

Consider \(f : \R^2 \to \R\). \(L^p\) measures the width and the height of the function. For example, if we have
\[f_1 = A 1_W\]
then
\[\norm{f_1}_{L^p} = \abs{A}\vol(W)^{1/p} = \abs{A}V^{1/p}\]
Now consider \(\phi \in C_c^\infty(\R^n)\) with \(\phi(x) \le 1\), and let
\[f_2(x) = \phi(x)e^{i\omega \cdot x}\]
Then we know \(\abs{f_2}(x) \le 1\), and \(\Supp(f_2) \sse C\) is uniformly bounded in \(\omega\). With this,
\[\partial_1f_2 =  \phi'e^{i\omega \cdot x} + i\phi \omega_1e^{i\omega\cdot x}\]
Thus, \(Df_2\) is \emph{not} uniformly bounded in \(\omega\).

Next, consider
\[f_3(x) = \abs{\omega}^{-k}\phi(x)e^{i \omega \cdot x}\]
and so, we have a uniform bound in \(\omega\) of \(D^\ell f_3\) for \(\ell \le k\).

Finally, let
\[f_4(x) = A\phi(x/R)e^{i\omega \cdot x}\]
Then
\[\norm{f_4}^p_{W^{1, p}} \sim \int_{\abs{x} \le R}\abs{A\phi e^{i\omega \vdot x}}^p + \int_{\abs{x} \le R}\abs{\frac{A}{R}\phi'e^{i\omega \cdot x} + A\phi \omega e^{i\omega\vdot x}}^p \sim \abs{A}C^{1/p}\abs{\omega}\]
Now recall the uncertainty principle:
\[\delta_x \delta_p \ge \frac{\bar h}{2} > 0\]
Thus,
\[\text{volume} \times \text{frequency} \ge c > 0\]
Thus, a function with frequency \(\omega\) must be spread out on a ball of radius \(\ge 1/\omega\). Thus, the support must have measure \(\ge \omega^{-n}\). With this, \(\omega \ge V^{-n}\), and so
\[\norm{f}_{W^{1, p} \ge \abs{A}V^{1/p - 1/n}} = \abs{A}V^{p^*} = \norm{f}_{L^{p^*}}\]

See ``Tery Tao uncertainty principle'' for more details.

\lec

\begin{remark}
    \begin{itemize}
        \item If \(u \equiv 1\), then it wouldn't satisfy \cref{eq:gns}, and so it is essential that we are in \(W^{1, p}\).
        \item Proof follows from density of \(C_c^\infty(\R^n)\) in \(W^{k, p}_0(\R^n) = W^{k,p}(\R^n)\).
    \end{itemize}
\end{remark}

\begin{proof}
    \textbf{Step 1:} We can assume \(u \in C_c^\infty(\R^n)\), and first consider the case \(p = 1\). By FTC and compact support,
    \[u(x) = \int_{-\infty}^{x_i}\partial_{x_i}u(x_1, \dots, x_{i-1}, y_i, x_{i+1}, \dots, x_n)\dd y_i\]
    which means that
    \[\abs{u(x)} \le \underbrace{\int_{-\infty}^\infty \abs{Du(x_1, \dots, x_{i-1}, y_i, x_{i+1}, \dots, x_n)}}_{=f_i(\tilde x_i)\dd y_i}\]
    Thus,
    \[\abs{u(x)}^n \le f_1(\tilde x_1)\cdots f_n(\tilde x_n)\]
    Integrating over \(x \in \R^n\),
    \[\norm{\abs{u}}^{n/(n-1)}_{L^1(\R^n)} \le \norm{\prod_{i=1}^n \left(f_i(\tilde x_i)\right)^{1/(n-1)}}_{L^1{\R^n}} \le \prod_{i=1}^n \norm{f_i^{1/(n-1)}}_{L^{n-1}(\R^{n-1})} = \norm{Du}_{L^1(\R^n)}^{n/(n-1)}\]
    With this,
    \[\norm{u}_{L^{n/(n-1)}(\R^n)} \le \norm{Du}_{L^1(\R^n)}\]
    and in this case, \(p^* = n/(n-1)\). Since \(C_c^\infty(\R^n)\) is dense in \(W^{1, 1}(\R^n)\), result follows by density.
    
    \textbf{Step 2:} Now suppose \(p > 1\), consider \(v(x) = \abs{u(x)}^\gamma\). Then
    \[Dv = \gamma \sign(u)\abs{u}^{\gamma - 1}Du\]
    and
    \begin{align*}
        \left(\int_{\R^n}\abs{u}^{\gamma n/(n-1)} \dd x\right)^{(n-1)/n} &= \norm{\abs{u}^\gamma}_{L^{n/(n-1)}(\R^n)} \\
        &\le \norm{D(\abs{u}^\gamma)}_{L^1(\R^n)} \\
        &\le \gamma\int_{\R^n}\abs{u}^{\gamma - 1}\abs{D u}\dd x \\
        &\underbrace{\le}_{\text{H\"older}} \gamma \left(\abs{u}^{(\gamma - 1)p/(p-1)}\dd x\right)^{1 - \frac{1}{p}}\left(\int_{\R^n}\abs{D u}^p\right)^{1/p}
    \end{align*}
    Choose \(\gamma\) to match the exponents of \(u\) in the integrals, i.e.
    \[\gamma = \frac{p(n-1)}{n-p}\]
    In particular,
    \[\frac{\gamma n}{n-1} = p^*\]
    and so we have that
    \begin{align*}
        \left(\int_{\R^n}\abs{u}^{p^*}\right)^{(n-1)/n} \le \frac{p(n-1)}{n-p}\left(\int_{\R^n}\abs{u}^{p^*}\right)^{(p-1)/p}\norm{Du}_{L^p(\R^n)}
    \end{align*}
    Which then implies that
    \[\norm{u}_{L^{p^*}(\R^n)} \le \underbrace{\frac{p(n-1)}{n-p}}_{=: C}\norm{Du}_{L^p(\R^n)} \le C\norm{u}_{W^{1, p}(\R^n)}\]
    We can then conclude by density.
\end{proof}

Note in particular \(C \to \infty\) as \(p \to n\).

\begin{corollary}
    [GNS for \(U \sse \R^n\)]\index{Gagliado-Nirenberg-Sobolev inequality!for \(U \sse \R^n\)} Suppose \(U \sse \R^n\) is open and bounded, with \(C^1\) boundary. Let \(1 \le p < n\). If \(p^* = \frac{np}{n-p}\), then
    \[W^{1, p}(U) \sse L^{p^*}(U)\]
    and the embedding is continuous. That is, there exists \(C = C(U, p, n)\) such that
    \[\norm{u}_{L^{p^*}(U)} \le C\norm{u}_{W^{1, p}(U)}\]
    for all \(u \in W^{1, p}(U)\).
\end{corollary}

\begin{proof}
    Exercise. Use the extension theorem and the GNS inequality for \(\R^n\).
\end{proof}

\begin{corollary}
    [Poincar\'e inequality]\index{Poincar\'e inequality}Let \(U \sse \R^n\) be open and bounded. Suppose \(u \in W_0^{1, p}(U)\), for some \(1 \le p < n\). Then there exists a constant \(C = C(p, q, n, U)\) such that
    \[\norm{u}_{L^q(U)} \le C\norm{Du}_{L^p(U)}\]
    for each \(1 \le q \le p^*\). In particular, as \(1 \le p \le p^*\), we get
    \[\norm{u}_{L^p(U)} \le C\norm{Du}_{L^p(U)}\]
\end{corollary}

\begin{remark}
    \begin{enumerate}[(i)]
        \item On \(W^{1, p}(U)\) with \(U\) bounded, the \(W^{1, p}\) norm is equivalent to \(\norm{Du}_{L^p(U)}\).
        \item We do need that \(u \in W^{1, p}_0\), to kill off constant functions, which have derivative zero.
    \end{enumerate}
\end{remark}

\begin{proof}
    We will use that \(W^{1, p}_0(U)\) is the closure of \(C_c^\infty(U)\) under the \(W^{1, p}(U)\) norm. That is, given \(u \in W^{1, p}_0(U)\), there exists \(u_n \in C_c^\infty(U)\) such that \(\norm{u_n - u}_{W^{1, p}(U)} \to 0\). Since \(u_n\) is smooth, and vanishes near \(\partial U\), we can extend \(\overline{u_n} = 0\) on \(\R^n \setminus U\), with \(\overline{u_n} \in C_c^\infty(\R^n)\).

    Applying \cref{thm:Gagliado-Nirenberg-Sobolev},
    \[\norm{\overline{u_n}}_{L^{p^*}(\R^n)} \le C \norm{D\overline{u_n}}_{L^p(\R^n)}\]
    Sending \(n \to \infty\) and noting that \(\overline u\) vanishes on \(\R^n \setminus U\), we get the result for \(q = p^*\). In general, we use H\"older: since \(\abs{U} < \infty\),
    \[\norm{u}_{L^q(U)} \le C\norm{u}_{L^{p^*}(U)} \le C'\norm{Du}_{L^p(U)}\]  
\end{proof}

\subsubsection*{Case 2: \(p = n\)}

In this case, \(p^* \to \infty\), and so we may expect
\[\norm{u}_{L^\infty(U)} \le C\norm{u}_{W^{1, n}}\]
But this is false for \(n > 1\). One dimensional PDEs are boring, so we won't continue with this case.

\subsubsection*{Case 3: \(n < p < \infty\)}

We might expect in this case that it is ``better than \(L^\infty\)'', i.e. continuous.

\begin{theorem}
    [Morrey's inequality] \index{Morrey's inequality}Let \(n < p < \infty\), then there exists \(C = C(p, n)\) such that for \(u \in C_c^\infty(\R^n)\),
    \[\norm{u}_{C^{0, \gamma}(\R^n)} \le C \norm{u}_{W^{1, p}(\R^n)} \qqtext{for} \gamma = 1 - \frac{n}{p} < 1\]
    That is, we have an embedding
    \[W^{1, p}(\R^n) \inj C^{0, \gamma}(\R^n)\]
\end{theorem}

\begin{proof}
    Let \(Q\) be an open cube of side length \(r\), containing \(0\), and set
    \[\overline u = \frac{1}{\abs{Q}}\int_Q u(x)\dd x\]
    for the average of \(u\) over \(Q\). Then
    \[\abs{\overline u - u(0)} \le \frac{1}{\abs{Q}}\int_Q\abs{u(x) - u(0)}\dd x\]
    Since \(u \in C_c^\infty(\R^n)\), by the fundamental theorem of calculus and the chain rule,
    \begin{align*}
        u(x) - u(0) &= \int_0^1 \dv{t}(u(tx))\dd t \\
        &= \sum_{i=1}^n \int_0^1 x^i \pdv{u}{x^i}(tx)\dd t
    \end{align*}
    Thus,
    \[\abs{u(x) - u(0)} \le r\sum_{i=1}^n \int_0^1 \abs{\partial_{x_i}u(tx)}\dd t\]
    This then gives us that
    \begin{align*}
        \abs{\overline u - u(0)} &\le \frac{r}{\abs{Q}}\int_Q\int_0^1 \sum_{i=1}^n \abs{\partial_{x_i}u(tx)}\dd t \dd x \\
        &= \frac{r}{\abs{Q}}\int_0^1 t^{-n}\left(\sum_{i=1}^n \int_{tQ} \partial_{x_n}u(y)\dd y\right) \dd t \\
        &\le \frac{r}{\abs{Q}}\int_0^1 t^{-n}\left(\sum_i \norm{\partial_{x_i}u}_{L^p(tQ)}\abs{tQ}^{1/q}\right)\dd t \\
        &\le \frac{r}{r^n}\int_0^1 t^{-n}\norm{Du}_{L^p(\R^n)}t^{n/q}r^{n/q}\dd t \\
        &= \frac{r^{1-n/p}}{1 - n/p}\norm{Du}_{L^p(\R^n)}
    \end{align*}
    That is,
    \[\abs{\overline u - u(0)} \le \frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    By translation invariance,
    \[\abs{\overline u - u(x)} \le \frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    for all \(x \in Q\). Thus, by the triangle inequality,
    \[\abs{u(x) - u(y)} \le \abs{u(x) - \overline u} + \abs{\overline u - u(y)} \le 2\frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    for all \(x, y \in Q\). But for \(x, y \in Q\), there exists a cube \(Q\) of side length \(r = 2\abs{x - y}\) such that \(x, y \in Q\), which means that
    \[\frac{\abs{u(x) - u(y)}}{\abs{x - y}^\gamma} \le C \norm{Du}_{L^p(\R^n)}\]
    Note the left hand side is independent of \(r\), and so the inequality is true for all \(x, y \in \R^n\). Thus,
    \[[u]_{C^{0, \gamma}(\R^n)} \le C\norm{Du}_{L^p(\R^n)}\]
    Finally, we would like to control \(\sup_{x \in \R^n}\abs{u(x)}\), note that any \(x \in \R^n\) belongs to a cube with side length \(1\). In particular,
    \begin{align*}
        \abs{u(x)} &\le \abs{\overline u} + \abs{u(x) - \overline u} \\
        &\le \int_Q\abs{u(x)}\dd x + C \norm{Du}_{L^p} \\
        &\le \norm{u}_{L^p(\R^n)}\norm{1}_{L^q(Q)} + C\norm{Du}_{L^p(\R^n)} \\
        &\le C\norm{u}_{W^{1, p}(\R^n)}
    \end{align*}
    Note the constant is independent of the choice of \(x\). That is,
    \[\norm{u}_{C^{0, \gamma}(\R^n)} \le C\norm{u}_{W^{1, p}(\R^n)}\]
\end{proof}

\lec

\begin{corollary}
    For \(n < p < \infty\), and \(u \in W^{1, p}(\R^n)\), then there exists a unique \(u^*\) with \(u^* = u\) a.e., and \(u^*\) is continuous with
    \[\norm{u^*}_{C^{0, \gamma}(\R^n)} \le C\norm{u}_{W^{1, p}(\R^n)}\]
\end{corollary}

\begin{corollary}
    Suppose \(n < p < \infty\), \(u \in W^{1, p}(U)\) for \(U \sse \R^n\) open bounded, with \(\partial U\) being \(C^1\). Then there exists a unique \(u^* \in C^{0, \gamma}(\R^n)\), \(\gamma = 1 - \frac{n}{p}\), \(u = u^*\) a.e. on \(U\), and
    \[\norm{u^*}_{C^{0, \gamma}(U)} \le C\norm{u}_{W^{1, p}(U)}\]
    where \(C\) depends only on \(U, p, n\).
\end{corollary}

\begin{proof}
    By the extension theorem, there exists \(\overline u \in W^{1, p}(\R^n)\), with \(\Supp(u)\) compact, \(\overline{u} = u\) a.e. on \(U\). Thus, there exists a sequence \((u_j) \in C_c^\infty(\R^n)\) with \(u_j \to \overline u\) in \(W^{1,p}(\R^n)\). Note by Morrey's inequality,
    \[\norm{u_m - u_j}_{C^{0, \gamma}(\R^n)} \le C\norm{u_m - u_j}_{W^{1, p}(\R^n)}\]
    and so \((u_j)\) is Cauchy in the Banach space \(C^{0, \gamma}(\R^n)\), and so there exists a limit \(\overline u^* \in C^{0, \gamma}(\R^n)\). Then \(u^* = \overline u^*\vert_U\) satisfies the requirements.
\end{proof}

In summary, if \(U \sse \R^n\) is open, bounded and has \(C^1\) boundary, then:

\begin{itemize}
    \item if \(1 \le p < n\), then we have a continuous embedding
    \[W^{1, p}(U) \inj L^{p^*}(U)\]
    where
    \[\frac{1}{p_*} = \frac{1}{p} - \frac{1}{n}\]
    and so \(p^* > p\).
    \item If \(n < p < \infty\), then
    \[W^{1, p}(U) \inj C^{0, \gamma}(U)\]
    where
    \[\gamma = 1 - \frac{n}{p}\]
\end{itemize}

\begin{example}
    Let \(n = 3\), \(u \in W^{2, 2}\). Then \(u, Du \in W^{1,2}\). \(p = 2 < 3\), and we have \(p^* = 6\), hence \(u, Du \in L^6\). Thus, \(u \in W^{1, 6}\), and \(6 > 3 = n\) so \(u \in C^{0, 1/2}\).
\end{example}

\section{Second order elliptic boundary value problems}

In this section, let \(U\) be an open bounded subset of \(\R^n\), with \(C^1\) boundary.

For \(u \in C^2(\overline U)\), define
\begin{equation}
    \label{eq:elliptic-operator-div-form}
    Lu = -\sum_{i, j=1}^n(a^{ij}(x)u_{x_i})_{x_j} + \sum_{i=1}^n b^i(x)u_{x_i} + c(x)u
\end{equation}
where \(a^{ij}, b^i, c\) are functions on \(U\). We assume \(a^{ij}, b^i, c\) are \(L^\infty\), and \(a^{ij} = a^{ji}\). This form is called \emph{divergence form}\index{divergence form}, since it looks like
\[\grad\cdot(A \grad u)\]
If \(a^{ij} \in C^1(\overline U)\), then we can rewrite \(L\) in \emph{non-divergence form}\index{non-divergence form}
\[Lu = -\sum_{i, j=1}^n a^{ij}(x)u_{x_ix_j} + \sum_{i=1}^n \tilde b^i(x)u_{x_i} + cu\]

We will study the divergence form, since it is adapted to Hilbert space methods. The non-divergence form is better for maximum principles, and Dirichlet energies. The second form is the topic of the Part III Elliptic PDEs course.

\begin{definition}
    We say \(L\) is \emph{elliptic}\index{elliptic} if
    \[\sum_{i, j}a_{ij}\xi_i\xi_j > 0\]
    for all \(x \in U, \xi \in \R^n \setminus 0\). We say that \(L\) is \emph{uniformly elliptic}\index{uniformly elliptic} if there exists a constant \(\theta > 0\), such that
    \[\sum_{i, j}a^{ij}(x)\xi_i\xi_j \ge \theta\norm{\xi}^2\]
    for all \(x \in U, \xi \in \R^n\). 
\end{definition}

Note some references call uniformly elliptic: strongly or strictly elliptic.

\subsection{Weak formulation and Lax-Milgram}

We will consider the boundary value problem
\begin{equation}
    \label{eq:elliptic-bvp}
    \begin{cases}
        Lu = f &\text{on }U \\
        u\vert_{\partial U} = 0
    \end{cases}
\end{equation}
with \(f \in L^2(U), a^{ij}, b^i, c \in L^\infty(U)\).

Suppose \(u \in C^2(\overline U)\) solves \cref{eq:elliptic-bvp} pointwise a.e.. Take any \(v \in C^2(\overline U)\) with \(v\vert_{\partial U} = 0\), we get (using summation convention)
\begin{align*}
    \int_U f v \dd x &= \int_U -v(a^{ij}u_{x_i})_{x_j} + vb^j u_{x_j} + cuv \dd x \\
    &= \underbrace{-\int_{\partial U}va^{ij}u_{x_i}n_j \dd S}_{=0} + \int_U a^{ij}u_{x_i}v_{x_j} + bu_{x_i}v + cuv \dd x
\end{align*}
and so,
\begin{equation}
    \label{eq:weak-elliptic}
    \int_U vf \dd x = B[u, v]
\end{equation}
for all \(v \in C^2(\overline U)\) with \(v\vert_{\partial U} = 0\), where
\[B[u, v] = \int_U a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + cuv \dd x\]
With this, if \(u \in C^2(\overline U)\) solves \cref{eq:elliptic-bvp}, then \cref{eq:weak-elliptic} holds. Conversely, if \(u \in C^2(\overline U)\) with \(u\vert_{\partial U} = 0\) and satisfying \cref{eq:weak-elliptic}, then by integration by parts, we get that
\[\int_U (f - Lu)v \dd x = 0\]
for all \(v \in C_c^\infty(U)\). Thus, \(Lu = f\) pointwise a.e. on \(U\).

In conclusion, if \(u \in C^2(\overline U)\), with \(u\vert_{\partial U} = 0\), then \(u\) satisfies \cref{eq:elliptic-bvp} if and only if it satisfies \cref{eq:weak-elliptic}.

But we note that \cref{eq:weak-elliptic} makes sesne for \(v \in H_0^1(U)\) and \(u \in H^1\). To encode the boundary conditions, we can assume \(u \in H_0^1(U)\).

\begin{definition}
    [weak solution] We say that \(u \in H^1_0(U)\) is a \emph{weak solution}\index{weak solution!to elliptic boundary value problem} of \cref{eq:elliptic-bvp} for given \(f \in L^2(U)\) if
    \[B[u, v] = \inner{f, v}_{L^2(U)}\]
    for all \(v \in H_0^1(U)\).
\end{definition}

\begin{theorem}
    [Lax-Milgram] \label{thm:lax-milgram} Let \(H\) be a real Hilbert space, with inner product \(\inner{\cdot, \cdot}\). Suppose \(B : H \times H \to \R\) is bilinear, such that there exists constants \(\alpha, \beta > 0\) such that
    \begin{enumerate}[(i)]
        \item (Boundedness) \(\abs{B[u, v]} \le \alpha \norm{u}\norm{v}\) for all \(u, v \in H\), 
        \item (Coercivity) \(\beta\norm{u}^2 \le B[u, u]\) for all \(u \in H\).
    \end{enumerate}
    Then if \(f \in H^*\), there exists a unique \(u \in H\) such that
    \[B(u, v) = \inner{f, v}\]
    for all \(v \in H\).
\end{theorem}

We will defer the proof to the next lecture.

\begin{example}
    Recall that \(H^k = W^{k, 2}\) is a Hilbert space. Consider the boundary value problem
    \[\begin{cases}
        Lu = -\Delta u + cu = f & \text{on }U \\
        u = 0 & \text{on }\partial U
    \end{cases}\]
    where \(c \ge 0\), \(f \in L^2(U)\). In this case,
    \[B[u, v] = \int_U\grad u \cdot \grad v + cuv \dd x\]
    For boundedness, by H\"older (or Cauchy-Schwarz),
    \[\abs{B[u, v]} \le (1 + c)\norm{u}_{H^1}\norm{v}_{H^1}\]
    For coercivity,
    \[B[u, u] = \norm{\grad u}_{L^2(U)}^2 + c\norm{u}_{L^2(U)}^2 \ge \norm{\grad u}_{L^2(U)}^2 \ge \tilde C\norm{u}_{H^1(U)}^2\]
    where for the last inequality, we used the Poincare inequality. Thus, we can apply Lax-Milgram with \(H = H_0^1\).
\end{example}

\begin{corollary}
    [of Lax-Milgram, stability] With the assumptions of Lax-Milgram, let \(u_i\) be the unique solution to
    \[B[u_i, v] = \inner{f_i, v}\]
    for all \(v \in H\). Then
    \[\norm{u_1 - u_2}_H \le \frac{1}{\beta}\norm{f_1 - f_2}_{H^*}\]
\end{corollary}

\begin{proof}
    Since \(B[u_i, v] = \inner{f_i, v}\), by bilinearity,
    \[B[u_1 - u_2, v] = \inner{f_1 - f_2, v}\]
    Choosing \(v = u_1 - u_2\), then
    \[\beta\norm{v}^2 \le B[u_1 - u_2, v] = \inner{f_1 - f_2, v} \le \norm{f_1 - f_2}\norm{v}\]
    where we use Cauchy-Schwarz for the last inequality.
\end{proof}

\begin{proof}
    [Proof of \cref{thm:lax-milgram}]

    \textbf{Step 1:} For each fixed \(u \in H\), define \(\varphi_u(v) = B[u, v]\). This is a bounded linear functional on \(H\), i.e. \(\varphi_u \in H^*\). Applying the Riesz representation theorem, there exists a unique \(w_u \in H\) such that
    \[\varphi_u(v) = \pinner{w_u, v} = B[u, v]\]
    for all \(v \in H\). In particular, we have a map
    \begin{align*}
        A : H &\to H \\
        u &\mapsto w_u
    \end{align*}
    and we have that \(B[u, v] = \pinner{Au, v}\) for all \(v \in H\).

    \textbf{Step 2:} \(A\) is bounded. If \(\lambda_1, \lambda_2 \in \R\), \(u_1, u_2 \in H\), then for each \(v \in H\), we have the following:
    \begin{align*}
        \pinner{A(\lambda_1 u_1 + \lambda_2 u_2), v} &= B[\lambda_1u_1 + \lambda_2 u_2, v] \\
        &=\lambda_1B[u_1, v] + \lambda_2B[u_2, v] \\
        &= \lambda_1\pinner{Au_1, v} + \lambda_2\pinner{Au_2, v} \\
        &= \pinner{\lambda_1 Au_1 + \lambda_2 Au_2, v}
    \end{align*}
    and so \(A\) is linear. Moreover,
    \[\norm{Au}^2 = \pinner{Au, Au} = B[u, Au] \le \alpha\norm{u}\norm{Au}\]
    Hence \(\norm{Au} \le \alpha\norm{u}\). Thus \(A\) is bounded, with \(\norm{A} \le \alpha\).

    \textbf{Step 3:} We will show that \(A\) is injective and \(A(H)\) is closed.
    \[\beta\norm{u}^2 \le B[u, u] = \pinner{Au, u} \le \norm{Au}\norm{u}\]
    With this, \(\beta\norm{u} \le \norm{Au}\). That is, \(A\) is bounded below, hence \(A\) is injective and the image is closed\footnote{If \((Au_j)\) is Cauchy, then so is \((u_j)\)}.

    \textbf{Step 4:} We will show that \(A\) is surjective. Since \(A(H)\) is a closed subspace of \(H\), which is a Hilbert space, and so we can write
    \[H = A(H) \oplus A(H)^\perp\]
    With this, it suffices to show \(A(H)^\perp = 0\). For \(w \in A(H)^\perp\),
    \[\beta\norm{w}^2 \le B[w, w] = \pinner{Aw, w} = 0\]
    \(\norm{w} = 0\), and so \(A(H)^\perp = 0\). With this, \(A\) is bijective with bounded inverse.

    \textbf{Step 5:} We would like to solve the following problem: Given \(f \in H^*\), we would like to find \(u\) such that \(B[u, v] = \inner{f, v}\) for all \(v \in H\). By the Riesz representation theorem, there exists a unique \(w_f \in H\) such that \(\inner{f, v} = \pinner{w_f, v}\) for all \(v \in H\). Now let \(u = A^{-1}w_f\). Then
    \[B[u, v] = \pinner{Au, v} = \pinner{w_f, v} = \inner{f, v}\]
    i.e. \(B[u, \cdot] = f\).

    \textbf{Step 6:} For uniqueness, if \(u_1, u_2\) satisfy \(B[u_j, \cdot] = f\), then
    \[B[u_1 = u_2, v] = 0\]
    for all \(v \in H\). Setting \(v = u_1 - u_2\), and using coercivity we are done.
\end{proof}

\begin{theorem}
    [energy estimates for \(B\)] 
    \index{Energy estimates}
    Suppose
    \[Lu = -(a^{ij}u_{x_i})_{x_j} + b^iu_{x_i} + cu\]
    where \(a^{ij} = a^{ji}, b^i, c \in L^\infty(U)\), and suppose \(L\) is uniformly elliptic. If
    \[B[u, v] = \int a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + cuv \dd x\]
    is the associated bilinear form. Then there exists \(\alpha, \beta > 0, \gamma \ge 0\), such that for all \(u, v \in H_0^1(U)\).
    \begin{enumerate}[(i)]
        \item \(\abs{B[u, v]} \le \alpha\norm{u}_{H_0^1(U)}\norm{v}_{H_0^1(U)}\),
        \item (Garding's inequality\index{Garding's inequality}) \(\beta\norm{u}^2_{H_0^1(U)} \le B[u, u] + \gamma\norm{u}_{L^2(U)}^2\)
    \end{enumerate}
\end{theorem}

\begin{remark}
    In PDE theory, ``energy'' refers to \(L^2\).
\end{remark}

\begin{proof}
    For (i),
    \begin{align*}
        \abs{B[u, v]} &\le \sum_{i, j}\norm{a^{ij}}_{L^\infty(U)}\int_U\abs{Du}\abs{Dv}\dd x + \sum_i \norm{b^i}_{L^\infty(U)}\int_U \abs{Du}\abs{v}\dd x + \norm{c}_{L^\infty(U)}\int_U \abs{u}\abs{v}\dd x \\
        &\le \alpha\norm{u}_{H^1_0}\norm{v}_{H^1_0}
    \end{align*}
    where in the last step we used Cauchy-Schwarz, and collecting terms.

    For (ii), we will use uniform ellipticity.
    \begin{align*}
        \theta \int_U \abs{Du}^2 \dd x &\le \int_U a^{ij}u_{x_i}u_{x_j} \dd x \\
        &= B[u, u] - \int_U(b^i u_{x_i}u + cu^2)\dd x\\
        &= \le B[u, u] + \sum_i \norm{b^i}_{L^\infty}\int_U \abs{Du}\abs{u}\dd x + \norm{c}_{L^\infty(U)}\int_U u^2 \dd x
    \end{align*}
    By Young's inequality with
    \[\abs{ab} = \sqrt{2\epsilon}\abs{a}\frac{\abs{b}}{\sqrt{2\epsilon}}\]
    we get that
    \[\int_U \abs{Du}\abs{u}\dd x \le \epsilon\int_U\abs{Du}^2 \dd x + \frac{1}{2\epsilon}\int_U \abs{u}^2 \dd x\]
    Choose \(\epsilon\) such that
    \[\epsilon\sum_i\norm{\beta_i}_{L^\infty(U)} \le \frac{\theta}{2}\]
    This then gives us that
    \[\frac{\theta}{2} \int_U\abs{D u}^2 \dd x \le B[u, u] + c\int_U\abs{u}^2\dd x\]
    Adding to this the Poincar\'e inequality, we get that
    \[\beta\norm{u}^2_{H^1_0(U)} \le B[u, u] + \gamma\norm{u}^2_{L^2(U)}\]
\end{proof}

\begin{remark}
    If \(B\) is a bilinear form for the operator \(L\) with \(b^i = c = 0\), then \(\gamma = 0\). In this case, we get
    \[\theta\int_U \abs{D u}^2 \dd x \le B[u, u]\]
    and if we add the Poincar\'e inequality, we get
    \[\norm{u}^2_{H^1_0(U)} \le cB[u, u]\]
    which is Garding's inequality with \(\gamma = 0\). In this case, we can apply Lax-Milgram directly.
\end{remark}

On the other hand, if \(\gamma > 0\), we can't apply Lax-Milgram.

\begin{theorem}
    \label{thm:gamma-exists-unique}
    Let \(L\) be as above, then there exists a \(\gamma \ge 0\) such that for any \(\mu \ge \gamma\), and any \(f \in L^2(U)\), there exists a unique solution \(u \in H^1_0(U)\) to the boundary value problem
    \begin{equation}
        \label{eq:elliptic-bvp-modified}
        \begin{cases}
            L_\mu u = Lu + \mu u = f & \text{in }U \\
            u = 0 & \text{on }\partial U
        \end{cases}
    \end{equation}
    Moreover, there exists \(C > 0\) such that
    \[\norm{u}_{H^1(U)} \le C\norm{f}_{L^2(U)}\]
\end{theorem}

\begin{proof}
    Let \(\gamma\) be from Garding's inequality, i.e.
    \[\beta\norm{u}^2_{H_0^1} \le B[u, u] + \gamma\norm{u}^2_{L^2(U)}\]
    Let \(\mu \ge \gamma\), and set
    \[B_\mu[u, v] = B[u, v] + \mu\pinner{u, v}_{L^2}\]
    which is the bilinear form for the operator \(L_\mu\) in \cref{eq:elliptic-bvp-modified}. In this case, \(B_\mu\) satisfies the conditions of Lax-Milgram.

\lec

    Given \(f \in L^2(U)\), and set \(\inner{f, v} = (f, v)_{L^2(U)}\). This is a bounded linear functional on \(L^2(U)\), i.e. \(f \mapsto (f, \cdot) \in (L^2(U))^*\). In particular, this is a bounded linear functional on \(H_0^1\). We can apply Lax-Milgtam to find a unique \(u \in H_0^1(U)\) with
    \[B_\mu[u, v] = \inner{f, v} = \pinner{f, v}_{L^2(U)}\]
    for all \(v \in H_0^1\). Finally,
    \[\beta\norm{u}_{H_0^1(U)}^2 \le B_\mu[u, u] = \pinner{f, u}_{L^2(U)} \le \norm{f}_{L^2(U)}\norm{u}_{L^2(U)} \le \norm{f}_{L^2(U)} \norm{u}_{H_0^1(U)}\]
\end{proof}

So far, the solutions only live in \(H_0^1\), and we need to pay a price for the \(\mu\).

\subsection{Compactness results in PDEs}

Recall the following results:

\begin{itemize}
    \item Bolzano-Weierstra\ss - closed unit ball in \(\R^n\) is sequentially compact. 
    \item Recall for a metric space, the following are equivalent:
    \begin{enumerate}[(i)]
        \item compactness,
        \item sequential compactness,
        \item completeness and totally boundedness
    \end{enumerate}
    \item If \(H\) is an infinite dimensional Hilbert space, then \(\{x \in H \mid \norm{x} \le 1\}\) is not compact.
\end{itemize}

We will consider a weaker topology to recover compactness, since the topology induced by the norm is too strong.

\begin{definition}
    [weak convergence]
    Suppose \(H\) is a Hilbert space, \((u_j) \sse H\) a sequence, then we say that
    \emph{\(u_j\) converges weakly to \(u \in H\)}\index{weak convergence} if for all \(w \in H\),
    \[\pinner{u_j, w} \to (u, w)\]
    and we write \(u_j \weak u\).
\end{definition}

\begin{remark}
    If the weak limit exists, then it is unique.
\end{remark}

\begin{proposition}
    [Banach-Alaoglu for a separable Hilbert space] \index{Banach-Alaoglu}Let \(H\) be a separable Hilbert space, and suppose we have a bounded sequence \((u_n) \sse H\). Then \((u_n)\) has a weakly convergent subsequence.

    That is, the closed unit ball in \(H\) is weakly sequentially compact.
\end{proposition}

\begin{proof}
    Diagonal argument, see AoF. Or deduce from the below, since any Hilbert space is reflexive, and so the weak and weak-\(*\) topologies agree.
\end{proof}

\begin{theorem}
    [Banach Alaoglu]
    Let \(X\) be a Banach space, then the closed unit ball in \(X^*\) is compact in the weak-\(*\) topology on \(X^*\).
\end{theorem}

\begin{lemma}
    [Poincar\'e again] 
    \index{Poincar\'e inequality}
    Suppose \(u \in H^1(\R^n)\), and \(Q = (\xi_1, \xi_1 + L) \times \cdots \times (\xi_n, \xi_n + L)\) be a cube with side lengths \(L\). Then
    \begin{enumerate}[(i)]
        \item \[\norm{u}_{L^2(Q)}^2 \le \frac{1}{\abs{Q}}\left(\int_Q u \dd x\right)^2 + \frac{nL^2}{2}\norm{Du}^2_{L^2(Q)}\]
        \item \[\norm{u - \overline u}_{L^2(Q)} \le \frac{nL^2}{2}\norm{Du}^2_{L^2(Q)}\]
        where
        \[\overline u = \frac{1}{\abs{Q}}\int_Q u(x)\dd x\]
    \end{enumerate}
    In particular, if \(\overline u = 0\) we recover the previous Poincar\'e inequality.
\end{lemma}

\begin{proof}
    For (i), since \(\partial Q\) is Lipschitz, we apply the approximation theorem, to get \(C^\infty(\overline Q)\) are dense in \(H^1(Q)\). Consider \(u \in C^\infty(\overline Q)\). For \(x, y \in Q\), we use the fundamental theorem of calculus to get
    \begin{align*}
        u(x) - u(y) = \int_{y_1}^{x_1}\dv{t}u(t, x_2, \cdots, x_n)\dd t + \int_{y_2}^{x_1}\dv{t}u(y_1, t, x_3, \cdots, x_n)\dd t + \cdots + \int_{y_n}^{x_n}\dv{t}u(y_1, \dots, y_{n-1}, t)\dd t
    \end{align*}
    Squaring, to get
    \begin{align*}
        u(x)^2 + u(y)^2 - 2u(x)u(y) \le n\left(\int_{y_1}^{x_1}\dv{t}u(t, x_2, \dots, x_n)\dd t\right)^2 + \cdots + n\left(\int_{y_1}^{x_1}\dv{t}u(y_1, \dots, y_{n-1}, t)\dd t\right)^2
    \end{align*}
    where we use Cauchy-Schwarz to get that
    \[(a_1 + \cdots + a_n)^2 \le n(a_1^2 + \cdots + a_n^2)\]
    Integrating over \(x, y \in Q\),
    \[\int_Q\int_Q(\text{LHS})\dd x \dd y = 2 \abs{Q}\norm{u}^2_{L^2(Q)} = 2\left(\int_Qu(x)\dd x\right)^2\]
    For the right hand side,
    \begin{align*}
        I_1 &= \left(\int_{y_1}^{x_1}\dv{t}u(t, x_2, \cdots, x_n)\dd t\right)^2 \le \left(\int_{y_1}^{x_1}\dd t\right)\left(\int_{y_1}^{x_1}\left(\dv{t}u(t, x_2, \dots, x_n)\right)^2 \dd t\right) \\
        &\le L\int_{\xi_1}^{\xi_1 + L}\left(\dv{t}u(t, x_2, \dots, x_n)\right)^2 \dd t
    \end{align*}
    Hence combining the terms,
    \[2\abs{Q}\norm{u}^2_{L^2(Q)} - 2\left(\int u(x)\dd x\right)^2 \le L^2n\abs{Q}\norm{D u}^2_{L^2(Q)}\]
    Rearranging gives the result.

    For (ii), consider \(\eta \in C_c^\infty\), with \(\eta = 1\) on \(Q\). Then
    \[\int_Q (U - \overline u \eta) \dd x = 0\]
    and we can then use (i).
\end{proof}

Recall if \(1 \le p < n\), we have an embedding \(W^{1, p} \inj L^{p^*}\).

\begin{theorem}
    [Rellich-Kondrachov] \index{Rellich-Kondrachov}Suppose \(U \sse \R^n\) be open with \(C^1\) boundary. Let \((u_n)\) be a bounded sequence in \(H^1(U)\). Then there exists \(u \in H^1(U)\), and a subsequence \((u_{n_j})\) such that \(u_{n_j} \weak u\) in \(H^1(U)\), and \(u_{n_j} \to u\) in \(L^2(U)\).
\end{theorem}

\begin{proof}
    By the extension theorem, we have an extension \(\overline{u_n} \in H^1(\R^n)\), with \(\Supp(\overline{u_n}) \sse Q\) for some cube \(Q\). Moreover, the extension operator \(E : H^1(U) \to H_0^1(Q)\) is bounded. In particular,
    \[\norm{\overline{u_n}}_{H^1(Q)} \le C\norm{u_n}_{H^1(U)} \le CK\]
    for some \(K\). Now \(H_0^1(Q)\) is a separable Hilbert space, so by Banach-Alaoglu there exists \(u \in H_0^1(Q)\), with \(\overline{u_{n_j}} \weak u\) in \(H_0^1(Q)\), and
    \[\norm{u}_{H^1(Q)} \le c\]
    We claim that \(w_j = \overline{u_{n_j}} \to u\) in \(L^2(Q)\).

    To see this, fix \(\delta > 0\) and divide \(Q\) into \(k\) subcubes \(\{Q_a\}_{a=1}^k\), of side lengths \(0 < \ell < \delta\), intersecting only on their faces. Then
    \[\norm{w_j - u}^2_{L^2(Q)} \le \sum_{a=1}^k\norm{w_j - u}^2_{L^2(Q)} \le \sum_{a=1}^k \left(\frac{1}{\abs{Q_a}}\left(\int_{Q_a}(w_j - u)\dd x\right)^2\right) + \frac{n^2\delta^2}{2}\norm{D w_j - Du}^2_{L^2(Q)}\]
    Fix \(\epsilon > 0\), since \(w_j, u \in H_0^1(Q)\), then \(\norm{Dw_j - Du}^2_{L^2(Q)} \le C\) for some \(C\). Fix \(\delta > 0\) such that
    \[\frac{n^2\delta^2}{2}\norm{Dw_j - Du}_{L^2(Q)}^2 < \frac{\epsilon}{2}\]
    This then fixes \(k\). Note that the map
    \[f \mapsto \int_Q f(x)\dd x\]
    is a bounded linear functional on \(H_0^1(Q)\), and so by weak convergence,
    \[\int_{Q_a}(w_j - u)\dd x \to 0\]
    This is true for all \(a\). Since \(k\) is fixed and finite, choose \(j\) large enough so that 
    \[\sum_{a=1}^k \left(\frac{1}{\abs{Q_a}}\left(\int_{Q_a}(w_j - u)\dd x\right)^2\right) < \frac{\epsilon}{2}\]
    Using this, \(\norm{w_j - u}_{L^2(Q)}^2 < \epsilon\).
\end{proof}

\subsection{Fredholm alternative and spectra of elliptic PDEs}

\begin{definition}
    [adjoint, compact]
    Let \(H\) be a Hilbert space, and consider \(K \in \mcB(H)\).
    \begin{enumerate}
        \item the \emph{adjoint of \(K\)}\index{adjoint}, denoted \(K^\dagger\) is the unique operator, such that
        \[\pinner{x, K^\dagger y} = \pinner{Kx, y}\]
        for all \(x, y \in H\).
        
        We say that \(K\) is \emph{self-adjoint}\index{self-adjoint} if \(K^\dagger = K\).
        \item \(K\) is \emph{compact}\index{compact operator} if for each bounded sequence \((u_j) \sse H\), there exists a subsequence \((u_{j_k})\) such that \(K(u_{j_k})\) converges strongly in \(H\). 
    \end{enumerate}
\end{definition}

\begin{example}
    [Key example] Let \(K : L^2(U) \to H^1(U)\) be a bounded linear operator. SInce \(H^1 \inj L^2\), we can think of \(K \in \mcB(L^2(U))\).

    \begin{claim}
        \(K \in \mcB(L^2(U))\) is compact.
    \end{claim}

    \begin{proof}
        Let \((u_j) \sse L^2(U)\) is a bounded sequence, then
        \[\norm{K(u_j)}_{H^1} \le \norm{K}\norm{u_j}_{L^2(U)}\]
        and so \((K(u_j))\) is a bounded sequence in \(H^1\). By Rellich-Kondrachov, there exists a subsequence \((u_{j_k})\), such that \(K(u_{j_k})\) converges strongly in \(L^2(U)\). 
    \end{proof}
\end{example}

The idea is that if we are looking at the equation
\[\Delta u = f\]
we can view this as a map
\begin{align*}
    \HH^1(U) &\to L^2(U) \\
    u &\mapsto f
\end{align*}
Finding a solution is the inverse map \(K : L^2(U) \to H^1(U)\), with \(K(f) = u\). This map will be compact.

\begin{theorem}
    [Fredholm alternative for compact operators] 
    \index{Fredholm alternative!for compact operators}
    Let \(H\) be a Hilbert space, \(K \in \mcB(H)\) compact. Then
    \begin{enumerate}[(i)]
        \item \(\ker(I - K)\) is finite dimensional,
        \item \(\im(I - K)\) is closed,
        \item \(\im(I - K) = \ker(I - K^\dagger)^\perp\),
        \item \(\ker(I - K) = 0\) if and only if \(\im(I - K) = H\),
        \item \(\dim(\ker(I - K)) = \dim(\ker(I - K^\dagger))\)
    \end{enumerate}
\end{theorem}

\begin{proof}
    Appendix D.5 of Evans.
\end{proof}

Note (iii) and (iv) are referred to as the \emph{Fredholm alternative}. Applied to linear algebra, we would like to consider the equation
\[Ax = b\]
We have the alternative:
\begin{enumerate}[(a)]
    \item \(A\) is invertible, \(A^{-1}\) exists, and so the inhomogeneous problem has a unique solution,
    \item \(\ker(A)\) is non-trivial. The homogeneous equation \(Ax = 0\) admits non-trivial solutions. Moreover, from (iii), \(\im(A) = \ker(A^\T)^\perp\), and so the inhomogeneous equation has a solution if and only if \(b \in \ker(A^\T)^\perp\), and so
    \[y^\T b = 0\]
    for all \(y \in \ker(A^\T)\), i.e. \(A^\T y = 0\).
\end{enumerate}

Restating (iii) and (iv), we have
\begin{enumerate}[(I)]
    \item for each \(f \in H\), \((I - K)u = f\) has a unique solution,
    \item or the homogeneous equation \((I - K)u = 0\) has a non-trivial solution. In this case, the space of solutions \((I - K)u\) is finite dimensional, and \((I - K)u = f\) has a solution if and only if \(f \in \ker(I - K^\dagger)^\perp\).
\end{enumerate}

\begin{definition}
    [resolvent and spectrum]
    Let \(H\) be a real Hilbert space, \(A \in \mcB(H)\). The \emph{resolvent (set)}\index{resolvent} of \(A\) is
    \[\rho(A) = \left\{\lambda \in \R \mid A - \lambda I \text{ is invertible}\right\}\]
    The \emph{real spectrum}\index{spectrum} of \(A\) is
    \[\sigma(A) = \R \setminus \rho(A)\]
    We also define the \emph{point spectrum}\index{point spectrum}
    \[\sigma_p(A) = \left\{\eta \in \sigma(A) \mid \ker(A - \eta I) \ne 0\right\}\]
    If \(Aw = \eta w\), we call \(w\) an \emph{eigenvector}\index{eigenvector}.
\end{definition}

\begin{remark}
    \(\rho(A)\) is open, and \(\sigma(A)\) is closed.
\end{remark}

\begin{theorem}
    [spectrum of compact operator] 
    \label{thm:compact-spectrum}
    Suppose \(H\) is a separable infinite dimensional Hilbert space, with \(K \in \mcB(H)\) compact. Then
    \begin{enumerate}[(i)]
        \item \(0 \in \sigma(K)\),
        \item \(\sigma(K) \setminus \{0\} = \sigma_p(K) \setminus \{0\}\),
        \item \(\sigma(K) \setminus \{0\}\) is countable. Say \(\sigma(K) \setminus \{0\} = \{\lambda_i\}_{i \in \N}\), then (up to reordering) \(\lambda_i \to 0\),
        \item if \(K\) is in addition self-adjoint, then there exists a countable orthonormal basis for \(H\) consisting of eigenvectors for \(K\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    II Linear Analysis.
\end{proof}

\subsubsection{Application to elliptic PDEs}

Consider \cref{eq:elliptic-operator-div-form} as before, with \(L\) uniformly elliptic on \(U \sse \R^n\). The bilinear form associated to \(L\) is
\[B[u, v] = \int_U a^{ij}u_{x_i}v_j + b^iu_{x_i}v + cuv \dd x\]

\begin{definition}
    [formal adjoint, adjoint bilinear form]
    We define the formal adjoint to \(L\) as
    \[L^\dagger v = -\sum_{i,j}(a^{ij}v_{x_i})_{x_j} - \sum_i b^i u_{x_i} + \left(c - \sum_i b^i_{x_i}\right)v\]
    and the adjoint bilinear form is given by
    \[B^\dagger[v, u] = B[u, v]\]
\end{definition}

We say \(v \in H_0^1(U)\) is a \emph{weak solution} of the \emph{adjoint problem}
\[\begin{cases}
    L^\dagger v = f & \text{ in }U \\
    v = 0 & \text{ on }\partial U
\end{cases}\]
if
\[B^\dagger[w, v] = \pinner{f, w}_{L^2}\]
for all \(w \in H_0^1(U)\).

Note if \(b^i \in C^1(U)\), then \(B^\dagger\) is the same bilinear form as \(B\).

\begin{theorem}
    [Fredholm alternative for elliptic boundary value problem]
    \index{Fredholm alternative!for elliptive boundary value problems}
    \label{thm:fredholm-elliptic-bvp}
    Consider for bounded \(U\) with \(C^1\) boundary,
    \begin{equation}
        \label{eq:fredholm-alt-elliptic-bvp}
        \begin{cases}
            Lu = f & \text{in }U \\
            u = 0 & \text{on }\partial U
        \end{cases}
    \end{equation}
    Then
    \begin{enumerate}[(I)]
        \item for each \(f \in L^2(U)\), \cref{eq:fredholm-alt-elliptic-bvp} admits a unique weak solution, or
        \item there exists a non-trivial weak solution to the homogeneous problem (i.e. \(f = 0\)), and \(\dim(N) = \dim(N^\dagger)\), where
        \[N = \left\{\text{weak solutions to homogeneous equation}\right\} \sse H_0^1(U)\]
        and
        \[N^\dagger = \left\{\text{weak solutions to the homogenous adjoint equation}\right\}\]
        With this, \cref{eq:fredholm-alt-elliptic-bvp} has a unique solution if and only if
        \[\pinner{f, v}_{L^2} = 0\]
        for all \(v \in N^\dagger\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    By \cref{thm:gamma-exists-unique}, there exists \(\gamma > 0\) such that for every \(f \in L^2(U)\), there exists a unique weal solution \(u \in H_0^1(U)\) to
    \[\begin{cases}
        L_\gamma u = f &\text{in }U \\
        u = 0 &\text{on }\partial U
    \end{cases}\]
    where \(L_\gamma u =L u + \gamma u\). We also have an associated bilinear form
    \[B_\gamma[u, v] = B[u, v] + \gamma\pinner{u, v}_{L^2}\]
    and we have that
    \[B_\gamma[u, v] = \pinner{f,v}_{L^2}\]
    for all \(v \in H^1_0\), and \(\norm{u}_{H^1} \le C\norm{f}_{L^2}\).

    Write \(L_\gamma^{-1}(f) = u\) for the solution operator. This is well defined as the solution exists and is unique. We can check that this is linear. The inequality above shows that
    \[\norm{L_\gamma^{-1}(f)}_{\HH^1} \le C\norm{f}_{L^2}\]
    and so
    \[L_\gamma^{-1} : L^2 \to H_0^1\]
    is bounded, hence \(L_\gamma^{-1} : L^2 \to L^2\) is compact.

    Observe that for \(g \in L^2\), then \(L_\gamma^{-1}(g) = w\) if and only if \(B_\gamma[w, v] = \pinner{g, v}\) for all \(v \in H_0^1(U)\). Now suppose \(u \in H_0^1\) is a weak solution to \cref{eq:fredholm-alt-elliptic-bvp}, that is,
    \[B[u, v] = \pinner{f, v}\]
    for all \(v \in H_0^1\), and so
    \[B_\gamma[u, v] = \inner{f + \gamma u, v}\]
    for all \(v \in H_0^1\). Thus, \(u\) solves \cref{eq:fredholm-alt-elliptic-bvp} weakly if and only if
    \[u = L_\gamma^{-1}(f + \gamma u) = L_\gamma^{-1}(f) + \gamma L_\gamma^{-1}(u)\]
    which is true if and only if \((I - K)u = h\), where
    \[K = \gamma L_\gamma^{-1} \qqtext{and} h = L_\gamma^{-1}(f)\]
\lec

Observe the map \(K : L^2 \to L^2\) is also compact, and so we can apply the Fredholm theorem for compact operators, and either:
\begin{enumerate}[(I)]
    \item for all \(h \in L^2\), \(u - Ku = h\) has a unique solution \(u \in L^2\),
    \item there exists \(0 \ne u \in L^2\) with \(u - Ku = 0\).
\end{enumerate}
Suppose (I) holds. Set \(h = L_\gamma^{-1}(f)\), then there exists a unique \(u \in L^2\) with
\[u = \gamma L_\gamma^{-1}(u) + L_\gamma^{-1}(u)\]
Since \(L_\gamma^{-1} : L_2 \to H_0^1\), \(u \in H_0^1\) and by the above, \(u\) is a weak solution of \cref{eq:fredholm-alt-elliptic-bvp}.

Now suppose (II) holds. Then there exists \(u \in L^2\) non-zero, with \(u = Ku = \gamma L_\gamma^{-1}(u)\). As above, \(u \in H^1_0\). Use the definition of \(L_\gamma^{-1}\) to see that
\[B[u, v] + \gamma\pinner{u, v}_{L^2} = \pinner{\gamma u, v}_{L^2}\]
for all \(v \in L^2\). Hence \(B[u, v] = 0\) for all \(v \in H^1_0\). That is, \(u \in N\). Moreover, \(\dim(N) = \dim(\ker(I - K)) = \dim(\ker(I - K^\dagger))\).

\begin{claim}
    Let \(v \in L^2\), then \((I - K^\dagger)v = 0\) if and only if \(B^\dagger[v, w] = 0\) for all \(w \in H^1_0\).
\end{claim}
\begin{proof}
    \begin{align*}
        (I - K^\dagger)v = 0 &\iff \pinner{v, w}_{L^2} = \pinner{v, Kw}_{L^2} \text{ for all }w \in L^2 \\
        &\iff \pinner{v, w}_{L^2} = \pinner{v, \gamma L_\gamma^{-1}(w)}_{L^2} \text{ for all }w \in L^2
    \end{align*}
    But note that any weak solution to
    \[\begin{cases}
        L_\gamma \overline w = \overline f &\text{on }U \\
        \overline w = 0 &\text{on }\partial U
    \end{cases}\]
    obeys
    \[B[\overline w, \varphi] + \gamma\pinner{\overline w, \varphi}_{L^2} = \pinner{\overline f, \varphi}_{L^2}\]
    So if we take \(\overline f = w\),  then we have \(\overline{w} = L_\gamma^{-1}(w)\). Hence
    \[B[L_\gamma^{-1}(w), v] + \gamma\pinner{L_\gamma^{-1}(w), v} = \pinner{w, v}_{L^2}\]
    Hence the above is true if and only if
    \begin{align*}
        &\iff 
        B[L_\gamma^{-1}(w), v] + \gamma\pinner{L_\gamma^{-1}(w), v}_{L^2} = \pinner{v, \gamma L_\gamma^{-1}(w)}_{L^2} \text{ for all }w \in L^2 \\
        &\iff B[L_\gamma^{-1}(w), v] = 0 \text{ for all }w \in L^2 \\
        &\iff B^\dagger[v, L_\gamma^{-1}(w)] = 0 \text{ for all }w\in L^2
    \end{align*}
    On examples sheet 3, \(\im(L_\gamma^{-1})\) is dense, and so we have that
    \[v = K^\dagger v \iff B^\dagger[v, w] = 0\]
\end{proof}
It remains to prove that \cref{eq:fredholm-alt-elliptic-bvp} has a weak solution if and only if \(\pinner{f, v}_{L^2} = 0\) for all \(v \in N^\dagger\). Now note
\[\text{\cref{eq:fredholm-alt-elliptic-bvp} has a solution} \iff (I - K)u = L_\gamma^{-1}f \iff L_\gamma^{-1}(f) \in \Im(I - K) = \ker(I - K^\dagger)^\perp\]
That is, we need \(\pinner{v, L_\gamma^{-1}(f)}_{L^2} = 0\) for all \(v \in \ker(I - K^\dagger)\). But for all \(v \in \ker(I - K^\dagger)\),
\[0 = \pinner{v, L_\gamma^{-1}(f)}_{L^2} = \pinner{v, \frac{1}{\gamma}Kf}_{L^2} - \frac{1}{\gamma}(K^\dagger v, f) = \frac{1}{\gamma}\pinner{v, f}_{L^2}\]
and so \(\pinner{v, f}_{L^2} = 0\) for all \(v \in \ker(I - K^\dagger)\).
\end{proof}

\begin{remark}
    In this proof, given \(L\), we see that for \(\gamma\) large, \(L_\gamma\) is a bounded inverible linear map, the map \(L_\gamma^{-1} = (L + \gamma I)^{-1}\) is called the \emph{resolvent of \(L\)}. The fact that \(L_\gamma^{-1} : L^2 \to L^2\) is compact is expressed as \emph{\(L\) has compact resolvent}.
\end{remark}

\begin{theorem}
    \index{spectum!of ellipic differential operators}
    Under the same assumptions as in \cref{thm:fredholm-elliptic-bvp},
    \begin{enumerate}[(i)]
        \item there exists a countable set \(\Sigma \sse \R\) such that the boundary value problem
        \begin{equation}
            \label{eq:elliptic-bvp-spectrum}
            \begin{cases}
                Lu = \lambda u + f &\text{in }U \\
                u = 0 & \text{on }\partial U
            \end{cases}
        \end{equation}
        has a weak solution for all \(f \in L^2(U)\) if and only if \(\lambda \notin \Sigma\).
        \item if \(\Sigma\) is infinite, then \(\Sigma = \{\lambda_k\}_{k \in \N}\). After reordering, then
        \[\lambda_1 < \lambda_2 < \cdots\]
        with \(\lambda_k \to \infty\) as \(k \to \infty\).
        \item for each \(\lambda \in \Sigma\), there exists a finite dimensional space
        \[\mcE(\lambda) = \left\{u \in H^1 \mid u \text{ is a weak solution to the homog. problem }Lu = \lambda u\right\}\]
    \end{enumerate}
    We call \(\lambda \in \Sigma\) an \emph{eigenvalue} of \(L\), and elements of \(\mcE(\lambda)\) are the corresponding \emph{eigenfunctions}.
\end{theorem}

\begin{proof}
    Choose \(\gamma > 0\) as in \cref{eq:fredholm-alt-elliptic-bvp}. Choose \(\mu \ge \gamma\), then \(L_\mu = L + \mu I\) is invertible, and
    \[L_\mu^{-1} : L^2 \to L^2\]
    is compact. If \(\lambda \le -\gamma\), then the problem
    \[\begin{cases}
        Lu - \lambda u = f & \text{in }U \\
        u = 0&\text{on }\partial U
    \end{cases}\]
    Thus, \(\Sigma \sse (-\gamma, \infty)\). If \(\lambda > -\gamma\), then solving \cref{eq:elliptic-bvp-spectrum} is equivalent to solving
    \begin{equation}
        \label{eq:elliptic-bvp-spectrum-2}
        \begin{cases}
            (L - \lambda I)u = f &\text{in }U \\
            u = 0 &\text{on }\partial U
        \end{cases}
    \end{equation}
    Applying \cref{thm:fredholm-elliptic-bvp} to \(L - \lambda I\), \cref{eq:elliptic-bvp-spectrum-2} has a unique weak solution for all \(f \in L^2\) if and only if \(u = 0\) is the unique solution to
    \[\begin{cases}
        (L - \lambda I)u = 0 &\text{in }U \\
        u = 0 &\text{on }\partial U
    \end{cases}\]
    That is, case (II) in \cref{thm:fredholm-elliptic-bvp} does not occur. This is true if and only if \(u = 0\) is the only solution to
    \[\begin{cases}
        Lu + \gamma u = (\lambda + \gamma)u &\text{in }U \\
        u = 0 &\text{on }\partial U
    \end{cases}\]
    which is true if and only if \(u =0\) is the only solution to
    \[u = L_\gamma^{-1}((\lambda + \gamma)u) = \frac{\gamma + \lambda}{\gamma}K(u)\]
    which is saying \(u = 0\) is the only solution to
    \[K(u) = \frac{\gamma}{\gamma + \lambda}u\]
    That is, \(\gamma/(\gamma + \lambda)\) is \emph{not} an eigenvalue of \(K\). So
    \[\lambda \in \Sigma \iff \mu = \frac{\gamma}{\gamma + \lambda}\text{ is an eigenvalue of }K\]
    But recall \cref{thm:compact-spectrum}, the set of eigenvalues of \(K\) is either finite, or countably infinite and converging to zero. In the second case, if
    \[\mu_k \to 0 \text{ then }\lambda_k \to \infty\]
    The fact that \(\mcE(\lambda)\) is finite dimensional follows from the Fredholm alternative.
\end{proof}

\begin{remark}
    If \(\lambda \notin \Sigma\), then there exists \(C(\lambda) > 0\) such that
    \[\norm{u}_{L^2} \le C(\lambda)\norm{f}_{L^2}\]  
    As \(\lambda\) approaches an eigenvalue, \(C(\lambda) \to \infty\).  
\end{remark}

\subsection{Self-adjoint positive operators}

\begin{definition}
    [formally self adjoint]
    The operator \(L\) is \emph{formally self-adjoint}\index{formally self-adjoint} if \(L = L^\dagger\). Equivalently, \(b^i = 0\) for all \(i\).
\end{definition}

If \(L\) is self adjoint, then \(B[u, v] = B[v, u]\).

\begin{definition}
    [positive] We say \(L\) is \emph{positive}\index{positive} if there exists \(\beta > 0\) such that
    \[\beta\norm{u}^2_{H^1} \le B[u, u]\]
    for all \(u \in H^1_0\).
\end{definition}
That is, \(B\) is coercive.

\lec

\begin{theorem}
    [eigenvalues of symmetric elliptic operators] Let \(L\) be uniformly elliptic, formally self-adjoint, positive operator on \(U\). Then we can represent the eigenvalues of \(L\) as a sequence
    \[0 < \lambda_1 \le \lambda_2 \le \cdots\]
    with multiplicity, i.e \(\lambda\) appears \(\dim(\mcE(\lambda))\) times. Moreover, there exists an orthonormal basis of \(L^2\), consisting of eigenfunctions \(\{w_k\}\), with
    \[\begin{cases}
        Lw_k = \lambda w_k & \text{on }U \\
        w_k = 0 &\text{on }\partial U
    \end{cases}\]
    and each \(w_k \in H^1_0(U)\).
\end{theorem}

\begin{proof}
    By positivity, Lax-Milgram implies that \(L\) is invertible. Moreover, \(L^{-1} : L^2(U) \to H^1_0(U)\) is bounded. Denote \(S = L^{-1} : L^2(U) \to L^2(U)\). Then \(S\) is compact.

    \begin{claim}
        \(S\) is self-adjoint.
    \end{claim}

    \begin{proof}
        Choose \(f, g \in L^2(U)\), then \(Sf = u\) means \(u \in H^1_0(U)\) is the unique weak solution to
        \[\begin{cases}
            Lu = f &\text{in }U \\
            u = 0&\text{on }\partial U
        \end{cases}\]
        The same statement applies to \(Sg = v\). That is,
        \[B[u, w] = \pinner{f, w}_{L^2}\]
        for all \(w \in H_0^1\), and
        \[B[v, \varphi] = \pinner{g, \varphi}_{L^2}\]
        for all \(\varphi \in H_0^1\). With this,
        \begin{align*}
            \pinner{Sf, g}_{L^2} &= \pinner{u, g}_{L^2} \\
            &= B[v, u] \\
            &= B[u, v] \\
            &= \pinner{f, v}_{L^2} \\
            &= \pinner{f, Sg}_{L^2}
        \end{align*}
    \end{proof}

    Now by \cref{thm:compact-spectrum} (spectrum of compact self-adjoint operators), there exists a sequence of eigenvalues \((\mu_k)_k \sse \R\), such that \(\mu_k \to 0\), and there exists \(w_k \in L^2(U)\), such that \(\{w_k\}\) is an orthonormal basis of \(L^2\), with \(S w_k = \mu_k w_k\). Equivalently, \(L^{-1}w_k = \mu_k w_k \in H^1_0\), and so \(Lw_k = \lambda_k w_k\), where \(\lambda_k = 1/\mu_k\). Positivity of \(\lambda_k\) follows from positivity of \(L\), and so the positivity of \(S\).
\end{proof}

\subsection{Elliptic regularity}

In this section, we will assume \(U \sse \R^n\) is an open bounded domain, \(V \Subset U\). Our goal is to improve the regulaity of the weak solutions \(u \in H^1_0(U)\), to say \(u \in C^2(\overline U)\).

\begin{example}
    [motivating examples]
    Let \(u \in C_c^\infty(\R^n)\) be a solution to
    \[-\Delta u = f\]
    Then
    \begin{align*}
        \int_{\R^n}f^2 \dd x &= \int_{\R^n}(\Delta u)^2 \dd x\\
        &=\sum_{i, j}\int_{\R^n}(D_iD_i u)(D_j D_j u)\dd x
    \end{align*}
    Integrating by parts twice,
    \begin{align*}
        &= \sum_{i, j}\int_{\R^n}(D_iD_ju)(D_iD_ju) \dd x \\
        &= \norm{D^2 u}^2_{L^2(\R^n)}
    \end{align*}
    That is, we have that
    \[\norm{D^2u}_{L^2(\R^n)} \le \norm{\Delta u}_{L^2(\R^n)}\]
    and so all second derivatives are controlled in \(L^2\) by \(\Delta u\).
\end{example}

However, if \(u \in H^1\), then \(D^2 u\) may not exist (even weakly). Thus, we will approximate the derivatives.

\begin{definition}
    [difference quotient]
    For \(0 < \abs{h} < d(V, \partial U)\) (required so we stay awy from the boundary), define the \emph{difference quotient}\index{difference quotient}
    \[\Delta^h_iu(x) = \frac{u(x + he_i) - u(x)}{h}\]
    for \(x \in V, i = 1, \dots, n\). Write
    \[\Delta^h u = (\Delta^h_1u, \dots, \Delta^h_nu)\]
\end{definition}

\begin{remark}
    Suppose \(u \in L^2\), then \(\Delta^hu \in L^2(V)\), and
    \[D(\Delta^hu) = \Delta^h(Du)\]
    Hence if \(u \in H^1(U)\), then \(\Delta^hu \in H^1(V)\).
\end{remark}

\begin{lemma}
    \label{lem:diff-quot-equiv}
    Suppose \(u \in L^2(U)\), then \(u \in H^1(V)\) if and only if there exists \(C > 0\), such that for all \(h\) with \(0 < \abs{h} < \frac{1}{2}d(V, \partial U)\), with
    \[\norm{\Delta^hu}_{L^2(V)} \le C\]
    Moreover, there exists \(\widetilde C\) such that
    \[\frac{1}{\widetilde C}\norm{Du}_{L^2(V)} \le \norm{\Delta^hu}_{L^2(V)} \le \tilde C\norm{Du}_{L^2(V)}\]
    That is, the difference quotient is equivalent to the derivative, sometimes written
    \[\norm{Du}_{L^2(V)} \simeq \norm{\Delta^hu}_{L^2(V)}\]
\end{lemma}

\begin{proof}
    Examples sheet 3.
\end{proof}

\begin{theorem}
    [interior regularity] \index{interior regularity} \label{thm:int-regularity} Suppose \(L\) is uniformly elliptic on \(U\), and assume \(a^{ij} \in C^1(U)\) and \(b^i, c \in L^\infty(U)\), \(f \in L^2(U)\). Suppose \(u \in H^1(U)\) satisfies 
    \begin{equation}
        \label{eq:interior-regularity}
        B[u, v] = \pinner{f, v}_{L^2}
    \end{equation}
    for all \(v \in H^1_0(U)\), then \(u \in H^2_\loc(U)\), and for each \(V \Subset U\),
    \[\norm{u}_{H^2(V)} \le C\left(\norm{f}_{L^2(U)} + \norm{u}_{L^2(U)}\right)\]
    with \(C = C(a, b, c, V, U, n)\), but not on \(f\) or \(u\).
\end{theorem}

\begin{remark}
    The result says that we gain two weak derivatives by solving the equation. We can also write the inequality as
    \[\norm{u}_{H^2(V)} \le C\left(\norm{Lu}_{L^2(U)} + \norm{u}_{L^2(U)}\right)\]
\end{remark}

\begin{proof}
    \textbf{Step 1:} Fix \(V \Subset U\), and choose \(W\) such that \(V \Subset W \Subset U\). Let \(\xi \in C_c^\infty(W)\), such that \(0 \le \xi \le 1\), \(\xi\vert_V = 1, \xi\vert_{\partial W} = 0\). We can rewrite the weak equation as
    \[\int_U a^{ij}D_i u D_j v \dd x = \int_U \tilde f v \dd x\]
    where
    \[\tilde f = f - b^iD_i u - cu \in L^2(U)\]
    Let \(v = -\Delta_k^{-h}(\xi^2\Delta_k^hu)\) for \(k\) fixed, \(0 < \abs{h} < d(W, \partial U)\). Note by previous comments, \(v \in H^1_0(W)\), and approximates \(D^2u\). Set
    \begin{align*}
        A &= \int_U a^{ij}u_{x_i} v_{x_j} \dd x \\
        B &= \int_U \tilde f v \dd x
    \end{align*}
\lec
    Observe for \(\psi, \phi \in L^2(U)\) supported in \(W\), then
    \begin{align*}
        \int_U \psi(x)\left(\Delta_k^{-h}\phi(x)\right) \dd x = - \int_U (\Delta_k^h\psi(x))\phi(x)\dd x
    \end{align*}
    which is integration by parts for the difference quotient. Moreover,
    \[\Delta_j^h(\psi\phi)(x) = \frac{\psi(x + h e_k)\phi(x + he_k) - \psi(x)\phi(x)}{h} = (\tau_k^h\psi)(x) \Delta_k^h\phi(x) + (\Delta_k^h\psi)(x)\phi(x)\]
    where
    \[\tau_k^h\psi(x) = \psi(x + he_k)\]
    is the \emph{translation operator}.

    \textbf{Step 2 (Bounding \(A\)):} Using the above,
    \begin{align*}
        A &= -\int_U a^{ij}u_{x_i}\Delta_k^{-h}((\xi^2 \Delta_k^h u)_{x_j})\dd x \\
        &= \int_U \Delta_k^h(a^{ij}u_{x_i})(\xi^2 \Delta_k^h u)_{x_j} \dd x \\
        &= \int_U \left((\tau^h_k a^{ij})\Delta^h_k u_{x_i} + (\Delta^h_ka^{ij}u_{x_i})\right)\left(\xi^2 \Delta^h_ku_{x_j} + 2\xi \xi_{x_j}\Delta^h_k u\right)\dd x \\
        &= A_1 + A_2
    \end{align*}
    where
    \begin{align*}
        A_1 &= \int_U \xi^2(\tau_k^h a^{ij})(\Delta_k^h u_{x_i})(\Delta_k^h u_{x_j}) \dd x
    \end{align*}
    By uniform ellipticity,
    \[\tau_k^h a^{ij}\eta_i\eta_j \ge \theta \abs{\eta}^2\]
    Applying with \(\eta_i = \Delta_k^hu_{x_i}\), we get that
    \[A_1 \ge \theta \int_U \xi^2\abs{\Delta_k^h(D u)}^2 \dd x\]

    Next,
    \[A_2 = \int_U(\Delta_k^h a^{ij})u_{x_i}\xi^2\Delta_k^hu_{x_j} + 2 \xi(\Delta_k^h a_{ij})u_{x_i}\xi_j\Delta_k^h u + 2 \xi(\tau_k^ha^{ij})(\Delta_k^hu_{x_i})\xi_{x_i}\Delta_k^hu \dd x\]
    Since \(a^{ij} \in C^2(U)\), and \(\xi\) is bounded,
    \[\abs{A_2} \le C \int_W \xi\abs{Du}\abs{\Delta_k^h(Du)} + \xi\abs{Du}\abs{\Delta_k^hu} + \xi\abs{\Delta_k^h(Du)}\abs{\Delta_k^h u}\dd x\]
    We are interested in \(\Delta_k^h(Du)\). We can use Young's inequality
    \[\le \epsilon\int_W \xi^2\abs{\Delta_k^h(D u)}^2 \dd x + \frac{C}{\epsilon}\int_W\abs{Du}^2 + \abs{\Delta_k^hu}^2 \dd x\]
    By \cref{lem:diff-quot-equiv},
    \[\le \epsilon\int_W\xi^2\abs{\Delta_k^h(Du)}^2 \dd x + \frac{C}{\epsilon}\int_W\abs{Du}^2 \dd x\]
    Set \(\epsilon = \theta/2\), and use \(A_2 \ge -\abs{A_2}\), we find that
    \[A = A_1 + A_2 \ge \frac{\theta}{2}\int_W\xi^2\abs{\Delta_k^h(D u)}^2 \dd x - C \int_W\abs{Du}^2 \dd x\]

    \textbf{Step 3 (Bounding \(B\)):}
    \begin{align*}
        \abs{B} &\le C\int_W \left(\abs{f} + \abs{Du} + \abs{u}\right)\abs{\Delta_k^{-h}(\xi^2\Delta_k^h u)} \dd x
    \end{align*}
    Applying \cref{lem:diff-quot-equiv} again, and \((a + b)^2 \le 2a^2 + 2b^2\),
    \begin{align*}
        \int_W\abs{\Delta_k^h(\xi^2\Delta_k^h)}^2 \dd x &\le C\int_W\abs{D(\xi^2\Delta_k^hu)}^2 \dd x \\
        &\le C\int_W\abs{\xi}^2 \abs{D\xi}^2 \abs{\Delta_k^hu}^2 \dd x + C \int_W \xi^2 \abs{\Delta_k^h(Du)}^2 \\
        &\le C\int_W \abs{Du}^2 + C \int_W \xi^2 \abs{\Delta_k^h(Du)}^2 \dd x
    \end{align*}
    By Young's inequality on \(\abs{B}\),
    \[\abs{B} \le \epsilon\int_U \xi^2\abs{\Delta_k^h(Du)}^2 \dd x + \frac{C}{\epsilon}\int_W(\abs{f}^2 + u^2 + \abs{D u}^2) \dd x\]
    Set \(\epsilon = \theta/4\).

    \textbf{Step 4:} Since \(A = B\), we have that \(\abs{A} = \abs{B}\). Using the bounds that we have:
    \[\frac{\theta}{2}\int_U\xi^2\abs{\Delta_k^h(D u)}^2 \dd x - C \int_W\abs{Du}^2 \dd x \le \abs{A} = \abs{B} \le \frac{\theta}{4}\int_U \xi^2 \abs{\Delta_k^h(Du)}^2 \dd x + C \int_W(f^2 + u^2 + \abs{Du}^2)\dd x\]
    Rearranging,
    \[\int_U\xi^2 \abs{\Delta_k^h(Du)}^2 \dd x \le C \int_Wf^2 + u^2 + \abs{Du}^2 \dd x\]
    Since \(\xi\vert_V = 1\), we get that if \(u \in H^1(V)\) solves \cref{eq:interior-regularity}, then
    \[\int_V \abs{\Delta_k^h(Du)}^2 \dd x \le C \int_W f^2 + u^2 + \abs{Du}^2 \dd x\]
    Since \(C\) is independent of \(h\) (track every step), we can apply \cref{lem:diff-quot-equiv}, \(Du \in H^2(V)\) and so \(u \in H^2_\loc(U)\), with
    \[\norm{u}_{H^2(V)} \le C \left(\norm{f}_{L^2(W)} + \norm{u}_{H^1(W)}\right)\]

    \textbf{Step 5: Removing the dependency on \(\norm{Du}_{L^2(W)}\)}. Let \(\xi \in C_c^\infty(U)\) (be a different test function) with \(\xi\vert_W = 1\). Set \(v = \xi^2 u\) in \cref{eq:interior-regularity}, to get
    \[\int_U a^{ij}u_{x_i}(\xi^2u)_{x_j} + b^iu_{x_i} + cu^2\xi^2\dd x = \int_U \xi^2 fu \dd x\]
    By the same proof as in Garding's inequality, we can rearrange to get
    \[\norm{Du}^2_{L^2(W)} \le C\left(B[u, u] + \gamma\norm{u}^2_{L^2(W)}\right) \le C\left(\norm{f}^2_{L^2(W)} + \norm{u}^2_{L^2(W)}\right)\]
    Hence we have that
    \[\norm{u}_{H^1(W)} \le C\left(\norm{f}_{L^2(W)} + \norm{u}_{L^2(W)}\right)\]
    and so we have the expression
    \[\norm{u}_{H^2(V)} \le C\left(\norm{f}_{L^2(W)} + \norm{u}_{L^2(W)}\right)\]
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item This is a local result. To have \(u \in H^2(V)\), for \(V \Subset U\), it is enough to have \(f \in L^2(W)\), where \(V \Subset W \Subset U\). That is, if \(f\notin L^2\) near the boundary, we don't see this in our estimates.
        \item We can now show that the equation \(Lu = f\) holds pointwise a.e. To see this, \(u \in H^2_\loc(U)\), and so \(Lu \in L^2_\loc(U)\). Take \(V \Subset U\). For \(v \in C_c^\infty(U)\), then from \cref{eq:interior-regularity},
        \[\pinner{Lu - f, v}_{L^2(U)}= 0\]
        Since \(Lu - f \in L^2(V)\), it holds pointwise a.e. on \(V\).
    \end{enumerate}
\end{remark}

\begin{theorem}
    [higher order interior regularity]
    \index{interior regularity!higher order}
    If \(a^{ij}, b^i, c \in C^m(U)\), and \(f \in H^m(U)\), then \(u \in H^{m+2}(U)\) and for all \(V \Subset W \Subset U\),
    \[\norm{u}_{H^{m+2}(V)} \le C\left(\norm{f}_{H^m(U)} + \norm{u}_{L^2(U)}\right)\]
\end{theorem}

\begin{remark}
    We also have a H\"older theory of elliptic regularity, i.e. if \(f \in C^{k, \gamma}(U)\) then \(u \in C^{k+2, \gamma}(U)\).
\end{remark}

\lec

\begin{remark}
    Recall if \(m\) is large enough, i.e. \(m > n/2\), then
    \[H^{m+2}_\loc(U) \inj C^2_\loc(U)\]
    and so if \(a^{ij}, b^i, c, f \in C^\infty(U)\), then \(u\) is also smooth.
\end{remark}

\begin{theorem}
    [boundary \(H^2\) regularity]
    \label{thm:boundary-regularity}
    \index{boundary \(H^2\) regularity}
    Suppose \(a^{ij} \in C^1(\overline U), b^i, c \in L^\infty(U)\), \(f \in L^2(U)\), and \(\partial U\) is \(C^2\). Suppose \(u \in H^1_0(U)\) is a weak solution to
    \begin{equation}
        \label{eq:int-reg-elliptic-bvp}
        \begin{cases}
            Lu = f & \text{in }U \\
            u = 0 &\text{on }\partial U
        \end{cases}
    \end{equation}
    Then \(u \in H^2(U)\), and
    \[\norm{u}_{H^2(U)} \le C\left(\norm{f}_{L^2(U)} + \norm{u}_{L^2(U)}\right)\]
\end{theorem}

\begin{proof}
    [Sketch proof] We focus on the case
    \[U = B_1(0) \cap \{x_n > 0\}\]
    Let \(V = B_{1/2}(0) \cap \{x_n > 0\}\) and \(\xi \in C_x^\infty(B_1(0))\) with \(\xi = 1\) on \(V\), \(0 \le \xi \le 1\). Since \(u \in H^1_0\) is a weak solution,
    \[\int_U a^{ij}u_{x_i}v_{x_j} = \int_U \tilde f v\]
    for all \(v \in H_0^1(U)\). Choose \(0 < \abs{h} < \frac{1}{4}d(\Supp(\xi), \partial B_1(0))\). Consider
    \[v = -\Delta_k^{-h(\xi^2\Delta_k^hu)}\]
    for \(k = 1, \dots, n-1\) fixed.

    \begin{claim*}
        \(v \in H_0^1(U)\).
    \end{claim*}

    \begin{proof}
        \begin{align*}
            v(x) &= \frac{-1}{h}\Delta_k^{-h}(\xi^2(x)(u(x + he_k) - u(x))) \\
            &= \frac{1}{h^2}\left(\xi^2(x - h e_k)(u(x) - u(x - h e_k)) - \xi^2(x)u(x + he_k - u(x))\right)
        \end{align*}
        The translation is horizontal, and \(Tu\vert_{x_n = 0} = 0\), and so
        \[T(u(x \pm he_k))\vert_{x_n = 0}\]
        for all \(\abs{x} < 1 - h\). When \(x_n = 0, \abs{x} \ge 1 - h\), we have that \(\xi(x) = 0\) and \(\xi(x - he_k) = 0\).
    \end{proof}
    Repeating the proof of \cref{thm:int-regularity} to conclude
    \[\int_V\abs{\Delta_k^h(Du)}^2\dd x \le C\int_U f^2 + u^2 + \abs{Du}^2 \dd x\]
    where \(C\) does not depend on \(h\). Hence
    \[D_ku \in H^1(V)\]
    for \(k = 1, \dots, n-1\), with
    \[\norm{D_kD_iu}_{L^2(V)} \le C(\norm{f}_{L_2(U)} + \norm{u}_{H^1(U)})\]
    where \(i = 1, \dots, n\).

    Hence it suffices to consider \(u_{x_nx_n}\). We will use the equation for this. Write the PDE as
    \[a^{nn}u_{x_nx_n} = F = - \sum_{i + j < 2n}a^{ij}u_{x_ix_j} - \sum_i b^iu_{x_i} - cu + f\]
    By uniform ellipticity,
    \[a^{nn} = \sum a^{ij}\xi_i\xi_j \ge \theta\norm{\xi}^2 = \theta > 0\]
    where \(\xi = (0, \dots, 0, 1)\).

    By the bound above, we can bound all of the terms of \(F\), and so \(F \in L^2(V)\), and
    \[u_{x_nx_n} = \frac{1}{a^{nn}}F \in L^2(V)\]
    and
    \[\norm{u}_{H^2(V)} \le C(\norm{f}_{L^2(U)} + \norm{u}_{H^1(U)})\]
    From the proof of Garding's inequality, we can replace \(\norm{u}_{H^1(U)}\) with \(\norm{u}_{L^2(U)}\) in the above.

    To finish, we cover the boundary with a finite union of \(V_i\)s, and sum using a partition of unity. See Evans for details.
\end{proof}

\begin{corollary}
    Under the assumptions of \cref{thm:boundary-regularity}, if \(u\) is the unique weak solution to the boundary value problem \cref{eq:int-reg-elliptic-bvp}, then we have that
    \[\norm{u}_{H^2(U)} \le C\norm{f}_{L^2(U)} = C\norm{Lu}_{L^2(U)}\]
    i.e. we can drop the \(\norm{u}_{L^2(U)}\) terms.
\end{corollary}

That is, the \(\norm{u}_{L^2(U)}\) measures the kernel of \(L\), and so if the solution is unique, the kernel is zero.

\begin{remark}
    We can get higher regularity. If \(a^{ij}, b^i, c \in C^{m+1}(\overline U), f \in H^m(\overline U)\), \(\partial U\) is \(C^{m+2}\),  and \(u \in H^1_0\) a weak solution, then \(u \in H^{m+2}(U)\) and
    \[\norm{u}_{H^{m+2}(U)} \le C(\norm{f}_{H^m(U)} + \norm{u}_{L^2(U)})\]
\end{remark}

\begin{remark}
    If everything is smooth, then \(u\) is smooth. For example, if \(Lu = \lambda u\), then \(L - \lambda I\) is uniformly elliptic, and
    \[(L - \lambda I)u  = 0 \in C^\infty\]
    and so the eigenfunctions are smooth.
\end{remark}

\section{Hyperbolic PDEs}

We will consider \emph{second order linear PDE} of the form
\begin{equation}
    \label{eq:hyp-pde}
    \sum_{i=1}^{n+1} (a^{ij}u_{y_i})_{y_j} + \sum_{i=1}^{n+1} a^i(u)u_{y_i} + a(y)u = f
\end{equation}
with \(y \in \R^{n+1}\), \(a^{ij} = a^{ji}\), \(a^{ij}, a^i, a \in C^\infty(\R^{n+1})\). This equation is \emph{hyperbolic} if the quadratic form
\[q(\xi) = \sum_{i, j=1}^{n+1}a^{ij}(y)\xi_i \xi_j\]
has signature \((+, -, \dots, -)\) for all \(y \in \R^{n+1}\). That is, at each \(y \in \R^{n+1}\), after a change of basis, we can write \(q\) as
\[\lambda_{n+1}^2\xi_{n+1}^2 - \sum_{i=1}^n \lambda_i^2\xi_i^2\]
where each \(\lambda_i > 0\).

We call \(q\) the \emph{principal symbol}\index{principal symbol} of the PDE. By a coordinate transformation, locally we can put \cref{eq:hyp-pde} in the form
\begin{equation}
    \label{eq:hyperbolic-local-form}
    u_{tt} - \sum_{i, j=1}^n (a^{ij}(x, t)u_{x_i})_{x_j} + \sum_{i=1}^{n+1} b^i(x, t)u_{x_i} + c(x, t)u
\end{equation}
where \((y_1, \dots, y_{n+1}) = (x_1, \dots, x_n, t)\).

Note if we assume
\[\sum_{i, j}a^{ij}\xi_i\xi_j \ge \theta\norm{\xi^2}\]
then since the coefficient of \(u_{tt}\) is \(1\), which is non-zero, then we see that
\[\{(x, t) \mid t = 0\}\]
is a non-characteristic surface of the PDE. In principle, we can solve the PDE with analytic data \(u, u_t\) at \(t = 0\).

\subsection{Hyperbolic initial boundary value problems}

Suppose \(U \sse \R^n\) is open bounded with \(C^1\) boundary. Define
\[U_T = (0, T) \times U \qquad \Sigma_t = \{t\} \times U \qquad \partial^*U_T = [0, T] \times \partial U\]

Using this,
\[\partial U_T = \Sigma_0 \cup \Sigma_1 \cup \partial^*U_T\]
Let \(u \in C^2(U_T)\), which satisfies the initial boundary value problem
\[\begin{cases}
    u_{tt} = \Delta u & \text{in }U_T \\
    u = \psi_0 & \text{on }\Sigma_0 \\
    u_t = \psi_1 & \text{on }\Sigma_0 \\
    u = 0 & \text{on }\partial^*U_T
\end{cases}\]

We will perform an energy estimate. Multiply the PDE by \(u_t\), integrating by parts over \(U_t = (0, t) \times U\) for \(t \in (0, T)\), we get
\[0 = \int_{U_t}u_{tt}u_t - u_t \Delta u \dd x \dd t\]
In what follows, \(D\) will denote the derivative with respect to the space variables only. Recall \(\grad \cdot (g \grad h) = \grad g \cdot \grad h + g \Delta h\), and so we get
\begin{align*}
    &= \int_{U_t}\left(\frac{1}{2}\partial_t(u_t^2) - \div_x(u_tDu) + Du_t Du\right) \dd x \dd t \\
    &= \int_{U_t}\frac{1}{2}\partial_t\left((u_t)^2 + \abs{Du}^2\right) - \div_x(u_tDu) \dd x \dd t \\
    &= \frac{1}{2}\int_{\Sigma_t}(u_t^2 + \abs{Du}^2)\dd x - \frac{1}{2}\int_{\Sigma_0}(u_t)^2 + \abs{Du}^2 \dd x
\end{align*}
where we use the divergence theorem, and the fact that \(u\) vanishes on \(\partial^*U_t\). Hence we have that
\[\int_{\Sigma_t}u_t^2 + \abs{Du}^2 \dd x = \int_{\Sigma_0} \psi_1^2 + \abs{D\psi_0}^2 \dd x\]
Ww call this an energy estimate as the energy is conserved, where \(u_t\) is kinetic energy and \(\abs{Du}^2\) is potential energy.

\lec

We call ths estimate above an \emph{a priori estimate}. These are very useful.

Let \(v, \overline v \in C^2(U_T)\) be two solutions with initial data \(\phi_i, \overline{\phi_i}\). Let \(u = v - \overline{v}\), \(\psi_0 = \phi_0 - \overline{\phi_0}\) and \(\psi_1 = \phi_1 - \overline{\phi_1}\). Then there exists \(C > 0\) such that
\[\sup_{t \in [0, T]}\left(\norm{u(\cdot, t)}^2_{H^1(\Sigma_t)} + \norm{u_t(\cdot, t)}^2_{L^2(\Sigma_t)}\right) \le C\left(\norm{\psi_0}_{H^1(\Sigma_t)}^2 + \norm{\psi_1}_{L^2(\Sigma_0)}^2\right)\]
Thus, in this case, we have uniqueness and continuous dependence on initial conditions.

Define
\[Lu = -\sum_{i, j=1}^n(a^{ij}(x, t)u_{x_i})x_j + \sum_{i=1}^n b^i(x, t) u_{x_i} + b(x, t)u_t + c(x, t)u\]
with \(a^{ij} = a^{ji}, b^i, b, c \in C^1(\overline{U_T})\). Suppose there exists \(\theta > 0\) such that
\[\sum_{i, j=1}^n a^{ij}(x, t)\xi_i\xi_j \ge \theta\abs{\xi}^2\]
for all \((x, t) \in U_T\), \(\xi \in \R^n\).

We will consider the initial boundary value problem
\begin{equation}
    \label{eq:hyperbolic-ibvp}
    \begin{cases}
        u_tt + Lu = f & \text{in }U_T \\
        u = \psi_0, u_t = \psi_1 & \text{on }\Sigma_0 \\
        u = 0 &\text{on }\partial^*U_T
    \end{cases}
\end{equation}
We would like to find the weak formulation.

Suppose \(u \in C^2(\overline{U_T})\) is a solution to \cref{eq:hyperbolic-ibvp}. Multiply by \(v \in C^2(\overline{U_T})\), such that \(v = 0\) on \(\partial^*U_T \cup \Sigma_T\).

Integrating over \(U_T\),
\begin{align*}
    \int_{U_T}fv\dd x \dd t &= \int_{U_T} (u_{tt}v + Lu v) \dd x \dd t \\
    &= \int_{U_T}(-u_tv_t + a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + bu_t v + cuv)\dd x \dd t
    + \left[\int_{\Sigma_t}u_t v \dd x\right]_{t=0}^T - \int_{0}^T \int_{\partial\Sigma_t}a^{ij}u_{x_i}v \dd \textbf{S}\dd t \\
    &= \int_{U_T}(-u_tv_t + a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + bu_t v + cuv)\dd x \dd t - \int_{\Sigma_0}\psi_1(x)v(x, 0)\dd x
\end{align*}
Thus, we have the equation
\begin{equation}
    \label{eq:hyperbolic-ibvp-weak}
    \int_{U_T}fv\dd x \dd t = \int_{U_T}(-u_tv_t + a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + bu_t v + cuv)\dd x \dd t - \int_{\Sigma_0}\psi_1(x)v(x, 0)\dd x
\end{equation}
Now suppose \cref{eq:hyperbolic-ibvp-weak} holds for all \(v \in C^2(\overline{U_T})\) with \(v = 0\) on \(\partial^*U_T \cup \Sigma_T\). If \(v \in C_c^\infty(\overline{U_T})\), then we can undo the integration by parts, and we get that
\[\int_{U_T}v(u_{tt} + Lu - f)\dd x \dd t = 0\]
Since \(v\) is arbitrary, \(u_{tt} + Lu = f\) on \(U_T\).

Now if \(v \in C^\infty(\overline{U_T})\), then we get that
\[\int_{U_T}(u_{tt} + Lu - f)\dd x \dd t = \int_{\Sigma_0}(\psi_1 - u_t)v \dd x\]
Hence
\[\int_{\Sigma_0}(\psi_1 - u_t)v\dd x = 0\]
for all \(v \in C^\infty(\overline{U_T})\) with \(v = 0\) on \(\partial^*U_T \cup \Sigma_T\). Now let \(v(x, t) = \chi(t)\varphi(x)\), with \(\chi \in C^\infty([0, T])\) and \(\varphi \in C_c^\infty(\Sigma_0)\). We require that \(\chi = 1\) near \(t = 0\) and \(\chi = 0\) near \(t = T\), hence
\[v\vert_{\Sigma_0} = \varphi\]
Hence
\[\int_{\Sigma_0}(\psi_1(x) - u_t(x, 0))\varphi(x)\dd x = 0\]
and so \(\psi_1 = u_t\) on \(\Sigma_0\).

\begin{definition}
    [weak solution]
    Suppose \(f \in L^2(U_T)\), \(\psi_0 \in H_0^1(\Sigma_0)\), \(\psi_1 \in L^2(\Sigma_0)\), \(a^{ij}= a^{ji}, b^i, b, c \in C^1(\overline{U_T})\). We say that \(u \in H^1(U_T)\) is a \emph{weak solution to the hyperbolic initial boundary value problem \cref{eq:hyperbolic-ibvp}}\index{weak solution to hyperbolic initial boundary value problem} if \(u\vert_{\Sigma_0} = \psi_0\), \(u \vert_{\partial^*U_T} = 0\) in the trace sense, and \cref{eq:hyperbolic-ibvp-weak} for all \(v \in H^1(U_T)\), with \(v = 0\) on \(\partial^*U_T \cup \Sigma_T\) in the trace sense.
\end{definition}

\begin{theorem}
    [uniqueness] A weak solution, if it exists, is unique.
\end{theorem}

\begin{proof}
    If \(v, \overline v\) are two weak solutions with the same initial data, then we can use the linearity of the PDE problem, \(u = v - \overline v\) is a weak solution with \(f = 0\), \(\psi_0 = 0\) and \(\psi_1 = 0\).

    The idea is to use an energy to show that \(\norm{u} = 0\) to show that \(u = 0\). We would like to pick \(v = u_t\), as we did for the wave equation, but
    \begin{enumerate}
        \item \(v\) may not be in \(H^1(U_T)\), since we only know that \(u \in H^1\).
        \item \(v\) may not vanish on \(\Sigma_T\).
    \end{enumerate}
    We set
    \[v(x, t) = \int_t^T e^{-\lambda s}u(x, s)\dd t\]
    where we will choose \(\lambda\) later. We can see that \(v\) is in \(H^1(U_T)\), and \(v = 0\) on \(\partial^*U_T \cup \Sigma_T\). Moreover,
    \[v_t = -e^{-\lambda t}u(x, t)\]
    We will take this \(v\) as the test function. This gives us
    \begin{align*}
        \int_{U_T}u_t ue^{-\lambda t} - e^{\lambda t}a^{ij}v_{tx_i}v_{x_j} + b^iu_{x_i}v + bu_t v + (c - 1)uv - e^{\lambda t}v v_t\dd x \dd t = 0
    \end{align*}
    Integrating by parts,
    \begin{align*}
        \int_{U_T}u_tue^{-\lambda t} - e^{\lambda t}a^{ij}v_{tx_j}v_{x_i} + \underbrace{(b^iu v)_{x_i}}_a + \underbrace{(buv)_t}_b - (b^i_{x_i}uv + b^iuv_{x_i} + b_tuv + buv_t) + (c-1)uv - \frac{1}{2}\partial_t(v^2e^{\lambda t}) + \frac{1}{2}\lambda v^2e^{\lambda t}\dd x \dd t
    \end{align*}
    The terms \(a\) and \(b\) vanish by boundary conditions. Hence we have that
    \begin{align*}
        \int_{U_T}\frac{1}{2}\pdv{t}\left(u^2e^{-\lambda t} = a^{ij}v_{x_i}v_{x_j}e^{\lambda t} - v^2e^{\lambda t}\right)\dd x \dd t + \frac{\lambda}{2}\int_{U_T}(u^2 e^{-\lambda t} + a^{ij}e^{\lambda t}v_{x_i}v_{x_j} + v^2e^{\lambda t}) \dd x \dd t \\
        = \int_{U_T}\frac{1}{2}a^{ij}_j v_{x_i}v_{x_j}e^{\lambda t} + (b^i_{x_i} + b_t + 1 - c)uv + b^iv_{x_i}u + buv_t \dd x \dd t 
    \end{align*}
    Call the first line \(A\) and the second \(B\). For \(A\),
    \[A = e^{\lambda T}\int_{\Sigma_T}\frac{1}{2}u^2 \dd x + \frac{1}{2}\int_{\Sigma_0}a^{ij} v_{x_i}v_{x_j} + v^2 \dd x + \frac{\lambda}{2}\int_{U_T}u^2e^{-\lambda t} + e^{\lambda t}a^{ij}v_{x_i}v_{x_j} + v^2 e^{\lambda t}\dd x \dd t\]
    Hence we have that
    \[A \ge \frac{\lambda}{2}\int_{U_T}u^2e^{-\lambda t} + \theta \abs{Du}^2 e^{\lambda t} + v^2e^\lambda t \dd x \dd t\]
    Also,
    \begin{align*}
        B &\le C(a^{ij}_t)\int_{U_T}e^{\lambda t}\abs{Dv}^2\dd x + C(b, b^i, c)\int_{U_T}\abs{u}\abs{v}\dd x + C(b^i)\int_{U_T}\abs{u}\abs{Dv}\dd x + C(b)\int_{U_T}u^2e^{-\lambda t} \dd x \\
        &\le \frac{C}{\theta}\int_{U_T}e^{\lambda t}\theta\abs{Dv}^2 + C\int_{U_T}e^{-\lambda t}\abs{u^2} + e^{\lambda t}(\abs{v}^2 \abs{Dv}^2) \dd x \dd t \\
        &\le C \int_{U_T}\theta \abs{Dv}^2 e^{\lambda t} + u^2 e^{-\lambda t} + v^2 e^{\lambda t} \dd x \dd t
    \end{align*}
    Now using that \(\abs{A} = \abs{B}\),
    \[\left(\frac{\lambda}{2} - C\right)\int_{U_T}\underbrace{(u^2 e^{-\lambda t} + \theta\abs{Dv}^2 + v^2e^{-\lambda t})}_{\le 0}\dd x \dd t \le 0\]
    Taking \(\lambda > 2C\), the integral must be zero, and so
    \[\int_{U_T}u^2 e^{-\lambda t}\dd x\dd t = 0\]
    Hence \(u = 0\) a.e.
\end{proof}

\lec

\begin{theorem}
    [existence of solutions] 
    \index{existence of solutions to hyperbolic initial boundary value problems}
    Given \(\psi_0 \in H^1_0(U), \psi_1 \in L^2(U), f \in L^2(U_T)\), then there exists a unique weak solution of \cref{eq:hyperbolic-ibvp-weak} \(u \in H^1(U_T)\), with
    \[\norm{u}_{H^1(U_T)} \le C\left(\norm{\psi_0}_{H^1(U)} + \norm{\psi_1}_{H^1(U)} + \norm{f}_{L^2(U_T)}\right)\]
\end{theorem}

\begin{proof}
    [Proof (Galerkin's method)] We will project everything onto the finite dimensional subspace of \(L^2\), given by the first \(N\) eigenfunctions of the Dirichlet Laplacian. Taking \(N \to \infty\) gives the result.

    \textbf{Step 1:} Recall the eigenfunctions \(\{\varphi_k\}_{k=1}^\infty\) of \(L = -\Delta\)  with Dirichlet boundary conditions form an orthonormal basis of \(L^2(U)\). We have that \(\varphi_k \in H^1_0(U)\), and by elliptic regularity \(\varphi_k \in C^\infty(U)\) privided \(\partial U\) is \(C^\infty\). With this,
    \[\pinner{\varphi_k, \varphi_\ell}_{L^2(U)} = \delta_{k\ell}\]
    and if \(u \in L^2(U)\) then
    \[u = \sum_{k=1}^\infty(u, \varphi_k)_{L^2(U)}\varphi_k\]
    with convergence in \(L^2(U)\).

    \textbf{Step 2:} First consider \(\psi_0, \psi_1 \in C_c^\infty(U)\), \(f \in C_c^\infty(U_T)\). These spaces are dense in \(\HH^1_0(U), L^2(U)\) and \(L^2(U_T)\) respectively. Define
    \[u^N(x, t) = \sum_{k=1}^N u_k(t)\varphi_k(x)\]
    Assume \(u_k(t) \in C^2(\overline{(0, T)})\), and that \(u^N\) is a weak solution to \cref{eq:hyperbolic-ibvp-weak}. Take \(v(x, t) = \rho(t)\varphi_\ell(x)\) for the test function, with \(\rho \in C_c^\infty((0, T))\) arbitrary. Substituting into \cref{eq:hyperbolic-ibvp-weak}, we get
    \begin{align*}
        \int_{U_T}-(u^N_t \dot\rho \varphi_\ell + a^{ij}u^N_{x_j}(\varphi_\ell)_{x_i}\rho + b^i u^N_{x_i}\rho \varphi_\ell + b u^N_t \rho\varphi_\ell + cu\rho\varphi_\ell - f\rho\varphi_\ell)\dd x \dd t = 0
    \end{align*}
    Note
    \[\int_{U_T} -u^N_t \dot\rho \varphi_\ell \dd x \dd t = \int_{U_T}u^N_{tt}\rho\varphi_\ell \dd x\dd t\]
    and so our identity looks like
    \[\int_0^t\int_{\Sigma_t}G(x, t)\rho(t)\dd x \dd t = 0\]
    But \(\rho\) is arbitrary, and so
    \[\int_{\Sigma_t}G(x, t) \dd x = 0\]
    for all \(t\). With this,
    \begin{equation}
        \label{eq:hyper-exist-int-over-space}
        \pinner{u^N_{tt}, \varphi_\ell}_{L^2(\Sigma_t)} + \int_{\Sigma_t}a^{ij}u^N_{x_j}(\varphi_\ell)_{x_i} + b^i(u^N)_{x_i} \varphi_\ell + bu^N_t \varphi_\ell + cu^N\varphi_\ell \dd x = \pinner{f, \varphi_\ell}_{L^2(\Sigma_t)}
    \end{equation}
    But \cref{eq:hyper-exist-int-over-space} holds for all \(t\), and \(\ell = 1, \dots, N\). By orthogonality,
    \[\pinner{u^N_{tt}, \varphi_\ell}_{L^2(\Sigma_t)} = \sum_{k=1}^M\left(\ddot u_k(t)\varphi_k, \varphi_\ell\right)_{L^2(\Sigma_t)} = \ddot u_\ell(t)\]
    With this, we get that for \(\ell = 1, \dots, N\), then
    \[\ddot u_\ell(t) + \sum_{k=1}^N (\alpha_{\ell, k}(t)u_k(t) + \beta_{\ell, k}(t)\dot u_k(t)) = f_\ell(t)\]
    where
    \begin{align*}
        \alpha_{\ell, k}(t) &= \int_{\Sigma_t}a^{ij}(\varphi_\ell)_{x_j}(\varphi_k)(x_i) + b^i(\varphi_\ell)_{x_i}\varphi_k + c\varphi_\ell\varphi_k \dd x \\
        \beta_{\ell, k}(t) &= \int_{\Sigma_t}b(x, t)\varphi_k\varphi_\ell \dd x \\
        f_\ell(t) &= \int_{\Sigma_t}f(x, t)\varphi_\ell(x) \dd x
    \end{align*}
    and
    \begin{align*}
        u_\ell(0) &= \pinner{\psi_0, \varphi_\ell}_{L^2(\Sigma_0)} \\
        \dot u_\ell(0) &= \pinner{\psi_1, \varphi_\ell}_{L^2(\Sigma_0)} \\
    \end{align*}
    This is a system of \(N\) second order ODEs, linear in \(u_k\), with coefficients which are bounded uniformly in \(C^1\) for \(t \in [0, T]\). By Picard-Lindel\"of, there exists a unique solution \(u_k \in C^2([0, T])\). Moreover,
    \[u^N, u^N_t \in H^1(U_T)\]

    \textbf{Step 3:} We would like a uniform estimates
    \[\norm{u^N}_{\HH^1(U_T)} \le C\]
    which are independent of \(N\). Multiply \cref{eq:hyper-exist-int-over-space} by \(e^{-\lambda t}\dot u_\ell(t)\), sum over \(1, \dots, n\), and integrate over \([0, \tau] \sse [0, T]\). 

    For example,
    \[\sum_{\ell = 1}^N\int_{-\lambda t}\dot u_\ell(t)\int_{\Sigma_t}u^N_{tt}\varphi_\ell \dd x \dd t = \int_{U_\tau}e^{-\lambda t}u^N_{tt}u^N_t \dd x\ dd t\]
    We find that
    \[\int_{U_\tau}\left(u^N_{tt}u^N_t + a^{ij}u^N_{x_i}u^N_{tx_j} + b^iu^N_{x_i}u^N_t + b(u^N_t)^2 + cu^Nu^N_t\right) e^{-\lambda t}\dd x \dd t =\int_{U_\tau}f u^N_t e^{-\lambda t}\dd x \dd t\]
    Similar to the proof of uniqueness, we can rearrange this as
    \begin{align*}
        \tilde A &= \int_{U_\tau}\frac{1}{2}\dv{t}(Q_ae^{-\lambda t})\dd x \dd t + \frac{\lambda}{2}\int_{U_\tau}Q_ae^{-\lambda t}\dd x \dd t \\
        \tilde B &= \int_{U_\tau}\left(\frac{1}{2}a^{ij}_t u^N_{x_i}u^N_{x_j} - b^iu^N_{x_i} u^N_t = b(u^N_t)^2 + (1-c)u^N u^N_t + fu^N_t\right)e^{-\lambda t} \dd x \dd t \\
        \tilde A &= \tilde B
    \end{align*}
    where
    \[Q_a = (u^N_t)^2 + a^{ij}u^N_{x_i}u^N_{x_j} + (u^N)^2\]
    Let
    \[Q_\theta = (u^N_t)^2 + \theta \abs{Du^N}^2 + (u^N)^2\]
    Using uniform ellipticity, Young's inequality, \(e^{-\lambda t} \le 1\), and so on, we get that
    \begin{align*}
        \tilde B &\le \int_{U_\tau}Q_\theta e^{-\lambda t} \dd x \dd t + \norm{f}_{L^2(U_\tau)}^2 \\
        \tilde A&\ge e^{_\lambda\tau}\int_{\Sigma_\tau}Q_\theta\dd x - \frac{1}{2}\int_{\Sigma_\tau}Q_\theta \dd x + \frac{\lambda}{2}\int_{U_\tau}Q_\theta e^{-\lambda t}\dd x \dd t
    \end{align*}
    Note \(\abs{\tilde A} = \abs{\tilde B}\), for \(\lambda/2 - C \ge 1/2\), we get
    \begin{align*}
        e^{-\lambda \tau}\int_{\Sigma_\tau}Q_\theta \dd x + \int_0^\tau \int_{\Sigma_t}Q_\theta e^{-\lambda t} \dd x \dd t &\le \int_{\Sigma_0}Q_a \dd x + C\norm{f}^2_{L^2(U_\tau)} \\
        &\le C\left(\norm{u^N(\cdot, 0)}^2_{H^1(\Sigma_0)} + \norm{\dot u^N(\cdot, 0)}^2_{L^2(\Sigma_0)} + \norm{f}^2_{L^2(U_T)}\right)
    \end{align*}
    for all \(\tau \in [0, T]\). Taking \(\sup\),
    \begin{align*}
        \sup_\tau\left(\norm{u^N(\cdot, \tau)}^2_{H^1(\Sigma_t)} + \norm{\cdot u^N(\cdot, \tau)}^2_{L^2(\Sigma_\tau)}\right) + \int_{0}^T \left(\norm{u^N(\cdot, t)}^2_{H^1(\Sigma_t)} + \norm{\cdot u^N(\cdot, t)}^2-{L^2(\Sigma_t)}\right) \dd t \\
        \le Ce^{\lambda T}\left(\norm{u^N(\cdot, 0)}^2_{H^1(\Sigma_0)} + \norm{\dot u^N(\cdot, 0)}^2_{L^2(\Sigma_0)} + \norm{f}^2_{L^2(U_T)}\right)
    \end{align*}
    Since
    \[u^N(0) = \sum_{k=1}^N(\psi_0, \varphi_k)\varphi_k \to \psi_0\]
    as \(N \to \infty\), if \(\psi_0 \ne 0\), then for large \(N\), then
    \[\norm{u^N(0)}_{H^1(\Sigma_0)} \le 2\norm{\psi_0}\norm{\psi_0}_{H^1(\Sigma_0)}\]
    Similarly,
    \[\norm{\dot u^N}_{L^2(\Sigma_0)} \le 2\norm{\psi_1}_{L^2(\Sigma_0)}\]
    The right hand sides are independent of \(N\), and so
    \[\norm{u^N}_{H^1(U_T)} \le C_1 = C\left(\norm{\psi_0}_{H^1(\Sigma_0)} + \norm{\psi_1}_{H^1(\Sigma_0)} + \norm{f}_{L^2(U_T)}\right)\]
    The right hand side is the uniform estimate which we want.
\lec
    Now
    \[u^N \in H^1_\partial(U_T) = \left\{\phi \in H^1(U_T) \mid \phi\vert_{\partial^*U_T} = 0\right\}\]
    which is a closed subspace of \(H^1(U_T)\), and so it is weakly sequentially compact. Hence there exists a subsequence \((u^{N_i})\) such that
    \[u^{N_i} \weak u \in H^1_\partial(U_T)\]
    Moreover,
    \[\norm{u}_{H^1(U_T)} = \liminf_{i \to \infty}\norm{u^{N_i}}_{H^1(U_T)} \le C_1\]

    \textbf{Step 4:} We want to show that \(u\) is the desired weak solution. We can relabel the \(u^{N_i}\) as \(u^N\). Fix \(m \le N\), and consider
    \[v = \sum_{k=1}^m v_k(t)\varphi_k(x)\]
    where \(v_k \in H^1((0, T))\) with \(v_k(T) = 0\). Then \(v\) is a test function for the weak formulation. From \cref{eq:hyper-exist-int-over-space} (replace \(\ell\) with \(k\)), multiply the equation by \(v_k(t)\), and sum from \(k = 1, \dots, m\), we get that
    \begin{align*}
        \pinner{u^N_{tt}, v}_{L^2(\Sigma_t)} + \int_{\Sigma_t}a^{ij}u^N_{x_i}v_{x_j} + b^iu^M_{x_i}v + bu^N_tv+ cu^Nv \dd x &= \pinner{f, v}_{L^2(\Sigma_t)} 
    \end{align*}
    Now integrating over \([0, T]\), integrating by parts, and using the fact that \(v(T) = 0\), we get that
    \[-\int_{\Sigma_0}u^N_t v \dd x + \int_{U_T}-u_N^tv_t + a^{ij}u^N_{x_i}v_{x_j} + b^iu^N_{x_i}v + bu_tv + cu^Nv\dd x \dd t = \int_{U_T}fv \dd x \dd t\]
    But for the first term, since \(N > m\),
    \[\int_{\Sigma_0}u^N_tv \dd x = \int_{\Sigma_0}\psi_1 v \dd x\]
    Passing to the weak limit,
    \begin{equation}
        \label{eq:weak-limit-hyper}
        -\int_{\Sigma_0}\psi_1v\dd x + \int_{U_T}(-u_tv_t + a^{ij}u_{x_i}v_{x_j} + b^iu_{x_i}v + bu_t v + cuv)\dd x \dd t = \int_{U_T}fv\dd x\dd t
    \end{equation}
    But this is precisely the weak formulation. We leave as an exercise that the space of such \(v\) is dense in \(H^1_\partial(U_T)\), and so \cref{eq:weak-limit-hyper} holds for all \(v \in H^1_\partial(U_T)\).

    \textbf{Step 5:} It remains to show that \(u\vert_{\Sigma_0} = \psi_0\). For each fixed \(k\), define
    \begin{align*}
        \Phi_k : H^1(U_T) &\to \R \\
        w &\mapsto \int_{\Sigma_0}w\varphi_k\dd x
    \end{align*}
    This is a bounded linear map. To see this,
    \[\abs{\Phi_k(w)} \le \int_{\Sigma_0}\abs{w\varphi_k} \le \norm{w}_{L^2(\Sigma_0)}\norm{\varphi_k}_{L^2(\Sigma_0)} \le \norm{w}_{L^2(\partial U_T)} \le C\norm{w}_{H^1(U_T)}\]
    where in the last step we use the trace theorem. By weak convergence,
    \[\Phi_k(u^N) \to \Phi_k(u)\]
    Thus,
    \[\int_{\Sigma_0}\psi_0 \varphi_k\dd x = \int_{\Sigma_0}u^N(x, 0)\varphi_k(x)\dd x \to \int_{\Sigma_0}u(x, 0)\varphi_k(x)\dd x\]
    Hence
    \[\int_{\Sigma_0}\left(\psi_0 - u(x, 0)\right)\varphi_k \dd x = 0\]
    for all \(k\). Hence \(u = \psi_0\) on \(\Sigma_0\).
\end{proof}

\begin{definition}
    [Bochner space]
    If \(X\) is a Banach space, the \emph{Bochner space}\index{Bochner space} \(L^p((0, T); X)\) is 
    \begin{align*}
        L^p((0, T); X) = \left\{u : (0, T) \to X \mid \norm{u}_{L^p((0, T); X)} < \infty\right\}
    \end{align*}
    where
    \[\norm{u}_{L^p((0, T); X)} = \left(\int_0^T \norm{u}_X^p \dd t\right)^{1/p}\]
    for \(1 \le p < \infty\), and
    \[\norm{u}_{L^\infty((0, T); X)} = \esssup_{t \in (0, T)}\norm{u(t)}_X\]
\end{definition}

\begin{remark}
    In step 3, we showed that
    \[\norm{u}_{H^1(U_T)} \le C_1\]
    In fact, the weak solution satisfies
    \[\norm{u_t}_{L^\infty((0, T); L^2(U))} + \norm{u}_{L^\infty((0, T); H^1(U))} \le C_1\]
    Thus, instead of \(H^1(U_T)\), we can consider
    \[u \in L^\infty((0, T); H^1(U))\]
\end{remark}

\subsection{Finite speed of propagation}

A crucial feature of hyperbolic equation is that there is a finite speed of propagation.

\begin{definition}
    [spacelike, timelike]
    Let \(\Sigma \sse \R^{n+1}\) be a hypersurface, given by
    \[\Sigma = \left\{(x, t) \in \R^{n+1} \mid F(x, t) = 0\right\}\]
    Define
    \[w(F_{x_i}, F_t) = (F_t)^2 - a^{ij}F_{x_i}F_{x_j}\]
    We say that \(\Sigma\) \begin{itemize}
        \item spacelike if \(w > 0\),
        \item timelike if \(w < 0\),
        \item characteristic (in PDE theory) or null (in GR) if \(w = 0\).
    \end{itemize}
\end{definition}

\begin{example}
    The plane \(t = 0\) is spacelike.
\end{example}

\begin{example}
    The cylinder
    \[F = \abs{x - x_0}^2 - R^2\]
    is timelike
\end{example}

Let \(S_0 \sse U\) be an open set with smooth boundary. Let \(\tau : S_0 \to (0, T)\) be a smooth function such that \(\tau\vert_{\partial S_0} = 0\). Let 
\[S' = \Graph(\tau) = \left\{(x, \tau(x)) \mid x \in S_0\right\}\]
If \(F(x_1, \dots, x_n, t) = t = \tau(x)\), then we see that \(S'\) is spacelike if
\[1 - a^{ij}\tau_{x_i}\tau_{x_j} > 0\]
Equivalently,
\[a^{ij}(x)\tau_{x_i}\tau_{x_j} < 1\]
Let
\[D = \left\{(x, t) \in U_T \mid x \in S_0, 0 \le t \le \tau(x)\right\}\]

\underline{Exercise:} if \(a^{ij}\xi_i\xi_j \le \mu\abs{\xi}^2\), for some \(\mu > 0\), then we can show that such \(S_0, S'\) exists.

\begin{theorem}
    [domain of dependence] If \(S'\) is spacelike, \(u\) a weak solution to \cref{eq:hyperbolic-ibvp-weak}, then \(u\vert_D\) depends only on \(\psi_0\vert_{S_0}, \psi_1\vert_{S_0}, f\vert_D\).
\end{theorem}
