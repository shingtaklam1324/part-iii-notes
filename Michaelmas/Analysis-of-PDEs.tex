\section{Basics}

Let \(\mcU \subseteq \R^n\) be an open set. A \emph{PDE of order \(k\)}\index{PDE of order \(k\)} is an expression of the following form
\begin{equation}
    \label{eq:pde}
    F(x, u, Du, \dots, D^ku) = 0
\end{equation}
where \(u : \mcU \to \R\) is the unknown, \(F : \mcU \times \R \times \R^n \times \cdots \times \R^{n^k}\) is a general function. We say that \(u\) is a \emph{classical solution of}\index{classical solution} \cref{eq:pde} if \(u\) satisfies \cref{eq:pde} in \(\mcU\).

We may also consider the case where \(u(x) \in \R^p\) and \(F \in \R^q\), and we call this a \emph{system of PDEs}\index{system of PDEs}.

\subsection{Examples of PDEs}

\begin{example}
    [ODE system]
    One example of a model used in mathematical biology is the ODE system
    \begin{align*}
        \dv{u}{t} &= f(u, v) \\
        \dv{v}{t} &= g(u, v)
    \end{align*}
\end{example}

\begin{example}
    [Laplace's equation]
    \[\Delta u = \sum_{i=1}^n \partial_i^2 u = 0\]
    The Laplacian is an \emph{averaging} operator.
\end{example}

\begin{example}
    [Heat equation]
    \[u_t = D\Delta u\]
    This is also called the diffusion equation, \(D\) is called the diffusion constant.
\end{example}

\begin{example}
    [Navier-Stokes]
    The Navier-Stokes equations in fluid dynamics is
    \begin{align*}
        u_t &= \nu\Delta u - u \cdot \grad u - \grad \rho + f \\
        \div(u) &= 0
    \end{align*}
\end{example}

\begin{example}
    [Transport equation]

    The transport equation is
    \[u_t + vu_x = 0\]
    where \(v\) is a constant, corresponding to the velocity. A modification is the \emph{advection-diffusion} equation,
    \[u_t + v \vdot \grad u = D\Delta u + f\]
\end{example}

\begin{example}
    [Poisson equation]
    \[\Delta u = f\]
    Describes electric field due to some charge, or Newtonian gravity.
\end{example}

\begin{example}
    [Wave equation]
    \[\Box u = -u_tt + c^2 \Delta u = 0\]
    This models sound waves, seismic waves, ...
\end{example}

\begin{example}
    [KdV equation]
    This equation admits \emph{soliton} solution.
    \[u_t + \partial_x^3 u - 6u\partial_x u = 0\]
\end{example}

\begin{example}
    [Maxwell equations]
    \begin{align*}
        \div(E) &= \rho \\
        \div(B) &= 0 \\
        \partial_t E &= \curl B = J \\
        \partial_t B &= -\curl F
    \end{align*}
\end{example}

\begin{example}
    [Einstein's equations]
    \[\Ric(g) - \frac{1}{2}gR(g) = 0\]
\end{example}

\subsection{Data and Well-Posedness}

All of the examples from above need additional information to solve, which we call the \emph{data}\index{data}. For example, we might need \(u\vert_{\partial \mcU}\) and so on. A guiding principle to this process is called \emph{well-posedness (in the sense of Hadamard)}.

We say that a PDE problem (equation and the data) is well-posed\index{well-posed} if we have

\begin{enumerate}
    \item A solution exists (in some function space).
    \item Given some data, the solution should be unique (depends on the function space of choice).
    \item The solution depends continuously on the data.
\end{enumerate}

The aim is to find the largest space for which a solution exists, but small enough so that it is unique. (For example, strong cosmic censorship in GR?)

\begin{notation}
    [Multi-index notation] We will use multi-index notation\index{multi-index notation},
    \[\N = \{0, 1, \dots\}\]
    \(\alpha = (\alpha_1, \dots, \alpha_n) \in \N^n\) is called a \emph{multi-index}, and we define the \emph{order} of \(\alpha\)
    \[\abs{\alpha} = \alpha_1 + \alpha_2 + \dots + \alpha_n\]
    and the \(\alpha\)-th derivative is
    \[D^\alpha f(x) = \partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}f\]
    If \(x = (x_1, \dots, x_n)\), then \(x^\alpha = x_1^{\alpha_1}\cdots x_n^{\alpha_n}\), and
    \[\alpha! = \alpha_1!\cdots \alpha_n!\]
\end{notation}

\subsection{Classifying PDEs (of order \(k\))}

We say \cref{eq:pde} is \emph{linear}\index{linear PDE} if \(F\) is a linear function of \(x, u\) and its derivative. That is, we can write it as

\[\sum_{\abs{\alpha} \le k}a_\alpha(x) D^\alpha u = f(x)\]

Moreover, we say that a linear PDE is \emph{homogeneous}\index{linear PDE!homogeneous} if \(f = 0\). We say \cref{eq:pde} is \emph{semilinear}\index{semilinear PDE} if the highest order derivatives appear linearly with coefficients depending only on \(x\). That is, we have

\[\sum_{\abs{\alpha} = k}a_\alpha(x)D^\alpha u(x) + F(x, u, Du, \dots, D^{k-1}u) = 0\]

One example would be \(\Delta u = u_x^2\). Finally, we say \cref{eq:pde} is \emph{quasilinear}\index{quasilinear PDE} if the highest order derivatives appear linearly, but the coefficients depending on lower order derivatives. That is, we have

\[\sum_{\abs{\alpha} = k}a_\alpha(x, u, \dots, D^{k-1}u)D^k u + F(x, u, \dots, D^{k-1}u) = 0\]

For example, we can have \(u u_{xx} + u_{yy} - u_x^2 = 0\). Finally, we say \cref{eq:pde} is \emph{fully non-linear}\index{fully non-linear PDE} if it is non of the above.

\section{Cauchy-Kovalevskaya theorem}

\subsection{ODE theory}

Fix \(\mcU \sse \R^n\) open, and suppose \(f : \mcU \to \R^n\) is given. We would like to consider the ODE

\begin{equation}
    \label{eq:ode}
    \begin{cases}
        \dot u(t) = f(u(t)) \\
        u(0) = u_0
    \end{cases}
\end{equation}

\begin{theorem}
    [Picard-Lindel\"of]\index{Picard-Lindel\"of} Suppose we have \(r, K > 0\) such that \(B_r(u_0) \sse U\) and
    \[\norm{f(x) - f(y)} \le K\norm{x - y}\]
    for all \(x, y \in B_r(u_0)\). Then there exists \(\epsilon > 0\), depending on \(K, r\) and a unique \(C^1\) solution \(u : (-\epsilon, \epsilon) \to \mcU\) solving \cref{eq:ode}.
\end{theorem}

\lec

\begin{proof}
    [Sketch proof, see examples sheet 1] If \(u \in \C^1\) solves \cref{eq:ode}, then by the fundamental theorem of calculus, \(u\) satisfies the \emph{weak formulation}
    \begin{equation}
        \label{eq:weak-sol-ode}
        u(t) = u_0 + \int_0^t f(u(s))\dd s
    \end{equation}
    Moreover, if \(u \in C^0\) is a solution to \cref{eq:weak-sol-ode}, then it is a \(C^1\) solution to \cref{eq:ode}. Moreover, if \(u\) exists, then it is a fixed point of
    \[G(w) = u_0 + \int_0^t f(w(s))\dd s\]
    Let \(\mcS = \left\{w : (-\epsilon, \epsilon) \to \overline{B_{r/2}(u_0)}\right\} \text{ continuous}\). We want to show that \(\mcS\) is a complete metric space, \(G : \mcS \to \mcS\) is a contraction for \(\epsilon\) sufficiently small, and by the contraction mapping theorem \(G\) has a fixed point.
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item The solution (in general) can't be global. Consider for example
        \[\dot u(t) = u(t)^2 \qqtext{with} u(0) = u_0 > 0\]
        Solutions to this equation blow up in finite time.
        \item This does not apply to
        \[\dot u(t) = \sqrt{u(t)} \qqtext{with} u(0) = 0\]
        There are two solutions. Note we \emph{can} apply the Peano existence theorem.
    \end{enumerate}
\end{remark}

Now suppose \(f\) is smooth, and we have \(\dot u(t) = f(u(t))\) is \(C^1\). By the chain rule,
\[\ddot u(t) = Df(u(t))\vdot \dot u(t) = f_2(u(t), \dot u(t))\]
which is continuous. Hence \(\ddot u\) is continuous, and so \(u \in C^2\). Repeating this, we get that \(u \in C^k\) for all \(k\). That is, \(u\) is smooth.

In principle, given \(u_0 = u(0)\), we can determine
\[u^{(k)}(0) = F_k\left(u, u', \dots, u^{(k-1)}\right)\big\vert_{t=0}\]
and so we can write
\[\sum_{k\ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
We call this a \emph{formal power series solution}. Does our solution \(u(t)\) agree with this? That is, do we have
\[u(t) = \sum_{k \ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
in a neighbourhood of \(0\)?

\begin{theorem}
    [Cauchy-Kovalevskaya for simple ODEs] 
    \label{thm:c-k-simple-ode}
    \index{Cauchy-Kovalevskaya Theorem!for simple ODEs}If \(f(u)\) is real analytic in a neighbourhood of \(u_0\), then the series
    \[\sum_{k \ge 0}\frac{u^{(k)}(0)}{k!}t^k\]
    converges in a neighbourhood of \(0\) to the unique solution of \cref{eq:ode} given by Picard-Lindel\"of.
\end{theorem}

\subsection{Real analyticity and majorants}

Suppose \(f : (-\epsilon, \epsilon) \to \R\) is a smooth function. Therefore, \(f^{(n)}(0)\) exists for all \(n \ge 0\). Does the partial sums
\[\sum_{n \ge 0}\abs{f^{(n)}(0)}{n!}x^n\]
for some \(\abs{x} \le \delta\)? No, consider the function
\[f(x) = \begin{cases}
    \exp(-\frac{1}{x}) & x > 0 \\
    0 & x \le 0
\end{cases}\]
This is a smooth function, with \(f^{(n)}(0) = 0\) for all \(n\).

\begin{definition}
    [real analytic]
    Let \(\mcU \sse \R^n\) be open and \(f : \mcU \to \R\) is \emph{real analytic}\index{real analytic} (at \(x_0\)) if there exists \(r > 0\), \(f_\alpha \in \R\) such that
    \[f(x) = \sum_{\alpha}f_\alpha(x-x_0)^\alpha\]
    when \(\abs{x - x_0} < r\). 
\end{definition}

\begin{remark}
    \begin{enumerate}
        \item That is, \(f\) can be written as a convergent power series and
        \[f_\alpha = \frac{D^\alpha f(x_0)}{n!}\]
        \item Real analyticity is a local property.
        \item \(f\) is real analytic on an open set \(U\) if it is real analytic at each \(x_0 \in U\).
        \item We will denote the set of real analytic functions on \(\mcU\) by \(C^\omega(\mcU)\).
        \item If \(f\) is \(C^\omega\), then \(f\) is smooth (e.g. Weierstrass \(M\)-test).
        \item If \(f\) is real analytic, and \(\mcU\) is connected, then \(f\) is uniquely determined in \(\mcU\) by its derivatives \(D^\alpha f(x)\) at some point \(x \in \mcU\).
        \item In particular, \(f\) is real analytic if and only if for any compact \(K \sse U\), there exists \(C, r\) such that
        \[\sup_{x \in K}\abs{D^\alpha f(x)} \le C\abs{\abs{\alpha}!}{r^{\abs{\alpha}}}\]
    \end{enumerate}
\end{remark}

\underline{Exercise:} Show \(f(x) = 1/x\) and \(f(x) =\sqrt{x}\) are real analytic for \(x > 0\).

\begin{example}
    Recall
    \[\frac{1}{1-x} = \sum_{k \ge 0}x^k\]
    for \(\abs{x} < 1\). Let \(r > 0\), and consider
    \[f(x) = \frac{r}{r - (x_1 + \dots + x_n)} = \frac{1}{1- \frac{x_1 + \dots + x_n}{r}} = \sum_{k \ge 0}\left(\frac{x_1 + \dots + x_n}{r}\right)^k\]
    provided \(\abs{x_1 + \dots + x_n} \le \sqrt n\left(\sum_j \abs{x_j}^2\right)^{1/2} = \sqrt n \norm{x} < r\).
    By the multinomial theorem (sheet 1),
    \[f(x) = \sum_{k \ge 0}\frac{1}{r^k}\sum_{\abs{\alpha} = k}{\abs{\alpha}\choose \alpha}x^\alpha = \sum_\alpha \frac{\abs{\alpha}!}{\alpha!r^{\abs{\alpha}}}x^\alpha\]
    where
    \[{\abs{\alpha}\choose\alpha} = \frac{\abs{\alpha}!}{\alpha!}\]
    and so,
    \[f(x) = \sum_\alpha f_\alpha x^\alpha\]
    where
    \[f_\alpha = \frac{\abs{\alpha}!}{\alpha!}\frac{1}{r^{\abs{\alpha}}}\]

    This series is absolutely convergent near zero, since
    \[\sum_\alpha \frac{\abs{\alpha!}}{\alpha!}\frac{x^\alpha}{r^{\abs{\alpha}}} = \sum_{k \ge 0}\left(\frac{\abs{x_1} + \dots + \abs{x_n}}{r}\right)^k < \infty\]
\end{example}

\begin{definition}
    [majorise]
    Let \(f = \sum_\alpha f_\alpha x^\alpha\) and \(g = \sum_\alpha g_\alpha x^\alpha\), we say that \emph{\(g\) majorises \(f\)}\index{majorise}, or \emph{\(g\) is a mojorant of \(f\)}, written \(g \gg f\), if \(g_\alpha \ge \abs{f_\alpha}\) for all \(\alpha\).
\end{definition}

For vector valued functions, we require each component to be a majorant.

\begin{lemma}
    [properties of majorants]
    \label{lem:majorant-props}
    \begin{enumerate}
        \item If \(g \gg f\), and \(g\) converges for \(\norm{x} < r\), then \(f\) converges for \(\norm{x} < r\).
        \item If \(f = \sum f_\alpha x^\alpha\) converges for \(\norm{x} < r\), then for any \(s \in (0, r/\sqrt n)\), there exists a majorant of \(f\) which converges for \(\norm{x} < s/\sqrt n\).
    \end{enumerate}
\end{lemma}

\begin{proof}
    1. Looking at the partial sums
    \[\sum_{\abs{\alpha} \le k}\abs{f_\alpha x^\alpha} = \sum_{\abs{\alpha} \le k}\abs{f_\alpha}\abs{x_1}^{\alpha_1} \cdots \abs{x_n}^{\alpha_n} \le \sum_{\abs{\alpha}\le k}\abs{g_\alpha}\abs{x_1}^{\alpha_1}\cdots\abs{x_n}^{\alpha_n} \le \sum_\alpha\abs{x_1}^{\alpha_1} \cdots \abs{x_n}^{\alpha_n} = g(\tilde x)\]
    where \(\tilde(x) = (\abs{x_1}, \dots, \abs{x_n})\). So \(\norm{\tilde x} = \norm{x}\), and so if \(\norm{x} < r\), then \(g\) converges at \(\tilde x\). That is, \(g(\tilde x) < \infty\). Therefore, we have a uniform bound on the partial sum.

    2. Let \(s \in (0, r/\sqrt n)\), and set \(y = (s, \dots, s)\). Then \(\norm{y} = s\sqrt n\), and by assumption,
    \[f(y) = \sum_\alpha f_\alpha y^\alpha\]
    converges, as \(\norm{y} = s\sqrt n < r\). So there exists a constant \(c\) such that \(\abs{f_\alpha y^\alpha} \le c\). Hence \[\abs{f_\alpha} \le \frac{C}{\abs{y^\alpha}} = \frac{C}{\abs{y_1}^{\alpha_1}\cdots \abs{y_n}^{\alpha_n}} = \frac{C}{\abs{s}^{\abs{\alpha}}} \le \frac{C}{\abs{s}^{\abs{\alpha}}}\frac{\abs{\alpha}!}{\alpha!}\]
    
    So we define
    \[g(x) = \frac{Cs}{s - (x_1 + \dots + x_n)}\]
    From the above, \(g\) majorises \(f\).
\end{proof}

\subsection{Proof of Cauchy-Kovalevskaya for ODEs}

% Meet and greet 2:30 - 3:15 this afternoon.
\begin{proof}
    [Proof of \cref{thm:c-k-simple-ode}] We will use the method of majorants. Without loss of generality \(u_0 = 0\), and for simplicity, we can assume \(n = 1\). We need to find the series coefficients. So
    \[\dot u = f(u)\]
    and so \(\dot u(0) = f(u(0)) = f(0)\), that is, \(u_1 = f(0)\). Next,
    \[\ddot u(t) = f'(u(t))\dot u(t)\]
    and so \(\ddot u(0) = f'(0)f(0)\), that is, \(u_2 = f'(0)f(0) = \frac{1}{2!}f'(0)f(0)\). Repeating,
    \[u^{(3)}(0) = f''(0)f(0)^2 + f'(0)^2 f(0)\]
    and so
    \[u_3 = \frac{1}{3!}(f''(0)f(0)^2 + f'(0)^2 f(0))\]
    Iterating this procedure,
    \[u_k = P_k\left(f(0), \dots, f^{(k-1)}(0)\right)\]
    where \(P_k\) is a polynomial in \(k\)-variables, with nonnegative coefficients. For example,
    \begin{align*}
        P_1(x) &= x \\
        P_2(x,y) &= \frac{1}{2!}xy \\
        P_3(x,y,z) &= \frac{1}{3!}(x^2z + xy^2)
    \end{align*} 

    Since \(f\) is real analytic, we have that
    \[f(v) = \sum_{k \ge 0} f_k v^k\]
    where \(f_k = \frac{1}{k!}f^{(k)}(0)\). Hence we have that
    \[f^{(k)}(0) = k!\cdot f_k\]
    Substituting, we have that
    \[u_k = Q_k(f_0, \dots, f_{k-1})\]
    which again is a polynomial in \(k\)-variables and nonnegative coefficients. This polynomial is ``universal''.

    \underline{Aim:} We would like to show that the power series
    \[\sum_k u_k t^k\]
    converges in a neighbourhood of \(t = 0\), and solves the ODE, \cref{eq:ode}. Since \(f\) is analytic, we know that
    \[f(u) = \sum_k f_k u^k\]
    on \(\abs{u} < k\). Fixing some \(s < r\), there exists a majorant
    \[g(u) = \sum_k g_k u^k\]
    of \(f\), from \cref{lem:majorant-props} (ii). Consider the auxilliary differential equation
    \[\dot w(t) = g(w(t))\]
    and \(w(0) = 0\). Using the definition of \(g\), we that
    \[\dv{w}{t} = \frac{Cs}{s - w(t)}\]
    We get that
    \[w = s\pm\sqrt{s^2 - 2Cst}\]
    Due to the initial data, we take the \(-\) solution. That is,
    \[w = s - \sqrt{s^2 - 2Cst}\]
    This is real analytic, for \(\abs{t} < s/2C\). This tells us that
    \[w(t)  = \sum_k w_kt^k\]
    converges for \(\abs{t} < s/2C\). Moreover,
    \[w_k = Q_k(g_0, \dots, g_{k-1})\]
    since \(Q_k\) is ``universal''.

    \begin{claim}
        \(w\) majorises \(u\).
    \end{claim}

    By construction, \(g\) majorises \(f\), i.e. \(g_k \ge \abs{f_k}\) for all \(k\). Moreover, since \(Q_k\) has nonnegative coefficients,
    \[w_k = Q_k(g_0,\dots, g_{k-1}) \ge Q_k(\abs{f_0}, \dots, \abs{f_{k-1}}) \ge \abs{Q_k(f_0, \dots, f_{k-1})} = \abs{u_k}\]
    Hence by \cref{lem:majorant-props} (i), we know that the series
    \[\sum_k u_kt^k\]
    converges for \(\abs{t} < s/2C\).

    To conclude, set
    \[u(t) := \sum_{k\ge 0}u_kt^k\]
    and we need to check that it solves \cref{eq:ode}. Both sides are analytic, so suffices to check the derivatives on each side agree to all orders at \(t = 0\).
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item We can extend to systems, where we replace \(u_k\) with
        \[u_k^j = Q_k^j(D^\abs{\alpha} \mid \abs{\alpha} \le k)\]
    
        For \(w\), we can replace \(w^j = w^1\) as before.
        \item For the non-autonomous case,
        \begin{align*}
            u(t) &= f(u, t) \\
            u(0) &= 0
        \end{align*}
        Consider \(v(t) = (u(t), t)\), then \(\dot v(t) = (\dot u(t), 1) = (f(u, 1), 1) = (f(v), 1) = F(v)\)
        with \(v(0) = 0\), and we can apply the system version.
    \end{enumerate}
\end{remark}

\subsection{Cauchy-Kovalevskaya for PDEs}

Let \(u : \R^n \to R^m\), and choose \(r > 0\). Consider the following problem
\[u_t = \sum_{j=1}^n B_j(u, x)u_{x_j} + C(u, x)\]
on \(\norm{x}^2 + t^2 < r^2\), with
\[u = 0\]
on \(\norm{x}^2 + t^2 < r^2\) and \(t = 0\). The \(B_j\) are matrices, \(B_j\) and \(C\) are real analytic.

\begin{theorem}
    [Cauchy-Kovalevskaya for first order systems]
    \index{Cauchy-Kovalevskaya Theorem!for first order systems}
    Suppose \(B_j, C\) are real analytic, for small \(r > 0\). Then there exists a unique real analytic function
    \[u = \sum_\alpha u_\alpha x^\alpha\]
    solving the above PDE.
\end{theorem}

\begin{proof}
    [Idea] Compute
    \[u_\alpha = \frac{D^\alpha u}{\alpha!}\]
    in terms of \(B_j, C\), and show that the power series converges for small \(r\). We use the PDE to find all derivatives.
\end{proof}

\begin{example}
    Consider the system
    \begin{align*}
        u_t &= v_x - f \\
        v_t &= -u_x
    \end{align*}
    with \(u = v = 0\) on \(t = 0\). The boundary conditions give us that
    \[u(0, 0) = v(0, 0) = 0\]
    We would like to determine \(u_\alpha\) for all \(\alpha\). By differentiating the boudary conditions,
    \[\partial_x^nu(x, 0) = \partial_x^nv(x, 0) = 0\]
    for all \(n\). That is, for the case \(\alpha = (n, 0)\). From the PDE,
    \[u_t(x, 0) = 0 - f = -f \qquad v_t(x, 0) = 0\]
    This then means that
    \[\partial_x^n \partial_t u(x, 0) = -\partial_x^n f(x, 0)\]
    and
    \[\partial_x^n \partial_tv(x, 0) = 0\]
    for all \(n \ge 1\).
    
    Next, if \(\alpha = (n, 2)\) use the PDE and we get
    \[u_{tt}(x, 0) = f_t(x, 0)\]
    and
    \[v_{tt}(x, 0) = f_x(x, 0)\]
    The same method as above gives us that
    \begin{align*}
        \partial_x^n\partial_t^2 u(x, 0) &= -(\partial x)^n \partial_t f(x, 0) \\
        \partial_x^n\partial_t^2 v(x, 0) &= (\partial_x)^{n+1}f(x, 0)
    \end{align*}

    Repeating this, we can compute all of the derivatives.
\end{example}

\subsection{Reduction to first order systems}

\begin{example}
    Consider \(u : \R^3 \to \R\) satisfying
    \[u_{tt} = u u_{xy} - u_{xx} + u_t\]
    with conditions
    \[u\vert_{t=0} = u_0(x, y) \qqtext{and} u_t\vert_{t=0}u_1(x, y)\]

    where \(u_0, u_1\) are real analytic near \(0 \in \R^3\). Note
    \[f(t, x, y) = u_0 + tu_1\]
    is real analytic near \(0 \in \R^3\). Note
    \[f\vert_{t=0} = u_0 \qqtext{and} \partial_tf\vert_{t=0} = u_1\]
    Set \(w(t,x,y) = u - f\). Then we find that
    \[w_tt = ww_{xy} - w_{xx} + w_t + fw_{xy} + f_{xy}w + F\]
    where
    \[R = ff_{xy} -  f_{xx} + f_t\]
    where
    \[w\vert_{t=0} = \partial_tw\vert_{t=0} = 0\]
    Observe \(F\) is real analytic, and independent of \(w\) and its derivatives. Let \(x = (x, y, t) = (x^1, x^2, x^3)\), and set
    \[v = (w, w_x, w_y, w_z) = (v^1, v^2, v^3, v^4)\]
    Then
    \begin{align*}
        v^1_t &= w_t = v^4 \\
        v_t^2 &= w_{xt} = v^4_{x_1} \\
        v_t^3 &= w_{yt} = v^4_{x_2} \\
        v_t^4 &= v^1v_{x_2}^2 - v_{x_1}^2 + v^4 + fv_{x_2}^2 + f_{xy}v^1 + F
    \end{align*}
    Define
    \[B_1 = \begin{pmatrix}
        & & & 0\\
        & & & 1 \\
        & & & 0 \\
        0 & -1 & 0 & 0
    \end{pmatrix} \qquad B_2 = \begin{pmatrix}
        & & & \\
        & & & \\
        & & & 1 \\
        v^1 + f & 0 & 0 & 0
    \end{pmatrix}\]
    and
    \[c = \begin{pmatrix}
        v^4 \\
        0 \\
        0 \\
        v^4 + f_{xy}v^1 + F
    \end{pmatrix}\]
    We can see then
    \[\pdv{x_3}v = B_1v_{x_1} + B_2v_{x_2} + c\]
    In this case, \(B_1, B_2, c\) are real analytic functions of \(x, v\), and so we can apply Cauchy-Kovalevskaya.
\end{example}

More generally, consider the scalar quasilinear problem

\[\sum_{\abs{\alpha} = k}a_\alpha(D^{k-1}u, \dots, u, x)D^\alpha u + a_0(D^{k-1}u, \dots, u, x) = 0\]
where
\[u : B_r(0) \to \R \qqtext{and} \pdv{u}{x_n} = \dots = \left(\pdv{x_n}\right)^{k-1}u = 0\]
for \(\norm{x'} < r, x_n = 0\).

Define
\[v = \left(u, \pdv{u}{x_1}, \dots, \pdv{u}{x_n}, \pdv[2]{u}{x_1}{x_1}, \dots, \left(\pdv{x_n}\right)^{k-1}u\right) = (v^1, \dots, v^m) \in \R^m\]
We would like to get a first order system in \(v\). That is, express
\[\pdv{v^j}{x_n}\]
in terms of \(v^j\) and \(\pdv{v}{x_j}\) for \(j = 1, \dots, m-1\). If \(j = 1\), then
\[\pdv{v^1}{x_n} = \pdv{u}{x_n} = v^\ell\]
for some \(\ell\). If \(2 \le j \le m-1\), then
\[v^j = D^\alpha u\]
for some \(\abs{\alpha} \le k-1\), such that \(\alpha_n < k-1\). So
\[\pdv{v^j}{x_n} = D^\alpha \pdv{u}{x_n} = \frac{\partial^{\abs{\alpha} + 1}}{\partial_1^{\alpha_1}\cdots \partial_n^{\alpha_n + 1}}u\]
If \(\abs{\alpha} \le k-2\), then \(\abs{\alpha} + 1 \le k-1\),a nd so \(\pdv{v^j}{x_n} = v^\ell\) for some \(\ell\).

If \(\abs{\alpha} = k-1\), and \(\alpha_n < k-1\), then there exists \(p \ne n\) such that \(\alpha_p \ge 1\). So we have that
\[\pdv{v^j}{x_n} = \pdv{x_n}(\frac{\partial^{\abs{\alpha}}u}{\partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}}) = \pdv{x_p}(\frac{\partial^{\abs{\alpha}}u}{\partial_1^{\alpha_1}\cdots \partial_p^{\alpha_p - 1}\cdots \partial_n^{\alpha_n - 1}}) = \pdv{v^\ell}{x_p}\]
for some \(\ell\). Finally, to compute
\[\pdv{v^m}{x_n}\]
we will use the PDE. Recall the coefficients are \(a_\alpha(v, x)\) for \(v \in \R^m, x \in \R^n\). We assume \(a_\alpha : B_\rho(0) \to \R\) is real analytic, and suppose
\[a_c = a_{(0, \dots, 0, k)}(0) \ne 0\]
Since \(a_\alpha\) are real analytic near zero, they are continuous. Therefore, \(a_c(z, w) \ne 0\) for all \(\norm{z}^2 + \norm{w}^2 \le \delta^2\), where \(\delta < \rho\). Then
\[a_c \frac{\partial^k u}{\partial x_n^k} = -\left(\sum_{\abs{\alpha} = k, \alpha_n < k} a_\alpha D^\alpha u + a_0\right)\]
Dividing by \(a_c\), we get
\[\frac{\partial^k u}{\partial x_n^k} = -\frac{1}{a_c}\left(\sum_{\abs{\alpha} = k, \alpha_n < k} a_\alpha D^\alpha u + a_0\right)\]

The right hand side can be written in terms of
\[\pdv{v^\ell}{x_p}, v\]
for \(p < n\). Therefore, if \(a_c \ne 0\), we have turned the scalar quasilinear PDE into a first order PDE system, on which we can apply Cauchy-Kovalevskaya.

\begin{definition}
    If \(a_c = a_{(0, \dots, 0, k)}(0)\) is non-zero, then we say the plane
    \[\{x^n = 0\}\]
    is \emph{non-characteristic}\index{non-characteristic!equation}. Otherwise, it is \emph{characteristic}.
\end{definition}

\subsection{Exotic boundary conditions}

\begin{definition}
    [real analytic hypersurface]
    We say \(\Sigma \sse \R^n\) is a \emph{real analytic hypersurface near \(x_0 \in \Sigma\)}\index{real analytic hypersurface} if there exists \(\epsilon > 0\), and a real analytic function
    \[\Phi : B_\epsilon(x_0) \to U \sse \R^n\]
    where \(U\) is an open neighbourhood of \(0 \in \R^n\), and defining \(y = \Phi(x)\), with \(\Phi(x_0) = 0\). Moreover, we require
    \begin{enumerate}[(i)]
        \item \(\Phi\) is a bijection,
        \item \(\Phi^{-1} : U \to B_\epsilon(x_0)\) is real analytic,
        \item \(\Phi(\Sigma \cap B_\epsilon(x_0)) = \{y_n = 0\} \cap U\). 
    \end{enumerate}

    We can think of \(\Phi\) as ``straightening out'' \(\Sigma\).
\end{definition}

\begin{example}
    Spheres, planes, tori, etc. are real analytic (hyper)surfaces.
\end{example}

Let \(\gamma\) be unit normal to \(\Sigma\), and consider
\[\sum_{\abs{\alpha} = k} a_\alpha(D^{k-1}u, \dots, u, x)D^\alpha u + a_0(D^{k-1}u, \dots, u, x)\]
where
\begin{equation}
    \label{eq:surf-pde}
    u = \sum_i\gamma^i\partial_i u = (\gamma^i\partial_i)^{k-1}u = 0 \text{ on }\Sigma
\end{equation}

Define
\[v(y) = u(\Phi^{-1}(y))\]
for \(y \in U\). That is, \(u(x) = v(\Phi(x))\) for \(x \in B_\epsilon(x_0)\). Using the chain rule,
\[\pdv{u}{x_i} = \sum \pdv{u}{y_j} \pdv{\Phi^j}{x_i}\]
where \(\Phi = (\Phi^1, \dots, \Phi^n) \in \R^n\). So the PDE becomes

\[\sum_{\abs{\alpha} = k} b_\alpha D^\alpha v + b_0 = 0\]
where \(b_0, b_\alpha\) dependis on \(u\) and \(D^\alpha u\) for \(\abs{\alpha} \le k-1\), and also \(\Phi\) (which is given). The boundary conditions becomes
\[v = \pdv{y_n}v = \dots = \left(\pdv{y_n}\right)^{k-1}v = 0\]
on \(\{y_n = 0\}\). Since \(\Phi\) is real analytic, so are \(b_0, b_\alpha\).

\lec

We would like to apply Cauchy-Kovalevskaya, therefore we need to check that whether the hypersurface \(\{y_n = 0\}\) is non-characteristic. That is,
\[b_{(0, \dots, 0, k)}(D^{k-1}v = 0, \dots, Dv = 0, y = 0) \ne 0\]

Note if \(\abs{\alpha} = 2\), we can compute
\[D^\alpha u = \frac{\partial^kv}{\partial y_n^k}(D\Phi^n)^\alpha + \text{terms not involving} \frac{\partial^kv}{\partial y_n^k}\]

For example, if \(k = 2, n = 2, \alpha = (0, 2)\), then
\[D^\alpha u = u_{x_2x_2} = v_{y_2y_2}\underbrace{(\Phi^2_{x_2})(\Phi^2_{x_2})}_{=(D\Phi^2)^\alpha} + \text{terms not involving} v_{y_2y_2}\]
Thus,
\[b_{(0, \dots, 0, k)} = \sum_{\abs{\alpha} = k}a_\alpha \left(D \Phi^n\right)^\alpha\]

\begin{definition}
    We say that \(\Sigma\) is \emph{non-characteristic at \(x_0 \in \Sigma\)}\index{non-characteristic!hypersurface} if
    \[b_{(0, \dots, 0, k)}(0) = \sum_{\abs{\alpha} = k}a_\alpha(0, \dots, 0, x_0)\left(D\Phi^n(x_0)\right)^\alpha \ne 0\]

    Otherwise, \(\Sigma\) is characteristic at \(x_0\).
\end{definition}

\begin{remark}
    Note that \(\Sigma = \left\{x \in \R^n \mid \Phi^{(n)}(x) = y_n = 0\right\}\). This tells us that
    \[D\Phi^n(x) = c(x)\gamma(x)\]
    where \(\gamma\) is the unit normal of \(\Sigma\). In particular,

    \[D\Phi^n(x_0) = c(x_0)\gamma(x_0)\]
    and so the non-characteristic condition is equivalent to
    \[\sum_{\abs{\alpha} = k}a_\alpha\gamma^\alpha(x) \ne 0\]
\end{remark}

\begin{theorem}
    [Cauchy-Kovalevskaya for non-characteristic hypersurfaces] Suppose \(\Sigma \sse \R^n\) is a hypersurface, with normal \(\gamma\), and consider the PDE \cref{eq:surf-pde} as above. Suppose \(a_\alpha, a_0\) are real analytic near \(x_0 \in \Sigma\), and \(\Sigma\) is non-characteristic near \(x_0\). Then there exists a unique real analytic solution in a neighbourhood of \(x_0\).
\end{theorem}

\subsection{Characteristic surfaces}

Consider the linear operator
\[L = \sum_{i, j=  1}^n a_{ij}\pdv[2]{}{x_i}{x_j}\]
where \(a_{ij} \in \R\). Without loss of generality, we can assume \(a_{ij} = a_{ji}\). Consider the PDE problem
\begin{equation}
    \label{eq:plane-pde}
    \begin{cases}
        Lu = f & \\
        u = \gamma^i\partial_i u = 0 & \text{ on }\Pi_\gamma = \left\{x \mid x \vdot \gamma = 0\right\}
    \end{cases}
\end{equation}
That is, the boundary conditions are on the plane with unit normal \(\gamma\). In particular, \(\Pi_\gamma\) is non-characteristic for \cref{eq:plane-pde} if
\[\sum_{i, j= 1}^n a_{ij}\gamma^i\gamma^j \ne 0\]

We would like to find non-characteristic \(\Pi_\gamma\). Note that the left hand side is just \(\inner{A\gamma, \gamma}\), where \(A = (a_{ij})\) is a symmetric matrix, with the usual Euclidean inner product. In particular, \(A\) is diagonalisable, say \(A = P^\T\Lambda P\), where \(P\) is orthogonal and \(\Lambda\) is diagonal. Then
\[\inner{A\gamma, \gamma} = \inner{P^\T\Lambda P\gamma, \gamma} = \inner{\Lambda v, v}\]
where \(v = P\gamma\). If \(\{\lambda_i\}\) are the eigenvalues for \(A\), then the non-characteristic condition becomes
\[\sum_{i=1}^n \lambda_i(v_i)^2 \ne 0\]

\begin{example}
    [Laplacian]
    \[L = \Delta = \sum_{i=1}^n \partial^2_{1,1}\]
    gives
    \[A = \begin{pmatrix}
        1 \\
        &\ddots \\
        & & 1
    \end{pmatrix}\]
    The Laplacian is an elliptic operator.
\end{example}

\begin{example}
    [Wave equation]
    \[L = \partial_t^2 + \Delta\]
    gives
    \[A = \begin{pmatrix}
        1 \\
        & \ddots \\
        && 1 \\
        &&&-1
    \end{pmatrix}\]
\end{example}

\textbf{Case 1: All eigenvalues have the same sign (and are all non-zero)}. Since \(v\) is a unit vector, the characteristic condition is impossible. That is, there are no characteristic hyperplanes \(\Pi_\gamma\). In this case, we call \(L\) an \emph{elliptic operator}\index{elliptic operator}.

\textbf{Case 2: Say \(\lambda_n < 0\) and \(\lambda_j > 0\) for \(j \ne n\) (or vice versa).} In this case, we call \(L\) an \emph{hyperbolic operator}\index{hyperbolic operator}. In particular, the characteristic condition becomes
\[\sum \lambda_i v_i^2 = 0 \iff \lambda_1v_1^2 + \sum_{j=2}^n\lambda_jv_j^2 = 0\]
Considering the wave equation again, we have the condition that
\[v_n^2 = \sum_{j=1}^{n-1}v_j^2\]
subject to the condition
\[\norm{v} = 1\]

\begin{center}
    \includegraphics[scale=0.1]{diagrams/2023_10_14 (1).jpg}
\end{center}

\textbf{Note that these cases are \emph{not} exhaustive.}

Now we want to different features of elliptic and hyperbolic operators. We will forget about boundary conditions, and look for solutions of the form
\[u(x) = e^{ik \vdot x}\]
for \(k \in \R^n\). We are looking for \emph{wave-like}\index{wave-like solutions} solutions. Substituting,
\[L(e^{ik\vdot x}) = e^{ik\vdot x} = \sum_{j, \ell}a_{j\ell}k_jk_\ell\]
We would like to consider \(Lu = 0\). Taking \(k = c\gamma\), \(\norm{\gamma} = 1\), then the condition is equivalent to
\[c^2\sum a_{j\ell}\gamma_j\gamma_\ell = 0\]

If \(L\) is elliptic, then the only solution is when \(k = 0\). That is, there are no wave-like solutions. On the other hand, if \(L\) is hyperbolic, then we can have wave like solutions, that is,
\[\sum a_{ij}\gamma_i\gamma_j = 0\]
has a solution with \(\norm{\gamma} = 1\). In this case, we have
\[u(x) = e^{i\lambda \gamma \vdot x}\]
gives an infinite family of solutions, indexed by \(\lambda \in \R\).

As we take \(\abs{\lambda} \to \infty\), we see that \(u'(x)\) can grow large. In particular, solutions can be rough.

\begin{center}
    \includegraphics[scale=0.1]{diagrams/2023_10_14 (2).jpg}
\end{center}

By contrast, we will see that solutions to elliptic equations are smooth.

\lec

\begin{example}
    Consider the IVP for Laplace's equation
    \begin{align*}
        \begin{cases}
            u_{xx} + u_{yy} = 0 \\
            u(x, 0) = \varphi(x) \\
             \partial_y u(x, 0) = 0
        \end{cases}
    \end{align*}
    Is this problem well-posed? If \(\varphi(x) = 0\), then \(0\) is a solution. On the other hand, we don't have Cauchy stability. Consider
    \[u_k(x, y) = e^{-\sqrt k}\cos(kx)\cosh(ky)\]
    See typed notes for more details.
\end{example}

\section{Sobolev spaces}

\subsection{H\"older spaces \(C^{k, \gamma}\)}

Let \(U \sse \R^n\) be open, \(k \in \N\). 

\begin{definition}
    [\(C^k\) spaces]
    Define\index{\(C^k(U)\)}
    \[C^k(U) = \left\{f : U \to \R \mid u\text{ is }k\text{ times continuously differentiable}\right\}\]
    
    and define\index{\(C^k(\overline U)\)}
    \[C^k(\overline U) = \left\{u \in C^k(U) \mid u \text{ and its derivatives are bounded and uniformly continuous on }U\right\}\]

    We will define the norm
    \[\norm{u}_{C^k(\overline U)} = \sum_{\abs{\alpha} \le k}\sup_{x \in U}\abs{D^\alpha u(x)}\]
\end{definition}

The idea is that \(C^k(\overline U)\) is the space of functions which can be extended continuously to \(\partial U\). Note that this is is contained in, but not equal to
\[u : \overline U \to \R \text{ such that }u \text{ and its derivatives are continuous}\]

On examples sheet 2, we will show that \(C^k(\overline U)\) is a Banach space.

\begin{definition}
    [H\"older continuous]
    We say \(u : U \to \R\) is \emph{H\"older continuous of index \(\gamma\)}\index{H\"older continuous} with \(0 < \gamma \le 1\) if there exists a constant \(C > 0\), such that
    \[\abs{u(x) - u(y)} \le C\abs{x-y}^\gamma\]
    for all \(x, y \in U\).

    If \(\gamma = 1\), then we say that \(u\) is \emph{Lipschitz continuous}\index{Lipschitz continuous}.
\end{definition}

\begin{remark}
    If \(\gamma > 1\), and \(u\) is H\"older continuous of index \(\gamma\), then \(u\) is constant.
\end{remark}

\begin{definition}
    [\(0\)-H\"older space]
    For \(\gamma \in \Ioc{0, 1}\), we define the \emph{\(0\)-H\"older space}\index{H\"older space!\(C^{0, \gamma}\)}:
    \[C^{0, \gamma}(\overline U) = \left\{u \in C^0(\overline U) \ \big\vert\ u \text{ is }\gamma\text{-H\"older continuous}\right\}\]
    Define the \(\gamma\)-H\"older seminorm by
    \[[u]_{C^{0, \gamma}(\overline U)} = \sup_{x, y \in U}\frac{\abs{u(x) -  u(y)}}{\abs{x - y}}\]
    That is, the smallest \(C\) such that \(u\) is \(\gamma\)-H\"older continuous. Since constant functions vanish, we add the \(C^0\) norm, and define
    \[\norm{u}_{C^{0, \gamma}(\overline U)} = \norm{u}_{C^0(\overline U)} + [u]_{C^{0, \gamma}(\overline U)}\]
\end{definition}

The space \(C^{0, \gamma}\), with the H\"older norm \(\norm{\cdot}_{C^{0, \gamma}(\overline U)}\) is a Banach space. We can extend this to higher order, that is,

\begin{definition}
    [\(k\)-th H\"older space]
    Define the \emph{\(k\)-th H\"older space}\index{H\"older space!\(C^{k, \gamma}\)}
    \[C^{k, \gamma}(\overline U) = \left\{u \in C^k(\overline U) \mid D^\alpha u \in C^{0, \gamma}(\overline U) \text{ for all }\abs{\alpha} = k\right\}\]
    with norm
    \[\norm{u}_{C^{k, \gamma}(\overline U)} = \norm{u}_{C^k(\overline U)} + \sum_{\abs{\alpha} = k}[D^\alpha u]_{C^{0, \gamma}(\overline U)}\]
\end{definition}

As above, \(C^{k, \gamma}\) is a Banach space with the H\"older norm.

\subsection{The Lebesgue spaces}


\begin{definition}
    [\(L^p\) space]
    Let \(U \sse \R^n\) be open, and suppose \(1 \le p \le \infty\), define \index{Lebsgue space!\(L^p\)}
    \[L^p(U) = \frac{\left\{f : U \to \R \mid f \text{ measurable, and with} \norm{u}_{L^p(U)} < \infty\right\}}{\sim}\]
    where
    \[\norm{u}_{L^p} = \begin{cases}
        \left(\int_U \abs{u(x)}^p\dd x\right)^{1/p} & 1 \le p < \infty \\
        \esssup_{x \in U}\abs{u(x)} = \inf\left\{C > 0 \mid \abs{u(x)} \le C \text{ a.e.}\right\} & p = \infty
    \end{cases}\]
    and \(u \sim v\) if \(u = v\) almost everywhere.
\end{definition}

\(L^p(U)\) with the \(\norm{\cdot}_{L^p}\) norm is a Banach space. We also define local versions,

\begin{definition}
    [\(L^p_\loc\) space]
    We say that \(u \in L^p_\loc(U)\)\index{Lebesgue space!\(L^p_\loc\)} if \(f \in L^p(V)\) for every \(V \Subset U\) (\(V \sse K \sse U\), where \(K\) is compact). Equivalently,
    \[L^p_\loc(U) = \bigcap_{V \Subset U}L^p(V)\]
\end{definition}

Note that \(L^p_\loc(U)\) is not Banach, on the other hand, it is a Fr\'echet space.

\begin{remark}
    If \(K \sse U\) is compact, \(U\) is open, then
    \[d(K, \partial U) = \inf\{\abs{x - y} \mid x \in K, y \in \R^n \setminus U\} > 0\]
    We will use the space outside \(K\) as a ``buffer zone''.
\end{remark}

\subsection{Weak derivatives}

That is, a notion of derivative for \(L^p\).

\begin{definition}
    [weak derivative]
    Suppose \(u, v \in L^1_\loc(U)\), \(\alpha\) a multi-index. We say \(v\) is the \emph{\(\alpha\)-th weak derivative of \(U\)}\index{weak derivative} if
    \[\int_U u D^\alpha \phi \dd x = (-1)^{\abs{\alpha}}\int_Uv\phi\dd x\]
    for all \(\phi \in C_c^\infty(U)\). We will also call the space \(C_c^\infty(U)\) the space of \emph{test functions}\index{test function}.
\end{definition}

\begin{remark}
    \begin{enumerate}
        \item since \(\Supp(D^\alpha\phi)\) and \(\Supp(\phi)\) are compact, the integrals are finite,
        \item \(u, v\) obey the correct integration by parts formula
    \end{enumerate}
\end{remark}

\begin{example}
    \(u(x) = \abs{x}\) is not differentiable at \(x = 0\), but it is weakly differentiable with \(v(x) = \sign(x)\).
\end{example}

\begin{lemma}
    [uniqueness of weak derivative] Suppose \(v, \tilde v \in L^1_\loc(U)\) are both the \(\alpha\)-th weak derivative of \(u \in L^1_\loc(U)\), then \(v = \tilde v\) almost everywhere.
\end{lemma}

\begin{proof}
    For all \(\varphi \in C_c^\infty(U)\),
    \[\int_U v\phi \dd x = (-1)^\abs{\alpha}\int_U u D^\alpha \phi \dd x = \int_U\overline v \phi \dd x\]
    Thus,
    \[\int_U (v - \overline v)\phi \dd x = 0\]
    for all \(\phi \in C_c^\infty(U)\). Taking \(\phi\) to be a smooth approximation to an indicator, we get the required result.
\end{proof}

Suppose \(u\) is smooth, then the weak derivative agrees with the usual derivative almost everywhere.

\begin{notation}
    We will write \(v = D^\alpha u\).
\end{notation}

\begin{definition}
    [Sobolev space] Define the \emph{Sobolev space}\index{Sobolev space!\(W^{k, p}\)}
    \[W^{k, p}(U) = \left\{u \in L^1_\loc(U) \mid u \in L^p(U) \text{, the weak derivatives }D^\alpha u\text{ exists for }\abs{\alpha} \le k\text{, }D^\alpha u \in L^p(U)\right\}\]
    with the \emph{Sobolev norm}
    \[\norm{u}_{W^{k,p}(U)} = \begin{cases}
        \left(\sum_{\abs{\alpha} = k}\int_U\abs{D^\alpha u}^p \dd x\right)^{1/p} & p < \infty \\
        \sum_{\abs{\alpha} \le k}\esssup_U \abs{D^\alpha u} & p = \infty
    \end{cases}\]
    When \(p = 2\), we write \(H^k = W^{k, 2}\)\index{Sobolev space!\(H^k\)}.
\end{definition}

\begin{definition}
    [\(W_0^{k, p}\)]
    We denote by \(W^{k, p}_0(U)\)\index{Sobolev space!\(W^{k, p}_0\)} the completion of \(C_c^\infty(U)\) with respect to the \(W^{k,p}\) norm. Analogously, we define \(H^k_0 = W^{k, 2}_0(U)\)\index{Sobolev space!\(H^k_0\)}.
\end{definition}

The \(_0\) denotes that the function vanishes on the boundary.

\lec

\begin{example}
    Let \(n > 2, \lambda > 0\), and take \(U = B_1(0) \sse \R^n\) the open ball. Consider
    \[u(x) = \begin{cases}
        \abs{x}^{-\lambda} & \text{ for }x \ne 0 \\
        \text{anything} & x = 0
    \end{cases}\]
    When is \(u \in W^{p, 1}(U)\)?

    First of all, we compute
    \[\int_U\frac{1}{\abs{x}^\lambda}\dd x = C\int_0^1 r^{n-1-\lambda}\dd r\]
    which is finite if and only if \(\lambda < n\). Moreover, \(u \in L^p(U)\) if and only if \(\lambda p < n\).

    Let \(\phi \in C_c^\infty(B_1(0) \setminus \{0\})\), if \(u\) has a weak derivative \(v\), then
    \[v_i = D_iu = -\frac{\lambda x_i}{\abs{x}^{\lambda + 2}}\]
    on \(B_1(0)\setminus 0\). Thus,
    \[\abs{D u} = \frac{\abs{\lambda}}{\abs{x}^{\lambda + 1}}\]
    Hence \(v_i \in L^1\loc(U)\) if and only if \(\lambda + 1 < n\). Suppose \(\lambda + 1 < n\), then we claim that
    \begin{align*}
        v_i = \begin{cases}
            -\frac{\lambda x_i}{\abs{x}^{\lambda + 1}} & x \ne 0 \\
            \text{anything} & x = 0
        \end{cases}
    \end{align*}
    is a weak derivative of \(u\) on \(U\).

    For \(\phi \in C_c^\infty(U)\), by Stokes' theorem,
    \[(-1)\int_{U \setminus B_\epsilon(0)}u \phi_{x_i}\dd x = \int_{U\setminus B_\epsilon(0)}D_iu \phi \dd x - \int_{\partial B_\epsilon(0)} u \phi n \vdot \dd S\]
    Therefore, we can estimate
    \[\abs{\int_{\partial B_\epsilon}u \phi n \vdot \dd S} = \abs{\phi}_{L^\infty}\abs{\int_{\partial B_\epsilon}\epsilon^{-\lambda}n \vdot \dd S} \le C \epsilon^{n-1-\lambda} \to 0\]
    as \(\lambda \to 0\). Thus, by the dominated convergence theorem,
    \[-\int_U u \phi_{x_i}\dd x = \int_U v_i\phi \dd x\]
\end{example}

\begin{remark}
    \begin{enumerate}
        \item Weak derivatives can exist even when the function is not continuous.
        \item Since \(D_iu \in L^p(U)\) if and only if \(p(\lambda + 1) < n\), we see that
        \[u \in W^{1, p}(U) \iff \lambda < \frac{n}{p} - 1\]
        and if \(p > n\), we see that \(\lambda\) must be negative, and so it is continuous.

        Heuristically, larger \(p\) gives us nicer functions.
    \end{enumerate}
\end{remark}

\begin{theorem}
    \(W^{k, p}(U)\) is a Banach space for \(k \in \N, 1 \le p \le \infty\).
\end{theorem}

\begin{proof}
    First, we need to show that it is a normed vector space. This is straightforward, and for the triangle inequality we will need to use Minkowski's inequality
    \[\left(\sum_{i=1}^\ell (a_i + b_i)^p\right)^{1/p} \le \left(\sum_{i=1}^n a_i^p\right)^{1/p} + \left(\sum_{i=1}^n b_i^p\right)^{1/p}\]

    For completeness, we will use the completeness of \(L^p\). Let \((u_j)\) be a Cauchy sequence on \(W^{k, p}(U)\).

    Note that \(\norm{D^\alpha v}_{L^p(U)} \le \norm{v}_{W^{k, p}(U)}\) for \(\abs{\alpha} \le k\). Setting \(v = u_j\), we see that \((D^\alpha u_j)_j\) is a Cauchy sequence in \(L^p\). But we know that \(L^p\) is complete, and so there exists a function \(u^\alpha \in L^p(U)\), with \(D^\alpha u_j \to u^\alpha\) in \(L^p\) for all \(\abs{\alpha}\le k\). We will set \(u = u^{(0, \dots, 0)}\).

    \begin{claim}
        \(u^\alpha\) is the \(\alpha\)-th weak derivative of \(u\). That is, \(D^\alpha u\) exists and \(D^\alpha u = u^\alpha\).
    \end{claim}

    \begin{proof}
        Choose a test function \(\phi \in C_c^\infty(U)\). Since \(u_j \in W^{k, p}\), we know that \(D^\alpha u_j\) exists, and
        \[(-1)^{\abs{\alpha}}\int_U u_j D^\alpha \phi \dd x = \int_U(D^\alpha u_j)\phi \dd x\]
        for all \(j\). Taking \(j \to \infty\), using the fact that \(D^\alpha u_j \to u^\alpha\) (using H\"older, or the dominated convergence theorem), we get that
        \[(-1)^{\abs{\alpha}}\int_U uD^\alpha \phi \dd x = \int_U u^\alpha \phi \dd x\]
        and so \(D^\alpha u = u^\alpha \in L^p(U)\).
    \end{proof}

    Thus, \(u \in W^{k,p}(U)\).
\end{proof}

\subsection{Approximations of Sobolev spaces}

\subsubsection*{Convolution and mollifiers}

\begin{definition}
    [standard mollifier]
    Let
    \[\eta(x) = \begin{cases}
        C \exp(\frac{-1}{1-\abs{x}^2}) & \abs{x} < 1 \\
        0 & \abs{x} \ge 1
    \end{cases}\]
    where \(C\) is chosen such that
    \[\int_{\R^n}\eta(x)\dd x = 1\]
    For \(\epsilon > 0\), we denote
    \[\eta_\epsilon(x) = \frac{1}{\epsilon^n}\eta(x/\epsilon)\]
    We call \(\eta_\epsilon\) to be the \emph{standard mollifer}\index{standard mollifier}.
\end{definition}

\begin{remark}
    \begin{itemize}
        \item \(\eta_\epsilon \in C_c^\infty(\R^n)\),
        \item \(\Supp(\eta_\epsilon) = \overline{B_\epsilon(0)}\),
        \item \(\int_{\R^n}\eta_\epsilon(x)\dd x = 1\)
    \end{itemize}
\end{remark}

\begin{definition}
    Given \(U \sse \R^n\) open, define
    \[U_\epsilon = \left\{x \in U \mid d(x, \partial U) > \epsilon\right\}\]
\end{definition}

\begin{definition}
    [mollification]
    Given \(f \in L^1_\loc(U)\), the \emph{mollification of \(f\)}\index{mollification} is
    \begin{align*}
        f_\epsilon : U_\epsilon &\to \R \\
        f_\epsilon(x) &= \eta_\epsilon * f(x) = \int_{U}\eta_\epsilon(x-y)f(y)\dd y = \int_{B_\epsilon(0)}\eta_\epsilon(y)f(x-y) \dd y
    \end{align*}
    where \(*\) denotes the convolution.
\end{definition}

We can think of \(f_\epsilon(x)\) as the average of \(f\) in an \(\epsilon\)-ball, weighted by \(\eta\).

\begin{theorem}
    [properties of mollification]
    \index{mollification!properties of}
    Let \(f \in L^1_\loc(U)\), then
    \begin{enumerate}
        \item \(f_\epsilon \in C^\infty(U_\epsilon)\),
        \item \(f_\epsilon \to f\) a.e. on \(U\) as \(\epsilon\to 0\),
        \item if \(f \in C^0(U)\), then \(f_\epsilon \to f\) locally uniformly (i.e. uniformly on \(K \sse U\) compact).
        \item if \(1 \le p < \infty\), and \(f \in L^p_\loc(U)\), then \(f_\epsilon \to f\) in \(L^p_\loc(U)\). That is,
        \[\norm{f_\epsilon - f}_{L^p(V)} \to 0\]
        for all \(V \Subset U\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    See handout on moodle.
\end{proof}

In particular, there is a big improvement going from \(f \in L^1_\loc\) to \(f_\epsilon \in C^\infty\).

\begin{lemma}
    [local smooth approximation of Sobolev functions away from \(\partial U\)] \index{Sobolev approximation!local smooth approximation away from \(\partial U\)}\label{lem:local-smooth-approx-away-from-boundary}Let \(u \in W^{k, p}(U)\) for some \(1 \le p < \infty\). Set \(u_\epsilon = \eta_\epsilon * u\) in \(U_\epsilon\). Then
    \begin{enumerate}
        \item \(u_\epsilon \in C^\infty(U_\epsilon)\) for all \(\epsilon > 0\),
        \item \(u_\epsilon \to u\) in \(W^{k, p}_\loc(U)\). Note that for \(V \Subset U\), \(V \subseteq U_\epsilon\) for \(\epsilon\) sufficiently small.
    \end{enumerate}
\end{lemma}

\begin{proof}
    (i) follows from the theorem. For (ii),
    \begin{claim}
        \[D^\alpha u_\epsilon = D^\alpha(\eta_\epsilon * u) = \eta_\epsilon * D^\alpha u\]
    \end{claim}
    \begin{proof}
        Since \(u_\epsilon \in C^\infty\), we can compute the classical derivative as follows:
        \begin{align*}
            D^\alpha_x u_\epsilon(x) &= D^\alpha_x\int_U \eta_\epsilon(x - y)u(y) \dd y \\
            &= \int_U (D^\alpha_x\eta_\epsilon(x - y))u(y) \dd y \\
            &= (-1)^\abs{\alpha}\int_U(D^\alpha_u\eta_\epsilon(x - y))u(y) \dd y \\
            &= \int_U \eta_\epsilon(x - y)D^\alpha u(y) \dd y \\
            &= (\eta_\epsilon * u)(x)
        \end{align*}
        See handout for justification of swapping the integral and derivative.
    \end{proof}
    Note for \(V \Subset U\), by (iv) of the theorem, since \(D^\alpha u \in L^p(U)\), then
    \[D^\alpha u_\epsilon = \eta_\epsilon * D^\alpha u \to D^\alpha u\]
    in \(L^p(V)\) as \(\epsilon \to 0\). Thus, for all \(V \Subset U\), and \(\delta > 0\), there exists \(\epsilon_0 = \epsilon_0(\delta, V)\) such that
    \begin{align*}
        \norm{u_\epsilon - u}_{W^{k, p}(V)}^p = \sum_{\abs{\alpha} \le k}\norm{D^\alpha u_\epsilon - D^\alpha u}_{L^p(V)}^p \le \delta
    \end{align*}
    for \(0 < \epsilon \le \epsilon_0\).
\end{proof}

In conclusion, \(u \in W^{k, p}(U)\) can be approximated by \(C^\infty\) functions away from \(\partial U\).

\lec

\begin{theorem}
    [global Sobolev approximation globally away from \(\partial U\)]
    \index{Sobolev approximation!global Sobolev approximation globally away from \(\partial U\)}\label{thm:global-sobolev-approx-away-from-boundary}
    Suppose \(U \sse \R^n\) open bounded, and suppose \(u \in W^{k, p}(U)\), for \(1 \le p < \infty\). Then there exists a sequence \((u_j) \in C^\infty \cap W^{k, p}(U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\).
\end{theorem}

\underline{Exercise:} Drop the assumption that \(U\) is bounded.

\begin{remark}
    Note that we don't assume \(u \in C^\infty(\overline U)\).
\end{remark}

\begin{proof}
    \textbf{Step 1:} We have
    \[U = \bigcup_{j=1}^\infty\]
    where
    \[U_j = \left\{x \in U \mid d(x, \partial U) > 1/j\right\}\]
    and define \(V_j = U_{j+3} \setminus \overline{U_{j+1}} \Subset U\). Choose \(V_0 \Subset U\) such that \(U = \bigcup_{j=0}^\infty V_j\). Note in particular only the consecutive \(V_j\) intersect.

    Let \(\xi_j\) be a \emph{partition of unity subordinate to \(V_j\)}. That is,
    \begin{itemize}
        \item \(0 \le \xi_j \le 1\),
        \item \(\xi_j \in C_c^\infty(V_j)\),
        \item \(\sum_{j=0}^\infty \xi_j(x) = 1\) for all \(x \in U\). Note at any point at most two \(\xi_j\) are non-zero.
    \end{itemize}

    Given \(u \in W^{k, p}(U)\), then we see that \(\xi_j u \in W^{k, p}(U)\), and \(\Supp(\xi_j u)\in V_j\).

    \textbf{Step 2:} We would like to smooth out the split up function. Let \(W_j = U_{j+4} \setminus \overline{U_j} \supseteq V_j\). Let
    \[u_j = \eta_{\epsilon_j} * (\xi_ju)\]
    Fix \(\delta > 0\), for each \(j \ge 1\), we can choose \(\epsilon_j\) sufficiently small such that \(\Supp(u_j) \sse W_j\).

    By \cref{lem:local-smooth-approx-away-from-boundary}, we have that \(u_j \to \xi_j u\) in \(W^{k, p}(W_j)\). With this, we can make
    \[\norm{u_j -  \xi_j u}_{W^{k, p}(U)} = \norm{u_j - \xi_j u}_{W^{k, p}(U_j)} \le \frac{\delta}{2^{j+1}}\]

    Summing everything together, let \(v = \sum_{j=0}^\infty u_j\). Note that each \(u_j\) is non-zero on finitely many \(W_j\), and so at each point it is a finite sum. With this, \(v\) is smooth. Also note that \(u = \sum_j \xi_j u\) on \(U\), and so for any \(V \Subset U\),
    \[\norm{v - u}_{W^{k,p}(V)} \le \sum_{j=1}^\infty \norm{u_j - \xi_j u}_{W^{k, p}(V)} \le \delta \sum_{j=1}^\infty \frac{1}{2^{j+1}} = \delta\]
    where we applied the triangle inequality. Since \(\delta\) is independent of \(V\),  taking the \(\sup\) over all \(V \Subset U\), we get that
    \[\norm{v - u}_{W^{k, p}(U)} \le \delta\]
\end{proof}

\underline{Question:} Can we approximate \(u \in W^{k, p}(U)\) by \(u \in C^\infty(\overline U)\)?

The issue here is that \(\partial U\) could be a problem. For example, we can consider \(\partial U\) to be the Cantor set.

\begin{definition}
    [\(C^{k, \delta}\)-domain]
    Suppose \(U \sse \R^n\) is bounded and open. Then we say that \(\partial U\) is a \emph{\(\C^{k, \delta}\)-domain}\index{\(C^{k, \delta}\) domain} if for every \(p \in \partial U\), there exists \(r > 0\), and a function \(\gamma : \R^{n-1} \to \R\), with \(\gamma \in C^{k,\delta}(\R^{n-1})\), such that (after relabelling axes),
    \[U \cap B_r(p) = \left\{(x', x_n) \in B_(p) \mid x_n > \gamma(x')\right\}\]
\end{definition}

\begin{theorem}
    [smooth approximation of Sobolev functions up to \(\partial U\)]
    \index{Sobolev approximation!smooth approximation up to \(\partial U\)}\label{thm:smooth-approx-up-to-boundary}
    Let \(U \sse \R^n\) be open, bounded and \(\partial U\) a \(C^{0, 1}\) domain. Let \(u \in W^{k, p}(U)\), for some \(1 \le p < \infty\). Then there exists a sequence \((u_j) \in C^\infty(\overline U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\).
\end{theorem}

\begin{proof}
    \textbf{Step 1:} Fix \(x_0 \in \partial U\). Since \(\partial U\) is Lipschitz, there exists \(r > 0\) and a Lipschitz function \(\gamma : \R^{n-1} \to \R\), such that
    \[U \cap B_r(x_0) = \left\{x \in B_r(x_0) \mid x_n > \gamma(x')\right\}\]
    Let \(V = U \cap B_{r/2}(x_0)\).

    \textbf{Step 2:} Define the shifted point \(x^\epsilon = x + \lambda \epsilon e_n\), for \(x \in V, \epsilon > 0\).

    \begin{claim}
        For \(\lambda > 0\) large enough, \(B_\epsilon(x^\epsilon) \sse U \cap B_r(x_0)\) for all \(\epsilon > 0\).
    \end{claim}

    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_21.jpg}
    \end{center}

    That is, we need to show that for \(y \in B_\epsilon(x^\epsilon)\), \(y_n > \gamma(y')\). As \(\gamma\) is Lipschitz, there exists a constant \(L > 0\) such that
    \[\abs{\gamma(x') - \gamma(y')} \le L\abs{x' - y'}\]
    So we have that
    \[\abs{y' - (x^\epsilon)'} = \abs{y' - x'} < \epsilon\]
    and so,
    \[\gamma(y') \le \gamma(x') + L_\epsilon < x_n + L\epsilon\]
    by rearranging \(y_n > x_n^\epsilon - \epsilon = x_n + \lambda\epsilon - \epsilon = x_n + (\lambda - 1)\epsilon\), we see that
    \[y_n > \gamma(y')\]
    if \(\lambda \ge L + 1\).

    Define \(u_\epsilon(x) = u(x^\epsilon)\) for \(x \in V\). Set
    \[v_{\delta, \epsilon} = \eta_\delta * u_\epsilon\]
    for \(0 < \delta < \epsilon\). Then \(v_{\delta, \epsilon} \in C^\infty(\overline V)\). Fix \(\mu \ge 0\), then we note
    \[\norm{v_{\delta, \epsilon} - u}_{W^{k, p}(V)} \le \norm{v_{\delta, \epsilon} - u_\epsilon}_{W^{k, p}(V)} + \underbrace{\norm{u_\epsilon - u}_{W^{k, p}(V)}}_{\text{ translation is continuous on \(W^{k, p}\)}}\]
    We can choose \(\epsilon > 0\) such that the second term is at most \(\mu\). Fix \(\epsilon > 0\), we can choose \(\delta < \epsilon\) such that the first term is at most \(\mu\), using the same proof as in \cref{lem:local-smooth-approx-away-from-boundary}.

    \textbf{Step 3:} Let \(x_0\) vary over the boundary, then the \(V\)s which we get will cover the boundary, which is compact, and so we have a finite subcover. That is, finitely many points \(x_1, \dots, x_N \in \partial U\) and radii \(r_i\), where
    \[V_i = B_{r_i/2}(x_i) \cap U\]
    Choose \(V_0 \Subset U\) such that
    \[U = V_0 \cup V_0 \cup \cdots \cup V_N\]
    By step 2, we have \(v_i \in C^\infty(\overline{V_i})\), such that \(\norm{v_i - u}_{W^{k, p}(V_i)} \le \mu\). By \cref{lem:local-smooth-approx-away-from-boundary} there exists \(v_0 \in C^\infty(\overline{V_0})\) such that \(\norm{v_0 - u}_{W^{k, p}(V_0)} \le \mu\).

    \textbf{Step 4:} Summing everything together, using a partition of unity \(\xi_0, \dots, \xi_N\) subordinate to the open cover \(V_0, \dots, V_N\). Define
    \[v = \sum_{i=0}^n\xi_i v_i\]
    This sum is finite, and so \(v \in C^infty(\overline{U})\), and for \(\abs{\alpha} \le k\),
    \begin{align*}
        \norm{D^\alpha v - D^\alpha u}_{L^p(U)} &\le \sum_{i=0}^N\norm{D^\alpha(\xi_i(v_i - u))}_{L^p(V_i)} \\
        &\le C_k\sum_{i=0}^N\norm{v_i - u}_{W^{k, p}(V_i)} \\
        &= C_k(1+ N)\mu
    \end{align*}

    But \(\mu\) was arbitrary, and so we are done.
\end{proof}

\lec

To conclude, we consider some examples of functions:

\begin{itemize}
    \item \(\abs{x} \notin C^\infty(-1, 1)\), but it is in \(W^{1,1}(-1, 1)\).
    \item \(1/x\) is \(C^\infty\) and \(L^1_\loc\) on \((0, 1)\), but not \(C^\infty(\overline{(-1,1)})\) or \(W^{1, 1}\).
\end{itemize}
and so, \(C^\infty(U), C^\infty(\overline U) \not\subseteq W^{k,p}(U)\).

\subsection{Extensions and traces}

Suppose \(U \sse \R^n\) is open and bounded, \(u \in W^{k,p}(U)\). We would like to extend \(u\) to \(\overline{u} : \R^n \to \R\). What happens if we set
\begin{align*}
    \overline u = \begin{cases}
        u & \text{ on }U \\
        0 & \text{ on }U^c
    \end{cases}
\end{align*}
This is okay for \(L^p\), but not for \(W^{k, p}\) as the derivatives become an issue. Moreover, we can expect at most \(\overline u \in W^{k, p}(\R^n)\).


\begin{theorem}
    [Calderon, Stein] Suppose \(U \sse \R^n\) is open and bounded, and \(\partial U\) is \(C^1\). Choose \(V\) bounded in \(\R^n\), with \(U \Subset V\). Let \(1 \le p < \infty\). Then there exists a bounded linear operator
    \begin{align*}
        E : W^{1, p}(U) &\to W^{1, p}(\R^n) \\
        u &\mapsto \overline u
    \end{align*}
    such that for all \(u \in W^{1, p}(U)\),
    \begin{itemize}[(i)]
        \item \(\overline{u}\vert_U = u\) a.e.
        \item \(\Supp(E(u)) \sse V\),
        \item There exists a constant \(C\) depending only on \(U, V, p\), such that \[\norm{E(u)}_{W^{1,p}(\R^n)} \le C\norm{u}_{W^{1, p}(U)}\]
    \end{itemize}

    We call \(Eu\) the \emph{extension of \(U\) to \(R^n\)}\index{extension}.
\end{theorem}

\begin{proof}
    \textbf{Step 1:} Fix \(p \in \partial U\), and suppose \(\partial U\) is flat near \(p\). We may assume there exists \(r > 0\) such that
    \begin{align*}
        B^+ &= B_r(p) \cap \{x_n \ge 0\} \sse \overline{U} \\
        B^- &= B_r(p) \cap \{x_n < 0\} \sse \R^n \setminus \overline{U}
    \end{align*}
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_24 (1).jpg}
    \end{center}
    Suppose also that \(u \in C^1(\overline U)\).Denote \(x' = (x_1, \dots, x_{n-1})\). We define
    \[\overline{u}(x) = \begin{cases}
        u(x) & \text{ if }x \in B^+ \\
        -3u(x', -x_n) + 4u(x', -x_n/2) & \text{ if }x \in B^-
    \end{cases}\]
    which is called a \emph{higher order reflection of \(u\) from \(B^+\) to \(B^-\)}\index{higher order reflection}
    \begin{claim}
        \(\overline{u} \in C^1(B_r(p))\)
    \end{claim}
    \begin{proof}
        Clearly \(\overline{u}\) is continuous. Computing the derivatives:
        \[\partial_{x_n}\overline{u} = \begin{cases}
            \partial_{x_n}u(x) & x \in B^+ \\
            3\partial_{x_n}u(x', -x_n) - 2\partial_{x_n}u(x', -x_n/2) & x \in B^-
        \end{cases}\]
        Similarly,
        \[\partial_{x_i}\overline{u} = \begin{cases}
            \partial_{x_i}\overline(u) & x \in B^+ \\
            -3\partial_{x_i}u(x', -x_n) + 4\partial_{x_i}u(x', -x_n/2) & x \in B^-
        \end{cases}\]
        and so the derivative is continuous. 
    \end{proof}
    We can also check that the inequality holds in this case.

    \textbf{Step 2:} Suppose \(\partial U\) is not flat near \(p\). Since \(\partial U\) is \(C^1\), there exists \(r > 0\) and \(\gamma : \R^{n-1} \to \R\), such that
    \[U \cap B_r(p) = \left\{x \mid x_n > \gamma(x')\right\}\]
    
    Define
    \begin{align*}
        \Phi : \R^n &\to \R^n \\
        \Phi(x) &= (x_1, \dots, x_{n-1}, x_n - \gamma(x'))
    \end{align*}
    We can see that \(\Phi\) maps \(\partial U\) to \(\{y_n=0\}\), and it is invertible with \(C^1\) inverse
    \[\Psi(y) = (y_1, \dots, y_{n-1}, y_n + \gamma(y'))\]
    with
    \begin{itemize}
        \item \(\Phi(U \cap B_r(p)) \sse \{y_n > 0\}\)
        \item \(\det(D\Phi) = \det(D\Psi) = 1\)
    \end{itemize}
    \begin{center}
        \includegraphics[scale=0.15]{diagrams/2023_10_24 (2).jpg}
    \end{center}
    Moreover, there exists a neighbourhood \(W\) of \(p\), with \(\Phi(W) = B_s(\tilde p)\) for some \(s > 0\). In this case,
    \[\Phi(U \cap W) = B_s(\tilde p) \cap \{y_n > 0\} = B^+\]
    Define \(v(y) = u(\Psi(y))\) for \(y \in B_+\). Then \(v\) is \(C^1\), and so by step 1 there exists an extension \(\overline{v} \in C^1(B_s(\tilde p))\) with \(\overline v\vert_{B^+} = v\) and
    \[\norm{\overline v}_{W^{1, p}(B_s(\tilde p))} \le C\norm{v}_{W^{1, p}(B^+)}\]
    Define \(\overline u(x) = \overline{v}(\Phi(x))\), then \(\overline{u} \in C^1(W)\), and
    \[\norm{\overline u}_{W^{1, p}(W)} \le C\norm{u}_{W^{1, p}(U)}\]
    which we will see on examples sheet 1.

    \textbf{Step 3:} Now we have local extensions near all \(p \in \partial U\). We assume \(U\) is bounded, and so we have an open cover \(\{W_0, \dots, W_N\}\), with
    \[U \sse \bigcup_{i=0}^N W_i\]
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/2023_10_24 (3).jpg}
    \end{center}
    and we have extensions \(\overline u_i \in C^1(W_i)\). Let \((\xi_i)_{i=0}^N\) be a partition of unity subordinate to \(\{W_i\}\). Let
    \[\overline{u} = \sum_{i=0}^N \xi_i \overline{u_i}\]
    where \(\overline{u_0} = u\). Then \(\overline u\vert_U = u\) ae., and
    \[\overline{u} \in C_c^1(\R^n)\]
    with
    \[\norm{\overline u}_{W^{1, p}(\R^n)} \le C\norm{u}_{W^{1, p}(U)}\]

    We may assume \(\Supp(\overline u) \sse V\), since \(U \Subset V\), for example by using a cutoff function.

    \textbf{Step 4:} Given \(u \in W^{1, p}(U)\), by \cref{thm:smooth-approx-up-to-boundary}, there exists a sequence \((u_j) \in C^\infty(\overline U)\) with \(u_j \to u\) in \(W^{1,p}(U)\).

    \begin{claim}
        \((E(u_j))_j\) is a Cauchy sequence in \(W^{1, p}(\R^n)\).
    \end{claim}

    \begin{proof}
        By the previous steps, we have that \(E(u_j) \in W^{1, p}(\R^n)\). By linearity,
        \[\norm{E(u_j) - E(u_k)}_{W^{1, p}(\R^n)} = \norm{E(u_j - u_k)}_{W^{1, p}(\R^n)} \le C\norm{u_j - u_k}_{W^{1, p}(U)}\]
        But we know that \((u_j)\) is convergent, and thus Cauchy in \(W^{1, p}(U)\).
    \end{proof}
    Since \(W^{1, p}(\R^n)\) is complete, the sequence converges and we define
    \[E(u) = \lim_j E(u_j)\]
\end{proof}

\begin{remark}
    If \(\partial U\) is \(C^k\), then we have the extension operators
    \[E : W^{k, p}(U) \to W^{k, p}(\R^n)\]
    Given \(u \in C^k(\overline U)\), we set
    \[\overline{u} = \begin{cases}
        u(x) & x\in B_+ \\
        \sum_{j=1}^k c_ju(x', -x_n/j) & x \in B_-
    \end{cases}\] 
    To match at the boundary, we need
    \[\sum_{j=1}^kc_j\left(\frac{-1}{j}\right)^m = 1\]
    for all \(m = 0, \dots, k-1\).
\end{remark}

\lec

\subsubsection*{Traces}

If we have \(u \in C^0(\overline U)\), then \(u \vert_{\partial U}\) makes sense. But for \(u \in W^{k, p}(U)\), then \(u\vert_{\partial U}\) does not make sense, as \(\partial U\) has measure zero.

\begin{theorem}
    Let \(U\) be open bounded and \(\partial U\) is \(C^1\). Then there exists a bounded linear operator
    \[T : W^{1, p}(U) \to L^p(\partial U)\]
    called the \emph{trace of \(u\) on \(\partial U\)}\index{trace}, such that
    \begin{enumerate}[(i)]
        \item \(T(u) = u\vert_{\partial U}\) if \(u \in W^{1, p}(U) \cap C^0(\overline U)\),
        \item \(\norm{T(u)}_{L^p(\partial U)} \le C \norm{u}_{W^{1, p}(U)}\) for all \(u \in W^{k, p}(U)\), where \(C\) depends only on \(U, p\).
    \end{enumerate}
\end{theorem}

\begin{remark}
    We have \(u, Du \in L^p\), which is giving us the control of \(u\) on the boundary.
\end{remark}

\begin{proof}
    [Sketch proof] See examples sheet 2. Suppose \(u \in C^1(\overline U)\), and \(\partial U\) is flat near \(p\). Let
    \begin{align*}
        B^+ &= B_r(p) \cap \{x_n \ge 0\} \sse \overline U \\
        B^- &= B_r(p) \cap \{x_n < 0\} \sse \R^n \setminus U
    \end{align*}
    as before, and let \(\Gamma\) be the portion of \(\partial U\) within \(B_r(p)\). Choose \(\xi \in C_c^\infty(B_r(p))\) such that \(0 \le \xi \le 1\) on \(B_r(p)\), and \(\xi = 1\) on \(B_{r/2}(p)\). Then
    \begin{align*}
        \int_\Gamma \abs{u(x', 0)}^p \dd x' &\le \int_{B_r(p) \cap \{x_n = 0\}} \xi \abs{u(x', 0)}^p \dd x' \\
        &\underbrace{=}_{\text{FTC}} (-1)\int_{B^+}\partial x_n(\xi\abs{u}^p)\dd x_n \dd x' \\
        &= (-1)\int_{B^+}\abs{u^p}\partial_{x_n}\xi + p \abs{u}^{p-1}\sign(u)\partial_{x_n}u \xi \dd x \\
        &\underbrace{\le}_{\text{Young's inequality}} C_p\int_{B^+}\abs{u}^p + \abs{Du}^p \dd x \\
        &= C_p\norm{u}^p_{W^{1, p}(U)}
    \end{align*}

    In Sheet 2, we will extend to general \(\partial U\) using a partition unity, and the fact that it is compact. Then define
    \[T(u) = u\vert_{\partial U}\]
    for \(u \in C^1(\overline U)\), and we have that
    \[\norm{T(u)}_{L^p(\partial U)} \le C\norm{u}_{W^{1, p}(U)}\]
    Using density of \(C^\infty(\overline U)\) in \(W^{1, p}(U)\), we are done.
\end{proof}

\begin{remark}
    \begin{itemize}
        \item The map \(T\) above is not surjective, however in the case of \(T : \HH^s \to \HH^{s-1/2}\), it is surjective.
        \item Recall \(W^{k,p}_0(U)\) is the closure of \(C_c^\infty(U)\) in \(W^{k, p}(U)\). If \(u \in W^{k,p}_0(U)\), then there exists \((u_j) \sse C_c^\infty(U)\), such that \(u_j \to u\) in \(W^{k, p}(U)\). In particular,
        \[T(u) = T(\lim_j u_j) = \lim_j T(u_j) = 0\]
        In fact, the converse is true also. If \(T(u) = 0\), then \(u \in W^{1, p}_0(U)\).
        \item if \(u \in W^{k, p}(U)\), then we can define trace operators for \(Du, \dots, D^{k-1}(U)\).
    \end{itemize}
\end{remark}

\subsection{Sobolev inqualities}

In this case, the basic idea is that we can trade differentiability (measured by \(k\)) for integrability (measured by \(p\)). Note it does not work the other way. For example, if \(f' \in L^1(\R)\) then \(f \in L^\infty(\R)\), but the converse is not true.

The idea is that we will prove estimates of the form
\[\norm{u}_{L_q(\R^n)} \le C\norm{D u}_{L^p(\R^n)} \left(+ \norm{u}_{L^p(\R^n)}\right)\]

We have three cases:
\begin{enumerate}
    \item \(1 \le p < n\),
    \item \(p = n\),
    \item \(n < p \le \infty\).
\end{enumerate}

\subsubsection*{Case 1: \(1 \le p < n\)}

\begin{lemma}
    Let \(n \ge 2\), and \(f_1, \dots, f_n \in L^{n-1}(\R^{n-1})\). Then for \(1 \le i \le n\), define 
    \[\widetilde x_i = (x_1, \dots, x_{i-1}, x_{i+1}, \dots, x_n) \in \R^{n-1}\]
    and
    \[f(x) = \prod_{i=1}^n f_i(\widetilde x_i) : \R^n \to \R\]
    Then \(f \in L^1(\R^n)\), with
    \[\norm{f}_{L^1(\R^n)} \le \prod_{i=1}^n \norm{f_i}_{L^{n-1}(\R^{n-1})}\]
\end{lemma}

\begin{proof}
    We induct on \(n\). The case \(n = 2\) gives
    \[f(x_1, x_2) = f_1(x_1)f_2(x_2)\]
    But
    \begin{align*}
        \norm{f}_{L^1(\R^2)} &= \int_{\R^2}\abs{f_1(x_1)}\abs{f_2(x_2)}\dd x_1 \dd x_2 \\
        &= \int_\R \abs{f_1(x_1)}\dd x_1 \int \abs{f_2(x_2)}\dd x_2 \\
        &= \norm{f_1}_{L^1(\R)}\norm{f_2}_{L^1(\R)}
    \end{align*}
    Suppose the result is true for \(n\). Write
    \[F(x) = f_1(\widetilde x_1) \cdots f_n (\widetilde x_n)\]
    and so \(f(x) = F(x)f_{n+1}(\tilde x_{n+1})\). Fix \(x_{n+1}\) and integrate over \(x_1, \dots, x_n\):
    \begin{align*}
        \int_{\R^n}\abs{f(\xi_1, \dots, \xi_n, x_{n+1})} \dd \xi_1 \cdots \dd x_n &= \int_{\R^n}\abs{F(\xi, x_{n+1})} \abs{f_{n+1}(\xi)}\dd \xi \\
        &\underbrace{=}_{\text{H\"older}} \norm{F(\cdot, x_{n+1})}_{L^{n/(n-1)}(\R^n)}\norm{f_{n+1}}_{L^n(\R^n)}
    \end{align*}
    By the induction hypothesis, if \(q = n/(n-1)\), then
    \begin{align*}
        \norm{F(\cdot, x_{n+1})}_{L^{n/(n-1)}(\R^n)} &= \norm{F(\cdot, x_{n+1}^q)}^{1/q}_{L^1(\R^n)} \\
        &\le \prod_{i=1}^n \norm{f_i(\cdot, x_{n+1})^{n/(n-1)}}^{(n-1)/n}_{L^{n-1}(\R^n)} \\
        &= \prod_{i=1}^n \norm{f(\cdot, x_{n+1})}_{L^n(\R^{n-1})}
    \end{align*}
    Integrating over \(x_{n+1}\),
    \begin{align*}
        \norm{f}_{L1(\R^{n+1})} &\le \norm{f_{n+1}}_{L^n(\R^n)}\int_\R \prod_{i=1}^n \norm{f_i(\cdot, x_{n+1})}_{L^n(\R^{n-1})}\dd x_{n+1} \\
        &\le \norm{f_{n+1}}_{L^n(\R^n)}\prod_{i=1}^n\left(\int_\R \norm{f_i(\cdot, x_{n+1})}_{L^n(\R^{n-1})}^n \dd x_{n+1}\right)^{1/n} \\
        &= \prod_{i=1}^{n+1}\norm{f_i}_{L^n(\R^n)}
    \end{align*}
    Where we used the generalised H\"older inequality with \(p = n\), that is,
    \[\norm{\prod_i f_i}_{L^1} \le \prod_{i=1}^n \norm{f_i}_{L^{p_i}}\]
    where \(\sum \frac{1}{p_i} = 1\).
\end{proof}

\begin{theorem}
    [Gagliado-Nirenberg-Sobolev inequality]
    \label{thm:Gagliado-Nirenberg-Sobolev}
    \index{Gagliado-Nirenberg-Sobolev inequality}
    Suppose \(1\le p < n\), set
    \[p^* = \frac{np}{n-p}\]
    for the \emph{Sobolev conjugate of \(p\)}\index{Sobolev conjugate}. Then we have a continuous embedding
    \[W^{1, p}(\R^n) \sse L^{p^*}(\R^n)\]
    That is, there exists a constant \(C\) depending only on \(n, p\), such that
    \begin{equation}
        \label{eq:gns}
        \norm{f}_{L^{p^*}(\R^n)} \le C\norm{D u}_{L^q(\R^n)}
    \end{equation}
\end{theorem}

\begin{remark}
    \begin{enumerate}
        \item \(p^* > p\).
        \item Nothing is said about \(\norm{Du}_{L^{p^*}}\).
    \end{enumerate}
\end{remark}

\subsubsection*{Intuition}

Consider \(f : \R^2 \to \R\). \(L^p\) measures the width and the height of the function. For example, if we have
\[f_1 = A 1_W\]
then
\[\norm{f_1}_{L^p} = \abs{A}\vol(W)^{1/p} = \abs{A}V^{1/p}\]
Now consider \(\phi \in C_c^\infty(\R^n)\) with \(\phi(x) \le 1\), and let
\[f_2(x) = \phi(x)e^{i\omega \cdot x}\]
Then we know \(\abs{f_2}(x) \le 1\), and \(\Supp(f_2) \sse C\) is uniformly bounded in \(\omega\). With this,
\[\partial_1f_2 =  \phi'e^{i\omega \cdot x} + i\phi \omega_1e^{i\omega\cdot x}\]
Thus, \(Df_2\) is \emph{not} uniformly bounded in \(\omega\).

Next, consider
\[f_3(x) = \abs{\omega}^{-k}\phi(x)e^{i \omega \cdot x}\]
and so, we have a uniform bound in \(\omega\) of \(D^\ell f_3\) for \(\ell \le k\).

Finally, let
\[f_4(x) = A\phi(x/R)e^{i\omega \cdot x}\]
Then
\[\norm{f_4}^p_{W^{1, p}} \sim \int_{\abs{x} \le R}\abs{A\phi e^{i\omega \vdot x}}^p + \int_{\abs{x} \le R}\abs{\frac{A}{R}\phi'e^{i\omega \cdot x} + A\phi \omega e^{i\omega\vdot x}}^p \sim \abs{A}C^{1/p}\abs{\omega}\]
Now recall the uncertainty principle:
\[\delta_x \delta_p \ge \frac{\bar h}{2} > 0\]
Thus,
\[\text{volume} \times \text{frequency} \ge c > 0\]
Thus, a function with frequency \(\omega\) must be spread out on a ball of radius \(\ge 1/\omega\). Thus, the support must have measure \(\ge \omega^{-n}\). With this, \(\omega \ge V^{-n}\), and so
\[\norm{f}_{W^{1, p} \ge \abs{A}V^{1/p - 1/n}} = \abs{A}V^{p^*} = \norm{f}_{L^{p^*}}\]

See ``Tery Tao uncertainty principle'' for more details.

\lec

\begin{remark}
    \begin{itemize}
        \item If \(u \equiv 1\), then it wouldn't satisfy \cref{eq:gns}, and so it is essential that we are in \(W^{1, p}\).
        \item Proof follows from density of \(C_c^\infty(\R^n)\) in \(W^{k, p}_0(\R^n) = W^{k,p}(\R^n)\).
    \end{itemize}
\end{remark}

\begin{proof}
    \textbf{Step 1:} We can assume \(u \in C_c^\infty(\R^n)\), and first consider the case \(p = 1\). By FTC and compact support,
    \[u(x) = \int_{-\infty}^{x_i}\partial_{x_i}u(x_1, \dots, x_{i-1}, y_i, x_{i+1}, \dots, x_n)\dd y_i\]
    which means that
    \[\abs{u(x)} \le \underbrace{\int_{-\infty}^\infty \abs{Du(x_1, \dots, x_{i-1}, y_i, x_{i+1}, \dots, x_n)}}_{=f_i(\tilde x_i)\dd y_i}\]
    Thus,
    \[\abs{u(x)}^n \le f_1(\tilde x_1)\cdots f_n(\tilde x_n)\]
    Integrating over \(x \in \R^n\),
    \[\norm{\abs{u}}^{n/(n-1)}_{L^1(\R^n)} \le \norm{\prod_{i=1}^n \left(f_i(\tilde x_i)\right)^{1/(n-1)}}_{L^1{\R^n}} \le \prod_{i=1}^n \norm{f_i^{1/(n-1)}}_{L^{n-1}(\R^{n-1})} = \norm{Du}_{L^1(\R^n)}^{n/(n-1)}\]
    With this,
    \[\norm{u}_{L^{n/(n-1)}(\R^n)} \le \norm{Du}_{L^1(\R^n)}\]
    and in this case, \(p^* = n/(n-1)\). Since \(C_c^\infty(\R^n)\) is dense in \(W^{1, 1}(\R^n)\), result follows by density.
    
    \textbf{Step 2:} Now suppose \(p > 1\), consider \(v(x) = \abs{u(x)}^\gamma\). Then
    \[Dv = \gamma \sign(u)\abs{u}^{\gamma - 1}Du\]
    and
    \begin{align*}
        \left(\int_{\R^n}\abs{u}^{\gamma n/(n-1)} \dd x\right)^{(n-1)/n} &= \norm{\abs{u}^\gamma}_{L^{n/(n-1)}(\R^n)} \\
        &\le \norm{D(\abs{u}^\gamma)}_{L^1(\R^n)} \\
        &\le \gamma\int_{\R^n}\abs{u}^{\gamma - 1}\abs{D u}\dd x \\
        &\underbrace{\le}_{\text{H\"older}} \gamma \left(\abs{u}^{(\gamma - 1)p/(p-1)}\dd x\right)^{1 - \frac{1}{p}}\left(\int_{\R^n}\abs{D u}^p\right)^{1/p}
    \end{align*}
    Choose \(\gamma\) to match the exponents of \(u\) in the integrals, i.e.
    \[\gamma = \frac{p(n-1)}{n-p}\]
    In particular,
    \[\frac{\gamma n}{n-1} = p^*\]
    and so we have that
    \begin{align*}
        \left(\int_{\R^n}\abs{u}^{p^*}\right)^{(n-1)/n} \le \frac{p(n-1)}{n-p}\left(\int_{\R^n}\abs{u}^{p^*}\right)^{(p-1)/p}\norm{Du}_{L^p(\R^n)}
    \end{align*}
    Which then implies that
    \[\norm{u}_{L^{p^*}(\R^n)} \le \underbrace{\frac{p(n-1)}{n-p}}_{=: C}\norm{Du}_{L^p(\R^n)} \le C\norm{u}_{W^{1, p}(\R^n)}\]
    We can then conclude by density.
\end{proof}

Note in particular \(C \to \infty\) as \(p \to n\).

\begin{corollary}
    [GNS for \(U \sse \R^n\)]\index{Gagliado-Nirenberg-Sobolev inequality!for \(U \sse \R^n\)} Suppose \(U \sse \R^n\) is open and bounded, with \(C^1\) boundary. Let \(1 \le p < n\). If \(p^* = \frac{np}{n-p}\), then
    \[W^{1, p}(U) \sse L^{p^*}(U)\]
    and the embedding is continuous. That is, there exists \(C = C(U, p, n)\) such that
    \[\norm{u}_{L^{p^*}(U)} \le C\norm{u}_{W^{1, p}(U)}\]
    for all \(u \in W^{1, p}(U)\).
\end{corollary}

\begin{proof}
    Exercise. Use the extension theorem and the GNS inequality for \(\R^n\).
\end{proof}

\begin{corollary}
    [Poincar\'e inequality]\index{Poincar\'e inequality}Let \(U \sse \R^n\) be open and bounded. Suppose \(u \in W_0^{1, p}(U)\), for some \(1 \le p < n\). Then there exists a constant \(C = C(p, q, n, U)\) such that
    \[\norm{u}_{L^q(U)} \le C\norm{Du}_{L^p(U)}\]
    for each \(1 \le q \le p^*\). In particular, as \(1 \le p \le p^*\), we get
    \[\norm{u}_{L^p(U)} \le C\norm{Du}_{L^p(U)}\]
\end{corollary}

\begin{remark}
    \begin{enumerate}[(i)]
        \item On \(W^{1, p}(U)\) with \(U\) bounded, the \(W^{1, p}\) norm is equivalent to \(\norm{Du}_{L^p(U)}\).
        \item We do need that \(u \in W^{1, p}_0\), to kill off constant functions, which have derivative zero.
    \end{enumerate}
\end{remark}

\begin{proof}
    We will use that \(W^{1, p}_0(U)\) is the closure of \(C_c^\infty(U)\) under the \(W^{1, p}(U)\) norm. That is, given \(u \in W^{1, p}_0(U)\), there exists \(u_n \in C_c^\infty(U)\) such that \(\norm{u_n - u}_{W^{1, p}(U)} \to 0\). Since \(u_n\) is smooth, and vanishes near \(\partial U\), we can extend \(\overline{u_n} = 0\) on \(\R^n \setminus U\), with \(\overline{u_n} \in C_c^\infty(\R^n)\).

    Applying \cref{thm:Gagliado-Nirenberg-Sobolev},
    \[\norm{\overline{u_n}}_{L^{p^*}(\R^n)} \le C \norm{D\overline{u_n}}_{L^p(\R^n)}\]
    Sending \(n \to \infty\) and noting that \(\overline u\) vanishes on \(\R^n \setminus U\), we get the result for \(q = p^*\). In general, we use H\"older: since \(\abs{U} < \infty\),
    \[\norm{u}_{L^q(U)} \le C\norm{u}_{L^{p^*}(U)} \le C'\norm{Du}_{L^p(U)}\]  
\end{proof}

\subsubsection*{Case 2: \(p = n\)}

In this case, \(p^* \to \infty\), and so we may expect
\[\norm{u}_{L^\infty(U)} \le C\norm{u}_{W^{1, n}}\]
But this is false for \(n > 1\). One dimensional PDEs are boring, so we won't continue with this case.

\subsubsection*{Case 3: \(n < p < \infty\)}

We might expect in this case that it is ``better than \(L^\infty\)'', i.e. continuous.

\begin{theorem}
    [Morrey's inequality] \index{Morrey's inequality}Let \(n < p < \infty\), then there exists \(C = C(p, n)\) such that for \(u \in C_c^\infty(\R^n)\),
    \[\norm{u}_{C^{0, \gamma}(\R^n)} \le C \norm{u}_{W^{1, p}(\R^n)} \qqtext{for} \gamma = 1 - \frac{n}{p} < 1\]
    That is, we have an embedding
    \[W^{1, p}(\R^n) \inj C^{0, \gamma}(\R^n)\]
\end{theorem}

\begin{proof}
    Let \(Q\) be an open cube of side length \(r\), containing \(0\), and set
    \[\overline u = \frac{1}{\abs{Q}}\int_Q u(x)\dd x\]
    for the average of \(u\) over \(Q\). Then
    \[\abs{\overline u - u(0)} \le \frac{1}{\abs{Q}}\int_Q\abs{u(x) - u(0)}\dd x\]
    Since \(u \in C_c^\infty(\R^n)\), by the fundamental theorem of calculus and the chain rule,
    \begin{align*}
        u(x) - u(0) &= \int_0^1 \dv{t}(u(tx))\dd t \\
        &= \sum_{i=1}^n \int_0^1 x^i \pdv{u}{x^i}(tx)\dd t
    \end{align*}
    Thus,
    \[\abs{u(x) - u(0)} \le r\sum_{i=1}^n \int_0^1 \abs{\partial_{x_i}u(tx)}\dd t\]
    This then gives us that
    \begin{align*}
        \abs{\overline u - u(0)} &\le \frac{r}{\abs{Q}}\int_Q\int_0^1 \sum_{i=1}^n \abs{\partial_{x_i}u(tx)}\dd t \dd x \\
        &= \frac{r}{\abs{Q}}\int_0^1 t^{-n}\left(\sum_{i=1}^n \int_{tQ} \partial_{x_n}u(y)\dd y\right) \dd t \\
        &\le \frac{r}{\abs{Q}}\int_0^1 t^{-n}\left(\sum_i \norm{\partial_{x_i}u}_{L^p(tQ)}\abs{tQ}^{1/q}\right)\dd t \\
        &\le \frac{r}{r^n}\int_0^1 t^{-n}\norm{Du}_{L^p(\R^n)}t^{n/q}r^{n/q}\dd t \\
        &= \frac{r^{1-n/p}}{1 - n/p}\norm{Du}_{L^p(\R^n)}
    \end{align*}
    That is,
    \[\abs{\overline u - u(0)} \le \frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    By translation invariance,
    \[\abs{\overline u - u(x)} \le \frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    for all \(x \in Q\). Thus, by the triangle inequality,
    \[\abs{u(x) - u(y)} \le \abs{u(x) - \overline u} + \abs{\overline u - u(y)} \le 2\frac{r^\gamma}{\gamma}\norm{Du}_{L^p(\R^n)}\]
    for all \(x, y \in Q\). But for \(x, y \in Q\), there exists a cube \(Q\) of side length \(r = 2\abs{x - y}\) such that \(x, y \in Q\), which means that
    \[\frac{\abs{u(x) - u(y)}}{\abs{x - y}^\gamma} \le C \norm{Du}_{L^p(\R^n)}\]
    Note the left hand side is independent of \(r\), and so the inequality is true for all \(x, y \in \R^n\). Thus,
    \[[u]_{C^{0, \gamma}(\R^n)} \le C\norm{Du}_{L^p(\R^n)}\]
    Finally, we would like to control \(\sup_{x \in \R^n}\abs{u(x)}\), note that any \(x \in \R^n\) belongs to a cube with side length \(1\). In particular,
    \begin{align*}
        \abs{u(x)} &\le \abs{\overline u} + \abs{u(x) - \overline u} \\
        &\le \int_Q\abs{u(x)}\dd x + C \norm{Du}_{L^p} \\
        &\le \norm{u}_{L^p(\R^n)}\norm{1}_{L^q(Q)} + C\norm{Du}_{L^p(\R^n)} \\
        &\le C\norm{u}_{W^{1, p}(\R^n)}
    \end{align*}
    Note the constant is independent of the choice of \(x\). That is,
    \[\norm{u}_{C^{0, \gamma}(\R^n)} \le C\norm{u}_{W^{1, p}(\R^n)}\]
\end{proof}
