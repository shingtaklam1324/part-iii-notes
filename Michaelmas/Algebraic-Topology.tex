\section{Introduction}

Let \(X, Y\) be topological spaces, \(f, g : X \to Y\) be continuous functions. Note we will use the terminology `map := continuous function'. We say that \emph{\(f\) is homotopic to \(g\)}\index{homotopic}, written \(f \simeq g\), if there exists a map \(F : X \times I \to Y\), such that
\[F \vert_{X \times \{0\}}=  f \quad and \quad F \vert_{X \times \{1\}} = g\]
Here \(I = [0, 1]\) carries its Euclidean topology. An exercise (sheet 1) is that this defines an equivalence relation on the set of maps \(X \to Y\).

\begin{definition}
    [homotopy equivalence]
    We say that \(f : X \to Y\) is a \emph{homotopy equivalence}\index{homotopy equivalence} if there exists a function \(g : Y \to X\) such that \(f\circ g \simeq \id_Y\) and \(g \circ f \simeq \id_X\). We say that \(g\) is a \emph{homotopy inverse} to \(f\).
\end{definition}

The same exercise on sheet 1 shows that homotopy equivalence is an equivalence relation on topological spaces.

\begin{example}
    \begin{enumerate}
        \item If \(f : X \to Y\) is a homeomorphism, then \(f\) is also a homotopy equivalence.
        \item \(i : \{0\} \inj \R^n\) is a homotopy equivance.
        \item The inclusion \(i : S^{n-1} \inj \R^n \setminus 0\) is a homotopy equivalence. 
    \end{enumerate}
\end{example}

\begin{definition}
    A space \(X\) is \emph{contractible}\index{contractible} if it is homotopy equivalent to a point.
\end{definition}

\underline{Idea of Algebraic Topology:} Study spaces up to homotopy equivalence. Broadly, we are interested in ``connectivity properties'' of topological spaces.

\begin{example}
    \begin{enumerate}
        \item We say a space \(X\) is \emph{path-connected}\index{path connected} if any two maps \(\{*\} \to X\) are homotopic. For example, \(\R\) is path connected, \(\R \setminus \{0\}\) is not. A corollary of this is the intermediate value theorem.
        \item We say a path connected space \(X\) is \emph{simply connected}\index{simply connected} if every map \(f : S^1 \to X\) is homotopic to a constant map. Equivalently, every two maps \(S^1 \to X\) are homotopic. Or equivalently (again) every continuous map \(S^1 \to X\) extends to a continuous map \(D^2 \to X\). For example, \(\R^2\) is simply connected, by \(\R^2 \setminus \{0\}\) is not. If \(\gamma : S^1 \to \R^2 \setminus \{0\}\) is continuous, it has a \emph{winding number}, or \emph{degree} \(\deg(\gamma) \in \Z\), such that
        \begin{itemize}
            \item It is invariant under homotopy,
            \item If \(\gamma_n(t) = \exp(2n\pi it)\), then \(\deg(\gamma_n) = n\)
        \end{itemize}
        In particular, taking \(n = 0\) gives the constant map. Therefore, a constant map has degree \(0\). A corollary of this is the fundamental theorem of algebra.
        % Insert proof of FTA here
        \item We say a path connected space \(X\) is \emph{\(k\)-connected}\index{connected@\(k\)-connected}, if for all \(i \le k\), every map \(S^i \to X\) is homotopic to a constant map. In this case, \(\R^n\) is \((n-1)\)-connected, but \(\R^n \setminus 0\) is not. More precisely, maps from \(S^{n-1} \to \R^n \setminus 0\) have a degree \(\deg(f) \in Z\), such that degree is a homotopy invariant, and the degree of the inclusion map is \(1\), and the degree of the constant map is \(0\). A corollary of this is the Brouwer fixed point theorem.
        % Insert proof of Brouwer
    \end{enumerate}
\end{example}

\section{(Co)homology}

\subsection{Co(chain) complexes}

We will define invariants of topological spaces in two steps:
\begin{enumerate}[(a)]
    \item We associate to \(X\) a (co)chain complex,
    \item We take the (co)homology of the complex
\end{enumerate}
The topology enters in step (a), whereas (b) is just homological algebra.
\begin{definition}
    [chain complex]
    A \emph{chain complex}@\index{chain complex} \((C_*, \partial)\) consists of a sequence \(\{C_i\}_{i \in \Z}\) of abelian groups, along with \emph{boundary homomorphisms}\index{boundary homomorphism} \(\partial_n : C_n \to C_{n-1}\) for all \(n\), satisfying \(\partial_n \circ \partial_{n+1} = 0\). We also write this conditions as \(\partial^2 = 0\).
\end{definition}

\begin{definition}
    [homology group]
    The \emph{\(i\)-th homology group}\index{homology group} of a chain complex \((C_*, \partial)\) is
    \[\HH_i(C_*, \partial) = \frac{\ker(\partial_i : C_i \to C_{i-1})}{\im(\partial_{i+1} : C_{i+1} \to C_i)}\]
    We write
    \[\HH_*(C_*, \partial) = \bigoplus_{i \in \Z}\HH_i(C_*, \partial)\]
\end{definition}

\begin{definition}
    [cochain complex]
    A \emph{cochain complex}\index{cochain complex} \((C^*, \partial)\) consists of a sequence \(\{C^i\}_{i \in \Z}\) of abelian groups, along with \emph{boundary homomorphisms} \(\partial^n : C^n \to C^{n+1}\) for all \(n\), satisfying \(\partial^{n+1} \circ \partial^n = 0\). We also write this conditions as \(\partial^2 = 0\).
\end{definition}

\begin{definition}
    [cohomology group]
    The \emph{\(i\)-th cohomology group}\index{cohomology group} of a cochain complex \((C^*, \partial)\) is
    \[\HH^i(C^*, \partial) = \frac{\ker(\partial^i : C^i \to C^{i+1})}{\im(\partial^{i-1} : C^{i-1} \to C^i)}\]
    We write
    \[\HH^*(C^*, \partial) = \bigoplus_{i \in \Z}\HH_i(C^*, \partial)\]
\end{definition}

\begin{notation}
Elements of \(\ker(\partial)\) are called \emph{cycles}\index{cycle} (in a chain complex), \emph{cocycles}\index{cocycle} (in a cochain complex). Elements of \(\im(\partial)\) are called \emph{boundaries}\index{boundary} (and \emph{coboundary}\index{coboundary} resp.).

Elements of \(\HH_*\) are called \emph{homology classes}\index{homology class}, and elements of \(\HH^*\) are called \emph{cohomology classes}\index{cohomology class}.

We will call \(\partial\) the \emph{differential}\index{differential}.
\end{notation}

\begin{lemma}
    A chain map \(f : C_* \to D_*\) induces homomorphisms \(f_* : \HH_i(C_*) \to \HH_i(D_*)\).
\end{lemma}

\begin{proof}
    Let \(a \in \HH_i(C_*)\), and we can choose a cycle \(\alpha \in C_i(C_*)\), with \(\partial \alpha = 0\). In this case,
    \[\partial f_i(\alpha) = f_{i-1}(\partial\alpha) = f_{i-1}(0) = 0\]
    So \(f_i(\alpha) \in D_i\) is a cycle. We set \(f_*(a) = [f_i(\alpha)] \in \HH_i(D_*)\). Next, we need to show that this is independent of choices. Suppose \(a = [\alpha] = [\alpha']\). Then \(\alpha - \alpha' \in \im(\partial)\). Therefore, we can write \(\alpha - \alpha' = \partial\tau\). In this case,
    \[f_i(\alpha) - f_i(\alpha') = f_i(\partial\tau) = \partial f_{i+1}(\tau) \in \im(\partial)\]
    So we conclude \([f_i(\alpha)] = [f_i(\alpha')] \in \HH_i(D_*)\). It is clear that \(f_*\) defines a homomorphism.
\end{proof}

\underline{Goal:} To associate to a topological space \(X\) (co)chain complex \(C_*(X)\) and \(C^*(X)\). The definition works for any \(X\), but the theories are better behaved for ``nicer'' \(X\).

\subsection{Singular (co)homology}

We will develop singular (co)homology.

\begin{definition}
    [standard simplex, face]
    The \emph{standard simplex}\index{standard simplex} is
    \[\Delta^n = \left\{(t_0, \dots, t_n) \in \R^{n+1} \mid t_i \ge 0, \sum_i t_i = 1\right\}\]
    The \(i\)-th face \(\Delta^n\) is
    \[\Delta_i^n = \Delta^n \cap \{t_i = 0\}\]
    and we have a canonical homeomorphism
    \begin{align*}
        \delta_i : \Delta^{n-1} &\to \Delta^{n-1}_i \\
        (t_0, \dots, t_{n-1}) & \mapsto (t_0, \dots, t_{i-1}, 0, t_i, \dots, t_{n-1})
    \end{align*}
\end{definition}

\begin{definition}
    [singular simplex, singular chain complex]
    If \(X\) is a topological space, a \emph{singular \(n\)-simplex}\index{singular simplex} in \(X\) is a continuous map \(\sigma : \Delta^n \to X\).

    The \emph{singular chain complex}\index{singular chain complex} is given by the the free abelian groups
    \[C_i(X) = \bigoplus_{\sigma \text{ singular }n\text{-simplex}}\Z \sigma\]
    We define the boundary map \(C_i(X) \to C_{i-1}(X)\) via
    \[\partial \sigma = \sum_{j=0}^i (-1)^j \sigma\vert_{\Delta_j^i}\]
    where
    \[\sigma \vert_{\Delta_j^i} := \sigma \circ \delta_j : \Delta^{n-1} \to X\]
\end{definition}

Note if \(\{v_i\}_{i=0}^n\) are \(n+1\) ordered points in \(\R^{n+1}\), and if \(\{v_i - v_0\}_{i=1}^n\) are linearly independent, then the convex hull of \(v_0, \dots, v_n\) is an \(n\)-simplex, which we will call
\[[v_0, \dots, v_n]\]
which is given by the map
\begin{align*}
    \Delta^n &\to \R^{n+1} \\
    t &\mapsto \sum_i t_iv_i
\end{align*}
We orient the edges of the standard simplex (and thus any simplex \([v_0, \dots, v_n]\)) by saying that \(v_i < v_j\) if \(i < j\).

More concretely, for \(n = 2\), we have the oriented edges \(v_0 \to v_1, v_0 \to v_2, v_1 \to v_2\).

\begin{lemma}
    \(\partial^2 = 0\).
\end{lemma}

\begin{proof}
    If \(\sigma\) is an \(n\)-simplex, defined on \([v_0, \dots, v_n]\), then
    \[\partial\sigma = \sum_j(-1)^j\sigma\vert_{[v_0, \dots, \widehat{v_j}, \dots, v_n]}\]
    and so
    \[\partial(\partial\sigma) = \sum_{i < j}(-1)^i(-1)^j\sigma\vert_{[v_0, \dots, \widehat{v_i}, \dots, \widehat{v_j}, \dots, v_n]} + \sum_{i > j}(-1)^{i+1}{-1}^j\sigma\vert_{[v_0, \dots, \widehat{v_j}, \dots, \widehat{v_i}, \dots, v_n]} = 0\]
    since the signs cancel.
\end{proof}

\begin{definition}
    The \emph{singular homology groups of \(X\)}\index{singular homology}, \(\HH_i(X)\), are the homology groups of the singular chain complex.
\end{definition}

\begin{remark}
    Since all we used is the topology on \(X\), \(\HH_i(X)\) is clearly a topological (read homeomorphism) invariant of \(X\).

    Moreover, if \(f : X \to Y\) is continuous, \(\sigma : \Delta^n \to X\) is an \(n\)-simplex on \(X\), then \(f \circ \sigma\) is an \(n\)-simplex on \(Y\). The fact that
    \[(f \circ \sigma) \circ \delta_j = f \circ (\sigma \circ \delta_j)\]
    shows that \(f\) induces a chain map.
\end{remark}

\begin{definition}
    [singular cochain complex]
    The \emph{singular cochain complex}\index{singular cochain complex} of \(X\) has
    \[C^i(X) = \Hom_\Z(C_i(X), \Z)\]
    and the boundary operator
    \[\partial^*\psi(\sigma) = \psi(\partial \sigma)\]
    for \(\psi \in C^i(X), \sigma \in C_{i+1}(X)\). That is, it is adjoint to \(\partial : C_i(X) \to C_{i-1}(X)\).
\end{definition}

Since \(\partial^2 = 0, (\partial^*)^2 = 0\). The \emph{singular cohomology}\index{singular cohomology} is
\[\HH^*(X) = \bigoplus_{i \ge 0}\HH^i(C^*, \partial^*)\]

If \(f : X \to Y\) continuous, then \(f\) induces a chain map \(f_* : C_*(X) \to C_*(Y)\) via \(f_*(\sigma) = f \circ \sigma\), satisfying \(\partial f_* = f_*\partial\). Similarly, \(f : X \to Y\) defines a \emph{pullback map} \(f^* : C^*(Y) \to C^*(X)\), given by
\[f^*(\psi)(\tau) = \psi(f_*(\tau))\]

Again, we have that
\[\partial^* f^*(\psi)(\tau) = f^*\psi(\partial\tau) = \psi(f_*\partial\tau) = \psi(\partial f_*\tau) = (\partial^*\psi)(f_*\tau) = f^*(\partial^*\psi)(\tau)\]
So \(f^*\) is a cochain map, i.e. \(\partial^*f^* = f^*\partial^*\). Therefore, we have an induced homomorphism on cohomology,
\[f^* : \HH^*(Y) \to \HH^*(X)\]

\underline{Warning:} by definition,
\[C^i(X) = \Hom_\Z(C_i(X), \Z)\]
but it is not true (in general) that
\[\HH^i(X) = \Hom_\Z(\HH_i(X), \Z)\]

\begin{lemma}
    If we write
    \[X = \bigsqcup_{\alpha \in A}X_\alpha\]
    as a disjoint union of path components, then
    \[\HH_i(X) = \bigoplus_{\alpha \in A}\HH_i(X_\alpha)\]

    Moreover, if \(X\) is path connected (and nonempty), then \(\HH_0(X) = \Z\).
\end{lemma}

\begin{proof}
    Since \(\Delta^i\) is path connected, the image of a continuous map \(\sigma : \Delta^i \to X\) must lie within some \(X_\alpha\). Moreover, so does the image of all of the boundary faces \(\sigma \circ \delta_j\). So in fact
    \[(C_i(X), \partial) = \left(\bigoplus_{\alpha \in A}C_i(X_\alpha), \partial\right)\]
    Note that a general element of \(C_i(X)\) is a \underline{finite} linear combination of \(i\)-simplicies in \(X\). So its image can only meet finitely many of the \(X_\alpha\).

    Now suppose \(X\) is path connected. Define a function
    \begin{align*}
        \epsilon : C_0(X) &\to \Z \\
        \sum n_i\sigma_i &\mapsto \sum n_i
    \end{align*}

    Since \(X\) is nonempty, \(\epsilon\) is surjective. On the other hand, if \(\tau  \in C_1(X)\), then \(\epsilon(\tau(1) - \tau(0)) = 0\). So by linearity, 
    \[\im(\partial : C_1(X) \to C_0(X)) \sse \ker(\epsilon)\]
    Recall
    \[\HH_0(X) = \frac{C_0(X)}{\im(\partial : C_1(X) \to C_0(X))}\]
    and so \(\epsilon\) descends to a map \(\HH_0(X) \to \Z\). Now we will use the fact that \(X\) is path connected. Suppose
    \[\sum n_i\sigma_i \in \ker(\epsilon)\]
    and fix a base point \(p \in X\). Since the \(\sigma_i\) are \(0\)-simplicies, they correspond to points in \(X\). Moreover, since \(X\) is path connected, we can choose paths
    \[\tau_i : \Delta^1 \to X\]
    such that \(\tau_i(1) = \sigma_i\) and \(\tau_i(0) = p\). In this case,
    \[\partial\left(\sum n_i\tau_i\right) = \sum n_i\partial \tau_i = \sum n_i\sigma_i - (\sum n_i) p\]
    But we assumed \(\sum n_i = 0\), and so \(\ker(\epsilon) \sse \im(\partial : C_1(X) \to C_0(X))\). This then gives the required result, as the induced map
    \[\HH_0(X) \to \Z\]
    is then an isomorphism.
\end{proof}

\subsection{Fundamental properties}

(Co)homology is useful by virtue of various structural properties.

\begin{theorem}
    [homotopy invariance]
    \index{homotopy invariance!of (co)homology}
    If \(f, g : X \to Y\) are homotopic, then the induced maps \(f_*, g_*\) on homology, and the induced maps \(f^*, g^*\) on cohomology agree.
\end{theorem}

\begin{corollary}
    If \(X \simeq Y\), then \(H_*(X) \simeq Y\), and if \(f\) is a homotopy equivalence, then \(f_*\) induces the isomorphism. For cohomology we have the similar statement, but with \(f^*\).
\end{corollary}

\begin{proof}
    [Proof of the corollary] \(X \simeq Y\) via \(f\) is saying that there exists \(g : Y \to X\), such that \(g \circ f \simeq \id_X\), and \(f\circ g\simeq \id_Y\). That is,
    \[g_* \circ f_* = \id \qqtext{and} f_* \circ g_* = \id\]
\end{proof}

We think of this as saying that (co)homology is ``insensitive to inessential deformations''.

\begin{example}
    Recall \(\{0\} \inj \R^n\) is a homotopy equivalence, and so
    \[\HH_*(\R^n) = \begin{cases}
        \Z & * = 0 \\
        0 & \text{otherwise}
    \end{cases}\]
\end{example}

The other key structural property relies on some homological algebra.

\begin{corollary}
    \(\R^n\) is homeomorphic to \(\R^m\) if and only if \(m = n\).
\end{corollary}

\begin{proof}
    Suppose \(f : \R^n \to \R^m\) is a homeomorphism. This gives us a homeomorphism
    \[f : \R^n \setminus 0 \to \R^m \setminus \{f(0)\}\]
    But \(\R^n\setminus \{*\}\) is homotopy equivalent to \(\S^{n-1}\). Thus \(f\) induces an isomorphism in homology
    \[\HH_i(S^{n-1}) = \HH_i(S^{m-1})\]
    But by our homology computations, this implies that \(n = m\).
\end{proof}

Note in the below we assume \(n > 0\).

\begin{definition}
    [degree]
    Suppose \(f : S^n \to S^n\) is a continuous map, then \(f\) induces a map
    \[f_* : \HH_n(S^n) \to \HH_n(S^n)\]
    which is multiplication by some \(d \in \Z\). We call \(\deg(f) = d\) the \emph{degree}\index{degree!of a map \(S^n \to S^n\)} of \(f\).
\end{definition}

\begin{remark}
    Since \(\HH_n(S^n) = \Z\) which is free, therefore \(f_*\) is determined by \(f_*(1)\). Moreover, we will need to use the \emph{same} identification of \(\HH_n(S^n) = \Z\), and so \(\deg(f)\) is well defined. Note that if we use different isomorphisms, then \(\deg(f)\) is only defined up to a sign.
\end{remark}

Note that 
\begin{itemize}
    \item if \(f \simeq g\), then \(\deg(f) = \deg(g)\),
    \item \(\deg(\id) = 1\).
    \item \(\deg(\text{const}) = 0\), since the constant map factors through \(\{*\}\), and so the induced map on homology factors through \(\HH^n(*) = 0\).
\end{itemize}

\begin{corollary}
    The antipodal map \(a_n : S^n \to S^n\) has degree \((-1)^{n+1}\).
\end{corollary}

\begin{proof}
    \(a_n\) is orthogonal. Or equivalently, it is a composition of \(n+1\) reflections, and we have that
    \[\deg(fg) = \deg(f)\deg(g)\]
\end{proof}

\begin{corollary}
    If \(f : S^n \to S^n\) has no fixed point, then \(f\) is homotopic to the antipodal map.
\end{corollary}

\begin{proof}
    In fact, we will see that \(f(x) \ne g(x)\) for all \(x \in S^n\), then \(f \simeq a_n \circ g\). Taking \(g = \id\) gives the result. Consider the map

    \[x \mapsto \frac{tf(x) - (1-t)g(x)}{\norm{tf(x) - (1-t)g(x)}}\]
    for \(0 \le t \le 1\). Note that the denominator never vanishes. This then defines a homotopy from \(f\) to \(-g\).
\end{proof}

Using this, the degree of any map without fixed points is \((-1)^{n+1}\). Another corollary is that if \(f(x) \ne -x\) for all \(x \in S^n\), then \(f \simeq a \circ a = \id\).

\begin{definition}
    [vector field on \(\S^n\)]
    A \emph{vector field on \(S^n\)}\index{vector field on \(S^n\)} is a map \(v : S^n \to \R^{n+1}\), such that for all \(x \in S^n\),
    \[\inner{x, v(x)} = 0\]
\end{definition}

\begin{proposition}
    [Hairy ball theorem] \(S^n\) has a nowhere vanishing vector field if and only if \(n\) is odd.
\end{proposition}

\begin{proof}
    If \(n = 2k-1\),
    \[v(x_1, y_1, \dots, x_k, y_k) = (-y_1, x_2, \dots, -y_k, x_k)\]
    works.

    \lec

    Now suppose a nowhere vanishing vector field \(v : S^n \to \R^{n+1}\) exists. By normalising, i.e. considering
    \[\frac{v(x)}{\norm{v(x)}}\]
    we can consider \(v : S^n \to S^n \sse \R^{n+1}\). Now consider the family
    \[v_t(x) = \cos(t)x + \sin(t)v(x)\]
    for \(0 \le t \le \pi\), which has unit length as we assumed \(v(x)\) and \(x\) are orthogonal. This family has \(v_0 = \id\), and \(v_\pi = a_n\). Thus, we have that
    \[1 = \deg(\id) = \deg(a_n) = (-1)^{n+1}\]
\end{proof}

Recall if \(f, g : X \to Y\) are homotopic, we would like to show that the induced maps on homology and cohomology are the same. Recall also that the maps are induced by (co)chain maps on (co)chain complexes.

\begin{lemma}
    Suppose \(f_*, g_* : C_* \to D_*\) are chain homotopic. Then they induce the same map on homology.
\end{lemma}

\begin{proof}
    Let \(\alpha \in C_i\) be a cycle, that is, \(\partial \alpha = 0\). By definition, \(f_*[\alpha] = [f(\alpha)]\). Consider
    \[f_i(\alpha) - g_i(\alpha) = (f_i - g_i)(\alpha) = (\partial P + P\partial)(\alpha) = \partial(P\alpha)\]
    But this is in the image of \(\partial\), and so the corresponding homology classes are the same.
\end{proof}

\lec

\underline{Exercise:} Chain homotopy is an equivalence relation on chain maps \(C_* \to D_*\).

\begin{theorem}
    [homotopy invariance of homology]
    \index{homotopy invariance!of (co)homology!proof}
    If \(f, g : X \to Y\) are homotopic, then the induced maps on homology are the same.
\end{theorem}

\begin{proof}
    Let \(i_j : X \inj X \times \{j\}\) be the natural map, \(F : X \times I \to Y\) the homotopy between \(f, g\). But then
    \[f_* = (F \circ i_0)_* = F_* \circ i_{0*}\]

    Therefore, it suffices to show that \(i_{0*}\) and \(i_{1*}\) are chain homotopic.

    Thus, we want
    \[P : C_n(X)  \to C_{n+1}(X \times [0, 1])\]
    such that \(\partial P + P \partial = i_{0*} - i_{1*}\). \(P\) is a \emph{prism operator}\index{prism operator} from a universal way of decomposing \(\Delta^n \times [0, 1]\) into a  finite collection of \(n+1\)-simplices.

    \begin{center}
        \includegraphics[scale=0.15]{diagrams/2023_10_20 (3).jpg}
    \end{center}

    Consider the ordered collections\([v_0, \dots, v_i, w_i, \dots, w_n]\), where \([v_0,\dots, v_n] = \Delta^n \times \{0\}\) and \([w_0, \dots, w_n] = \Delta^n \times \{1\}\). We define \(P\) by
    \[P(\sigma) = \sum_{i=0}^n (-1)^i(\sigma \times I)\vert_{[v_0, \dots, v_i, w_i, \dots, w_n]}\]

    \begin{claim}
        \(\partial P + P \partial = i_{i*} - i_{0*}\). That is,
        \[\partial P = i_{1*} - i_{0*} - P\partial\]
        i.e. ``boundary of prism is the top and bottom faces, as well as the prism of the boundary''.
    \end{claim}

    \begin{proof}
        \begin{align*}
            \partial P(\sigma) = \sum_{j \le i}(-1)^i(-1)^j(\sigma \times I)\vert_{[v_0, \dots, \widehat{v_j}, \dots, v_i, w_i, \dots, w_n]} + \sum_{j \ge i}(-1)^i(-1)^{j+1}(\sigma \times I)\vert_{[v_0, \dots, v_i, w_i, \dots, \widehat{w_j}, \dots, w_n]}
        \end{align*}
        The terms when \(j = i\) cancel, except for
        \[(\sigma \times I) \vert_{[w_0, \dots, w_n]} - (\sigma \times I)\vert_{[v_0, \dots, v_n]}\]
        which are the top and bottom faces. The terms with \(i > j\) and \(i < j\) comprise, up to a sign, \(P(\partial \sigma)\). Just compute (in a dark room with some gin).
    \end{proof}
\end{proof}

\begin{remark}
    If \(C^*, D^*\) are cochain complexes, cochain maps \(f^*, g^*\) are cochain homotopic if we have \(P^i : C^i \to D^{i-1}\), such that
    \[\partial P + P\partial = f^* - g^*\]
    It is easy to check that in this case, \(f^*\) and \(g^*\) induce the saem map \(\HH^*(C^*) \to \HH^*(D^*)\).

    With this, our prism operator \(P : C_n(X) \to C_{n+1}(X \times [0, 1])\) has a dual
    \begin{align*}
        P^* : C^{n+1}(X \times [0, 1]) &\to C^n(X) \\
        P^*(f) &= f \circ P
    \end{align*}

    Dualising everything, the relation \(\partial P + P\partial = i_{1*} - i_{0*}\) becomes
    \[\partial P^* + P^*\partial = i_1^* - i_0^*\]
    and thus homotopic maps induce the same map on cohomology.
\end{remark}

\subsection{Excision}

\begin{theorem}
    [Excision] \index{excision}\label{thm:excision} Let \(X\) be a topological space, \(A \sse X\) a subspace, \(Z \sse X\) a subspace with \(\Cl(Z) \sse \Int(A)\). Then the inclusion
    \[i : (X\setminus \Z, A \setminus Z) \to (X, A)\]
    induces an isomorphism on relative homology
    \[i_* : \HH_n(X \setminus Z, A \setminus Z) \to \HH_n(X, A)\]
\end{theorem}

\begin{lemma}
    [five lemma] \index{five lemma} Suppose we have a diagram

	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow["\epsilon"{description}, from=1-5, to=2-5]

	\arrow["\gamma"{description}, from=1-3, to=2-3]

\begin{corollary}
    If \(f : (X, A) \to (Y, B)\) is a map of pairs, and the maps \(f_* : \HH_*(X) \to |HH_*(Y)\) and \(f_* : \HH_*(A) \to \HH_*(B)\) are isomorphisms, then so are the induced maps on relative homology
    \[f_* : \HH_*(X, A) \to \HH_*(Y, B)\]
\end{corollary}

\begin{proof}
    Follows from the five lemma and the LES of relative homology. 
\end{proof}

\subsection{Relative homology}

\begin{definition}
    [Reduced homology] If \(X\) is a topological space, \(x_0 \in X\) is a base point, then we define the \emph{reduced homology of \(X\)}\index{reduced homology, \(\tilde \HH_*(X)\)} as
    \[\tilde \HH_*(X) = \HH_*(X, x_0)\]
    It is easy to see that \(\tilde \HH_i(X) = \HH_i(X)\) for \(i > 0\), and \(\tilde\HH_i(X) \oplus \Z = \HH_i(X)\). 
\end{definition}

\begin{definition}
    [good pair]
    A pair \((X, A)\) is \emph{good}\index{good pair} if \(A \sse X\) is closed, and \(A\) a neighbourhood deformation retract. That is, there exists an open neighbourhood \(A \sse U \sse X\) of \(A\), and a homotopy \(H : U \times [0, 1] \to U\), with
    \begin{itemize}
        \item \(H(u, 0) = u\),
        \item \(H(u, 1) \in A\) for all \(u \in U\),
        \item \(H(a, t) = a\) for all \(a \in A, t \in [0, 1]\).
    \end{itemize}
\end{definition}

\begin{proposition}
    If \((X, A)\) is a good pair, the natural map \((X, A) \to (X/A, A/A = \{\bullet\})\), induces isomorphism

    is also a neighbourhood deformation retract. With this,

\subsection{Small simplices theorem}

Both the Mayer-Vietoris property and excision will follow from the \emph{small simplices theorem}.

\begin{definition}
    [chain complex adapted to a cover]
    Let \(X\) be a topological space,
    \[\mcU = \left\{U_\alpha\right\}_{\alpha \in I}\]
    be such that \(\{\Int(U_\alpha)\}_{\alpha \in I}\) is an open cover of \(X\). Let
    \[C_j^\mcU(X) = \left\{\sum_\text{finite} a_j \sigma_i \ \big\vert\ a_i \in \Z, \sigma_i : \Delta^i \to X, \im(\sigma_i) \sse U_{\alpha(\sigma_i)} \text{ for some }\alpha(\sigma_i) \in I\right\}\]
    This defines a subcomplex of \(C_*(X)\) with the usual boundary map.
\end{definition}

\begin{theorem}
    [small simplices] \index{small simplices theorem} The inclusion \(C_*^\mcU(X) \inj C_*(X)\) induces an isomorphism on homology.
\end{theorem}

\begin{remark}
    If \(X, Y\) are spaces with covers \(\mcU, \mcV\), and if \(f : X \to Y\) sends each \(U_\alpha\) to some \(V_{f(\alpha)}\), then \(f\) induces a map
    \[f_* : C_j^\mcU(X) \to C_*^\mcV(Y)\] 
    But \((X, \mcU) \mapsto C_*^\mcU(X)\) is \emph{not} functorial for arbitrary maps.
\end{remark}

We will use small simplices to prove Mayer-Vietoris and excision.

\begin{proof}
    [Proof of Mayer-Vietoris, \cref{thm:mayer-vietoris}]
    \index{Mayer-Vietoris!proof}
    Let \(\mcU = \{A, B\}\), where \(A, B \sse X\) open. Then we have an SES of chain complexes
    % https://q.uiver.app/#q=WzAsNSxbMCwwLCIwIl0sWzEsMCwiQ18qKEEgXFxjYXAgQikiXSxbMiwwLCJDXyooQSlcXG9wbHVzIENfKihCKSJdLFszLDAsIkNfKl5cXG1jVShYKSJdLFs0LDAsIjAiXSxbMCwxXSxbMSwyXSxbMiwzXSxbMyw0XV0=
\[\begin{tikzcd}
	0 & {C_*(A \cap B)} & {C_*(A)\oplus C_*(B)} & {C_*^\mcU(X)} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}\]
    The last map is surjective because we have taken the subgroup \(C_*^\mcU(X)\). By the Snake lemma and the small simplices theorem, we get the Mayer-Vietoris sequence.
\end{proof}
