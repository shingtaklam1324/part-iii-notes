\section{Introduction}

A \emph{Lie group} is fundamentally a group, which also a (smooth) manifold. For example, \(\GL_n, \SL_n, \SO_n, \Sp_{2n}\).

\begin{example}
    A prototypical example of a Lie group is the circle group \(S^1\).
\end{example}

Let \(G\) be a Lie group. Then the \emph{Lie algebra}\index{Lie algebra!as tangent space} of \(G\) is the tangent space at the identity \(e\) of \(G\). That is, 
\[\mfg = \T_eG\]
\(\mfg\) is a vector space, with additional structure, which we will see later.

By taking a derivative, we turn the conjugation map
\begin{align*}
    G &\to \Aut(G) \\
    g &\mapsto g(\cdot)g^{-1}
\end{align*}
into a map
\begin{align*}
    \ad : \mfg \to \End(\mfg)
\end{align*}
called the \emph{adjoint}\index{adjoint}. This gives a bilinear map
\begin{align*}
    [\cdot, \cdot] : \mfg \times \mfg &\to \mfg \\
    [x, y] &= \ad(x)(y)
\end{align*}

\begin{example}
    If \(G = \GL_n(\R)\), then we have that \(\mfg = \gl_n(\R) = \Mat_n(\R)\), with
    \[[x, y] = xy - yx\]
\end{example}
What do Lie algebras tell us about the structure of the Lie group \(G\)? 
\begin{itemize}
    \item We will define the \emph{root system} of \(\mfg\), and this then tells us about commutator relations in \(G\) (see Carter's book).
    \item We will define the \emph{Weyl group} of \(\mfg\). For example, the Weyl group of \(\gl_n(\C)\) is isomorphic to \(S_n\), there is an embedding \(S_n \inj \GL_n(\C)\), vie permutation matrices. Let \(B\) denote the Borel subgroup of upper triangular matrices in \(\GL_n(\C)\), then there exists a \emph{Bruhat decomposition}
    \[G = \bigsqcup_{w \in S_n}BwB\]
\end{itemize}
Lie algebras also give us information of the representation theory of \(G\) and \(\mfg\). For example, there exists a bijection
\[\{\text{finite dimensional }\C\text{-representations of }\SL_n(\C)\} \bij \{\text{finite dimensional }\C\text{-representations of }\sl_n(\C)\}\]

Moreover, we can describe the right hand side completely.

In addition, Lie algebras have applications in Algebraic Geometry, for example, we can use Lie algebras to build families of surfaces, of equivalenrly, algebraic curves (See book by Slodovy).

We will define the \emph{Dynkin diagrams} of \emph{semisimple Lie algebras}. For example,

\section{Basic definitions and examples}

Let \(k\) be a field. Most of the time, \(k = \C\), but not always. We will sometimes point out how things can go wrong in characteristic \(p\).

\begin{definition}
    [Lie algebra]
    A \emph{Lie algebra}\index{Lie algebra} over \(k\) is a vector space \(\mfg\) over \(k\), together with a bilinear pairing
    \begin{align*}
        [\cdot, \cdot] : \mfg \times \mfg \to \mfg
    \end{align*}
    satisfying
    \begin{enumerate}
        \item \([xx] = 0\) for all \(x \in \mfg\),
        \item the Jacobi identity
        \[[x[yz]] + [y[zx]] + [z[xy]] = 0\]
    \end{enumerate}
\end{definition}

\begin{notation}
    Note that when clear, we will write \([xy] := [x, y]\).
\end{notation}

\begin{remark}
    In particular, we have antisymmetry, i.e.
    \[[xy] = -[yx]\]
\end{remark}

\begin{definition}
    [(Lie) subalgebra]
    A \(k\)-vector subspace \(\mfh\) of \(\mfg\) is a \emph{(Lie) subalgebra}\index{subalgebra@(Lie) subalgebra} if \(\mfh\) is closed under the Lie bracket of \(\mfg\). That is, for all \(x, y \in \mfh\), \([xy] \in \mfh\).
\end{definition}

\begin{example}
    Let \(V\) be a finite dimensional \(k\)-vector space, then
    \begin{enumerate}
        \item Let \(\gl(V) = \End(V)\), with \([xy] = xy-yx\). If we choose a basis for \(V\), then we can identify \(\gl(V)\) with \(\Mat_n(k)\). In this case, we will write \(\gl_n\) or \(\gl_n(k)\)\index{\(\gl_n\)}.
        \item Let
        \[\sl(V) = \left\{x \in \gl(V) \mid \tr(x) = 0\right\}\]
        This defines a subalgebra of \(\gl(V)\), called the \emph{special linear Lie algebra}, with \(\dim(\sl(V)) = \dim(V)^2 - 1\). The standard basis of \(\sl(V)\) is given by
        \begin{align*}
            E_{i,j} \text{ for }i \ne j \\
            E_{i, i} - E_{i+1, i+1}
        \end{align*}
        We will often write \(\sl_n\)\index{\(\sl_n\)} for this Lie algebra.       
    \end{enumerate}
\end{example}

        \item Suppose \(\Char(k) \ne 2\), and suppose \(V\) is endowed with a symmetric (non-degenerate?) bilinear form

        If \(M \in \GL(V)\) is such that \(\inner{v, w} = v^\T M w\), so

        We usually take
        \[M = \begin{cases}
            \begin{pmatrix}
                0 & I_\ell \\

            \end{pmatrix} & n = 2\ell \\
            \begin{pmatrix}
                1 & 0 & 0 \\
                0 & 0 & I_\ell \\
                0 & I_\ell & 0
            \end{pmatrix} & n = 2\ell + 1
        \end{cases}\]
        These are called the \emph{orthogonal Lie algebra}\index{\(\so_n\)}, denoted \(\so_n\).
        \begin{remark}
            \label{rem:sl2c}
            In the case \(n = 2\), let
            \[e = \begin{pmatrix}
                0 & 1 \\ 0 & 0
            \end{pmatrix} \quad f = \begin{pmatrix}
                0 & 0 \\

            \end{pmatrix} \quad h = \begin{pmatrix}
                1 & 0 \\ 0 & -1
            \end{pmatrix}\]
            viewed as matrices (in \(\sl_2(\C)\)). Indeed, this is the standard basis of \(\sl_2(\C)\). Note
            \begin{align*}
                [ef]&= h \\
                [he] &= 2e \\
                [hf] &= -2f
            \end{align*}
            We'll see that (in some sense) the structure of all semisimple Lie algebras comes from \(\sl_2(\C)\).
        \end{remark}
        \item Again suppose \(\Char(k) \ne 2\). Now suppose \(V\) is endowed with a non-degenerate skew-symmetric (or alternating) bilinear form \(\inner{\cdot, \cdot}\), then
        \[\sp(V) = \left\{x \in \gl(V) \mid \inner{xv, w} + \inner{v, xw} = 0\right\}\]
        In coordinates, we take the form \(\inner{\cdot, \cdot}\) to be the skew-symmetric form given by
        \[M = \begin{pmatrix}
            0 & I_\ell \\
            -I_\ell & 0
        \end{pmatrix}\]
        where \(n = 2\ell\). This is called the \emph{symplectic Lie algebra}\index{\(\sp_n\)}, denoted \(\sp_n\).
    \end{enumerate}
\end{example}

        \setcounter{enumi}{4}
        \item Any vector space \(V\) is a Lie algebra with \([vw] = 0\) for all \(v, w\). We call such Lie algebras \emph{abelian}\index{abelian Lie algebra}. It is named like this, since for \emph{linear Lie algebras}\index{linear Lie algebra}, that is, any subalgebra of \(\gl(V)\) where \(V\) is finite dimensional, \([xy] = xy - yx = 0\) is true if and only if \(x\) and \(y\) commute.
        \item \(\mfb_n\) is the \emph{Borel algebra}\index{\(\mfb_n\)} of upper triangular matrices
        \[\begin{pmatrix}
            * & \cdots & * \\
            & \ddots & \vdots \\
            0& & *
        \end{pmatrix}\]
        This is the Lie algebra associated to the Borel subgroup of upper triangular invertible matrices.
        \item \(\mfn_n\)\index{\(\mfn_n\)} is the Lie algebra of strictly upper triangular matrices,
        \[\begin{pmatrix}
            0 & \cdots & * \\
            & \ddots & \vdots \\
            0& & 0
        \end{pmatrix}\]
        \(\mfn\) stands for \emph{nilpotent}, see section 7.
    \end{enumerate}
\end{example}

\section{Basic representation theory}

\begin{definition}
    [homomorphism, isomorphism]
    A linear map \(\varphi : \mfg \to \mfh\) between two Lie algebras is a \emph{homomorphism}\index{homomorphism} if
    \[\varphi([xy]) = [\varphi(x),\varphi(y)]\]
    for all \(x, y \in \mfg\). If \(\varphi\) is a linear isomorphism, we call \(\varphi\) an \emph{isomorphism of Lie algebras}\index{isomorphism!of Lie algebras}.
\end{definition}

\begin{definition}
    [representation]
    A \emph{representation}\index{representation} of \(\mfg\) is a Lie algebra homomorphism
    \[\varphi : \mfg \to \gl(V)\]
    for some vector space \(V\).
\end{definition}

\begin{notation}
    We also call \(V\) itself a \emph{representation}, or a \(\mfg\)-module. We write \(\mfg \actson V\), and say \(\mfg\) acts on \(V\). We will write
    \[x \vdot v = xv := \varphi(x)(v)\]

    The \emph{dimension}\index{dimension} of the representation is the dimension of \(V\).
\end{notation}

\begin{example}
    \begin{enumerate}
        \item Let \(\dim(V) = 1\), then \(\mfg \actson V\) via \(xv = 0\). This is called the \emph{trivial representation}\index{trivial representation}.
        \item \(\mfg\) is a subalgebra of \(\gl(V)\), then the natural inclusion \(\mfg \inj \gl(V)\) is called the \emph{defining representation}\index{defining representation}.
        \item Let \(x \in \mfg\), define \(\ad_x : \mfg \to \mfg\) by \(\ad_x(y) = [xy]\). The map
        \begin{align*}
            \ad : \mfg &\to \gl(\mfg) \\
            x & \mapsto \ad_x
        \end{align*}
        is called the \emph{adjoint representation}\index{adjoint representation}.
        \begin{remark}
            Recall \(e, h, f\) from \cref{rem:sl2c}, the adjoint representation of \(\sl_2(\C)\) has matrices
            \[\ad_h = \begin{pmatrix}
                2 \\
                & 0 \\
                & & -2
            \end{pmatrix} \quad \ad_e = \begin{pmatrix}
                & -2 \\
                & & 1 \\
                &
            \end{pmatrix} \quad \ad_f = \begin{pmatrix}
                & \\
                -1 \\
                & 2 & \phantom{0}
            \end{pmatrix}\]
            with respect to the basis \(\{e,h,f\}\).
        \end{remark}
        \item If \(V, W\) are representations of \(\mfg\), then so is their \emph{direct sum}\index{direct sum} \(V \oplus V\), via
        \[x(v, w) = (xv, xw)\]
        \item If \(V\) is a representation of \(\mfg\), then so is the \emph{dual}\index{dual representation} \(V^*\),
        \[(xf)(v) = -f(xv)\]
        for all \(x \in \mfg, f \in V^*, v \in V\).
        \item If \(V, W\) are representations of \(\mfg\), then so is the \emph{homomorphisms}\index{hom representation@\(\Hom(V, W)\) representation} \(\Hom(V, W)\), via
        \[(x\vdot f)(v) = x\vdot f(v) - f(x \vdot v)\]
    \end{enumerate}
\end{example}

\begin{definition}
    [equivariant, isomorphism]
    If \(V, W\) are representations of \(\mfg\), then a linear map \(\varphi : V \to W\) is called \emph{\(\mfg\)-equivariant}\index{equivariant} for all \(x \in \mfg, v \in V\),
    \[x \vdot \varphi(v) = \varphi(x \vdot v)\]

    We say \(V, W\) are \emph{isomorphic}\index{isomorphism!of representations} if there exists a \(\mfg\) equivariant isomorphism (of vector spaces) \(V \to W\).
\end{definition}

\begin{definition}
    [subrepresentation, irreducible] A \emph{subrepresentation}\index{subrepresentation} \(W \le V\) is a subspace \(xw \in W\) for all \(x \in \mfg, w \in W\). A non-zero representation \(V\) is \emph{irreducible}\index{irreducible!representation} or \emph{simple}\index{simple representation} if the only subrepresentations of \(V\) are \(0\) and \(V\).
\end{definition}

\underline{Exercise:} The trivial representation is irreducible, and so are the defining and the adjoint representation of \(\sl_2(\C)\).

\lec

\begin{definition}
    [completely reducible, semisimple]
    A representation \(V\) is \emph{completely reducible}\index{completely reducible}, or \emph{semisimple}\index{semisimple} if it decomposes as the direct sum of irreducible representations.
\end{definition}

\underline{Exercise:} A representation \(V\) is completely reducible if and only if for every subrepresentation \(W\) of \(V\), there exists another subrepresentation \(W'\) such that
\[V = W \oplus W'\]

\begin{example}
    If \(V\) is a representation, \(W \le V\) a subrepresentation, then the \emph{quotient}\index{quotient representation} \(V/W\) is a representation of \(\mfg\), via
    \[x(v + W) = (xv) + W\]
\end{example}

\begin{remark}
    The isomorphism theorems hold for quotient representations as well.
\end{remark}

\section{Representation theory of \(\sl_2(\C)\)}

Recall
\[\sl_2(\C) = \left\{\begin{pmatrix}
    a & b \\
    c & -a 
\end{pmatrix}\ \bigg\vert\ a,b,c \in \C\right\}\]

\begin{example}
    We have
    \[\mfb_2 = \left\{\begin{pmatrix}
        a & b \\
        0 & c
    \end{pmatrix}\right\}\]
    which is the Borel subalgebra of \(2 \times 2\) matrices. Let \(V = \Span\left\{v_! = \begin{pmatrix}
        1 \\ 0
    \end{pmatrix}, v_2 = \begin{pmatrix}
        0 \\ 1
    \end{pmatrix}\right\}\), and the defining representation of \(\mfb_2\). This is not completely reducible. To see this, set
    \[V_1 = \Span\{v_1\}\]
    and this does not have a complement.
\end{example}

\begin{definition}
    [faithful]
    A representation \(V\) of \(\mfg\) is \emph{faithful}\index{faithful} if the map
    \[\mfg \to \gl(V)\]
    is injective.
\end{definition}

From now on, all Lie algebras and their representations will be over \(C\), unless stated otherwise.

Let \(V\) be a representation of \(\sl_2(\C)\). Recall the basis
\[e = \begin{pmatrix}
    0 & 1 \\ 0 & 0
\end{pmatrix} \quad f = \begin{pmatrix}
    0 & 0 \\ 1 & 0
\end{pmatrix} \quad h = \begin{pmatrix}
    1 & 0 \\ 0 & -1
\end{pmatrix}\]
of \(\sl_2(\C)\). We know three representations of \(\sl_2(\C)\) already.
\begin{center}
    \begin{tabular}{c c c}
        dimension & name & action of \(h\) \\\hline
        1 & trivial & \(0\) \\
        2 & defining & \(\begin{pmatrix}
            1 & 0 \\ 0 & -1
        \end{pmatrix}\) \\
        3 & adjoint & \(\begin{pmatrix}
            2 \\
            & 0\\
            &&-2
        \end{pmatrix}\)
    \end{tabular}
\end{center}

\begin{definition}
    [weight space]
    For \(\lambda \in \C\), the \emph{\(\lambda\)-weight space}\index{weight space} of \(V\) is
    \[V_\lambda = \left\{v \in V \mid hv = \lambda v\right\}\]
    is the \(\lambda\)-eigenspace of \(h\).
\end{definition}

The following are \underline{vector space sums}, not decompositions into subrepresentations.
\begin{itemize}
    \item For the trivial representation,
    \[V = V_0\]
    \item For the defining representation,
    \[V = V_1 \oplus V_{-1}\]
    \item For the adjoint representation,
    \[V = V_2 \oplus V_0 \oplus V_{-2}\]
    where \(V_2 = \inner{e}, V_0 = \inner{h}, V_{-2} = \inner{f}\).
\end{itemize}

For \(v \in V_\lambda\),
\[h(ev) = (he)v = ([he] + eh)v = 2ev + \lambda ev = (\lambda + 2)ev\]
Hence \(ev \in V_{\lambda + 2}\). Similarly, \(fv \in V_{\lambda -2 }\).

That is, we have

\begin{definition}
    [highest weight vector]
    A non-zero
    \[v \in V_\lambda \cap \ker(e)\]
    for some \(\lambda\) is called a \emph{highest weight vector (of weight \(\lambda\))}\index{highest weight vector}
\end{definition}

\begin{example}
    In the adjoint representation, \(e\) is a highest weight vector.
\end{example}

\begin{lemma}
    \label{lem:e-f-v}
    Suppose \(v \in V_\lambda\) is a highest weight vector. Then for all \(n \ge 1\),
    \[ef^nv = n(\lambda - n + 1)f^{n-1}v\]
\end{lemma}
\begin{proof}
    Induction on \(n\). For \(n = 1\),
    \[(ef)v = ([ef] + fe)v = (h + fe)v = \lambda v = 1(\lambda - 1 + 1)f^0v\]
    The inductive step follows similarly.
\end{proof}

\begin{lemma}
    \label{lem:span-f-subrep}
    Suppose \(v \in V_\lambda\) is a highest weight vector. Then
    \[W = \Span\{v, fv, f^2v, \dots\}\]
    is a sub-representation of \(V\),
\end{lemma}

\begin{proof}
    Suffices to show that for \(w = f^n v \in W\), then \(ew, fw, hw \in W\). By definition, \(fw \in W\) is obvious. \(ew \in W\) follows by \cref{lem:e-f-v}, and the \(n = 0\) case is just \(ew = 0\).

    For \(hw\),
    \[f^nv \in V_{\lambda - 2n}\]
    and so
    \[hw = (\lambda - 2n)w \in W\]
\end{proof}

\begin{proposition}
    \label{prop:highest-weight-vector-exists}
    If \(V\) is finite dimensional, then a highest weight vector exists.
\end{proposition}

\begin{proof}
    Choose any nonzero eigenvector \(v\) of \(h\) (always exists as we are working over \(\C\)). Consider
    \[v, ev, e^2v, \dots\]
    These are eigenvectors for \(h\), with distinct eigenvalues. Hence the set
    \[\{e^nv \mid e^n v \ne 0\}\]
    is linearly independent. As \(V\) is finite dimensional, this set must be finite. Hence there must exists \(n\) such that
    \[e^n v \ne 0 \qqtext{and}e^{n+1}v= 0\]
    Then \(e^nv\) is a highest weight vector.
\end{proof}

\begin{lemma}
    \label{lem:highest-weight-integer}
    Suppose \(V\) is finite dimensional, and \(v \in V_\lambda\) is a highest weight vector, then \(\lambda \in \Z_{\ge 0}\).
\end{lemma}

\begin{proof}
    Any non-zero vectors of the form \(f^n v\) must be linearly independent, so there exists \(n \ge 0\) such that \(f^n v \ne 0, f^{n+1}v = 0\). By \cref{lem:e-f-v},
    \[0 = ef^{n+1}v = (n+1)(\lambda - n)f^n v\]
    Hence we must have that \(\lambda = n\), since \(n + 1\ne 0, f^n v \ne 0\).
\end{proof}

\underline{Conclusion:} Suppose \(V\) is irreducible, of dimension \(n+1\). Then by \cref{prop:highest-weight-vector-exists}, a highest weight vector \(v \in V_\lambda\) exists. By \cref{lem:span-f-subrep}, we have a subrepresentation
\[W = \Span\{v, fv, \dots\}\]
So by irreducibility,
\[\{v, fv, \dots, f^nv\}\]
is a basis, as the \(f^iv\) are linearly independent, and we know from \cref{lem:highest-weight-integer} that \(\lambda = n\).

\begin{corollary}
    \label{cor:sl2-irrep-actions}
    If \(V\) is an irreducible representation of \(\sl_2\), of dimension \(n+1\), then there exists a basis \(v_0, \dots, v_n\) of \(V\), such that the actions are:
    \[h v_i = (n-2i)v_i \qquad fv_i = \begin{cases}
        v_{i+1} & i+1 \le n \\
        0 & i = n
    \end{cases} \qquad e v_i = \begin{cases}
        i(n-i+1)v_{i-1} & i -1 \ge 0 \\
        0 & i = 0
    \end{cases}\]
    In particular, there is a unique irreducible representation of \(\sl_2\) with dimension \(n+1\) for all \(n \ge 0\).
\end{corollary}

\lec

\begin{remark}
    Let \(V\) be the \(n+1\) dimensional irreducible representation of \(\sl_2\), and let \(v \in V\) be a highest weight vector. Note
    \[\left(ef + fe + \frac{1}{2}h^2\right)(v) = \left(n + \frac{n^2}{2}\right)v\]
\end{remark}

\section{Irreducible modules for \(\sl_2\)}

\begin{notation}
    We will write \(V(n)\) for the \(n+1\) dimensional irreducible representation of \(\sl_2\).
\end{notation}

\begin{definition}
    [weights]
    Given a representation \(V\) for \(\sl_2\),
    \[\{\lambda \in \C \mid V_\lambda \ne 0\}\]
    are the \emph{weights}\index{weights} of \(V\).
\end{definition}

We will show \emph{Weyl's theorem}.

\begin{theorem}
    \label{thm:weyl}
    \index{Weyl's theorem!for \(\sl_2\)}
    Every finite dimensional representation of \(\sl_2\) is completely reducible.
\end{theorem}

This result, along with \cref{cor:sl2-irrep-actions}, implies that the action of \(h\) completely determines a finite dimensional representation of \(\sl_2\).

\begin{example}
    Suppose \(V\) is a 5-dimensional representation of \(\sl_2\), and there exists \(v \in V\) of weight \(3\). This means that by counting dimensions, the possible weights are \(\{3, 1, -1, -3\}\), and \(\{0\}\). Thus,
    \[V \cong V(3) \oplus V(0)\]
\end{example}

We will need a few facts. Let \(\mfg\) be a Lie algebra, \(\varphi : \mfg \to \gl(V)\) be a representation of \(\mfg\), and suppose there exists \(\sigma\) commuting with \(\varphi(x)\) for all \(x \in \mfg\). Then:

\underline{Fact 1:}
\[\ker(\sigma - \lambda \id_V)\]
is a subrepresentation of \(V\), for all \(\lambda \in \C\).

\underline{Fact 2:}
If \(V\) is irreducible, then \(\sigma\) is a scalar multiple of \(\id_V\). That is, Schur's Lemma.

\begin{definition}
    [Casimir element]
    Let \(V\) be a finite dimensional representation of \(\sl_2\). Then
    \[\Omega = ef + fe + \frac{1}{2}h^2 \in \gl(V)\]
    is called the \emph{Casimir element (of \(\sl_2\))}\index{Casimir element!of \(\sl_2\)}.
\end{definition}

In fact, \(\Omega\) is \emph{central}.

\begin{lemma}
    \label{lem:sl2-casimir-central}
    If \(\varphi : \sl_2 \to \gl(V)\) is a representation, then \(\Omega\) commutes with \(\varphi(x)\) for all \(x \in \sl_2\).
\end{lemma}

\begin{proof}
    To show \(\Omega\) is central, suffices to show \(\Omega e = e\Omega, \Omega f = f\Omega, \Omega h = h\Omega\). Just compute.
\end{proof}

\begin{corollary}
    If \(V\) is an irreducible finite dimensional representation of \(\sl_2\), then \(\Omega \actson V\) by a scalar.
\end{corollary}

\begin{proof}
    By Schur's lemma and \cref{lem:sl2-casimir-central}. Moreover, the scalar is
    \[\frac{n^2}{2} + n\]
\end{proof}

\begin{proof}
    [Proof of \cref{thm:weyl}] Let \(\varphi : \sl_2 \to \gl(V)\) be a finite dimensional representation of \(\sl_2\). Let \(W \le V\) be a subrepresentation. We need to find a subrepresentaion \(U \le V\), such that
    \[V \cong W \oplus U\]

    \textbf{Case 1: \(W\) has codimension \(1\)}. So \(V/W \cong V(0)\).

    \textbf{Subcase (i): \(W\) is trivial.} In this case, \(\dim(V) = 2\), and so we have a basis \(v_1, v_2\) of \(V\), with respect to which \(\sl_2\) acts on \(V\), by matrices
    \[\begin{pmatrix}
        0 & * \\
        0 & 0
    \end{pmatrix}\]
    We will show \(V\) is isomorphic to \(\underbrace{V(0)}_{=W} \oplus \underbrace{V(0)}_{=U}\).
    Note that
    \[\left[\begin{pmatrix}
        0 & x \\
        0 & 0
    \end{pmatrix}, \begin{pmatrix}
        0 & y \\
        0 & 0
    \end{pmatrix}\right] = 0\]
    for all \(x, y\). Since \(\varphi\) is a representation, it respects the Lie bracket. We must have that
    \[\varphi(h) = [\varphi(e),\varphi(f)] = 0\]
    and so
    \[\varphi(e) = \frac{1}{2}[\varphi(h), \varphi(e)] = 0\]
    and \(\varphi(f) = 0\) similarly. Thus, the action of \(\sl_2\) is trivial.

    \textbf{Subcase (ii): \(W = V(n)\) is irreducible, \(n > 0\).} Now we have \(\Omega \in \gl(V)\), and we will show
    \[V = V(n) \oplus \ker(\Omega)\]
    By Schur's lemma, and the fact that \(W\) is irreducible, and that \(\Omega\) acts on \(V/W\) trivially, there is a basis for \(V\), such that
    \[\Omega = \begin{pmatrix}
        \lambda I & * \\
        0 & 0 
    \end{pmatrix}\]
    \(W\) is non-trivial by assumption, and so \(\ker(\Omega) \ne 0\), and \(W \cap \ker(\Omega) = 0\). Hence
    \[V = W \oplus \Omega\]

    \textbf{Subcase (iii): For a general \(W\).} We do this by induction on \(\dim(V)\). If \(\dim(V) = 1\), the result is clearly true. So we can assume \(\dim(V) \ge 2\). Let \(W' \le W\) be a non-zero subrepresentation. As \(\dim(W/W') < \dim(V)\), and \(\codim_{V/W'}(W/W') = 1\), by induction, this implies that we have a splitting
    \begin{equation}
        \label{eq:weyl-decomp-1}
        \frac{V}{W'} = \frac{W}{W'} \oplus \frac{W''}{W'}
    \end{equation}
    for some subspace \(W'' \le V\), with \(W' \sse W''\), and \(W''/W'\) is a subrepresentation of \(V/W'\). Moreover, \(W''/W'\) is codimension \(1\), and \(\dim(W') \le \dim(V)\). \(W''\) being a subrepresentation of \(V\) follows from the fact that \(W''/W'\) is a subrepresentation. By induction again, there exists a subrepresentation \(U \le W''\) such that
    \[W'' = W' \oplus U\]
    We know that
    \[V = W \oplus U\]
    as \(W \cap U = 0\) since \cref{eq:weyl-decomp-1} is a direct sum, and so \(W \cap U \le W' \cap U = 0\). Using \(\dim(U) = 1\), and counting dimensions we are done.

    \textbf{Case 2: Let \(W\) be arbitrary.} Recall the action on \(\Hom(V, W)\), is given by
    \[(x\varphi)(v) = x\varphi(v) - \varphi(xv)\]
    Define
    \[\van = \left\{\psi \in \Hom(V, W) \mid \psi\vert_W = \lambda\id_W \text{ for some }\lambda\right\}\]
    and we have a subspace
    \[\bb W = \left\{\psi \in \van \mid \psi\vert_W = 0\right\} \le \van\]
    Since we lose one degree of freedom going from \(\bb V\) to \(\bb W\), and so \(\codim_{\bb V}(\bb W) = 1\). Suppose \(\psi\vert_W = \lambda \id_W\), \(x \in \sl_2, w\in W\), then
    \[(x\psi)(w) = x\psi(w) - \psi(xw) = x(\lambda w) - \lambda(xw) = 0\]
    So \(\bb V\) is a subrepresentation of \(\Hom(V, W)\), and so \(\bb W\) is a subrepresentation as well. By case 1, there exists a one-dimensional subrepresentation \(\bb U \le \bb V\), such that
    \[\bb V = \bb U \oplus \bb W\]
    Write \(\bb U = \inner{\gamma}\), for some \(\gamma \in \bb V\), and so \(\gamma\vert_W = \lambda \id_W\), for some non-zero \(\lambda\).

    \begin{claim}
        We have a vector space decomposition:
        \[V = W \oplus \ker(\gamma)\]
    \end{claim}

    \lec

    \begin{proof}
        [Proof of claim]
        By construction, \(W \cap \ker(\gamma) = 0\), and by dimension counting, \(\dim(V) = \dim(W) + \dim(\ker(\gamma))\), as \(W = \im(\gamma)\). Since dimensions add up and the intersection is zero, we have a direct sum of vector spaces.
    \end{proof}

    Finally, it remains to show that \(\ker(\gamma)\) is a subrepresentation of \(V\). Let \(v \in \ker(\gamma)\), \(x \in \sl_2\). Since \(\bb U\) is one-dimensional of \(\sl_2\), it must be the trivial representation. Thus,
    \[0 = (x\gamma)(v) = x\gamma(v) - \gamma(x v) = -\gamma(xv)\]
    as \(\gamma(v) = 0\). This means that \(xv \in \ker(\gamma)\) as required.


\end{proof}

In Humphreys' book \S6.3, Humphreys proves \cref{thm:weyl} for a general semisimple Lie algebra. Or see Henderson \S5.2.1, \S7.5.1.

\begin{remark}
    \begin{enumerate}
        \item The proof only needed (in terms of representation theory)
        \begin{itemize}
            \item existence of a Casimir element \(\Omega\),
            \item the only one-dimensional representation is the trivial representation.
        \end{itemize}
        \item Complete reducibility is rare, it can fail for infinite dimensional Lie algebras, or simple Lie algebras in positive characteristic. For example, the adjoint representation of \(\sl_n(\bb F_p)\) on \(\gl_n(\bb F_p)\) is not completely reducible if \(p \mid n\).
    \end{enumerate}
\end{remark}

\section{Tensor products}

Given vector spaces \(V, W\), with bases \(v_1, \dots, v_n\) and \(w_1, \dots, w_m\) respectively. We define the \emph{tensor product}\index{tensor product!of vector spaces} \(V \otimes_\C W\) as the \(\C\)-vector space, with basis
\[\{v_i \otimes w_j\}_{i, j}\]
subject to the usual bilinearity conditions.

\begin{definition}
    [tensor product of representations]
    If \(V, W\) are representations of a Lie algebra \(\mfg\), then so is \(V \otimes W\)\index{tensor product!of representations}, with
    \[x(v \otimes w) = (xv) \otimes w + v \otimes (xw)\]
\end{definition}

\begin{example}
    If \(V, W\) are \(\sl_2\) representations, \(v \in V_\lambda, w \in W_\mu\), then
    \[h(v \otimes w) = (\lambda + \mu)(v \otimes w)\]
    That is,
    \[v \otimes w = (V\otimes W)_{\lambda + \mu}\]
    Thus, the weights of \(V \otimes W\) are just \(\lambda + \mu\), where \(\lambda\) is a weight for \(V\), \(\mu\) is a weight for \(W\).  
\end{example}

\begin{example}
    For
    \[V(2) \otimes V(2) = V(2)^{\otimes 2}\]
    we have the weights:
    \begin{center}
        \begin{tabular}{c|c c c}
            & 2 & 0 & -2 \\
            \hline
            2 & 4 & 2 & 0 \\
            0 & 2 & 0 & -2 \\
            -2 & 0 & -2 & -4
        \end{tabular}
    \end{center}
    and so
    \[V(2)^{\otimes 2} = V(4) \oplus V(2) \oplus V(0)\]
    In particular, if \(v_n\) is a highest weight vector in \(V(n)\), then \(v_n \otimes v_m\) is a highest weight vector for \(V(n) \otimes V(m)\).
\end{example}

We would like a general formula for decomposing \(V(n) \otimes V(m)\). The answer, as in Part II Representation theory, is a \emph{Clebsch-Gordon}\index{Clebsch-Gordon} formula
\[V(n) \otimes V(m) = \bigoplus_{r = \abs{n - m}, r \cong n - m \pmod 2}^{n + m}V(r)\]
We won't need this though.

\begin{definition}
    The \emph{\(n\)-th symmetric power}\index{symmetric power} is
    \[S^nV = \Sym^n(V) = \frac{V^{\otimes n}}{M_n}\]
    where \(M_n\) is the span of
    \[u_1 \otimes \cdots \otimes u_n - u_{\sigma(1)}\otimes \cdots \otimes u_{\sigma(n)}\]
    where \(\sigma \in S_n\), and \(u_i \in V\).
\end{definition}

For example, \(M_2\) is the span of \(v \otimes w - w \otimes v\). In particular, note that \(M_n\) is a subrepresentation of \(V^{\otimes n}\) of \(V\). So \(\Sym^n(V)\) is a representation of \(V\).

\begin{example}
    In \(S^2V\), \(v \otimes w = w \otimes v\), and so \(S^2 V\) has basis
    \[v_i \otimes v_j\]
    for \(i \le j\). Decomposing \(S^2V(2)\), we see that \(e \otimes e \in S^2V(2)\) is nonzero (note \(V(2)\) is the adjoint representation), and so \(V(4)\) is a subrepresentation of \(S^2V(2)\). In particular, we have a splitting
    \[S^2V(2) = V(4) \oplus V(0)\]
\end{example}

\begin{definition}
    The \emph{\(n\)-th exterior (or alternating) power}\index{exterior power} is
    \[\Lambda^n V = \frac{V^{\otimes n}}{N_n}\]
    where \(N_n\) is the span of
    \[u_1 \otimes \cdots \otimes u_n\]
    where \(u_i \in V\), and \(u_i = u_j\) for some \(i \ne j\).
\end{definition}

Again, \(N_n\) is a subrepresentation of \(V^{\otimes n}\), and so \(\Lambda^n V\) is a representation.

\begin{example}
    With \(n = 2\), \(N_2 = \Span\{v \otimes v\}\).
\end{example}

\begin{notation}
    We write (the coset of) \(u_1 \otimes \cdots \otimes u_n\) in \(\Lambda^n V\) as
    \[u_1 \wedge \cdots \wedge u_n\]
\end{notation}

\underline{Exercises:}
\begin{enumerate}
    \item Decompose \(\Lambda^2V(2) = V(2)\), with basis \(e \wedge f, e \wedge h, h \wedge f\).
    \item Find the dimensions of \(S^n V\) and \(\Lambda^n V\).
\end{enumerate}

\section{Results about semisimple Lie algebras}

Let \(\mfg\) be a Lie algebra over \(\C\).

\begin{definition}
    [ideal] A subspace \(I \sse \mfg\) is an \emph{ideal}\index{ideal} of \(\mfg\) if
    \[[xy]\in I\]
    for all \(x \in \mfg, y \in I\).
\end{definition}

\begin{remark}
    Any ideal is automatically a subalgebra.
\end{remark}

Suppose \(I\) is an ideal, then \(\mfg/I\) is a Lie algebra under
\[[x + I, y + I] = [x,y] + I\]
Moreover, \(I\) is an ideal if and only if it is a subrepresentation of the adjoint representation \(\ad : \mfg \to \gl(\mfg)\).

\begin{example}
    The \emph{centre}\index{centre} of \(\mfg\) is
    \[Z(\mfg) = \left\{x \in \mfg \mid [x, y] = 0 \text{ for all }y \in \mfg\right\}\]
    and by definition, \(Z(\mfg) = \ker(\ad : \mfg \to \gl(\mfg))\).
\end{example}

If \(Z(\mfg) = 0\), then \(\ad : \mfg \to \gl(\mfg)\) is faithful, and thus we have an embedding
\[\ad : \mfg \to \gl(\mfg)\]
and as such, \(\mfg\) can be regarded as a Lie subalgebra of \(\gl_n\), where \(n = \dim(\mfg)\).

\begin{theorem}
    [Ado] Suppose \(\Char(k) = 0\), then any finite dimensional Lie algebra \(\mfg\) embeds as a Lie subalgebra of \(\gl_m\) for some \(m\).
\end{theorem}

\begin{proof}
    Omitted.
\end{proof}

Note that the embedding need not be via the adjoint representation. In fact, this is true for \(\Char(k) = p > 0\), due to Iwasawa.

\begin{example}
    The \emph{derived subalgebra}\index{derived subalgebra} of \(\mfg\) is
    \[D(\mfg) = [\mfg, \mfg] = \left\{[x, y] \mid x,y \in \mfg\right\}\]
    This is an ideal of \(\mfg\).
\end{example}

Recall \(\sl_n\) is a subalgebra of \(\gl_n\). In fact,
\[D(\gl_n) = \sl_n\]

\begin{example}
    Suppose \(\varphi : \mfg \to \mfh\) is a homomorphism  of Lie algebras, then
    \[\ker(\varphi) = \left\{x \in \mfg \mid \varphi(x) = 0\right\}\]
    is an ideal of \(\mfg\). In fact, every ideal arises in this way.
\end{example}

\lec

\begin{definition}
    [simple] A Lie algebra \(\mfg\) is \emph{simple}\index{simple!Lie algebra} if \([\mfg, \mfg] \ne 0\), and the only ideals are \(0\) and \(\mfg\).
\end{definition}

\begin{example}
    We can show that \(\sl_n\) (for \(n \ge 2\)), \(\so_n\) (for \(n \ge 2\)) and \(\sl_{2\ell}\) (for \(\ell \ge 1\)) are simple.
\end{example}

\begin{remark}
    \begin{enumerate}
        \item if \(\mfg\) is simple, then \([\mfg, \mfg] = \mfg\).
        \item if \(\mfg\) is simple, every representation of \(\mfg\) is either faithful, or the direct sum of trivial representations. 
        \item \(\mfg\) is simple if and only if the adjoint representation is irreducible.
    \end{enumerate}
\end{remark}

\begin{definition}
    [semisimple] A Lie algebra \(\mfg\) is \emph{semisimple!Lie algebra} if it is the direct sum of simple ideals. That is, ideals which are simple as Lie algebras.
\end{definition}

\begin{example}
    \[\so_4 \cong \sl_2 \oplus \sl_2\]
\end{example}

We will state a more `standard' definition of semisimple Lie algebras, and show that these are equivalent.

\begin{definition}
    [(lower) central series] The \emph{(lower) central series}\index{central series} of a Lie algebra \(\mfg\) is the sequence of subalgebras
    \[\mfg = \mfg^0 \supseteq \mfg^1 \supseteq \mfg^2 \supseteq \cdots\]
    with \(\mfg^n = [\mfg, \mfg^{n-1}]\). That is,
    \[\mfg \supseteq [\mfg, \mfg] \supseteq [\mfg, [\mfg, \mfg]] \supseteq \cdots\]
\end{definition}

\begin{definition}
    [derived series, upper central series] The \emph{derived series}\index{derived series}, or \emph{upper central series} for a Lie algebra \(\mfg\) is the sequence
    \[\mfg^{(0)} \supseteq \mfg^{(1)} \supseteq \mfg^{(2)} \supseteq \cdots\]
    where \(\mfg^{(0)} = \mfg\), and \(\mfg^{(n)} = [\mfg^{(n-1)}, \mfg^{(n-1)}]\). That is,
    \[\mfg \supseteq [\mfg, \mfg] \supseteq [[\mfg, \mfg], [\mfg, \mfg]] \supseteq \cdots\]
\end{definition}

\begin{remark}
    \begin{itemize}
        \item \(\mfg^{(n)} \sse \mfg^n\),
        \item \(\mfg^n\) and \(\mfg^{(n)}\) are ideals.
    \end{itemize}
    To see the second point, we induct on \(n\). The case \(n = 0\) is clear. Let \(x \in \mfg, y \in \mfg^n\). Then
    \[[x, y] \in \mfg^{n-1}\]
    since \(\mfg^{n-1}\) is an ideal, and \(\mfg^n \sse \mfg^{n-1}\). As
    \[\mfg^n = \left\{[x, y] \mid x \in \mfg, y \in \mfg^{n-1}\right\}\]
    this clearly contains \([x, y]\).
\end{remark}

\begin{example}
    \begin{enumerate}
        \item if \(\mfg\) is simple, then \(\mfg^n = \mfg^{(n)} = \mfg\) for all \(n\).
        \item if \(\mfg\) is abelian, then \(\mfg^1 = \mfg^{(1)} = 0\).
        \item let \(\mfn_n \sse \gl_n\) be the Lie algebra of strictly upper triangular \(n \times n\) matrices. The central series for \(\mfn_n\) is
        \[\left\{\begin{pmatrix}
            0 & & * \\
            & \ddots \\
            & & 0
        \end{pmatrix}\right\} \supseteq \left\{\begin{pmatrix}
            0 & 0 & & * \\
            & \ddots & \ddots & \\
            & & \ddots & 0 \\
            & & & 0
        \end{pmatrix}\right\} \supseteq \cdots  \supseteq 0\]
        Note \(\mfn_3\) is the \emph{Heisenberg Lie algebra}\index{Heisenberg Lie algebra}.
    \end{enumerate}
\end{example}

\begin{definition}
    [nilpotent] If \(\mfg^n = 0\) for some \(n\), then \(\mfg\) is called \emph{nilpotent}\index{nilpotent}.
\end{definition}

\begin{definition}
    [solvable, soluble] If \(\mfg^{(n)} = 0\) for some \(n\), then \(\mfg\) is called \emph{solvable}\index{solvable} (or \emph{soluble} in BrE).
\end{definition}

\underline{Exercise:} Let \(\mfb_n \sse \gl_n\) be the Borel Lie algebra of upper triangular matrices. Then \(\mfb_n\) is solvable but not nilpotent (for \(n \ge 2\)).

Note on the other hand that as \(\mfg^{(n)} \sse \mfg^n\), nilpotent implies solvable.

The next result allows the theory of complex semisimple Lie algebras to go `far' with minimal work.

\begin{theorem}
    [Lie's theorem] \label{thm:lie}Let \(k = \C\) (or an algebraically closed field with characteristic \(0\)). Let \(\mfg \sse \gl(V)\) be a Lie subalgebra, and suppose \(\mfg\) is solvable. Then there exists a common eigenvector for all elements of \(\mfg\).

    That is, there exists \(v \in V\) non-zero, such that for all \(x \in \mfg\), \(xv = \lambda_xv\) for some \(\lambda_x \in \C\).
\end{theorem}

\begin{proof}
    Omitted. See Humphreys Theorem 4.1.
\end{proof}

\begin{corollary}
    There exists a basis for \(V\) such that every element is upper triangular. 
\end{corollary}
In fact,  using \cref{thm:lie} and induction on \(\dim(V)\), we can show that there exists a chain of subspsaces
\[0 = V_0 \le V_1 \le \cdots \le V_n = V\]
with \(\dim(V_i) = i\), and \(\mfg(V_i) \sse V_i\). By considering a basis for \(V_i\), we get the corollary.

Using this result, we can consider \(\mfg \sse \mfb_n\). 

\underline{Aside:} We call the sequence
\[0 = V_0 \le V_1 \le \cdots \le V_n\]
a \emph{maximal flag}\index{maximal flag}, and there is interesting geometry related to this.

\begin{proposition}
    Suppose \(I, J\) are ideals of \(\mfg\). Then
    \begin{enumerate}[(i)]
        \item if \(\mfg\) is solvable, then any subalgebra or quotient of \(\mfg\) is solvable.
        \item if \(I\) is solvable, and \(\mfg/I\) is solvable, then so is \(\mfg\).
        \item if \(I, J\) are solvable, then so is \(I + J\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    (i) is clear from definitions. For (ii), choose \(n\) such that \((g/I)^{(n)} = 0\). Then this forces \(\mfg^{(n)} \sse I\). But then we have \(\mfg^{(n+m)} \sse I^{(m)}\) for each \(m \ge 0\). Since we know that \(I\) is solvable, we are done.

    For (iii), note that 
    \begin{align*}
        \frac{I + J}{J} \cong \frac{I}{I \cap J}
    \end{align*}
    and the right hand side is solvable, by (i), and \(J\) is solvable by assumption, and so by (ii), \(I + J\) is solvable.
\end{proof}

\begin{definition}
    [radical] The \emph{radical}\index{radical} of \(\mfg\) is \(\Rad(\mfg)\) is the maximal solvable ideal of \(\mfg\). That is, it is the sum of all solvable ideals of \(\mfg\).
\end{definition}

\begin{definition}
    [trace form]
    Suppose \(\varphi : \mfg \to \gl(V)\) is a finite dimensional representation of \(\mfg\). Then the \emph{trace form}\index{trace form} of \(V\) (or \(\varphi\)) is the symmetric bilinear form
    \begin{align*}
        (\cdot, \cdot) : \mfg \times \mfg &\to \C \\
        (x, y) &= \tr(\varphi(x)\varphi(y))
    \end{align*}
\end{definition}

\underline{Exercise:} We have the invariance relation \(([x, y], z) = (x, [y, z])\)

\begin{definition}
    [Killing form] The \emph{Killing form}\index{Killing form} \(K(\cdot, \cdot)\) is the trace form of \(\ad\). That is,
    \[K(x, y) = \tr(\ad(x)\ad(y)) = (x, y)_{\ad}\]
\end{definition}

\lec

\begin{theorem}
    [Cartan-Killing criterion] 
    \label{thm:cartan-killing}\index{Cartan-Killing criterion}
    For a finite dimensional Lie algebra \(\mfg\), the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(\mfg\) is semisimple,
        \item \(\Rad(\mfg) = 0\),
        \item the Killing form of \(\mfg\) is non-degenerate.
    \end{enumerate}
\end{theorem}

\begin{remark}
    \(\Rad(\mfg/\Rad(\mfg)) = 0\), since a suitable ideal of \(\mfg/\Rad(\mfg)\) would lift to give an ideal \(J\) of \(\mfg\), containing \(\Rad(\mfg)\), with \(J/\Rad(g)\) solvable. Hence \(J\) is solvable, and \(J \sse \Rad(\mfg)\).

    Using this, \(\mfg/\Rad(\mfg)\) is semisimple.
\end{remark}

    splits. This is called the \emph{Levi decomposition of \(\mfg\)}\index{Levi's theorem!Levi decomposition}, and \(\mfg'\) the \emph{Levi subalgebra of \(\mfg\)}\index{Levi's theorem!Levi subalgebra}.
\end{theorem}

\begin{proof}
    [Not proven in the course] See Fulton-Harris appendix E.
\end{proof}

\begin{lemma}
    Let \(\mfg\) be a Lie algebra,
    \begin{enumerate}[(i)]
        \item if \(I\) is an ideal of \(\mfg\), then so is \([I, I]\),
        \item \(\Rad(\mfg) = 0\) if and only if \(\mfg\) has no non-trivial abelian ideals.
    \end{enumerate}
\end{lemma}

\begin{proof}
    For (i), if \(x, y \in I, z \in \mfg\), we need to show that
    \[[z,[x,y]] \in [I,I]\]
    Using the Jacobi identity,
    \[[z, [x, y]] = -[x, [y, z]] - [y, [x, z]] \in [I, I]\]
    as \(I\) is an ideal.

    For (ii), it is clear that any abelian ideal is solvable. Conversely, if \(I\) is solvable, then the last non-zero term in the derived series of \(I\) is abelian.
\end{proof}

\begin{notation}
    Define\index{orthogonal, \(\mfg^\perp\)}
    \[\mfg^\perp = \left\{x \in \mfg \mid K(x, y) = 0 \text{ for all }x \in \mfg\right\}\]
\end{notation}

\begin{lemma}
    \(\mfg^\perp\) is an ideal.
\end{lemma}

\begin{proof}
    For \(x \in \mfg^\perp, y, z \in \mfg\), then \(K([x, y], z) = K(x, [y, z]) = 0\)
\end{proof}

\begin{lemma}
    \label{lem:killing-form-on-ideal}
    Let \(I\) be an ideal of \(\mfg\), and let \(K_I\) denote the Killing form of \(I\). Then
    \[K_I(x, y) = K(x, y)\]
    for all \(x, y \in I\).
\end{lemma}

\begin{proof}
    Choose a basis for \(I\), and extend it to a basis of \(\mfg\). Given \(x, y \in I\), with respect to this basis,
    \[\ad(x) = \begin{pmatrix}
        A & * \\
        0 & 0
    \end{pmatrix}\]
    where \(A = (\ad(x))\vert_I\), and similarly for \(\ad(y)\). Set \(B = (\ad(y))\vert_I\). Then
    \[K_I(x, y) = \tr(AB) = \tr(\ad(x)\ad(y)) = K(x, y)\]
\end{proof}

\begin{theorem}
    [Cartan's criterion (for solvability)] 
    \label{thm:cartan-criterion}
    \index{Cartan's criterion (for solvability)}
    Suppose \(\mfg\) is a subalgebra of \(\gl(V)\), for \(V\) a finite dimensional vector space over \(\C\). \(\mfg\) is solvable if and only if \(\tr(xy) = 0\) for all \(x \in \mfg, y \in [\mfg, \mfg]\).

    In other words, \(\mfg^{(1)} \le \mfg^\perp\).
\end{theorem}

\begin{proof}
    See Humphreys 4.3 using Lie's theorem and the Jordan decomposition. For the Jordan decomposition, see \S8.
\end{proof}

\begin{corollary}
    \begin{enumerate}[(i)]
        \item if \(\mfg = \mfg^\perp\), then \(\mfg\) is solvable.
        \item if \(\mfg\) is simple, then \(\mfg^\perp = 0\),
        \item \(\mfg^\perp\) is solvable for any finite dimensional Lie algebra \(\mfg\).
    \end{enumerate}
\end{corollary}

\begin{proof}
    (i) Consider the adjoint \(\ad : \mfg \to \gl(V)\). The image \(\ad(\mfg) = \mfg/\ker(\mfg) = \mfg/Z(\mfg)\). \(Z(\mfg)\) is solvable, since it is abelian, and by assumption \(\mfg = \mfg^\perp\). By \cref{thm:cartan-criterion}, \(\ad(\mfg)\) is solvable, and so \(\mfg\) is solvable.

    (ii) Since \(\mfg^\perp\) is an ideal, and \(\mfg\) is simple, \(\mfg^\perp = 0\) or \(\mfg^\perp = \mfg\). In the second case, by (i) \(\mfg\) is solvable, contradicting the fact that \(\mfg\) is simple, as \([\mfg, \mfg] = \mfg\).

    (iii) By \cref{lem:killing-form-on-ideal}, \((\mfg^\perp)^\perp = \mfg^\perp\), so by (i), \(\mfg^\perp\) is solvable.
\end{proof}

With this, we can now prove the Cartan-Killing criterion.

\begin{proof}
    [Proof of \cref{thm:cartan-killing}]

    First we show (ii) \(\implies\) (iii). In this case, \(\mfg\) is solvable, and so \(\mfg^\perp \le \Rad(\mfg) = 0\). Thus \(K\) is non-degenerate.

    For (iii) \(\implies\) (ii), let \(A\) be an abelian ideal of \(\mfg\). We will show that \(A \sse \mfg^\perp\). Choose \(a \in A, y \in \mfg\). Choose a basis for \(A\), and extend it to a basis of \(\mfg\). That is,
    \[\ad(a) = \begin{pmatrix}
        0 & * \\
        0 & 0
    \end{pmatrix} \qqtext{and} \ad(y) = \begin{pmatrix}
        * & * \\
        0 & *
    \end{pmatrix}\]
    Computing \(\tr(\ad(a)\ad(y))\), we find that it is zero. Thus \(A = 0\). Therefore, \(\Rad(\mfg)\) is non-zero, then \(\mfg^\perp\) is non-zero.

    Now assume (ii) and (iii) hold. If \(\mfg\) is simple, then we are done. If not, choose a minimal non-zero ideal \(I\). Let
    \[\mfg_I = \left\{x \in \mfg \mid K(x, y) = 0\text{ for all }y \in I\right\}\]
    This is an ideal of \(\mfg\).

    \begin{claim}
        \(\mfg = \mfg_I \oplus I\).
    \end{claim}

    \begin{proof}
        Since \(I\) is simple (by minimality), and non-abelian (by (ii)),
        \[I \cap \mfg_I \sse I^\perp = 0\]
        Consider the map

\[\begin{tikzcd}
	\mfg & {\mfg^*} & {I^*}

	\arrow[from=1-2, to=1-3]
\end{tikzcd}\]
        given by
        \[x \mapsto K(x, \cdot)\]
        The kernel of this map is \(\mfg_I\). Thus, \(\mfg/\mfg_I \cong I^* \cong I\) as vector spaces.
    \end{proof}

    Repeat this argument, with \(\mfg_I\) (choosing some minimal ideal of \(\mfg_I\)). We can do this as any ideal of \(\mfg_I\) is an ideal of \(\mfg\), and so \(\Rad(\mfg_I) = 0\).

    \begin{claim}
        \((\mfg_I)^\perp = 0\)
    \end{claim}

    \begin{proof}
        Since if \(x \in (\mfg_I)^\perp\), then \(x \in \mfg^\perp\).
    \end{proof}

    This proves (i).

    Finally, to show (i) \(\implies\) (ii), write
    \[\mfg = \bigoplus_j I_j\]
    where the \(I_j\) are simple ideals. Let \(\pi_j : \mfg \to I_j\) denote the projection.

\lec

    \begin{claim}
        If \(J\) is an ideal of \(\mfg\), then \(\pi_j(J)\) is an ideal of \(I_j\).
    \end{claim}

    Now if \(A\) is an abelian ideal of \(\mfg\), then \(\pi_j(A)\) is an abelian ideal of \(I_j\), and so \(\pi_j(A) = 0\) for all \(j\). With this, \(A = 0\).
\end{proof}

\end{proof}

\underline{Exercise:} Any ideal or quotient of a semisimple Lie algebra is semisimple.

For the Casimir element, let \(\varphi : \mfg \to \gl(V)\) be an irreducible representation of a semisimple Lie algebra \(\mfg\). Without loss of generality, assume \(\varphi\) is faithful (if not, we can consider \(\mfg/\ker(\varphi)\)). We know by \cref{thm:cartan-killing} that the trace form \((\cdot, \cdot)_V\) is non-degenerate. Choose a basis \(x_1, \dots, x_n\) for \(\mfg\). With respect to the trace form, we have a dual basis \(y_1, \dots, y_n\) for \(\mfg\). That is,
\[(x_i,y_j)_V = \delta_{ij}\]

\begin{definition}
    [Casimir element]
    Define the \emph{Casimir element associated with \(\varphi\)}\index{Casimir element!of a representation of a semisimple Lie algebra}
    \[\Omega_{\varphi} = \sum_i \varphi(x_i)\varphi(y_i)\]
\end{definition}

\begin{remark}
    \begin{itemize}
        \item \(\Omega_\varphi \in \gl(V)\),
        \item \(\Omega_\varphi\) commutes with \(\varphi(x)\) for all \(x \in \mfg\). In particular, by Schur's lemma, \(\Omega_\varphi\) is a scalar multiple of \(\id_V\), and
        \[\tr(\Omega_\varphi) = \sum\tr(\varphi(x_i)\varphi(y_i)) = \dim(\mfg)\]
        From this, we also see that \(\Omega_\varphi\) is independent of the choice of basis of \(\mfg\) which we chose. 
    \end{itemize}
\end{remark}

\begin{example}
    If \(\mfg = \sl_2 \le \gl_2\), let \(V = \C^2\) and \(\varphi = \id\) is the defining representation. Recall the basis
    \[e = \begin{pmatrix}
        0 & 1 \\
        0 & 0
    \end{pmatrix} \quad f = \begin{pmatrix}
        0 & 0 \\
        1 & 0 
    \end{pmatrix} \quad h = \begin{pmatrix}
        1 & 0 \\
        0 & -1
    \end{pmatrix}\]
    Some easy linear algebra gives a dual basis with respect to the trace form, which is \(\{f, \frac{1}{2}h, e\}\) (in the same order). With this,
    \[\Omega_{\varphi} = ef + fe + \frac{1}{2}h^2 = \begin{pmatrix}
        3/2 & 0 \\
        0 & 3/2
    \end{pmatrix}\]
    which is the same as the one we obtained earlier.
\end{example}

\section{Jordan decomposition}

Two observations:

\begin{enumerate}
    \item If \(\mfg\) is a simple Lie algebra, \(\varphi : \mfg \to \gl(V)\) is a finite dimensional representation, then \(\varphi(\mfg) \sse \sl(V)\). This is because \([\mfg, \mfg] = \mfg\), and so 
    \[\varphi(\mfg) = \varphi([\mfg, \mfg]) = [\varphi(\mfg), \varphi(\mfg)] \le [\gl(V), \gl(V)] = \sl(V)\]
    \item Recall from Linear algebra that if \(x \in \gl(V)\), then there exists a basis of \(V\) such that \(x\) a block diagonal matrix, with Jordan blocks of the form
    \[\begin{pmatrix}
        \lambda & 1 & \cdots & 0 \\
        & \ddots & \ddots & \vdots \\
        & & \ddots & 1 \\
        & & & \lambda
    \end{pmatrix}\] 
\end{enumerate}

\begin{definition}
    [nilpotent, semisimple] We say \(x \in \gl(V)\) is \emph{nilpotent}\index{nilpotent} if \(x^n = 0\) for some \(n\). We say \(x\) is \emph{semisimple}\index{semisimple} if the roots of its minimal polynomial are distinct, that is, it is diagonalisable.
\end{definition}


\begin{proposition}
    [Jordan decomposition]
    \label{prop:jordan-decomposition}
    If \(x \in \gl(V)\), where \(V\) is finite dimensional. Then
    \begin{enumerate}[(i)]
        \item there exists unique elements \(x_s, x_n \in \gl(V)\), where \(x_s\) is semisimple and \(x_n\) is nilpotent, with
        \[x = x_s + x_n\]
        and \([x_s, x_n] = 0\).
        \item there exists polynomials \(p_s, p_t \in \C[t]\), without constant terms, such that \(x_s = p_s(x)\) and \(x_n = p_n(x)\). In particular, \(x_s\) and \(x_n\) will commute with any \(y \in \gl(V)\) with \([x, y] = 0\).
        \item if \(A \le B \le V\) are subspaces, and \(x(B) \sse A\), then \(x_s(B) \sse A\) and \(x_n(B) \sse A\).
    \end{enumerate}
    The decomposition \(x = x_s + x_n\) is called the \emph{(additive) Jordan(-Chevalley) decomposition of \(x\)}\index{Jordan decomposition}. \(x_s\) and \(x_n\) are called the \emph{semisimple part}\index{semisimple!part} and the \emph{nilpotent part}\index{nilpotent!part} of \(x\) respectively.
\end{proposition}

\begin{proof}
    Routine linear algebra. See Humphreys \S4.2.
\end{proof}

\begin{example}
    If \(x\) is represented by a single Jordan block
    \[x = \begin{pmatrix}
        \lambda & 1 & \cdots & 0 \\
        & \ddots & \ddots & \vdots \\
        & & \ddots & 1 \\
        & & & \lambda
    \end{pmatrix}\] 
    then the Jordan decomposition is
    \[x_s = \begin{pmatrix}
        \lambda &  & & 0 \\
        & \ddots &  &  \\
        & & \ddots &  \\
        & & & \lambda
    \end{pmatrix} \qquad x_n = \begin{pmatrix}
        0 & 1 & \cdots & 0 \\
        & \ddots & \ddots & \vdots \\
        & & \ddots & 1 \\
        & & & 0
    \end{pmatrix}\]
    i.e. the diagonal and nilpotent parts. 
\end{example}

\subsubsection*{Why is this valuable?}

% Suppose \(\mfg\) is a finite dimensional linear Lie algebra. Then we have an adjoint representation \(\ad : \mfg \to \gl(\mfg)\). If \(x \in \gl(V)\) is nilpotent, then so is \(\ad(x)\). Similarly, if \(x\) is semisimple, then so is \(\ad(x)\).

Let \(V\) be a finite dimensional vector space. We can consider the adjoint representation \(\ad : \gl(V) \to \gl(\gl(V))\). If \(x \in \gl(V)\) is semisimple, then so is \(\ad(x)\). Similarly, if \(x\) is nilpotent, then so is \(\ad(x)\).

\begin{lemma}
    Let \(x \in \mfg \le \gl(V)\), where \(V\) is finite dimensional. Let \(x = x_s + x_n\) be its Jordan decomposition. Then \(\ad(x) = \ad(x_s) + \ad(x_n)\) is the Jordan decomposition of \(\ad(x) \in \gl(\mfg)\).
\end{lemma}

\begin{proof}
    \(\ad(x_s)\) and \(\ad(x_n)\) are semisimple and nilpotent respectively, they commute since
    \[[\ad(x_s),\ad(x_n)] = \ad([x_s, x_n]) = \ad(0) = 0\]
    Thus, by uniqueness in \cref{prop:jordan-decomposition}, the Jordan decomposition must be as stated.
\end{proof}

\begin{theorem}
    \label{thm:semisimple-nilpotent-parts-in}
    Suppose \(\mfg\) is a semisimple Lie algebra, which is a subalgebra of \(\gl(V)\). Let \(x \in \mfg\), then \(x_s, x_n \in \mfg\).
\end{theorem}

\begin{proof}
    Let
    \[N(\mfg) = \left\{y \in \gl(V) \mid [y, z] \in \mfg \text{ for all }z \in \mfg\right\}\]
    be the \emph{normaliser of \(\mfg\) in \(\gl(V)\)}\index{normaliser, \(N(\mfg)\)}.

    \begin{claim}
        \begin{enumerate}[(i)]
            \item \(N(\mfg)\) is a subalgebra of \(\gl(V)\), containing \(\mfg\) as an ideal.
            \item \(x_s, x_n \in N(\mfg)\).
        \end{enumerate}
    \end{claim}

    \begin{proof}
        (i) is clear from the definition of the normaliser. For (ii), let \(z \in \mfg\), we have that
        \[[x_s, z] = \ad(x_s)(z) = \ad(x)_s(z)\]
        By \cref{prop:jordan-decomposition} (ii), this is in \(\mfg\), as \(\ad(x)_s\) is a polynomial in \(\ad(x)\) with no constant term. But for \(z \in \mfg\), as \(x \in \mfg\), \(\ad(x)(z) = [x,z] \in \mfg\).
    \end{proof}

    Let \(W\) be an irreducible subrepresentation of \(V\), and define
    \[\mfg_W = \left\{y \in \gl(V) \mid yw \in W \text{ for all }w \in W \text{ and }\tr(y\vert_W) = 0\right\}\]

    \begin{claim}
        \(\mfg\) is a subalgebra of \(\mfg_W\).
    \end{claim}

    \begin{proof}
        \(W\) is a subrepresentation of \(\mfg\), and so it is stabilised by \(\mfg\), and also the image of \(\mfg\) in \(\gl(W)\), say \(\overline{\mfg}\), is also semisimple. 
        
\lec
        
        With this, \([\overline{\mfg}, \overline{\mfg}] = \overline{\mfg}\), and so every element of \(\overline{\mfg}\) is a sum of commutators, and all of the traces are zero, and so \(\mfg \le \mfg_W\).
    \end{proof}

    Using this, \(\tr(x \vert_W) = 0\). Note \(x_s, x_n\) are polynomials in \(x\), and so they stabilise everything that \(x\) does. Moreover, \(\tr(x_n\vert_W) = 0\) as \(x_n \vert_W\) is nilpotent. Thus, \(\tr(x_s\vert_W) = \tr(x\vert_W) -\tr(x_n\vert_W) = 0\). Using this, \(x_s, x_n \in \mfg_W\) for all \(W\).

    To finish, define 
    \[\mfg' = N(\mfg) \cap \bigcap_{W \le V \text{ irred subrep}}\mfg_W\]
    \begin{claim}
        \(\mfg = \mfg'\).
    \end{claim}

    \begin{proof}
        Since \(\mfg' \le N(\mfg)\), \(\mfg\) is an ideal of \(\mfg'\). Then \(\mfg\) is a subrepresentation of \(\mfg'\) under the adjoint action of \(\mfg\). By Weyl's theorem\footnote{applied to the representation \(\mfg'\) of the semisimple Lie algebra \(\mfg\)}, \(\mfg' = \mfg \oplus U\) as representations. It suffices to show \(U = 0\). 
        
        Choose \(u \in U\), then as \(\mfg\) is an ideal, \([u,\mfg] \sse \mfg\). But \(\ad(\mfg)(U) \sse U\), and so \([u, \mfg] \sse U\). Hence \([u, \mfg] = 0\), and so \(u\) commutes with every element of \(\mfg\). Using this, \(u\) is a \(\mfg\)-endomorphism \(V \to V\), and so it stabilises every irreducible subrepresentation \(W\).

        By Schur's lemma, \(u \vert_W = \lambda \id_W\) for some scalar \(\lambda \in \C\). But \(\tr(u\vert_W) = 0\) since \(u \in \mfg_W\), and so \(\lambda = 0\). But every representation splits as a direct sum of irreducibles, and so \(u\) must be zero.
    \end{proof}
    By the above, we see that \(x_s, x_n\) is in each set on the right hand side, and so \(x_s, x_n \in \mfg\).
\end{proof}

For \(\mfg \le \gl(V)\) a semisimple Lie algebra, we can define an \emph{abstract Jordan decomposition}\index{abstract Jordan decomposition}
\[\ad(x) = \ad(x)_s + \ad(x)_n\]
Since \(\ad\) is faithful, as \(\mfg\) is semisimple, \(\mfg\) is isomorphic to \(\ad(\mfg) \le \gl(\mfg)\). By \cref{thm:semisimple-nilpotent-parts-in}, \(\ad(x)_s, \ad(x)_n \in \ad(\mfg)\), and so there exists \(x_s, x_n \in \mfg\) such that \(x = x_s + x_n\).

Suppose \(\mfg \le \gl(V)\) for some \(V\), with \(x = x_s + x_n\). Then since \(\ad(x_n) = \ad(x)_n\) and \(\ad(x_s) = \ad(x)_s\), the abstract Jordan decomposition is just the usual Jordan decomposition.

\begin{corollary}
    Let \(\varphi : \mfg \to \gl(V)\) be any representation of a semisimple Lie algebra \(\mfg\). Choose \(x \in \mfg\), and let it have a Jordan decomposition \(x = x_s + x_n\), then
    \[\varphi(x_s) = \varphi(x)_s \qqtext{and}\varphi(x_n) = \varphi(x)_n\]
    defines a Jordan decomposition of \(\varphi(x)\).
\end{corollary}

\begin{proof}
    See Corollary 5.11 of the notes by David Stuart on Moodle. It needs semisimplicity, and the fact that we are working over \(\C\). It fails if we work over a field with positive characteristic.
\end{proof}

\section{Cartan subalgebras and root space decompositions}

In this section, \(\mfg\) is a finite dimensional semisimple Lie algebra over \(\C\).

\begin{definition}
    [toral subalgebra] A subalgebra \(\mft\) of \(\mfg\) is \emph{toral}\index{toral subalgebra} if
    \begin{enumerate}
        \item \(\mft\) is abelian,
        \item \(\ad(x)\) is semisimple for all \(x \in \mft\).
    \end{enumerate}

    A maximal toral subalgebra is called a \emph{maximal torus}\index{maximal torus}, or a \emph{Cartan subalgebra}\index{Cartan subalgebra} (CSA).
\end{definition}

\begin{remark}
    Many authors, including Humphreys define Cartan subalgebras as a nilpotent subalgebra which equals its normaliser in \(\mfg\). That is,
    \[\mft = \left\{x \in \mfg \mid [x, \mft] \sse \mft\right\}\]
    This is equivalent to our definition.
\end{remark}

\begin{example}
    If \(\mfg \le \sl_n, \gl_n\), with \(\mft\) being the set of diagonal matrices, then \(\mft\) is a maximal torus. It is true for \(\so_n\) and \(\sp_{2n}\) as well.
\end{example}

\begin{lemma}
    \label{lem:simultaneuous-eigenspace}
    Let \(t_1, \dots, t_n \in \End(V)\) which pairwise commute, and are all semisimple. Suppose \(\lambda = (\lambda_1, \dots, \lambda_n) \in \C^n\), and define
    \[V_\lambda = \left\{v \in V \mid t_iv = \lambda_i v \text{ for all }i\right\}\]
    That is, the simultaneuous eigenspaces for the \(t_i\). Then
    \[V = \bigoplus_{\lambda \in \C^n}V_\lambda\]
\end{lemma}

\begin{proof}
    By induction on \(n\). \(n=1\) is true by definition. For \(n > 1\), we know by induction that
    \[V = \bigoplus_{\lambda' \in \C^{n-1}}V_{\lambda'}\]
    for the action of \(t_1, \dots, t_{n-1}\). Then since the \(t_i\) commute,
    \[t_n(V_{\lambda'}) \sse V_{\lambda'}\]
    for all \(\lambda'\). By decomposing each \(V_{\lambda'}\) in terms of \(t_n\) eigenspaces, we are done.
\end{proof}

\begin{lemma}
    Any \(\mfg\) contains a Cartan subalgebra.
\end{lemma}

\begin{proof}
    Needs Engel's theorem (Examples sheet 2), and Zorn's lemma. See David Stuart's notes.
\end{proof}

Recasting \cref{lem:simultaneuous-eigenspace}, suppose we have \(\mfh \le \gl(V)\) with a basis of commuting semisimple elements \(t_1, \dots, t_n\). Take \(\lambda \in \C^n\), this corresponds the element of \(\mfh^*\), given by
\[t_i \mapsto \lambda_i\]
Then
\[V_\lambda = \left\{v \in V \mid h v = \lambda(h)v \text{ for all }h \in \mfh\right\}\]
In our situation, fix a Cartan subalgebra \(\mft \le \mfg\), then
\[\mfg = \bigoplus_{\lambda \in \mft^*} \mfg_\lambda\]
where 
\[\mfg_\lambda = \left\{x \in \mfg \mid [t, x] = \lambda(t)x \text{ for all }x \in \mft\right\}\]

\lec

\begin{definition}
    [root] Let
    \[\Phi = \{\alpha \in \mft^* \setminus \{0\} \mid \mfg_\alpha \ne 0\}\]
    The elements of \(\Phi\) are the \emph{roots of \(\mfg\) with respect to \(\mft\)}\index{root}. If \(\alpha \in \Phi\), then \(\mfg_\alpha\) is called a \emph{root space}\index{root space}.
\end{definition}

With this, we have
\[\mfg = \mfg_0 \oplus \bigoplus_{\alpha \in \Phi}\mfg_\alpha\]
which is the \emph{root space decomposition}\index{root space decomposition}, or the \emph{Cartan decomposition}\index{Cartan decomposition} of \(\mfg\).

\begin{proposition}
    \label{prop:root-space-properties}
    \begin{enumerate}[(i)]
        \item For all \(\alpha, \beta \in \mft^*\), \([\mfg_\alpha, \mfg_\beta] \sse \mfg_{\alpha + \beta}\),
        \item If \(\alpha \in \Phi\), \(x \in \mfg_\alpha\), then \(\ad(x)\) is nilpotent.
        \item If \(\alpha + \beta \ne 0\), then \(K(\mfg_\alpha, \mfg_\beta) = 0\) for all \(\alpha, \beta \in \mfg^*\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), let \(x \in \mfg_\alpha, y \in \mfg_\beta\), \(t \in \mft\). We have the Jacobi identity:
    \[[t,[x,y]] = - [x, [y, t]] - [y, [t, x]] = [x, [t, y]] - \alpha(t)[y, x] = \beta(t)[x, y] - \alpha(t)[y, x] = (\alpha + \beta)(t)[x, y]\]
    Note Fulton-Harris calls this the \emph{fundamental calculation}\index{fundamental calculation}.

    For (ii), use (i) and the fact that \(\mfg\) is finite dimensional.

    For (iii), if \(\alpha + \beta \ne 0\), we can find \(t \in \mft\) such \((\alpha + \beta)(t) \ne 0\). Fix such a \(t \in \mft\), \(x \in \mfg_\alpha\) and \(y \in \mfg_\beta\). Then
    \[\alpha(t)K(x, y) = K([t,x], y) = -K([x,t],y) = -K(x, [t, y]) = -\beta(t)K(x,y)\]
    and so \((\alpha + \beta)(t)K(x, y) = 0\). But by assumption \(\alpha(t) + \beta(t) \ne 0\), and so \(K(x, y) = 0\).
\end{proof}

\begin{corollary}
    \begin{enumerate}[(i)]
        \item The Killing form restricted to \(\mfg_0\) is non-degenerate.
        \item If \(\alpha \in \Phi\), then \(-\alpha \in \Phi\).
    \end{enumerate}
\end{corollary}

\begin{proof}
    For (i), if \(z \in \mfg_0\), with \(K(z, x) = 0\) for all \(x \in \mfg_0\), then by (iii), we know that \(\mfg_0\) is orthogonal to all \(\mfg_\alpha\) with \(\alpha \ne 0\). If \(x \in \mfg\), we can write it as
    \[x = x_0 + \sum_{\alpha \in \Phi}x_\alpha\]
    with \(x_\alpha \in \mfg_\alpha\). Using this, we see that \(K(z, x) = 0\) for all \(x \in \mfg\). By non-degeneracy of the Killing form (as \(\mfg\) is semisimple), we must have that \(z = 0\).
    
    For (ii), the proof is similar.
\end{proof}

\begin{lemma}
    \([\mfg_\alpha, \mfg_{-\alpha}]\) is one-dimensional.
\end{lemma}

\begin{proof}
    Take \(x \in \mfg_\alpha, y \in \mfg_\alpha\), then \([x, y] \in \mft\). Let \(t \in \mft\), so
    \[K([x, y], t) = K(x, [y, t]) = -K(x, [t, y]) = \alpha(t)K(x, y)\]
    Hence \([x, y] = K(x, y)t_\alpha\). With this,
    \[[\mfg_\alpha, \mfg_{-\alpha}] \sse \Span\{t_\alpha\}\]
    and so the dimension is at most one. But by non-degeneracy, we can find \(x, y\) such that \(K(x, y) \ne 0\), and so the dimension is one.
\end{proof}

\begin{lemma}
    \(\alpha(t_\alpha) \ne 0\).
\end{lemma}

\begin{proof}
    Since the Killing form is non-degenerate, and rescaling is possible, we may choose \(x \in \mfg_\alpha, y \in \mfg_{-\alpha}\) such that \(K(x, y) = 1\). Then
    \begin{align*}
        [x, y] &= t_\alpha \\
        [t_\alpha, x] &= \alpha(t_\alpha)x \\
        [t_\alpha, y] &= -\alpha(t_\alpha)y
    \end{align*}
    Thus, the space \(\mfh = \Span\{x, y, t_\alpha\}\) is a subalgebra of \(\mfg\). Suppose \(\alpha(t_\alpha) = 0\). Then, \([\mfh, \mfh] = \Span\{t_\alpha\}\), and so \(\mfh\) is solvable. Now consider the adjoint representation
    \[\ad : \mfg \to \gl(\mfg)\]
    This shows that \(\mfh\) embeds as a solvable Lie subalgebra of \(\gl(\mfg)\). By Lie's theorem, we can assume \(\ad(\mfh)\) is a subset of the space of upper triangular matrices. With this, \(\ad(t_\alpha) = [\ad(x), \ad(y)]\) is strictly upper triangular. Hence \(\ad(t_\alpha)\) is nilpotent, but \(\ad(t_\alpha)\) is semisimple\footnote{This follows from \(t_\alpha \in \mft\).}, and so \(\ad(t_\alpha) = 0\). Thus,
    \[t_\alpha \in Z(\mfg) = 0\]
    Contradiction, as \(\alpha \ne 0\) implies \(t_\alpha \ne 0\).
\end{proof}

\begin{proposition}
    Let \(\alpha \in \Phi\). Then the root spaces \(\mfg_{\pm\alpha}\) are 1-dimensional. Moreover, the if \(c\alpha \in \Phi\), for some \(c \in \C\), then \(c = \pm 1\).
\end{proposition}

\begin{proof}
    Suppose \(c\alpha \in \Phi\), then \(h_\alpha\) takes \(c\alpha(h_\alpha)= 2c\) as an eigenvalue. The eigenvalues of \(h_\alpha\) are integers, and so either \(c \in \Z\), or \(c \in \Z + \frac{1}{2}\).

    Write
    \[V = \mft \oplus\bigoplus_{c\alpha \in \Phi}\mfg_\alpha\]
    Let \(K = \ker(\alpha) \le \mft\). We can check that \(K + \mfm_\alpha\) is an \(\mfm_\alpha\)-subrepresentation of \(V\). By Weyl's theorem, as a representation of \(\mfm_\alpha\),
    \[V = K \oplus \mfm_\alpha \oplus W\]
    where \(W\) is a complementary subrepresentation. Suppose either of the conclusions in the statement are false. Then \(W \ne 0\). Let \(W_0 \le W\) be an irreducible subrepresentation. We know \(W_0 \cong V(s)\) for some \(s\). Then \(W_0\) has a highest weight vector \(w_0\), with \(w_0 \in \mfg_{c\alpha}\) for some \(c\), and
    \[[h_\alpha, v_0] = sv_0\]
    \textbf{Case 1: \(s\) is even.} In this case, \(0\) is an eigenvalue of \(h_\alpha\). Let \(e\) be the eigenvector. But the zero eigenspace of \(\mfh_\alpha\) on \(V\) is \(\mft\), which is contained in \(K \oplus \mfm_\alpha\). Thus, \(e \in (K \oplus \mfm_\alpha) \cap W_0 = 0\). Contradiction.

    \underline{Aside:} If \(2\alpha\) is a root, then \(h_\alpha\) has \(2\alpha(h_\alpha) = 4\) as an eigenvalue, but the eigenvalues of \(h_\alpha\) on \(K \oplus m_\alpha\) are \(0, 2, -2\). So the only way this could happen is if \(W\) contains an irreducible subrepresentaion \(V(s)\), where \(s\) is even. With this, if \(\alpha\) is a root, then \(2\alpha\) is not a root.

    \textbf{Case 2: \(s\) is odd.} In this case, \(1\) is an eigenvalue of \(h_\alpha\). As \(\alpha(h_\alpha) = 2\), this means that \(\frac{1}{2}\alpha\) is a root. But then by the above,
    \[\alpha = 2\cdot\frac{1}{2}\alpha\]
    is not a root. Contradiction.
\end{proof}

\begin{proposition}
    \label{prop:root-string-properties}
    Let \(\alpha, \beta \in \Phi\) such that \(\alpha \ne \pm \beta\). Then
    \begin{enumerate}[(i)]
        \item \(\beta(h_\alpha) \in \Z\), and we call these the \emph{Cartan integers}\index{Cartan integers}.
        \item there exists integers \(p, q \ge 0\) such that if \(r \in \Z\), then
        \[\beta + r\alpha \in \Phi \iff -p \le r \le q\]
        Moreover, \(p - q = \beta(h_\alpha)\).
        \item \([\mfg_\alpha, \mfg_\beta] = \mfg_{\alpha + \beta}\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), consider
    \[V = \bigoplus_{r \in \Z}\mfg_{\beta + r\alpha}\]
    and let \(\mfm_\alpha\) act on \(V\) by the adjoint action. Let
    \[q = \max\left\{r \in \Z \mid \beta + r\alpha \in \Phi\right\}\]
    In particular, \(\beta + q\alpha\) is a root. Choose \(v \in \mfg_{\beta + q\alpha}\) non-zero. Then
    \[[e_\alpha, v] \in \mfg_{\beta + (q+1)\alpha} = 0\]
    and
    \[[h_\alpha, v] = (\beta + q\alpha)(h_\alpha)(v) \in \Span\{v\}\]
    Thus, \(v\) is a highest weight vector, with weight \((\beta + q\alpha)(h_\alpha)\). With this, by \(\sl_2\) representation theory,
    \[\beta(h_\alpha) + q\alpha(h_\alpha) \in \Z_{\ge 0}\]
    and so
    \[\beta(h_\alpha) + 2q \in \Z_{\ge 0}\]
    which means that \(\beta(h_\alpha) \in \Z\).

    For (ii), recall from \cref{lem:span-f-subrep} that
    \[W = \Span\{v, fv, f^2v, \dots\}\]
    is an irreducible representation of \(V\), and so \(h_\alpha\) acts by
    \[\begin{pmatrix}
        (\beta + q\alpha)(h_\alpha) \\
        & (\beta + (q-1)\alpha)(h_\alpha) \\
        & & \ddots \\
        & & & -(\beta + q\alpha)(h_\alpha)
    \end{pmatrix}\]
    In particular,
    \[W = \bigoplus_{r = -p}^q \mfg_{\beta + r\alpha}\]
    for some \(p\), where
    \[p = \min\{r \in  \Z \mid \beta - r\alpha \in \Phi\}\]

    Suppose \(W' \le V\) is a proper subrepresentation. Then \(W'\) contains a highest weight vector \(w \in \mfg_\gamma\), for some \(\gamma\). Then
    \[0 \le \gamma(h_\alpha) < -(\beta + q\alpha)(h_\alpha) \le 0\]
    Contradiction. Finally,
    \[(\beta - p\alpha)(h_\alpha) = -(\beta + q\alpha)(h_\alpha)\]
    and rearranging,
    \[\beta(h_\alpha) = p - q\]

    \lec

    For (iii), we already know that \([\mfg_\alpha, \mfg_\beta] \sse \mfg_{\alpha + \beta}\). Thus, if \(\alpha + \beta\) is not a root we are done. On the other hand, if \(\alpha + \beta\) is a root, choose \(v \in \mfg_\beta\) non-zero. Suppose \([e_\alpha, v] = 0\), then \(v\) is a highest weight vector for \(v\). Contradiction. Thus, \([e_\alpha, v] \ne 0\), and \(\mfg_{\alpha + \beta}\) is spanned by it, as it is one-dimensional.
\end{proof}

\begin{definition}
    [reflection]
    For \(\alpha \in \Phi\), define the \emph{reflection at \(\alpha\)} by
    \begin{align*}
        w_\alpha : \mft^* &\to \mft^* \\
        w_\alpha(\beta)&=\beta - \beta(h_\alpha)\alpha
    \end{align*}
\end{definition}

\begin{corollary}
    [of \cref{prop:root-string-properties}] \(w_\alpha(\Phi) = \Phi\).
\end{corollary}

\begin{proof}
    Let \(\beta \in \Phi\), let \(p, q\) be as in \cref{prop:root-string-properties}. We need to show that \(\beta - \beta(h_\alpha)\alpha \in \Phi\). We have that
    \[\beta - \beta(h_\alpha)\alpha = \beta - (p-q)\alpha\]
    But
    \[-p \le -(p-q) \le q\]
    and so this lives in the root string.
\end{proof}

\subsection{Abstract root systems}

Let \((E, \pinner{\cdot, \cdot})\) be a real Euclidean space. If \(\alpha \in E\) is non-zero, define
\begin{align*}
    \alpha^\vee : E &\to \R \\
    \alpha^\vee(\lambda) &= \frac{2\pinner{\alpha, \lambda}}{\pinner{\alpha, \alpha}}
\end{align*}
and we define
\begin{align*}
    w_\alpha : E &\to E \\
    w_\alpha(\lambda) &= \lambda - \alpha^\vee(\lambda)
\end{align*}

\begin{definition}
    [root system] A finite subset \(\Phi \sse E\) is a \emph{root system}\index{root system} if
    \begin{enumerate}
        \item \(0 \notin \Phi\), \(\Phi\) spans \(E\),
        \item if \(\alpha, \beta \in \Phi\), then \(\beta^\vee(\alpha) \in \Z\),
        \item if \(\alpha \in \Phi\), then \(w_\alpha(\Phi) = \Phi\),
        \item if \(\alpha, c\alpha \in \Phi\), then \(c = \pm 1\).
    \end{enumerate}
    Each \(\alpha \in \Phi\) is called a \emph{root}.
\end{definition}

\begin{remark}
    Removing 4. gives a \emph{``non-reduced'' root system}.
\end{remark}

\begin{notation}
    If \(\mu \in E, \lambda \in E^*\), we will write
    \[\inner{\mu, \lambda} = \lambda(\mu)\]
    and so \(\inner{\beta, \alpha^\vee} = \alpha^\vee(\beta)\).
\end{notation}

\begin{example}
    If \(\mfg\) is a semisimple Lie algebra, \(\mft \le \mft\) a Cartan subalgebra, \(\Phi\) is the set of roots associated to \(\Phi\), then \(\Phi\) is a root system in \(\Span_\R(\Phi)\). 
\end{example}

\begin{definition}
    [rank] The \emph{rank}\index{rank} of a root system \((\Phi, E)\) is \(\dim_\R(E)\).
\end{definition}

\begin{definition}
    [isomorphism] Given root systems \((\Phi, E), (\Phi', E')\), an \emph{isomorphism}\index{isomorphism!of root systems} is a linear isomorphism \(\rho : E \to E'\), with
    \begin{enumerate}
        \item \(\rho(\Phi) = \Phi'\),
        \item \(\inner{\rho(\alpha), \rho(\beta)^\vee} = \inner{\alpha, \beta^\vee}\) for all \(\alpha, \beta \in \Phi\).
    \end{enumerate}
\end{definition}

\end{example}

\begin{lemma}
    [finiteness]
    \label{lem:finiteness}
    If \(\Phi\) is a root system, \(\alpha, \beta \in \Phi\), \(\alpha \ne \pm \beta\), then 
    \[\inner{\alpha, \beta^\vee}\inner{\beta, \alpha^\vee} \in \{0, 1, 2, 3\}\]
\end{lemma}

\begin{proof}
    \(\pinner{\alpha, \beta} = \norm{\alpha}\norm{\beta}\cos(\theta)\), where \(\theta\) is the angle between \(\alpha\) and \(\beta\). So
    \[\inner{\alpha, \beta^\vee}\inner{\beta, \alpha^\vee} = \frac{4\pinner{\alpha, \beta}^2}{\norm{\alpha}^2\norm{\beta^2}} = 4\cos^2(\theta) \in \Z\]
    Now \(\cos^2(\theta) \in [0, 1]\), and so \(\cos^2(\theta) \in \{0, 1/4, 1/2, 3/4, 1\}\). But they are not parallel, and so \(\cos^2(\theta) \ne 1\).
\end{proof}

\begin{corollary}
    If \(\Phi\) is a root system, and \(\alpha, \beta\) are roots, then
    \[\inner{\alpha, \beta^\vee} \in \{0, \pm 1, \pm 2, \pm 3\}\] 
\end{corollary}

\underline{Exercise:} The only rank 2 systems are, up to isomorphism, \(A_1 \times A_1, A_2, B_2, G_2\).

\begin{corollary}
    If \(\Phi\) is an irreducible root system, then \(\norm{\alpha}^2\) can take at most two values as \(\alpha \in \Phi\) varies.
\end{corollary}

\begin{proof}
    Exercise. Suppoe not, then we get a contradiction due to the fact that the \(\inner{\alpha, \beta^\vee}\) are integers.
\end{proof}

\begin{definition}
    [simply laced] An irreducible root system \(\Phi\) is \emph{simply-laced}\index{simply-laced} if all the roots are of the same length.
\end{definition}

\begin{example}
    \(A_1, A_2\) are simply laced, \(B_2, G_2\) are not.
\end{example}

\underline{Exercise:} If \(\Phi\) is simply laced, then \((\Phi, E)\) is isomorphic to a root system \((\Phi', E')\), where \(\inner{\alpha, \beta^\vee} \in \{0, \pm 1\}\) for all \(\alpha, \beta \in \Phi', \alpha \ne \pm \beta\).

\section{Weyl chambers and root bases}

Throughout, \((\Phi, E)\) is a root system.

For a root \(\alpha \in \Phi\), we have the \emph{root hyperplane}\index{root hyperplane}
\[H_\alpha = \{\lambda \in E \mid \inner{\lambda, \alpha^\vee} = 0\}\]
corresponding to \(\alpha\). The connected components of
\[E \setminus \bigcup_{\alpha \in \Phi} H_\alpha\]
are called the \emph{Weyl chambers}\index{Weyl chambers}.

A subset \(\Delta = \{\alpha_1, \dots, \alpha_\ell\} \sse \Phi\) is called a \emph{root basis}\index{root basis}, or a \emph{base}\index{base}, if 
\begin{enumerate}
    \item \(\Delta\) is a basis for \(E\),
    \item if \(\alpha \in \Phi\), with \[\alpha = \sum_{i=1}^\ell c_i \alpha_i\]
    then all of the \(c_i\) have the same sign (or are zero).
\end{enumerate}

\begin{definition}
    [simple roots, positive and negative roots]
    The elements of \(\Delta\) are called \emph{simple roots}\index{simple roots}. If
    \[\alpha = \sum_i c_i \alpha_i\]
    with all \(c_i \ge 0\), then we call \(\alpha\) a \emph{positive root}\index{positive root}, denoted \(\alpha > 0\) (or \(\alpha \succ 0\)). If all \(c_i \le 0\), we call \(\alpha\) a \emph{negative root}\index{negative root}.

    The set of all positive roots is denoted \(\Phi^+\), and \(\Phi^- = -\Phi^+\) the set of negative roots.
\end{definition}

\begin{remark}
    \(\Delta\) defines a partial order on \(E\), by saying
    \[\mu < \lambda \iff \lambda - \mu \text{ is a sum of positive root or }\lambda = \mu\]
\end{remark}

\begin{lemma}
    Let \(W\) be the Weyl group of \((\Phi, E)\). Then if \(\Delta\) is a base, and \(w \in W\), then \(w(\Delta)\) is a base.
\end{lemma}

\begin{proof}
    We know that \(w\) is invertible, so \(w(\Delta)\) is a basis for \(E\), and as \(w\) acts on \(\Phi\), \(w(\Delta) \sse \Phi\). If \(\alpha \in \Phi\) is a root, with
    \[\alpha = \sum_ic_i\alpha_i\]
    with all \(c_i \ge 0\) (without loss of generality). Then
    \[w(\alpha) = \sum_i c_i w(\alpha_i)\]
    with all \(c_i \ge 0\).
\end{proof}

That is, the Weyl group acts on the set of root bases. It remains to show how to construct a root basis.

\begin{itemize}
    \item Choose \(\gamma \in E \setminus \bigcup_\alpha H_\alpha\) (i.e. \(\gamma\) in a Weyl chamber). Define
    \[\Phi_\gamma^+ = \{\alpha \in \Phi \mid \inner{\gamma, \alpha^\vee} > 0\}\]
    and \(\Phi_\gamma^- = -\Phi^+_\gamma\). Note \(\Phi = \Phi_\gamma^+ \cup \Phi_\gamma^-\).
    \item Define
    \[\Delta_\gamma = \{\alpha \in \Phi_\gamma^+ \mid \alpha \ne \beta_1 + \beta_2 \text{ for all }\beta_1 + \beta_2 \in \Phi^+_\gamma\}\]
\end{itemize}

\begin{theorem}
    \begin{enumerate}[(i)]
        \item \(\Delta_\gamma\) is a root basis,
        \item every root basis is of this form \(\Delta_\gamma\) for some \(\gamma\) in a Weyl chamber.
    \end{enumerate}
\end{theorem}

\begin{proof}
    For (i),
    \begin{claim}
        \label{claim:root-basis-1}
        If \(\alpha, \beta \in \Delta_\gamma\), then \(\alpha - \beta \notin \Delta_\gamma\).
    \end{claim}

    \begin{proof}
        Suppose \(\alpha, \beta \in \Delta_\gamma\). Without loss of generality \(\alpha - \beta \in \Phi_\gamma^+\). Otherwise, take \(\beta - \alpha\). Then
        \[\alpha = (\alpha - \beta) + \beta\]
        Contradicting the definition.
    \end{proof}
\lec

    \begin{claim}
        If \(\alpha, \beta \in \Delta_\gamma\) are distinct, then \(\inner{\alpha, \beta^\vee} = 0\).
    \end{claim}

    \begin{proof}
        Recall from \cref{lem:finiteness} that
        \[\inner{\alpha, \beta^\vee}\inner{\beta, \alpha^\vee} \in \{0, 1, 2, 3\}\]
        Suppose \(\inner{\alpha, \beta^\vee} > 0\). Without loss of generality, assume \(\inner{\alpha, \beta^\vee} = 1\). Otherwise, consider \(\inner{\beta, \alpha^\vee}\). Now
        \[w_\beta(\alpha) = \alpha - \inner{\alpha, \beta^\vee}\beta = \alpha - \beta \in \Delta_\gamma\]
        But \(w_\beta\) preserves \(\Delta_\gamma\), contradicting \cref{claim:root-basis-1}.
    \end{proof}

    \begin{claim}
        Let \(\Delta_\gamma = \{\alpha_1, \dots, \alpha_\ell\}\), \(\alpha \in \Phi_\gamma^+\),
        \[\alpha = \sum_{i-1}^n c_i\alpha_i\]
        then \(c_i \ge 0\) for all \(i\).
    \end{claim}

    \begin{proof}
        Suppose not. Choose an \(\alpha\) which cannot be written this way, and with \(\pinner{\alpha, \gamma}\) minimal. By construction, \(\alpha \notin \Delta_\gamma\), hence \(\alpha = \beta_1 + \beta_2\) where \(\beta_1, \beta_2 \in \Phi_\gamma^+\). With this, \(\pinner{\alpha, \gamma} = \pinner{\beta_1, \gamma} + \pinner{\beta_2, \gamma}\). By definition, \(\beta_i \in \Phi_\gamma^+\), and so \(\pinner{\beta_i, \gamma} > 0\), hence \(\pinner{\alpha, \gamma} > \pinner{\beta_i, \gamma}\). With this, \(\beta_1, \beta_2\) can be written as a \(\Z_{\ge 0}\)-linear combination of the \(\alpha_i\). But then so can \(\alpha = \beta_1 + \beta_2\). Contradiction.
    \end{proof}

    If we worked with \(\Phi_\gamma^-\) instead, this means that every element in \(\Phi_\gamma^-\) is a non-positive linear combination instead. Also, \(\Delta_\gamma\) spans \(E\) as \(\Phi\) spans \(E\).

    \begin{claim}
        \(\Delta_\gamma\) is a linearly independent set.
    \end{claim}

    \begin{proof}
        Suppose for some \(c_i \in \R\), \(\sum_i c_i \alpha_i = 0\). Without loss of generality, we can assume \(c_i \ge 0\) for \(1 \le i \le m\) and \(c_i \le 0\) for \(m+1 \le i \le \ell\). Set
        \begin{align*}
            v &= \sum_{i=1}^m c_i\alpha_i = -\sum_{j=m+1}^\ell c_j\alpha_j
        \end{align*}
        Now consider
        \begin{align*}
            \pinner{v, v} = -\sum_{i=1}^m\sum_{j=m+1}^\ell \underbrace{c_ic_j}_{\le 0} \underbrace{\pinner{\alpha_i, \alpha_j}}_{\le 0} \le 0
        \end{align*}
        Hence \(v = 0\). With this,
        \begin{align*}
            0 = \pinner{\gamma, v} = \sum_{i=1}^m c_i\underbrace{\pinner{\gamma, \alpha_i}}_{> 0}
        \end{align*}
        Hence \(c_i = 0\) for \(1 \le i \le m\). Similarly, the other \(c_i\) are zero as well.
    \end{proof}

    For (ii), see Humphreys \S10.1.
\end{proof}

\begin{corollary}
    We have a bijection
    \[\{\text{Weyl chambers}\} \bij \{\text{root bases}\}\]
\end{corollary}

\begin{proof}
    Given a Weyl chamber \(C\), we can choose \(\gamma \in C\), and we have a root basis \(\Delta_\gamma\). Conversely given \(\Delta\), \(\Delta = \Delta_\gamma\) for some \(\gamma\), which in turn is in a Weyl chamber \(C\).
\end{proof}

\begin{notation}
    Write \(C_\Delta = C_\gamma\) if \(\Delta = \Delta_\gamma\), where \(C_\gamma\) is the Weyl chamber containing \(\gamma\). We call it the \emph{fundamental Weyl chamber relative to \(\Delta\)}\index{fundamental Weyl chamber}.
\end{notation}

\begin{definition}
    [height] If \(\Delta = \{\alpha_1, \dots, \alpha_\ell\}\) is a root basis, \(\alpha \in \Phi\), say
    \[\alpha = \sum_{i=1}^\ell c_i \alpha_i\]
    The \emph{height}\index{height} of \(\alpha\) is
    \[\sum_{i=1}^\ell c_i\]
\end{definition}

\begin{lemma}
    If \(\Delta = \{\alpha_1, \dots, \alpha_\ell\}\) is a root basis, \(\beta \in \Phi^+\setminus \Delta\), then there exists \(i\) such that \(\beta - \alpha_i \in \Phi\).
\end{lemma}

\begin{proof}
    Given \(\beta\), if \(\pinner{\beta, \alpha_i} \le 0\) for all \(i\), then \(\Delta \cup \{\beta\}\) is a linearly independent set. So there exists \(i\) such that
    \[\inner{\beta, \alpha_i^\vee} > 0\]
    Since \(\inner{\beta, \alpha_i^\vee}\inner{\alpha_i, \beta^\vee} \in \{0, 1, 2, 3\}\), then \(\inner{\beta, \alpha_i^\vee} = 1\) or \(\inner{\alpha_i, \beta^\vee} = 1\). That is,
    \[w_{\alpha_I}(\beta) = \beta - \alpha_i \text{ or }w_\beta(\alpha_i) = \alpha_i - \beta\]
    In either case, \(\beta - \alpha_i \in \Phi\). 
\end{proof}

\begin{corollary}
    If \(\beta \in \Phi^+\), then \(\beta\) can be written as a sum
    \[\beta = \sum_{j=1}^n\alpha_{i(j)}\]
    where \(\alpha_{i(j)}\) are not-necessarily distinct simple roots, and each partial sum is a root, i.e.
    \[\sum_{j=1}^k\alpha_{i(j)} \in \Phi\]
\end{corollary}

\begin{proof}
    Use the lemma and induction on the height of \(\beta\).
\end{proof}

Recall that \(W\) acts on the set of root bases, and so it preserves the Weyl chambers.
\begin{lemma}
    If \(w \in W\), \(\lambda, \mu \in E\), then

\end{lemma}
Using this, we can deduce that
\begin{proposition}
    If \(\Delta\) is a root basis, and \(w \in W\), then
    \[C_{w(\Delta)} = w(C_\Delta)\]
\end{proposition}

\begin{lemma}
    For \(\Phi\) a root system, \(\Delta\) a root basis, and \(W\) the Weyl group, \(\alpha \in \Delta\), then \(w_\alpha\) permutes \(\Phi^+ \setminus \Delta\).
\end{lemma}

\begin{proof}
    Take \(\alpha_1 \in \Delta\), where \(\Delta = \{\alpha_1, \dots, \alpha_\ell\}\), take \(\beta \in \Phi^+\), \(\beta \ne \alpha_1\). In particular, we can write
    \[\beta = \sum_i c_i \alpha_i\]
    with \(c_i \in \Z_{\ge 0}\). In this case,
    \begin{align*}
        w_{\alpha_1}(\beta) &= \beta - \inner{\beta, \alpha_1^\vee}\alpha_1 \\

\section{Classification of irreducible root systems}

Throughout, let \((\Phi, E)\) be a root system, and a root basis \(\Delta = \{\alpha_1, \dots, \alpha_\ell\}\). Let \(W\) be the Weyl group of \(\Phi\).

\begin{definition}
    [Cartan matrix] The \emph{Cartan matrix}\index{Cartan matrix} of \(\Phi\) is the \(\ell \times \ell\) matrix,
    \[C = \begin{pmatrix}
        \inner{\alpha_i, \alpha_j^\vee}
    \end{pmatrix}_{1 \le i, j \le \ell}\]
    This is independent of the choice of root basis (up to permutation), since given \(\Delta'\) another root basis, there exists \(w \in W\) with \(w(\Delta) = \Delta'\), and the action of \(W\) preserves \(\inner{\cdot, \cdot}\).
\end{definition}

Note \(\det(C) \ne 0\).

\begin{example}
    Recall the root system \(G_2\), given by
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/root-systems/G2.jpg}
    \end{center}
    Let \(\alpha_1 = \alpha, \alpha_2 = \beta\). In this case, \(\inner{\alpha_1, \alpha_2^\vee} = -1, \inner{\alpha_2, \alpha_1^\vee} = -3\). Hence the Cartan matrix is
    \[\begin{pmatrix}
        2 & -1 \\
        -3 & 2
    \end{pmatrix}\]
\end{example}

\begin{example}
    Similarly, for \(A_1 \times A_1\), we have
    \[\begin{pmatrix}
        2 & 0 \\
        0 & 2
    \end{pmatrix}\]
    For \(A_2\) we have
    \[\begin{pmatrix}
        2 & -1 \\
        -1 & 2
    \end{pmatrix}\]
    and for \(B_2\), we have
    \[\begin{pmatrix}
        2 & -2 \\
        -1 & 2
    \end{pmatrix}\]
\end{example}

\begin{proposition}
    Suppose \((\Phi', E')\) is another root system, with root basis \(\{\alpha_1', \dots, \alpha_\ell'\}\), with
    \[\inner{\alpha_i, \alpha_j^\vee} = \inner{\alpha_i', (\alpha_j')^\vee}\]
    Then the linear map \(\alpha_i \mapsto \alpha_i'\) induces an isomorphism \(\phi\) of root systems, with
    \[\inner{\phi(\alpha), \phi(\beta)^\vee} = \inner{\alpha, \beta^\vee}\]
    for all \(\alpha, \beta \in \Phi\). Hence the Cartan matrix of \(\Phi\) determines \(\Phi\) up to isomorphism.
\end{proposition}

\begin{proof}
    Since \(\Delta\) is a basis for \(E\), and \(\Delta'\) a basis for \(E'\), we have a unique linear isomorphism \(\phi : E \to E'\), with \(\phi(\alpha_i) = \alpha_i'\). If \(\alpha, \beta \in \Delta\), then
    \begin{align*}
        w_{\phi(\alpha)}(\phi(\beta)) &= w_{\alpha'}(\beta') = \beta' - \inner{\beta', (\alpha')^\vee}\alpha' \\
        &= \phi(\beta) - \inner{\beta, \alpha^\vee}\phi(\alpha) \\
        &= \phi(\beta - \inner{\beta, \alpha^\vee}\alpha) \\
        &= \phi(w_\alpha(\beta))
    \end{align*}
    That is, we have a commutative diagram
    % https://q.uiver.app/#q=WzAsNCxbMCwwLCJFIl0sWzIsMCwiRSciXSxbMCwyLCJFIl0sWzIsMiwiRSciXSxbMCwxLCJcXHBoaSJdLFsyLDMsIlxccGhpIiwyXSxbMCwyLCJ3X1xcYWxwaGEiLDJdLFsxLDMsIndfe1xccGhpKFxcYWxwaGEpfSJdXQ==
    \[\begin{tikzcd}
        E && {E'} \\
        \\
        E && {E'}
        \arrow["\phi", from=1-1, to=1-3]
        \arrow["\phi"', from=3-1, to=3-3]
        \arrow["{w_\alpha}"', from=1-1, to=3-1]
        \arrow["{w_{\phi(\alpha)}}", from=1-3, to=3-3]
    \end{tikzcd}\]
    Now use \cref{thm:weyl-group-props}, the respective Weyl groups \(W, W'\) are generated by simple reflections, and so the map
    \[w \mapsto \phi \circ w \circ \phi^{-1}\]
    is an isomorphism \(W \to W'\), sending \(w_\alpha\) to \(w_{\phi(\alpha)}\) for each \(\alpha \in \Delta\). Each \(\beta \in \Phi\) is conjugate under \(W\) to a simple root, say \(\beta = w(\alpha)\) for some \(\alpha \in \Delta\). This implies that
    \[\phi(\beta) = \phi(w(\alpha)) = (\phi w \phi^{-1})\phi(\alpha) \in \Phi'\]
    Hence \(\phi\) maps \(\Phi\) onto \(\Phi'\). Using the formula for reflections, \(\phi\) preserves the Cartan integers, i.e.
    \[\inner{\alpha, \beta^\vee} = \inner{w(\phi(\alpha)), w(\phi(\beta))^\vee}\]
\end{proof}

\begin{remark}
    The proposition suggests it is possible, in principle, to recover the root systems \(\Phi\) from the Cartan integers. See Humphreys for a reference.
\end{remark}

\begin{definition}
    [Coxeter graph]
    Recall if \(\alpha \ne \pm \beta\), then
    \[\inner{\alpha, \beta^\vee}\inner{\beta, \alpha^\vee} \in \{0, 1, 2, 3\}\] The \emph{Coxeter graph}\index{Coxeter graph} of \(\Phi\) is a graph with \(\ell\) vertices, for \(i \ne j\), we join the \(i\)-th vertex to the \(j\)-th with
    \(\inner{\alpha_i, \alpha_j^\vee}\inner{\alpha_j, \alpha_i^\vee}\)
    edges.
\end{definition}
The Coxeter graph determines \(\inner{\alpha_i, \alpha_j^\vee}\) in the case where all the roots have the same length (i.e. it is simply-laced). In the case where more than one root length occurs (e.g. for \(B_2, G_2\)), the graph fails to tell us which of a pair of vertices should correspond to a short simple root, and which corresponds to a long simple root (in the case where the vertices are joined by two or three edges).

\begin{remark}
    The Coxeter graph determines \(W\) as it determines the order of products of the generators of \(W\).
\end{remark}

\begin{definition}
    The \emph{Dynkin diagram}\index{Dynkin diagram} of \(\Phi\) is the Coxeter graph, but if a multiple edge between vertices occurs, we add an arrow to point to the shorter root.
\end{definition}

\begin{example}
    For ranks \(1\) and \(2\), we have the Dynkin diagrams
    \begin{enumerate}
        \item \(A_1\): \dynkin A1
        \item \(A_1 \times A_1\): \dynkins{A1|A1}
        \item \(A_2\): \dynkin A2 
        \item \(B_2\): \dynkin B2 \footnote{in lectures \(B_2\) was drawn the other way around, i.e. \dynkin[reverse arrows] B2}
        \item \(G_2\): \dynkin G2
    \end{enumerate}
\end{example}
