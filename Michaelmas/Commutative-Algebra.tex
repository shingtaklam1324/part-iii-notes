\lec

In this course, a ring is a commutative unital ring \(R\). One non-commutative exception is the ring \(\End(M)\), where \(M\) is an abelian group. This is a ring with pointwise addition, and composition as multiplication.

\begin{definition}
    [module] An \emph{\(R\)-module}\index{module} \(M\) is an abelian group \(M\) with an fixed ring homomorphism \(\rho : R \to \End(M)\). We will write \(r \cdot m := \rho(r)(m)\).
\end{definition}

\begin{remark}
    By definition, this implies that \((r_1 + r_2) \cdot m = r_1 \cdot m + r_2 \cdot m\), \(r \cdot (m_1 + m_2) = r \cdot m_1 + r \cdot m_2\) and \(r_1 \cdot (r_2 \cdot m) = (r_1r_2)\cdot m\).
\end{remark}

\begin{example}
    [Examples of modules]
    \begin{itemize}
        \item Let \(k\) be a field. Then a \(k\)-module is the same as a \(k\)-vector space.
        \item Every abelian group is a \(\Z\)-module in a unique way, since we must have that \(\rho(1) = \id_M\). Therefore, abelian groups and \(\Z\)-modules are the same thing.
        \item Every ring \(R\) is (trivially) an \(R\)-module.
        \item More generally, \(R^{\oplus \N}\) (direct sum) and \(R^\N\) (direct product) are \(R\)-modules.
    \end{itemize}
\end{example}

\section{Chain conditions}

\begin{definition}
    [Noetherian, Artinian module]
    An \(R\)-module \(M\) is \emph{Noetherian}\index{Noetherian!module} if one of the following (equivalent) conditions hold:
    \begin{enumerate}
        \item Every ascending chain of submodules \(M_0 \sse M_1 \sse M_2 \sse \cdots\) stabilises. That is, it is eventually constant.
        \item Every non-empty set \(\Sigma\) of submodules of \(M\) has a maximal element.
    \end{enumerate}

    \(M\) is \emph{Artinian}\index{Artinian!module} if we replace in the above: ascending with descending, maximal with minimal.
\end{definition}

\begin{lemma}
    An \(R\)-module \(M\) is Noetherian if and only if every submodule of \(M\) is finitely generated.
\end{lemma}

In particular, every Noetherian module is finitely generated. If \(R = \Z[T_1, T_2, \dots]\), with \(M = R\) as an \(R\)-module. Then \(M\) is finitely generated. On the other hand, \(M' = \inner{T_1, T_2, T_3, \dots}\), is not finitely generated.

\begin{definition}
    [Noetherian, Artinian ring]
    A ring \(R\) is Noetherian\index{Noetherian!ring} (resp. Artinian\index{Artinian!ring}) if it is Noetherian (resp. Artinian) as an \(R\)-module.
\end{definition}

\begin{example}
    \begin{enumerate}
        \item \(\Z\) is Noetherian (as it is a PID), but not Artinian (e.g. \(\inner{2} \supseteq \inner{4} \supseteq \inner{8} \supseteq \cdots\)).
        \item \(\Z[1/2]/\Z\) is Artinian, but not Noetherian as a \(\Z\)-module.
        \item A ring \(R\) is Artinian if and only if \(R\) is Noetherian and \(R\) has Krull dimension \(0\).
    \end{enumerate}
\end{example}

\begin{lemma}

	\arrow[from=1-4, to=1-5]
\end{tikzcd}\]
be an SES of \(R\)-modules. Then \(M\) is Noetherian (resp. Artinian) if and only if \(N\) and \(L\) are Noetherian (resp. Artinian).
\end{lemma}

\begin{proof}
    We may assume without loss of generality that \(N\) is a submodule of \(M\). Let \(P_1 \sse P_2 \sse \dots\) be an increasing (resp. decreasing) sequence of submodules of \(M\). In this case,
    \[N \cap P_1 \sse N \cap P_2 \sse \cdots\]
    is an increasing (resp. decreasing) sequence of submodules of \(N\), hence eventually constant. Similarly,
    \[\frac{N + P_1}{N} \sse \frac{N + P_2}{N} \sse \cdots\]
    is an increasing (resp. decreasing) sequence of submodules of \(L = M/N\), hence eventually constant. For large \(n\), we will have
    \[P_n \subseteq P_{n+1} \quad N \cap P_n = N \cap P_{n+1} \quad N + P_n = N + P_{n+1}\]
    Hence \(P_n = P_{n+1}\) for large enough \(n\).
\end{proof}

\begin{corollary}
    If \(M_1, \dots, M_n\) are Noetherian (resp. Artinian) \(R\)-modules, then \(M_1 \oplus \cdots \oplus M_n\) is Noetherian (resp. Artinian).
\end{corollary}

\begin{proof}
    By the lemma and induction.
\end{proof}

Recall a module homomorphism
\[\varphi : M_1 \oplus \cdots \oplus M_n \to L\]
is the same as a collection of module homomorphism \(\varphi_i : M_i \to L\). This is also true for infinite direct sums (but not products!).

\begin{proposition}
    For a Noetherian (resp. Artinian) ring \(R\), every finitely generated \(R\)-module is Noetherian (resp. Artinian).
\end{proposition}

\begin{proof}
    \(M\) is finitely generated if and only if there exists a surjection \(R^n \surj M\) for some \(n \in \N\). The fact that \(R^n\) is Noetherian (resp. Artinian) implies that \(M\) is Noetherian (resp. Artinian), as quotients of Noetherian (resp. Artinian) modules are Noetherian (resp. Artinian).
\end{proof}

\begin{definition}
    [algebra]
    An \emph{\(R\)-algebra}\index{algebra} \(A\) is a ring \(A\) with a fixed ring homomorphism \(\rho : R \to A\). We will write \(r \cdot a := \rho(r)a\).
\end{definition}

\begin{definition}
    [noetherian algebra] An \(R\)-algebra is \emph{Noetherian}\index{Noetherian!algebra} if it is Noetherian as a ring.
\end{definition}

\begin{remark}
    Every \(R\)-algebra is an \(R\)-module.
\end{remark}

\begin{example}
    The polynomial ring \(k[T_1, \dots, T_n]\) is a \(k\)-algebra. Do note however that it is a finitely generated by \(T_1, \dots, T_n\) as a \(k\)-algebra, but it is infinite dimensional as a \(k\)-vector space.
\end{example}

\begin{definition}
    [algebra homomorphism] \(\varphi : A \to B\) is an \emph{\(R\)-algebra homomorphism}\index{algebra!homomorphism} if \(\varphi\) is a ring homomorphism and \(\varphi(r \cdot 1_A) = r \cdot 1_B\).
\end{definition}

\begin{definition}
    [finitely generated algebra]
    An \(R\)-algebra \(A\) is \emph{finitely generated}\index{algebra!finitely generated} if there exists a surjective \(R\)-algebra homomorphism \(R[T_1, \dots, T_n] \surj A\) for some \(n \in \N\).
\end{definition}

\lec

\begin{theorem}
    [Hilbert basis theorem] \index{Hilbert basis theorem}Every finitely generated algebra \(A\) over a Noetherian ring \(R\) is Noetherian (as a ring).
\end{theorem}

For example, if \(k\) is a field, then \(k[T_1, \dots, T_n]\) is Noetherian.

\begin{proof}
    It suffices to prove for \(A = R[T_1, \dots, T_n]\), since every finitely generated algebra is a quotient of \(R[T_1, \dots, T_n]\). Moreover, by induction, suffices to prove the result for \(A = R[T]\).

    Let \(\mfa\) be an ideal of \(A = R[T]\). For every \(i \ge 0\), define
    \[\mfa(i) = \left\{c_0 \mid c_0 t^i + \dots + c_it^0 \in \mfa\right\}\]
    for the set of all leading coeffients of elements of degree \(i\) in \(\mfa\) (and containing \(0\)). In this case, \(\mfa(i) \sse R\) is an ideal, and we have an ascending chain of ideals
    \[\mfa(i) \sse \mfa(i+1) \sse \cdots\]
    Since \(R\) is Noetherian, each \(\mfa\) is finitely generated (as an ideal), and the ascending sequence of ideal stabilises. That is,
    \[\mfa(m') = \mfa(m)\]
    for all \(m' \ge m\). We write \(\mfa(i) = \inner{b_{i, 1}, \dots, b_{i, m_i}}\), where \(b_{i, j} \in R\). Let \(f_{i, j} \in \mfa\) be a polynomial of degree \(i\), with leading coefficient \(b_{i, j}\). Define the new ideal
    \[\mfb = \inner{f_{i, j} \mid i \le m, 1 \le j \le m_i} \idealof R[T]\]
    In this case, \(\mfb(i) = \mfa(i)\) for all \(i\). By construction, \(\mfb \sse \mfa\). 
    
    Suppose for contradiction that \(\mfa \not\sse \mfb\). Take \(f \in \mfa \setminus \mfb\) of minimal degree \(i\). But \(\mfb(i) = \mfa(i)\), and so there exists \(g \in \mfb\), of degree \(i\), and with the same leading coefficient as \(f\). That is, \(\deg(f - g) < i\). By minimality, \(f - g \in \mfb\), and so \(f = (f - g) + g \in \mfb\). Contradiction.
\end{proof}

Therefore, if we have a subset \(S \sse R[T_1, \dots, T_n]/I\), then \(\inner{S} = \inner{S_0}\), where \(S_0 \sse S\) is finite.

\section{Tensor products}

Let \(M, N\) be \(R\)-modules. An informal definition of their tensor product is
\[M \otimes_R N = \left\{\sum_{i=1}^\ell m_i \otimes n_i \ \bigg\vert\ m_i \in M, n_i \in N\right\}\]
where we have the relations \((m_1 + m_2) \otimes n = m_1 \otimes n + m_2 \otimes n\), \(m \otimes (n_1 + n_2) = m \otimes n_1 + m \otimes n_2\), and that for \(r \in R\), \((rm) \otimes n = r (m \otimes n) = m \otimes (rn)\).

For example, consider \(\Z/2 \otimes_\Z \Z/3\). Then
\[x \otimes y = (3x)\otimes y = x \otimes (3y) = x \otimes 0 = 0\]
and so, \(\Z/2 \otimes_\Z \Z/3 = 0\). On the other hand, if we have vector spaces, then
\[\R^m \otimes_\R \R^\ell \cong \R^{m\ell}\]

Recall \(f : M \times N \to L\) is \emph{\(R\)-bilinear} if \(n \mapsto f(m_0, n)\) and \(m \mapsto f(m, n_0)\) are \(R\)-linear for all \(m_0 \in M, n_0 \in N\).

\begin{definition}
    [tensor product of modules] Let \(M, N\) be \(R\)-modules, let
    \[\mcF = R^{\oplus (M \times N)} = \Span_R\left\{e_{(m, n)} \mid m \in m, n \in N\right\}\]
    be the free module indexed by \(m \times n\), and define \(\mcK \sse \mcF\) for the submodule generated by the relations (where we write \((m, n)\) for \(e_{(m, n)}\))
    \begin{align*}
        (m, n_1) + (m, n_2) &= (m, n_1 + n_2) \\
        (m_1, n) + (m_2, n) &= (m_1 + m_2, n) \\
        r(m, n) &= (rm, n) \\
        r(m, n) &= (m, rn)
    \end{align*}
    The \emph{tensor product}\index{tensor product!of \(R\)-modules} is
    \[M \otimes_R N := \frac{\mcF}{\mcK}\]
    We have an \(R\)-bilinear map
    \begin{align*}
        i_{M \otimes N} : M \times N &\to M \otimes_R N \\
        (m, n) & \mapsto m \otimes n
    \end{align*}
\end{definition}

\[\begin{tikzcd}
	{M \times N} && {M \otimes_R N} \\
	\\
	&& L
	\arrow["f"', from=1-1, to=3-3]
	\arrow["{i_{M \otimes N}}", from=1-1, to=1-3]

commute.
\end{proposition}

\begin{proof}
    Uniqueness is clear, since we must have that
    \[h(m \otimes n) = f(m, n)\]
    since the pure tensors generate, \(h\) must be unique, if it exists. Therefore, suffices to show the above extends to an \(R\)-linear map \(M \otimes_R N \to L\). This follows from the map
    \begin{align*}
        R^{\oplus (M \times N)} &\to L \\
        e_{(m, n)} &\mapsto f(m, n) 
    \end{align*}
    extending to a linear map (by the universal property of the direct sum), and that this map vanishes on \(\mcK\). Therefore, \(h\) extends to \(M \otimes_R N\) from the pure tensors.
\end{proof}

\begin{proposition}
    \label{prop:tensor-sum-eq-zero-iff}
    Suppose \(M, N\) are \(R\)-modules, then
    \[\sum_i m_i \otimes n_i = 0 \in M\otimes_R N\]
    if and only if for all \(R\)-modules \(L\), and every \(R\)-bilinear map \(f : M \times N \to L\) has
    \[\sum_i f(m_i, n_i) = 0\]
\end{proposition}

\begin{proof}
    Suppose \(\sum m_i \otimes n_i = 0\), let \(f : M \times N \to L\) be bilinear. Then \(f\) factors through \(M \times N \to M \otimes_R N\), and we can write
    % https://q.uiver.app/#q=WzAsMyxbMCwwLCJNXFx0aW1lcyBOIl0sWzIsMCwiTSBcXG90aW1lcyBOIl0sWzIsMiwiTCJdLFswLDIsImYiLDJdLFswLDEsImkiXSxbMSwyLCJoIl1d
\[\begin{tikzcd}
	{M\times N} && {M \otimes N} \\
	\\
	&& L
	\arrow["f"', from=1-1, to=3-3]
	\arrow["i_{M \otimes N}", from=1-1, to=1-3]
	\arrow["h", from=1-3, to=3-3]
\end{tikzcd}\]
    In this case, we have that
    \[\sum_i f(m_i, n_i) = \sum_i h(i(m, n)) = \sum_i h(m_i \otimes n_i) = h \left(\sum_i m_i \otimes n_i\right) = h(0) = 0\]
    Conversly, if
    \[\sum_i m_i \otimes n_i \ne 0\]
    then by definition,
    \[\sum_i i_{m \otimes n}(m_i, n_i) \ne 0\]
\end{proof}

\begin{example}
    Let \(k\) be a field, and consider the tensor product
    \[k^m \otimes k^\ell\]
    Suppose \(k^m\) has basis \(\{e_1, \dots, e_m\}\), and \(k^\ell\) has basis \(\{f_1, \dots, f_\ell\}\), then
    \[k^m \otimes k^\ell = \Span_k\{v \otimes w \mid v \in k^m, w \in k^\ell\} = \Span_k\left\{e_i \otimes f_j\right\}\]

    \begin{claim}
        \(\{e_i \otimes f_j\}\) is a basis.
    \end{claim}

    \begin{proof}
        Suppose we have
        \[\sum_{ij} \alpha_{ij}(e_i \otimes f_j) = 0\]
        For every \(1 \le a \le m, 1 \le b \le \ell\), define a bilinear map
        \begin{align*}
            T_{ab} : k^m \times k^\ell &\to k \\
            T_{ab}((v_i), (w_j)) = v_aw_b
        \end{align*}
        This is a \(k\)-bilinear map. By \cref{prop:tensor-sum-eq-zero-iff},
        \[0 = \sum_{i,j}\alpha_{ij}T_{ab}(e_i, f_j) = \sum_{i, j}\alpha_{ij}\delta_{ia}\delta_{jb} = \alpha_{ab}\]
    \end{proof}
\end{example}

\begin{example}
    More concretely, let us consider \(\R^2 \otimes \R^2\). We have a basis of size \(4\), given by
    \[e_1 \otimes f_1, e_1 \otimes f_2, e_2\otimes f_1, e_2 \otimes f_2\]
    What do pure tensors look like?
    \begin{align*}
        (\alpha e_1 + \beta e_2) \otimes (\gamma f_1 + \delta f_2) &= \alpha\gamma (e_1 \otimes f_1) + \alpha\delta (e_1 \otimes f_2) + \beta\gamma (e_2 \otimes f_1) + \beta\delta (e_2 \otimes f_2)
    \end{align*}
    These are not generic elements of \(\R^2 \otimes \R^2\), since the vectors
    \[(\alpha\gamma, \alpha\delta) \qqtext{and} (\beta\gamma, \beta\delta)\]
    are linearly dependent. In particular,
    \[e_1 \otimes f_1 + 2e_1 \otimes f_2 + 3e_2 \otimes f_1 + 4e_2 \otimes f_2\]
    is \emph{not} a pure tensor.
\end{example}

\begin{example}
    [warning]
    First consider
    \[\Z \otimes_\Z \Z/2\]
    In this case,
    \[2 \otimes 1 = 1 \otimes 2 = 1 \otimes 0 = 0\]
    Now consider
    \[2\Z \otimes_\Z \Z/2\]
    But in this case,
    \[2 \otimes 1 \ne 0\]
    since we can define a bilinear map
    \begin{align*}
        B : 2\Z \times \Z/2 &\to \Z/2 \\
        B(2m, x) &= mx
    \end{align*}
    In this case,
    \[B(2, 1) = 1 \cdot 1 = 1 \ne 0\]

    However, if \(M' \le M, N' \le N\) are submodules, and
    \[\sum_i m_i \otimes n_i = 0\]
    in \(M' \otimes N'\), then
    \[\sum_i m_i \otimes n_i = 0\]
    in \(M \otimes N\).
\end{example}

\begin{proposition}
    \label{prop:tensor-eq-zero-in-fg-submod}
    If
    \[\sum m_i \otimes n_i = 0 \in M \otimes_R N\]
    then there are finitely generated \(R\)-submodules \(M' \le M, N' \le N\), such that
    \[\sum m_i \otimes n_i = 0 \in M' \otimes_R N'\]
\end{proposition}

Intuitively, a proof that the sum is zero is finite, and so it can only involve finitely many expressions. We can take them to be the generators.

\begin{proof}
    \[\sum m_i \otimes n_i 0 = \in M \otimes N = \frac{R^{\oplus(M \times N)}}{\mcK}\]
    then
    \[\sum_i e_{(m_i, n_i)} = 0 \in \mcK\]
    This means that we can write the left hand side as a finite sum of the generators of \(\mcK\). Taking all the elements of \(M\) and \(N\) which appear, gives the result. 
\end{proof}

\begin{corollary}
    Let \(A, B\) be torsion-free abelian groups, then \(A \otimes_\Z B\) is torsion free.
\end{corollary}

\begin{proof}
    Suppose
    \[n \cdot \left(\sum_i a_i \otimes b_i\right) = 0 \in A \otimes B\]
    for some \(n \ge 1\). By \cref{prop:tensor-eq-zero-in-fg-submod}, there exists finitely generated subgroups \(A' \le A, B' \le B\), such that
    \[n \cdot \left(\sum_i a_i \otimes b_i\right) = 0 \in A' \otimes B'\]
    By the structure theorem of finitely generated abelian groups, \(A' \cong \Z^r, B' \cong \Z^s\), and so we have that
    \[A' \otimes B' \cong \Z^{rs}\]
    which is torsion free. Contradiction.
\end{proof}

\begin{example}
    \[\C^2 \otimes_\C \C^3 \cong \C^6\] 
    as \(\C\)-vector spaces, and we also have that \(\C^6 \cong \R^12\) as \(\R\)-vector spaces. On the other hand,
    \[\C^2 \otimes_\R \C^3 \cong \R^4 \otimes_\R \R^6 \cong \R^{24}\]
\end{example}

\begin{proposition}
    \label{prop:tensor-properties}
    \begin{enumerate}
        \item \(M \otimes N \cong N \otimes N\)
        \item \((M \otimes N) \otimes P \cong M \otimes (N \otimes P) \cong M \otimes N \otimes P\),
        where we define \(M \otimes N \otimes P\) using trilinear maps.
        \item \(\left(\bigoplus_i M_i\right) \otimes P \cong \bigoplus_i(M_i \otimes P)\)
        \item \(R \otimes_R M \cong M\),
    \end{enumerate}
\end{proposition}

\begin{proof}
    See examples sheet 1.
\end{proof}

\begin{example}
    Using \cref{prop:tensor-properties}, we can compute
    \begin{align*}
        R^m \otimes R^\ell &\cong \left(\oplus_{i=1}^m R\right) \otimes \left(\oplus_{j=1}^\ell R\right) \\
        &\cong \oplus_{i, j}R \\
        &\cong R^{m\ell}
    \end{align*}
\end{example}

\subsection{Tensor product of \(R\)-linear maps}

\begin{proposition}
    For \(R\)-linear maps \(f : M \to M', g : N \to N'\), then there exists a unique \(R\)-linear map\index{tensor product!of \(R\)-linear maps}
    \[f \otimes g : M \otimes N \to M' \otimes N'\]
    with
    \[(f \otimes g)(m \otimes n) = f(m) \otimes g(n)\]
\end{proposition}

\begin{proof}
    Uniqueness is clear since the pure tensors generate. For existence, we can use the universal property on the \(R\)-bilinear map
    \begin{align*}
        T : M \times N &\to M' \otimes N' \\
        T(m, n) &= f(m) \otimes g(n)
    \end{align*}
\end{proof}

\lec

\underline{Exercise:} \((f \otimes g) \circ (h \otimes i) = (f \circ h) \otimes (h \circ i)\).

\begin{example}
    Let \(T : k^a \to k^c\) and \(S : k^b \to k^d\) be linear. Then
    \[(T \otimes S)(e_i \otimes e_j) = T(e_i) \otimes S(e_j) = \sum_{\ell, t}[T]_{\ell i}[S]_{tj}(f_\ell \otimes f_t)\]
    where \([T]\) is the matrix representation of \(T\). If we order the basis of \(k^a \otimes k^b\) by
    \[e_1 \otimes e_1, \dots, e_1 \otimes e_c, e_2\otimes e_1, \dots, e_2\otimes e_c, \dots, e_a \otimes e_c\]
    and a similar ordering for the range, then
    \[[T \otimes S] = \begin{pmatrix}
        [T]_{11}S & \cdots & [T]_{1a}S \\
        \vdots &\ddots&\vdots \\
        [T]_{c1}S & \cdots & [T]_{ca}S
    \end{pmatrix}\]
    is the \emph{Kronecker product}\index{Kronecker product} of \([T]\) and \([S]\).
\end{example}

\begin{proposition}
    Let \(f : M \to M', g : N \to N'\) be \(\R\)-linear.
    \begin{enumerate}[(i)]
        \item If \(f, g\) are isomorphisms, then so is \(f \otimes g\),
        \item if \(f\) and \(g\) are surjective, so is \(f \otimes g\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), \((f^{-1} \otimes g^{-1}) = (f \otimes g)^{-1}\), since we have that \((f \otimes g) \circ (h \otimes i) = (f \circ h) \otimes (h \circ i)\).

    For (ii), notice that \(\im(f \otimes g)\) contains all pure tensors in \(M' \otimes N'\).
\end{proof}

\begin{example}
    If \(f: \Z \to \Z\), \(f(n) = pn\), then we have
    \[(f \otimes \id) : \Z \otimes \Z/p \to \Z \otimes \Z/p\]
    is the zero map, as
    \[(f \otimes \id)(a \otimes b) = (pa) \otimes b = a \otimes (pb) = a \otimes 0 = 0\]
    But \(\Z \otimes \Z/p \cong \Z/p\) which is nonzero.
\end{example}

\subsection{Tensor product of algebras}

Let \(B, C\) be \(R\)-algebras. Then we have \(B \otimes_R C\) as an \(R\)-module. We would like to define the multiplication by
\[(b \otimes c)(b' \otimes c') = (bb') \otimes (cc')\]
This is well-defined. Fix \((b, c) \in B \times \C\), then we have a bilinear map
\begin{align*}
    B \times C &\to B\otimes C \\
    (b', c') &\mapsto (bb') \otimes (cc')
\end{align*}
which gives us a map \(B \otimes C \to B'\otimes C'\), with
\[b' \otimes c' \mapsto (bb') \otimes (cc')\]
It is easy to show that this then satisfies the ring axioms. Hence \(B \otimes C\) is a ring. The \(R\)-algebra structure\index{tensor product!of algebras} will be given by
\begin{align*}
    R &\to B \otimes C \\
    r &\mapsto (r1_B) \otimes 1_C = r(1_B \otimes 1_C) = 1_B \otimes (r1_C)
\end{align*}

\begin{example}
    There is an isomorphism
    \[\varphi : R[x_1, \dots, x_n] \otimes_R R[t_1, \dots, t_r] \cong R[x_1, \dots, x_n, t_1, \dots, t_r]\]

    \begin{proof}
        We have an \(R\)-basis for the left hand side, which is
        \[x^k \otimes t^\ell\]
        and we also have a \(R\)-basis for the right hand side,
        \[x^kt^\ell\]
        Define
        \[\varphi(x^k \otimes t^\ell) = x^kt^\ell\]
        which gives us a \(R\)-module isomorphism. Moreover,
        \[\varphi(r \otimes 1) = r1 = 1\]
        and by distributivity, suffices to show
        \[\varphi((x^k \otimes t^\ell)(x^m \otimes t^n)) = x^kt^\ell x^mt^n\]
        which is clear by definition.
    \end{proof}

    More generally,
    \begin{align*}
        \frac{R[x_1, \dots, x_n]}{I} \otimes \frac{R[t_1, \dots, t_r]}{J} \cong \frac{R[x_1, \dots, x_n, t_1, \dots, t_r]}{L} \cong \frac{R[x_1, \dots, x_n, t_1, \dots, t_r]}{I^e + J^e}
    \end{align*}
    where \(I^e = \inner{I} \idealof R[x_1, \dots, x_n, t_1, \dots, t_r]\) denotes the extension of \(I\).
\end{example}

\begin{example}
    \(\frac{\C[x,y,z]}{\inner{f, g}} \otimes \frac{\C[w,u]}{h}\) is isomorphic as \(\C\)-algebras to
    \[\frac{\C[x,y,z,w,u]}{\inner{f,g,h}}\]
\end{example}

\begin{proposition}
    [universal property of tensor product of algebras]\index{universal property of!tensor product of algebras} Let \(A, B\) be \(R\)-algebras, for every \(R\)-algebra \(C\), and \(R\)-algebra homomorphisms \(f_1 : A \to C\) and \(f_2 : B \to C\), there exists a unique \(R\)-algebra map
    \[h : A \otimes B \to C\]
    such that

\[\begin{tikzcd}
	A & {A \otimes B} & B \\
	& C
	\arrow["{i_A}", from=1-1, to=1-2]
	\arrow["{i_B}"', from=1-3, to=1-2]
	\arrow["{f_1}"', from=1-1, to=2-2]
	\arrow["{f_2}", from=1-3, to=2-2]
	\arrow["h"{description}, from=1-2, to=2-2]
\end{tikzcd}\]
commutes, where \(i_A(a) = a \otimes 1\), \(i_B(b) = 1 \otimes b\). Moreover, this characterises \((A \otimes B, i_A, i_B)\) uniquely (up to isomorphism).
\end{proposition}

\begin{proof}
    \(A \otimes B\) is generated, as an \(R\)-algebra, by
    \[\{a \otimes 1 \mid a \in A\} \cup \{1 \otimes b \mid b \in B\}\]
    This then implies the uniqueness of \(h\), as it defines \(h\) on the generators. For the existence, define the bilinear map \(A \times B \to C\), given by
    \[f(a, b) = f_1(a) f_2(b)\]
    Using the universal property of tensor product of modules, there exists \(h : A \otimes B \to C\) which is \(R\)-linear, with
    \[h(a \otimes b) = f_1(a)f_2(b)\]
    It is then easy to show that \(h\) is an algebra homomorphism.
\end{proof}

Consider \(R[x_1, \dots, x_n, t_1, \dots, t_r]\) from above. We have natural embeddings
from \(R[x_1, \dots, x_n]\) and \(R[t_1, \dots, t_n]\). Given \(f_1, f_2\) as above, we see that the image of the \(x_i\) is determined by \(f_1\), and the image of \(t_i\) is determined by \(f_2\). Therefore, 
\[R[x_1, \dots, x_n, t_1, \dots, t_r] \cong \R[x_1, \dots, x_n] \otimes R[t_1, \dots, t_r]\]
as it satisfies the universal property.

\lec

If we have \(f : A \to A', g : B \to B'\) which are algebra homomorphisms, then the tensor product of \(R\)-linear maps,
\[f \otimes g : A \otimes B \to A' \otimes B'\]
is an \(R\)-algebra homomorphism. Moreover, we have \(R\)-algebra isomorphisms
\begin{itemize}
    \item \((R/I) \otimes (R/J) \cong R/(I + J)\)
    \item \(A \otimes B \cong B \otimes A\),
    \item \((A \otimes B) \otimes C \cong A \otimes (B \otimes C)\),
    \item \(A \otimes B^n \cong (A \otimes B)^n\),
\end{itemize}

\subsection{Restriction and extension of scalars}

\subsubsection*{Restriction of scalars}
\index{restriction of scalars}

We will have a ring homomorphisms \(f : R \to S\), let \(M\) be an \(S\)-module, so \(M\) is also an \(R\)-module,
\[r \cdot m := f(r)m\]
for \(r \in R, m \in M\). The fact that this is a module is clear by our definition, since it is just the composition
% https://q.uiver.app/#q=WzAsMyxbMCwwLCJSIl0sWzIsMCwiUyJdLFs0LDAsIlxcRW5kKE0pIl0sWzAsMSwiZiJdLFsxLDJdXQ==
\[\begin{tikzcd}
	R && S && {\End(M)}
	\arrow["f", from=1-1, to=1-3]
	\arrow[from=1-3, to=1-5]
\end{tikzcd}\]

\begin{example}
    If we consider the embedding \(\R \inj \C\), then \(\C^n\) is a \(C\)-vector space, but also an \(\R\)-vector space, of dimension \(2n\).
\end{example}

\subsubsection*{Extension of scalars}
\index{extension of scalars}

Let \(f : R \to S\) be a ring homomorphism, \(M\) be an \(S\)-module (thus an \(R\)-module by restriction of scalars), \(N\) is an \(R\)-module. From this, we can form
\[M \otimes_R N\]
which is an \(R\)-module. In fact, \(M \otimes_R N\) is also an \(S\)-module, with
\[s \cdot (m \otimes n) := (sm) \otimes n\]
Is this well defined? We have an \(R\)-bilinear map
\begin{align*}
    M \times N &\to M \otimes_R N \\
    (m, n) &\mapsto (sm) \otimes n
\end{align*}
By the universal property, we have a map
\begin{align*}
    h_s : M \otimes_R N \to M \otimes_R N
\end{align*}
which is \(R\)-linear, and \(h_s(m \otimes n) = (sm) \otimes n\). Now define
\begin{align*}
    \varphi : S &\to \End(M \otimes_R N) \\
    \varphi(s) &= h_s
\end{align*}
Which is a ring homomorphism, and so, we have an \(S\)-module structure on \(M \otimes_R N\).

\begin{example}
    We know from before that \(S \otimes_R R \cong S\) as \(R\)-module, with
    \[s \otimes r \mapsto s \cdot f(r)\]
    But in fact, this is also \(S\)-linear, since
    \[s'\cdot(s \otimes r) = (s's) \otimes r \mapsto s's \cdot f(r)\]

    For example, this implies that
    \[\C \otimes_\R \R \cong \C\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    If \(M\) is an \(S\)-module, \(N_i\) are \(R\)-modules, then
    \[M \otimes_R \left(\bigoplus_i N_i\right) \cong \bigoplus_i(M \otimes_R N_i)\]
    as \(S\)-modules.

    In this case,
    \[\C \otimes_\R \R^n \cong \C^n\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    Consider \(\C^n\) as a \(\C\)-module. Restricting to \(\R\),
    \[\C^n \cong \R^{2n}\]
    as \(\R\)-vector spaces. Now extending scalars,
    \[\C \otimes_\R \R^{2n} \cong \C^{2n}\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    Now consider \(\R^n\) as an \(\R\)-vector space. Extending scalars,
    \[\R^n \otimes_\R \C \cong \C^n\]
    over \(\C\). Restricting to \(\R\),
    \[\C^n \cong \R^{2n}\]
\end{example}

\begin{example}
    Consider \(\Z^n\) as an \(\Z\)-module, and let \(f : \Z \to \Z/2\) be the quotient map. Extending scalars,
    \[(\Z/2) \otimes_\Z \Z^n \cong (\Z/2)^n\]
\end{example}

\begin{example}
    Consider
    \[\C^n \otimes_\R \R^\ell\]
    One way to compute this:
    \[\C^n \otimes_\R \R^\ell \cong_\R \R^{2n} \otimes \R^\ell \cong_\R \R^{2n\ell} \cong_\R \C^{n\ell}\]
    where \(\cong_\R\) denotes isomorphism as \(\R\)-vector spaces. Another way to do this:
    \[\C^n \otimes_\R \R^\ell \cong_\C \C^n \otimes_\C \left(\C \otimes_\R \R^\ell\right) \cong_\C \C^n \otimes \C^\ell \cong_\C \C^{n\ell}\]
    The first isomorphism is given by
    \[v \otimes u \mapsto v \otimes (1 \otimes u)\]
    Combining these, the isomorphism \(\C^n \otimes_\R \R^\ell \to \C^n \otimes \C^\ell\) sends
    \[v \otimes u \mapsto v\otimes u\]
    where we use the inclusion \(\R^\ell \inj \C^\ell\).
\end{example}

\begin{proposition}
    \label{prop:tensor-ext-scalar}
    Let \(M\) be an \(S\)-module, \(N\) be an \(R\)-module, then
    \[M \otimes_R N \cong M \otimes_S(S \otimes_R N)\]
    as \(S\)-modules. In particular, the isomorphism is given by
    \begin{align*}
        m \otimes n &\mapsto m \otimes (1 \otimes n) \\
        (sm) \otimes n &\mapsfrom m \otimes (s\otimes n)
    \end{align*}
\end{proposition}

\begin{proposition}
    \label{prop:tensor-ext-scalar-properties}
    Let \(M, M'\) be \(S\)-modules, \(N, N'\) be \(R\)-modules, then we have \(S\)-module isomorphisms
    \begin{enumerate}
        \item \(M \otimes_R N \cong N \otimes_R M\), via \(m \otimes n \to n \otimes m\)
        \item \((M \otimes_R N) \otimes_R N' \cong M \otimes_R (N \otimes_R N')\)
        \item \((M \otimes_R N) \otimes_S M' \cong M \otimes_S (N \otimes_R M')\)
        \item \(M \otimes_R (\bigoplus_i N_i) \cong \bigoplus_i (M \otimes_R N_i)\)
    \end{enumerate}
\end{proposition}

\begin{proof}
    We will prove (iii). Using \cref{prop:tensor-ext-scalar}, we have
    \begin{align*}
        (M \otimes_R N) \otimes_S M' &\cong \left(M \otimes_S (N \otimes_R S)\right) \otimes_S M' \\
        &\cong M \otimes_S \left((N \otimes_R S) \otimes_S M'\right) \\
        &\cong M \otimes_S (N \otimes_R M')
    \end{align*}
\end{proof}

\begin{example}
    As \(\C\)-vector spaces,
    \[\C \otimes_\R (\R^\ell \otimes_\R \R^k) \cong (\C \otimes_\R \R^\ell) \otimes_\C (\C \otimes_\R \R^k) \cong \C^\ell \otimes \C^k \cong \C^{\ell k}\]
\end{example}

\begin{corollary}
    If \(N, N'\) are \(R\)-modules, then 
    \[S \otimes_R (N \otimes N') \cong_S (S \otimes_R N) \otimes_S (S \otimes N')\]
\end{corollary}

\begin{proof}
    By \cref{prop:tensor-ext-scalar} and \cref{prop:tensor-ext-scalar-properties} (ii):
    \[S \otimes_R (N \otimes_\R N') \cong (S \otimes_R N) \otimes_R N' \cong (S \otimes_R N) \otimes_S (S \otimes_R N')\]
\end{proof}
