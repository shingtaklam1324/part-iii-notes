\lec

In this course, a ring is a commutative unital ring \(R\). One non-commutative exception is the ring \(\End(M)\), where \(M\) is an abelian group. This is a ring with pointwise addition, and composition as multiplication.

\begin{definition}
    [module] An \emph{\(R\)-module}\index{module} \(M\) is an abelian group \(M\) with an fixed ring homomorphism \(\rho : R \to \End(M)\). We will write \(r \cdot m := \rho(r)(m)\).
\end{definition}

\begin{remark}
    By definition, this implies that \((r_1 + r_2) \cdot m = r_1 \cdot m + r_2 \cdot m\), \(r \cdot (m_1 + m_2) = r \cdot m_1 + r \cdot m_2\) and \(r_1 \cdot (r_2 \cdot m) = (r_1r_2)\cdot m\).
\end{remark}

\begin{example}
    [Examples of modules]
    \begin{itemize}
        \item Let \(k\) be a field. Then a \(k\)-module is the same as a \(k\)-vector space.
        \item Every abelian group is a \(\Z\)-module in a unique way, since we must have that \(\rho(1) = \id_M\). Therefore, abelian groups and \(\Z\)-modules are the same thing.
        \item Every ring \(R\) is (trivially) an \(R\)-module.
        \item More generally, \(R^{\oplus \N}\) (direct sum) and \(R^\N\) (direct product) are \(R\)-modules.
    \end{itemize}
\end{example}

\section{Chain conditions}

\begin{definition}
    [Noetherian, Artinian module]
    An \(R\)-module \(M\) is \emph{Noetherian}\index{Noetherian!module} if one of the following (equivalent) conditions hold:
    \begin{enumerate}
        \item Every ascending chain of submodules \(M_0 \sse M_1 \sse M_2 \sse \cdots\) stabilises. That is, it is eventually constant.
        \item Every non-empty set \(\Sigma\) of submodules of \(M\) has a maximal element.
    \end{enumerate}

    \(M\) is \emph{Artinian}\index{Artinian!module} if we replace in the above: ascending with descending, maximal with minimal.
\end{definition}

\begin{lemma}
    An \(R\)-module \(M\) is Noetherian if and only if every submodule of \(M\) is finitely generated.
\end{lemma}

In particular, every Noetherian module is finitely generated. If \(R = \Z[T_1, T_2, \dots]\), with \(M = R\) as an \(R\)-module. Then \(M\) is finitely generated. On the other hand, \(M' = \inner{T_1, T_2, T_3, \dots}\), is not finitely generated.

\begin{definition}
    [Noetherian, Artinian ring]
    A ring \(R\) is Noetherian\index{Noetherian!ring} (resp. Artinian\index{Artinian!ring}) if it is Noetherian (resp. Artinian) as an \(R\)-module.
\end{definition}

\begin{example}
    \begin{enumerate}
        \item \(\Z\) is Noetherian (as it is a PID), but not Artinian (e.g. \(\inner{2} \supseteq \inner{4} \supseteq \inner{8} \supseteq \cdots\)).
        \item \(\Z[1/2]/\Z\) is Artinian, but not Noetherian as a \(\Z\)-module.
        \item A ring \(R\) is Artinian if and only if \(R\) is Noetherian and \(R\) has Krull dimension \(0\).
    \end{enumerate}
\end{example}

\begin{lemma}

	\arrow[from=1-4, to=1-5]
\end{tikzcd}\]
be an SES of \(R\)-modules. Then \(M\) is Noetherian (resp. Artinian) if and only if \(N\) and \(L\) are Noetherian (resp. Artinian).
\end{lemma}

\begin{proof}
    We may assume without loss of generality that \(N\) is a submodule of \(M\). Let \(P_1 \sse P_2 \sse \dots\) be an increasing (resp. decreasing) sequence of submodules of \(M\). In this case,
    \[N \cap P_1 \sse N \cap P_2 \sse \cdots\]
    is an increasing (resp. decreasing) sequence of submodules of \(N\), hence eventually constant. Similarly,
    \[\frac{N + P_1}{N} \sse \frac{N + P_2}{N} \sse \cdots\]
    is an increasing (resp. decreasing) sequence of submodules of \(L = M/N\), hence eventually constant. For large \(n\), we will have
    \[P_n \subseteq P_{n+1} \quad N \cap P_n = N \cap P_{n+1} \quad N + P_n = N + P_{n+1}\]
    Hence \(P_n = P_{n+1}\) for large enough \(n\).
\end{proof}

\begin{corollary}
    If \(M_1, \dots, M_n\) are Noetherian (resp. Artinian) \(R\)-modules, then \(M_1 \oplus \cdots \oplus M_n\) is Noetherian (resp. Artinian).
\end{corollary}

\begin{proof}
    By the lemma and induction.
\end{proof}

Recall a module homomorphism
\[\varphi : M_1 \oplus \cdots \oplus M_n \to L\]
is the same as a collection of module homomorphism \(\varphi_i : M_i \to L\). This is also true for infinite direct sums (but not products!).

\begin{proposition}
    For a Noetherian (resp. Artinian) ring \(R\), every finitely generated \(R\)-module is Noetherian (resp. Artinian).
\end{proposition}

\begin{proof}
    \(M\) is finitely generated if and only if there exists a surjection \(R^n \surj M\) for some \(n \in \N\). The fact that \(R^n\) is Noetherian (resp. Artinian) implies that \(M\) is Noetherian (resp. Artinian), as quotients of Noetherian (resp. Artinian) modules are Noetherian (resp. Artinian).
\end{proof}

\begin{definition}
    [algebra]
    An \emph{\(R\)-algebra}\index{algebra} \(A\) is a ring \(A\) with a fixed ring homomorphism \(\rho : R \to A\). We will write \(r \cdot a := \rho(r)a\).
\end{definition}

\begin{definition}
    [noetherian algebra] An \(R\)-algebra is \emph{Noetherian}\index{Noetherian!algebra} if it is Noetherian as a ring.
\end{definition}

\begin{remark}
    Every \(R\)-algebra is an \(R\)-module.
\end{remark}

\begin{example}
    The polynomial ring \(k[T_1, \dots, T_n]\) is a \(k\)-algebra. Do note however that it is a finitely generated by \(T_1, \dots, T_n\) as a \(k\)-algebra, but it is infinite dimensional as a \(k\)-vector space.
\end{example}

\begin{definition}
    [algebra homomorphism] \(\varphi : A \to B\) is an \emph{\(R\)-algebra homomorphism}\index{algebra!homomorphism} if \(\varphi\) is a ring homomorphism and \(\varphi(r \cdot 1_A) = r \cdot 1_B\).
\end{definition}

\begin{definition}
    [finitely generated algebra]
    An \(R\)-algebra \(A\) is \emph{finitely generated}\index{algebra!finitely generated} if there exists a surjective \(R\)-algebra homomorphism \(R[T_1, \dots, T_n] \surj A\) for some \(n \in \N\).
\end{definition}

\lec

\begin{theorem}
    [Hilbert basis theorem] \index{Hilbert basis theorem}Every finitely generated algebra \(A\) over a Noetherian ring \(R\) is Noetherian (as a ring).
\end{theorem}

For example, if \(k\) is a field, then \(k[T_1, \dots, T_n]\) is Noetherian.

\begin{proof}
    It suffices to prove for \(A = R[T_1, \dots, T_n]\), since every finitely generated algebra is a quotient of \(R[T_1, \dots, T_n]\). Moreover, by induction, suffices to prove the result for \(A = R[T]\).

    Let \(\mfa\) be an ideal of \(A = R[T]\). For every \(i \ge 0\), define
    \[\mfa(i) = \left\{c_0 \mid c_0 t^i + \dots + c_it^0 \in \mfa\right\}\]
    for the set of all leading coeffients of elements of degree \(i\) in \(\mfa\) (and containing \(0\)). In this case, \(\mfa(i) \sse R\) is an ideal, and we have an ascending chain of ideals
    \[\mfa(i) \sse \mfa(i+1) \sse \cdots\]
    Since \(R\) is Noetherian, each \(\mfa\) is finitely generated (as an ideal), and the ascending sequence of ideal stabilises. That is,
    \[\mfa(m') = \mfa(m)\]
    for all \(m' \ge m\). We write \(\mfa(i) = \inner{b_{i, 1}, \dots, b_{i, m_i}}\), where \(b_{i, j} \in R\). Let \(f_{i, j} \in \mfa\) be a polynomial of degree \(i\), with leading coefficient \(b_{i, j}\). Define the new ideal
    \[\mfb = \inner{f_{i, j} \mid i \le m, 1 \le j \le m_i} \idealof R[T]\]
    In this case, \(\mfb(i) = \mfa(i)\) for all \(i\). By construction, \(\mfb \sse \mfa\). 
    
    Suppose for contradiction that \(\mfa \not\sse \mfb\). Take \(f \in \mfa \setminus \mfb\) of minimal degree \(i\). But \(\mfb(i) = \mfa(i)\), and so there exists \(g \in \mfb\), of degree \(i\), and with the same leading coefficient as \(f\). That is, \(\deg(f - g) < i\). By minimality, \(f - g \in \mfb\), and so \(f = (f - g) + g \in \mfb\). Contradiction.
\end{proof}

Therefore, if we have a subset \(S \sse R[T_1, \dots, T_n]/I\), then \(\inner{S} = \inner{S_0}\), where \(S_0 \sse S\) is finite.

\section{Tensor products}

Let \(M, N\) be \(R\)-modules. An informal definition of their tensor product is
\[M \otimes_R N = \left\{\sum_{i=1}^\ell m_i \otimes n_i \ \bigg\vert\ m_i \in M, n_i \in N\right\}\]
where we have the relations \((m_1 + m_2) \otimes n = m_1 \otimes n + m_2 \otimes n\), \(m \otimes (n_1 + n_2) = m \otimes n_1 + m \otimes n_2\), and that for \(r \in R\), \((rm) \otimes n = r (m \otimes n) = m \otimes (rn)\).

For example, consider \(\Z/2 \otimes_\Z \Z/3\). Then
\[x \otimes y = (3x)\otimes y = x \otimes (3y) = x \otimes 0 = 0\]
and so, \(\Z/2 \otimes_\Z \Z/3 = 0\). On the other hand, if we have vector spaces, then
\[\R^m \otimes_\R \R^\ell \cong \R^{m\ell}\]

Recall \(f : M \times N \to L\) is \emph{\(R\)-bilinear} if \(n \mapsto f(m_0, n)\) and \(m \mapsto f(m, n_0)\) are \(R\)-linear for all \(m_0 \in M, n_0 \in N\).

\begin{definition}
    [tensor product of modules] Let \(M, N\) be \(R\)-modules, let
    \[\mcF = R^{\oplus (M \times N)} = \Span_R\left\{e_{(m, n)} \mid m \in m, n \in N\right\}\]
    be the free module indexed by \(m \times n\), and define \(\mcK \sse \mcF\) for the submodule generated by the relations (where we write \((m, n)\) for \(e_{(m, n)}\))
    \begin{align*}
        (m, n_1) + (m, n_2) &= (m, n_1 + n_2) \\
        (m_1, n) + (m_2, n) &= (m_1 + m_2, n) \\
        r(m, n) &= (rm, n) \\
        r(m, n) &= (m, rn)
    \end{align*}
    The \emph{tensor product}\index{tensor product!of \(R\)-modules} is
    \[M \otimes_R N := \frac{\mcF}{\mcK}\]
    We have an \(R\)-bilinear map
    \begin{align*}
        i_{M \otimes N} : M \times N &\to M \otimes_R N \\
        (m, n) & \mapsto m \otimes n
    \end{align*}
\end{definition}

\[\begin{tikzcd}
	{M \times N} && {M \otimes_R N} \\
	\\
	&& L
	\arrow["f"', from=1-1, to=3-3]
	\arrow["{i_{M \otimes N}}", from=1-1, to=1-3]

commute.
\end{proposition}

\begin{proof}
    Uniqueness is clear, since we must have that
    \[h(m \otimes n) = f(m, n)\]
    since the pure tensors generate, \(h\) must be unique, if it exists. Therefore, suffices to show the above extends to an \(R\)-linear map \(M \otimes_R N \to L\). This follows from the map
    \begin{align*}
        R^{\oplus (M \times N)} &\to L \\
        e_{(m, n)} &\mapsto f(m, n) 
    \end{align*}
    extending to a linear map (by the universal property of the direct sum), and that this map vanishes on \(\mcK\). Therefore, \(h\) extends to \(M \otimes_R N\) from the pure tensors.
\end{proof}

\begin{proposition}
    \label{prop:tensor-sum-eq-zero-iff}
    Suppose \(M, N\) are \(R\)-modules, then
    \[\sum_i m_i \otimes n_i = 0 \in M\otimes_R N\]
    if and only if for all \(R\)-modules \(L\), and every \(R\)-bilinear map \(f : M \times N \to L\) has
    \[\sum_i f(m_i, n_i) = 0\]
\end{proposition}

\begin{proof}
    Suppose \(\sum m_i \otimes n_i = 0\), let \(f : M \times N \to L\) be bilinear. Then \(f\) factors through \(M \times N \to M \otimes_R N\), and we can write
    % https://q.uiver.app/#q=WzAsMyxbMCwwLCJNXFx0aW1lcyBOIl0sWzIsMCwiTSBcXG90aW1lcyBOIl0sWzIsMiwiTCJdLFswLDIsImYiLDJdLFswLDEsImkiXSxbMSwyLCJoIl1d
\[\begin{tikzcd}
	{M\times N} && {M \otimes N} \\
	\\
	&& L
	\arrow["f"', from=1-1, to=3-3]
	\arrow["i_{M \otimes N}", from=1-1, to=1-3]
	\arrow["h", from=1-3, to=3-3]
\end{tikzcd}\]
    In this case, we have that
    \[\sum_i f(m_i, n_i) = \sum_i h(i(m, n)) = \sum_i h(m_i \otimes n_i) = h \left(\sum_i m_i \otimes n_i\right) = h(0) = 0\]
    Conversly, if
    \[\sum_i m_i \otimes n_i \ne 0\]
    then by definition,
    \[\sum_i i_{m \otimes n}(m_i, n_i) \ne 0\]
\end{proof}

\begin{example}
    Let \(k\) be a field, and consider the tensor product
    \[k^m \otimes k^\ell\]
    Suppose \(k^m\) has basis \(\{e_1, \dots, e_m\}\), and \(k^\ell\) has basis \(\{f_1, \dots, f_\ell\}\), then
    \[k^m \otimes k^\ell = \Span_k\{v \otimes w \mid v \in k^m, w \in k^\ell\} = \Span_k\left\{e_i \otimes f_j\right\}\]

    \begin{claim}
        \(\{e_i \otimes f_j\}\) is a basis.
    \end{claim}

    \begin{proof}
        Suppose we have
        \[\sum_{ij} \alpha_{ij}(e_i \otimes f_j) = 0\]
        For every \(1 \le a \le m, 1 \le b \le \ell\), define a bilinear map
        \begin{align*}
            T_{ab} : k^m \times k^\ell &\to k \\
            T_{ab}((v_i), (w_j)) = v_aw_b
        \end{align*}
        This is a \(k\)-bilinear map. By \cref{prop:tensor-sum-eq-zero-iff},
        \[0 = \sum_{i,j}\alpha_{ij}T_{ab}(e_i, f_j) = \sum_{i, j}\alpha_{ij}\delta_{ia}\delta_{jb} = \alpha_{ab}\]
    \end{proof}
\end{example}

\begin{example}
    More concretely, let us consider \(\R^2 \otimes \R^2\). We have a basis of size \(4\), given by
    \[e_1 \otimes f_1, e_1 \otimes f_2, e_2\otimes f_1, e_2 \otimes f_2\]
    What do pure tensors look like?
    \begin{align*}
        (\alpha e_1 + \beta e_2) \otimes (\gamma f_1 + \delta f_2) &= \alpha\gamma (e_1 \otimes f_1) + \alpha\delta (e_1 \otimes f_2) + \beta\gamma (e_2 \otimes f_1) + \beta\delta (e_2 \otimes f_2)
    \end{align*}
    These are not generic elements of \(\R^2 \otimes \R^2\), since the vectors
    \[(\alpha\gamma, \alpha\delta) \qqtext{and} (\beta\gamma, \beta\delta)\]
    are linearly dependent. In particular,
    \[e_1 \otimes f_1 + 2e_1 \otimes f_2 + 3e_2 \otimes f_1 + 4e_2 \otimes f_2\]
    is \emph{not} a pure tensor.
\end{example}

\begin{example}
    [warning]
    First consider
    \[\Z \otimes_\Z \Z/2\]
    In this case,
    \[2 \otimes 1 = 1 \otimes 2 = 1 \otimes 0 = 0\]
    Now consider
    \[2\Z \otimes_\Z \Z/2\]
    But in this case,
    \[2 \otimes 1 \ne 0\]
    since we can define a bilinear map
    \begin{align*}
        B : 2\Z \times \Z/2 &\to \Z/2 \\
        B(2m, x) &= mx
    \end{align*}
    In this case,
    \[B(2, 1) = 1 \cdot 1 = 1 \ne 0\]

    However, if \(M' \le M, N' \le N\) are submodules, and
    \[\sum_i m_i \otimes n_i = 0\]

    \[\sum_i m_i \otimes n_i = 0\]
    in \(M \otimes N\).
\end{example}

\begin{proposition}
    \label{prop:tensor-eq-zero-in-fg-submod}
    If
    \[\sum m_i \otimes n_i = 0 \in M \otimes_R N\]
    then there are finitely generated \(R\)-submodules \(M' \le M, N' \le N\), such that
    \[\sum m_i \otimes n_i = 0 \in M' \otimes_R N'\]
\end{proposition}

Intuitively, a proof that the sum is zero is finite, and so it can only involve finitely many expressions. We can take them to be the generators.

\begin{proof}
    \[\sum m_i \otimes n_i 0 = \in M \otimes N = \frac{R^{\oplus(M \times N)}}{\mcK}\]
    then
    \[\sum_i e_{(m_i, n_i)} = 0 \in \mcK\]
    This means that we can write the left hand side as a finite sum of the generators of \(\mcK\). Taking all the elements of \(M\) and \(N\) which appear, gives the result. 
\end{proof}

\begin{corollary}
    Let \(A, B\) be torsion-free abelian groups, then \(A \otimes_\Z B\) is torsion free.
\end{corollary}

\begin{proof}
    Suppose
    \[n \cdot \left(\sum_i a_i \otimes b_i\right) = 0 \in A \otimes B\]
    for some \(n \ge 1\). By \cref{prop:tensor-eq-zero-in-fg-submod}, there exists finitely generated subgroups \(A' \le A, B' \le B\), such that
    \[n \cdot \left(\sum_i a_i \otimes b_i\right) = 0 \in A' \otimes B'\]
    By the structure theorem of finitely generated abelian groups, \(A' \cong \Z^r, B' \cong \Z^s\), and so we have that
    \[A' \otimes B' \cong \Z^{rs}\]
    which is torsion free. Contradiction.
\end{proof}

\begin{example}
    \[\C^2 \otimes_\C \C^3 \cong \C^6\] 
    as \(\C\)-vector spaces, and we also have that \(\C^6 \cong \R^12\) as \(\R\)-vector spaces. On the other hand,
    \[\C^2 \otimes_\R \C^3 \cong \R^4 \otimes_\R \R^6 \cong \R^{24}\]
\end{example}

\begin{proposition}
    \label{prop:tensor-properties}
    \begin{enumerate}
        \item \(M \otimes N \cong N \otimes N\)
        \item \((M \otimes N) \otimes P \cong M \otimes (N \otimes P) \cong M \otimes N \otimes P\),
        where we define \(M \otimes N \otimes P\) using trilinear maps.
        \item \(\left(\bigoplus_i M_i\right) \otimes P \cong \bigoplus_i(M_i \otimes P)\)
        \item \(R \otimes_R M \cong M\),
    \end{enumerate}
\end{proposition}

\begin{proof}
    See examples sheet 1.
\end{proof}

\begin{example}
    Using \cref{prop:tensor-properties}, we can compute
    \begin{align*}
        R^m \otimes R^\ell &\cong \left(\oplus_{i=1}^m R\right) \otimes \left(\oplus_{j=1}^\ell R\right) \\
        &\cong \oplus_{i, j}R \\
        &\cong R^{m\ell}
    \end{align*}
\end{example}

\subsection{Tensor product of \(R\)-linear maps}

\begin{proposition}
    For \(R\)-linear maps \(f : M \to M', g : N \to N'\), then there exists a unique \(R\)-linear map\index{tensor product!of \(R\)-linear maps}
    \[f \otimes g : M \otimes N \to M' \otimes N'\]
    with
    \[(f \otimes g)(m \otimes n) = f(m) \otimes g(n)\]
\end{proposition}

\begin{proof}
    Uniqueness is clear since the pure tensors generate. For existence, we can use the universal property on the \(R\)-bilinear map
    \begin{align*}
        T : M \times N &\to M' \otimes N' \\
        T(m, n) &= f(m) \otimes g(n)
    \end{align*}
\end{proof}

\lec

\underline{Exercise:} \((f \otimes g) \circ (h \otimes i) = (f \circ h) \otimes (h \circ i)\).

\begin{example}
    Let \(T : k^a \to k^c\) and \(S : k^b \to k^d\) be linear. Then
    \[(T \otimes S)(e_i \otimes e_j) = T(e_i) \otimes S(e_j) = \sum_{\ell, t}[T]_{\ell i}[S]_{tj}(f_\ell \otimes f_t)\]
    where \([T]\) is the matrix representation of \(T\). If we order the basis of \(k^a \otimes k^b\) by
    \[e_1 \otimes e_1, \dots, e_1 \otimes e_c, e_2\otimes e_1, \dots, e_2\otimes e_c, \dots, e_a \otimes e_c\]
    and a similar ordering for the range, then
    \[[T \otimes S] = \begin{pmatrix}
        [T]_{11}S & \cdots & [T]_{1a}S \\
        \vdots &\ddots&\vdots \\
        [T]_{c1}S & \cdots & [T]_{ca}S
    \end{pmatrix}\]
    is the \emph{Kronecker product}\index{Kronecker product} of \([T]\) and \([S]\).
\end{example}

\begin{proposition}
    Let \(f : M \to M', g : N \to N'\) be \(\R\)-linear.
    \begin{enumerate}[(i)]
        \item If \(f, g\) are isomorphisms, then so is \(f \otimes g\),
        \item if \(f\) and \(g\) are surjective, so is \(f \otimes g\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), \((f^{-1} \otimes g^{-1}) = (f \otimes g)^{-1}\), since we have that \((f \otimes g) \circ (h \otimes i) = (f \circ h) \otimes (h \circ i)\).

    For (ii), notice that \(\im(f \otimes g)\) contains all pure tensors in \(M' \otimes N'\).
\end{proof}

\begin{example}
    If \(f: \Z \to \Z\), \(f(n) = pn\), then we have
    \[(f \otimes \id) : \Z \otimes \Z/p \to \Z \otimes \Z/p\]
    is the zero map, as
    \[(f \otimes \id)(a \otimes b) = (pa) \otimes b = a \otimes (pb) = a \otimes 0 = 0\]
    But \(\Z \otimes \Z/p \cong \Z/p\) which is nonzero.
\end{example}

\subsection{Tensor product of algebras}

Let \(B, C\) be \(R\)-algebras. Then we have \(B \otimes_R C\) as an \(R\)-module. We would like to define the multiplication by
\[(b \otimes c)(b' \otimes c') = (bb') \otimes (cc')\]
This is well-defined. Fix \((b, c) \in B \times \C\), then we have a bilinear map
\begin{align*}
    B \times C &\to B\otimes C \\
    (b', c') &\mapsto (bb') \otimes (cc')
\end{align*}
which gives us a map \(B \otimes C \to B'\otimes C'\), with
\[b' \otimes c' \mapsto (bb') \otimes (cc')\]
It is easy to show that this then satisfies the ring axioms. Hence \(B \otimes C\) is a ring. The \(R\)-algebra structure\index{tensor product!of algebras} will be given by
\begin{align*}
    R &\to B \otimes C \\
    r &\mapsto (r1_B) \otimes 1_C = r(1_B \otimes 1_C) = 1_B \otimes (r1_C)
\end{align*}

\begin{example}
    There is an isomorphism
    \[\varphi : R[x_1, \dots, x_n] \otimes_R R[t_1, \dots, t_r] \cong R[x_1, \dots, x_n, t_1, \dots, t_r]\]

    \begin{proof}
        We have an \(R\)-basis for the left hand side, which is
        \[x^k \otimes t^\ell\]
        and we also have a \(R\)-basis for the right hand side,
        \[x^kt^\ell\]
        Define
        \[\varphi(x^k \otimes t^\ell) = x^kt^\ell\]
        which gives us a \(R\)-module isomorphism. Moreover,
        \[\varphi(r \otimes 1) = r1 = 1\]
        and by distributivity, suffices to show
        \[\varphi((x^k \otimes t^\ell)(x^m \otimes t^n)) = x^kt^\ell x^mt^n\]
        which is clear by definition.
    \end{proof}

    More generally,
    \begin{align*}
        \frac{R[x_1, \dots, x_n]}{I} \otimes \frac{R[t_1, \dots, t_r]}{J} \cong \frac{R[x_1, \dots, x_n, t_1, \dots, t_r]}{L} \cong \frac{R[x_1, \dots, x_n, t_1, \dots, t_r]}{I^e + J^e}
    \end{align*}
    where \(I^e = \inner{I} \idealof R[x_1, \dots, x_n, t_1, \dots, t_r]\) denotes the extension of \(I\).
\end{example}

\begin{example}
    \(\frac{\C[x,y,z]}{\inner{f, g}} \otimes \frac{\C[w,u]}{h}\) is isomorphic as \(\C\)-algebras to
    \[\frac{\C[x,y,z,w,u]}{\inner{f,g,h}}\]
\end{example}

    \[h : A \otimes B \to C\]
    such that

\begin{proof}
    \(A \otimes B\) is generated, as an \(R\)-algebra, by
    \[\{a \otimes 1 \mid a \in A\} \cup \{1 \otimes b \mid b \in B\}\]
    This then implies the uniqueness of \(h\), as it defines \(h\) on the generators. For the existence, define the bilinear map \(A \times B \to C\), given by

    Using the universal property of tensor product of modules, there exists \(h : A \otimes B \to C\) which is \(R\)-linear, with

Consider \(R[x_1, \dots, x_n, t_1, \dots, t_r]\) from above. We have natural embeddings
from \(R[x_1, \dots, x_n]\) and \(R[t_1, \dots, t_n]\). Given \(f_1, f_2\) as above, we see that the image of the \(x_i\) is determined by \(f_1\), and the image of \(t_i\) is determined by \(f_2\). Therefore, 
\[R[x_1, \dots, x_n, t_1, \dots, t_r] \cong \R[x_1, \dots, x_n] \otimes R[t_1, \dots, t_r]\]
as it satisfies the universal property.

\lec

If we have \(f : A \to A', g : B \to B'\) which are algebra homomorphisms, then the tensor product of \(R\)-linear maps,
\[f \otimes g : A \otimes B \to A' \otimes B'\]
is an \(R\)-algebra homomorphism. Moreover, we have \(R\)-algebra isomorphisms
\begin{itemize}
    \item \((R/I) \otimes (R/J) \cong R/(I + J)\)
    \item \(A \otimes B \cong B \otimes A\),
    \item \((A \otimes B) \otimes C \cong A \otimes (B \otimes C)\),
    \item \(A \otimes B^n \cong (A \otimes B)^n\),
\end{itemize}

\subsection{Restriction and extension of scalars}

\subsubsection*{Restriction of scalars}
\index{restriction of scalars}

We will have a ring homomorphisms \(f : R \to S\), let \(M\) be an \(S\)-module, so \(M\) is also an \(R\)-module,
\[r \cdot m := f(r)m\]
for \(r \in R, m \in M\). The fact that this is a module is clear by our definition, since it is just the composition
% https://q.uiver.app/#q=WzAsMyxbMCwwLCJSIl0sWzIsMCwiUyJdLFs0LDAsIlxcRW5kKE0pIl0sWzAsMSwiZiJdLFsxLDJdXQ==
\[\begin{tikzcd}
	R && S && {\End(M)}
	\arrow["f", from=1-1, to=1-3]
	\arrow[from=1-3, to=1-5]
\end{tikzcd}\]

\begin{example}
    If we consider the embedding \(\R \inj \C\), then \(\C^n\) is a \(C\)-vector space, but also an \(\R\)-vector space, of dimension \(2n\).
\end{example}

\subsubsection*{Extension of scalars}
\index{extension of scalars!of modules}

Let \(f : R \to S\) be a ring homomorphism, \(M\) be an \(S\)-module (thus an \(R\)-module by restriction of scalars), \(N\) is an \(R\)-module. From this, we can form
\[M \otimes_R N\]
which is an \(R\)-module. In fact, \(M \otimes_R N\) is also an \(S\)-module, with
\[s \cdot (m \otimes n) := (sm) \otimes n\]
Is this well defined? We have an \(R\)-bilinear map
\begin{align*}
    M \times N &\to M \otimes_R N \\
    (m, n) &\mapsto (sm) \otimes n
\end{align*}
By the universal property, we have a map
\begin{align*}
    h_s : M \otimes_R N \to M \otimes_R N
\end{align*}
which is \(R\)-linear, and \(h_s(m \otimes n) = (sm) \otimes n\). Now define
\begin{align*}
    \varphi : S &\to \End(M \otimes_R N) \\
    \varphi(s) &= h_s
\end{align*}
Which is a ring homomorphism, and so, we have an \(S\)-module structure on \(M \otimes_R N\).

\begin{example}
    We know from before that \(S \otimes_R R \cong S\) as \(R\)-module, with
    \[s \otimes r \mapsto s \cdot f(r)\]
    But in fact, this is also \(S\)-linear, since
    \[s'\cdot(s \otimes r) = (s's) \otimes r \mapsto s's \cdot f(r)\]

    For example, this implies that
    \[\C \otimes_\R \R \cong \C\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    If \(M\) is an \(S\)-module, \(N_i\) are \(R\)-modules, then
    \[M \otimes_R \left(\bigoplus_i N_i\right) \cong \bigoplus_i(M \otimes_R N_i)\]
    as \(S\)-modules.

    In this case,
    \[\C \otimes_\R \R^n \cong \C^n\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    Consider \(\C^n\) as a \(\C\)-module. Restricting to \(\R\),
    \[\C^n \cong \R^{2n}\]
    as \(\R\)-vector spaces. Now extending scalars,
    \[\C \otimes_\R \R^{2n} \cong \C^{2n}\]
    as \(\C\)-vector spaces.
\end{example}

\begin{example}
    Now consider \(\R^n\) as an \(\R\)-vector space. Extending scalars,
    \[\R^n \otimes_\R \C \cong \C^n\]
    over \(\C\). Restricting to \(\R\),
    \[\C^n \cong \R^{2n}\]
\end{example}

\begin{example}
    Consider \(\Z^n\) as an \(\Z\)-module, and let \(f : \Z \to \Z/2\) be the quotient map. Extending scalars,
    \[(\Z/2) \otimes_\Z \Z^n \cong (\Z/2)^n\]
\end{example}

\begin{example}
    Consider
    \[\C^n \otimes_\R \R^\ell\]
    One way to compute this:
    \[\C^n \otimes_\R \R^\ell \cong_\R \R^{2n} \otimes \R^\ell \cong_\R \R^{2n\ell} \cong_\R \C^{n\ell}\]
    where \(\cong_\R\) denotes isomorphism as \(\R\)-vector spaces. Another way to do this:
    \[\C^n \otimes_\R \R^\ell \cong_\C \C^n \otimes_\C \left(\C \otimes_\R \R^\ell\right) \cong_\C \C^n \otimes \C^\ell \cong_\C \C^{n\ell}\]
    The first isomorphism is given by
    \[v \otimes u \mapsto v \otimes (1 \otimes u)\]
    Combining these, the isomorphism \(\C^n \otimes_\R \R^\ell \to \C^n \otimes \C^\ell\) sends
    \[v \otimes u \mapsto v\otimes u\]
    where we use the inclusion \(\R^\ell \inj \C^\ell\).
\end{example}

\begin{proposition}
    \label{prop:tensor-ext-scalar}
    Let \(M\) be an \(S\)-module, \(N\) be an \(R\)-module, then
    \[M \otimes_R N \cong M \otimes_S(S \otimes_R N)\]
    as \(S\)-modules. In particular, the isomorphism is given by
    \begin{align*}
        m \otimes n &\mapsto m \otimes (1 \otimes n) \\
        (sm) \otimes n &\mapsfrom m \otimes (s\otimes n)
    \end{align*}
\end{proposition}

\begin{proposition}
    \label{prop:tensor-ext-scalar-properties}
    Let \(M, M'\) be \(S\)-modules, \(N, N'\) be \(R\)-modules, then we have \(S\)-module isomorphisms
    \begin{enumerate}[(i)]
        \item \(M \otimes_R N \cong N \otimes_R M\), via \(m \otimes n \to n \otimes m\)
        \item \((M \otimes_R N) \otimes_R N' \cong M \otimes_R (N \otimes_R N')\)
        \item \((M \otimes_R N) \otimes_S M' \cong M \otimes_S (N \otimes_R M')\)
        \item \(M \otimes_R (\bigoplus_i N_i) \cong \bigoplus_i (M \otimes_R N_i)\)
    \end{enumerate}
\end{proposition}

\begin{proof}
    We will prove (iii). Using \cref{prop:tensor-ext-scalar}, we have
    \begin{align*}
        (M \otimes_R N) \otimes_S M' &\cong \left(M \otimes_S (N \otimes_R S)\right) \otimes_S M' \\
        &\cong M \otimes_S \left((N \otimes_R S) \otimes_S M'\right) \\
        &\cong M \otimes_S (N \otimes_R M')
    \end{align*}
\end{proof}

\begin{example}
    As \(\C\)-vector spaces,
    \[\C \otimes_\R (\R^\ell \otimes_\R \R^k) \cong (\C \otimes_\R \R^\ell) \otimes_\C (\C \otimes_\R \R^k) \cong \C^\ell \otimes \C^k \cong \C^{\ell k}\]
\end{example}

\begin{corollary}
    If \(N, N'\) are \(R\)-modules, then 
    \[S \otimes_R (N \otimes N') \cong_S (S \otimes_R N) \otimes_S (S \otimes N')\]
\end{corollary}

\begin{proof}
    By \cref{prop:tensor-ext-scalar} and \cref{prop:tensor-ext-scalar-properties} (ii):
    \[S \otimes_R (N \otimes_\R N') \cong (S \otimes_R N) \otimes_R N' \cong (S \otimes_R N) \otimes_S (S \otimes_R N')\]
\end{proof}

\lec

By induction, we have that
\[S \otimes_R (N_1 \otimes_R \cdots \otimes_R N_\ell) \cong (S \otimes_R N_1) \otimes_S \cdots \otimes_S (S \otimes_R N_1)\]

\subsubsection*{Extension of scalars for morphisms}
\index{extension of scalars!for morphisms}

Let \(f : N \to N'\) be \(R\)-linear, where \(N, N'\) are \(R\)-modules, \(M\) is an \(S\)-module. Then we have a map
\[\id \otimes f : M \otimes_R N \to M\otimes_R N'\]
In particular, it is \(S\)-linear, as
\[(\id\otimes f)(s (m \otimes n)) = (\id \otimes f)((sm) \otimes n) = (sm) \otimes f(n) = s(m \otimes f(n)) = s(\id \otimes f)(m \otimes n)\]

Given \(T : \R^n \to \R^\ell\) which is an \(\R\)-linear map, \(\R^n\) with basis \(e_1, \dots, e_n\) and \(\R^\ell\) with basis \(f_1, \dots, f_\ell\). In this case, consider
\[\id \otimes T : \C \otimes \R^n \to \C \otimes \R^\ell\]
Note that \(\C \otimes \R^n\) has basis \(1 \otimes e_1, \dots, 1 \otimes e_n\). In particular,
\[(\id \otimes T)(1 \otimes e_i) = 1 \otimes T(e_i) = 1 \otimes \sum_{j=1}^\ell T_{ji}f_j = \sum_{j=1}^\ell T_{ji}(1 \otimes f_j)\]
Thus, \(T\) and \(\id \otimes T\) have the same matrix representation.

\subsubsection*{Extension of scalars of algebras}
\index{extension of scalars!of algebras}

Let \(A, B\) be \(R\)-algebras. Recall that in this case, \(A \otimes_R B\) is also an \(R\)-algebra. In fact, \(A \otimes_R B\) is an \(A\)-algebra (and by symmetry a \(B\)-algebra). For example, we have
\begin{align*}
    A &\to A \otimes_R B \\
    a &\mapsto a \otimes 1
\end{align*}

\begin{example}
    \(S \otimes_R R[x_1, \dots, x_n] \cong_S S[x_1, \dots, x_n]\) (where \(\cong_S\) denotes isomorphism of \(S\)-algebras).

    \begin{proof}
        We already have an \(S\)-module isomorphism
        \[\varphi : S \otimes_R R[x_1, \dots, x_n] \to S[x_1, \dots, x_n]\]
        with \(\varphi(s \otimes f) = sf\). It is easy to show that
        \[\varphi(s \otimes 1) = s\]
        and that \(\varphi\) preserves multiplication.
    \end{proof}

    More generally, we have that
    \[S \otimes \left(\frac{R[x_1, \dots, x_n]}{I}\right) \cong \frac{S[x_1, \dots, x_n]}{I^e}\]
    where \(I^e = \inner{f(I)}\) is the ideal generated by \(I\) under the ring homomorphism \(f : R \to S\).
\end{example}

\begin{proposition}
    Suppose \(A\) is an \(R\)-algebra, \(B\) is an \(S\)-algebra, then \(A \otimes_R B\) is an \(S\)-algebra. Moreover,
    \[A \otimes_R B \cong_{S\text{-alg}}(A \otimes_R S) \otimes B\]
\end{proposition}

\begin{proof}
    \(A \otimes_R B\) is a \(B\)-algebra, and we can then restrict scalats to \(S\). The isomorphism is clear from the module case, as all we need to check is it preserves multiplication.
\end{proof}

\begin{proposition}
    Suppose \(A, B\) are \(R\)-algebras, then
    \[S \otimes_R (A \otimes_R B) \cong_{S\text{-alg}} (S \otimes_R A) \otimes_S (S \otimes_R B)\]
\end{proposition}

    \begin{align*}
        \Hom_R(Q, f) : \Hom_R(Q, N) &\to \Hom_R(Q, N') \\
        \varphi &\mapsto f \circ \varphi =: f_*(\varphi)

        \Hom_R(f, P) : \Hom_R(N', P) &\to \Hom_R(N, P) \\

\begin{definition}
    [flat module]
    An \(R\)-module \(M\) is \emph{flat}\index{flat module} if for any injective \(R\)-module homomorphism \(N \to N'\), the map \(\T_M(f) : \T_M(N) \to \T_M(N')\) is injective.
\end{definition}

\begin{example}

\subsection{Further examples of tensor products}

\begin{example}
    First consider \(x \otimes y \in \Q \otimes_Z (\Z/n)\). We can write
    \[x \otimes y = n\frac{x}{n} \otimes y = \frac{x}{n} \otimes ny = 0\]
    and so, \(\Q \otimes_\Z (\Z/n) = 0\). We used the fact that \(\Q\) is a \emph{divisible group}\index{divisible group}, that is, for all \(x \in \Q, n \in \N\), there exists \(y \in \Q\) such that \(ny = x\). Moreover, we also used the fact that \(\Z/n\) is torsion.

    More generally,
    \[\text{divisible} \otimes \text{torsion} = 0\]
    and so
    \[(\Q/\Z) \otimes_Z (\Q/\Z)\]

    But for an \(R\)-module \(M\) which is non-zero, if \(M\) is finitely generated, then \(M \otimes_R M \ne 0\).
\end{example}

\begin{example}
    Let \(V\) be a \(\Q\) vector space, then
    \[\Q \otimes_\Q V = V\]
    But in this case, we also have that
    \[\Q \otimes_\Z V = V\]
    with \(x \otimes v \mapsto xv\).

    \begin{proof}
        Every tensor in \(\Q \otimes_\Z V\) is pure, since we can write
        \[\sum \frac{a_i}{b_i} \otimes v_i = \sum \frac{1}{b_i} \otimes (a_iv_i) = \sum \frac{1}{b_i} \otimes \frac{a_i}{b_i}v_i = \sum 1 \otimes \frac{a_i}{b_i}v_i = 1 \otimes \sum \frac{a_i}{b_i}\]
        Clearly this map is surjective, and it is easy to see that if \(xv = 0\) then either \(x = 0\) or \(v = 0\).
    \end{proof}
\end{example}

\begin{example}

\begin{example}

    \[\C \otimes_\R \C = \C \otimes_\R \R^2 \cong \C^2\]
    and we have a basis \(\C \otimes_\R \C\), which is \(1 \otimes 1, 1 \otimes i\) as a \(\C\)-vector space.

    \[\C \otimes_\R \C \cong \C \otimes_\R \frac{\R[t]}{\inner{t^2 + 1}} \cong \frac{\C[t]}{\inner{t^2 + 1}} = \frac{\C[t]}{\inner{t-i}\inner{t+i}} \cong \frac{\C[t]}{\inner{t-i}} \times \frac{\C[t]}{\inner{t+i}} \cong \C \times \C\]
    where we used the Chinese remainder theorem. On a pure tensor, we have
    \[(a + bi) \otimes (c + di) \mapsto (a + bi) \otimes \underbrace{[c + dt]}_{\text{coset of }c + dt} \mapsto (a + bi)[c + dt]\]
    We can compute this, to get
    \[P = (ac + bdit) + (ibc + tad)\]
    and we then have
    \[P \mapsto (ac - bd + i(bc + ad), ac + bd + i(bc - ad))\]
    If we set \(x = a + bi, y = c + di\), then the result is just
    \((xy, x\overline y)\).
\end{example}

\section{Localisation}

\begin{definition}
    [multiplicative subset]
    A \emph{multiplicative(ly closed) subset}\index{multiplicative subset} \(S \sse \R\) such that
    \begin{enumerate}
        \item \(1 \in S\),
        \item if \(a, b \in S\), then \(ab \in S\).
    \end{enumerate}
\end{definition}

If \(U \sse R\) is any set, then the \emph{multiplicative closure}\index{multiplicative closure} \(S\) of \(U\) is the set of
\[\prod_{i=1}^n u_i\]
where \(u_i \in U\), \(n \ge 0\).

\begin{example}
    If \(R\) is an integral domain, then \(S = R \setminus \{0\}\) is multiplicative. More generally, if \(\mfp \idealof R\) is a prime ideal (of any ring \(R\)), then \(S = R \setminus \mfp\) is multiplicative.
\end{example}

\begin{example}
    If \(x \in R\), then \(S = \{1, x, x^2, \dots\}\) is multiplicative.
\end{example}

\begin{example}
    \(\Q\) is obtained from \(\Z\) by adding inverses for the elements of the multiplicative subset \(\Z \setminus \{0\}\), and we have a ring homomorphism \(\Z \inj \Q\).
\end{example}

We will generalise this example to general rings \(R\), and with arbitrary multiplicative subsets \(S \sse R\). But in general, we will lose injectivity.

\subsection{Construction}

\begin{definition}
    [localisation]
    Let \(S \sse R\) be a multiplicative set, \(M\) is an \(R\)-module. Consider the set \(M \times S\), with the relation \((m_1, s_1) \sim (m_2, s_2)\) if there exists \(u \in S\), such that
    \[u(s_2m_1 - s_1m_2) = 0\]
    This is an equivalence relation, and we  \(S^{-1}M\) for the set of equivalence classes. We write
    \[\frac{m}{s} = [(m, s)]\]
    for the equivalence class. Finally, we write
    \[\frac{m_1}{s_1} + \frac{m_2}{s_2} = \frac{m_1s_2 + m_2s_1}{s_1s_2}\]
    and
    \[r \cdot \frac{m}{s} = \frac{rm}{s}\]
    The above makes \(S^{-1}M\) into an \(R\)-module. We call \(S^{-1}M\) the \emph{localisation of \(M\) at \(S\)}\index{localisation!of a module}.

    If \(M = R\), we can make \(S^{-1}R\)\index{localisation!of a ring} into a ring by
    \[\frac{r_1}{s_1}\cdot\frac{r_2}{s_2}\]
    
    Next, we note that we have an \(S^{-1}R\)-module structure on \(S^{-1}M\), via
    \[\frac{r}{s}\cdot \frac{m}{t} = \frac{rm}{st}\]

    We have localisation maps:
    \begin{align*}
        R &\to S^{-1}R \\
        r &\mapsto \frac{r}{1}
    \end{align*}
    which is a ring homomorphism, and
    \begin{align*}
        M &\to S^[-1] M \\
        m &\mapsto \frac{m}{1}
    \end{align*}
    which is an \(R\)-linear map.
\end{definition}

We check that \(\sim\) above defines an equivalence relation: Reflexivity and symmetry are clear.  Say \((m_1, s_1) \sim (m_2, s_2)\) and \((m_2, s_2) \sim (m_3, s_3)\). That is, there exists \(u, v \in S\) such that
\[u(s_2m_1 - s_1m_2) = v(s_3m_2 - s_2m_3) = 0\]
Multiplying the first term by \(vs_3\) and the second by \(us_1\), we get
\begin{align*}
    uvs_2s_3m_1 &= uvs_3s_1m_2 \\
    uvs_1s_3m_2 &= uvs_1s_2m_3
\end{align*}
and so, we have that
\[uvs_2(s_3m_1 - s_1m_3) = 0\]
Since \(S\) is multiplicatively closed, we are done.

Let \(f : R \to B\) be a ring homomorphism, such that \(f(u)\) is a unit for all \(u \in U\). Then there exists a unique ring homomorphism \(h : S^{-1}R \to B\), such that the diagram

\[\begin{tikzcd}
	R && {S^{-1}R} \\
	\\
	&& B
	\arrow["f"', from=1-1, to=3-3]
	\arrow["{r \mapsto \frac{r}{1}}", from=1-1, to=1-3]
	\arrow["h", from=1-3, to=3-3]
\end{tikzcd}\]
    commutes. That is,
    \[f(r) = h\left(\frac{r}{1}\right)\]
\end{proposition}

Another way of thinking about this is that we have a bijection
\[\Hom_\text{Ring}(S^{-1}R, B) \bij \left\{\varphi : R \to B\text{ ring hom., with }\varphi(U) \sse B^\times\right\}\]
given by sending \(f\) to \(r \mapsto f\left(\frac{r}{1}\right)\).

\begin{proof}
    Let \(f : R \to B\) be a ring homomorphism, with \(f(U) \sse B^\times\). In this case, \(f(S) \sse B^\times\) as well. We want \(h : S^{-1}R \to B\), with
    \[f(r) = h\left(\frac{r}{1}\right)\]
    First, such \(h\) must satisfy:
    \[1 = h(1) = h \left(\frac{1}{s}\cdot \frac{s}{1}\right) = h\left(\frac{1}{s}\right)f(s)\]
    Thus, we must have that \(h(1/s) = f(s)^{-1}\). With this, we have
    \[h\left(\frac{r}{s}\right) = h\left(\frac{r}{1}\right)h\left(\frac{1}{s}\right) = f(r)f(s)^{-1}\]
    But we need to check if \(h\) is well defined. That is, if \(r_1/s_1 = r_2/s_2\), then there exists \(t \in S\) such that \(t(s_2r_1 - s_1r_2) = 0\), or equivalently,
    \[ts_2r_1 = ts_1r_2\]
    Applying \(f\), we get
    \[f(t)f(s_2)f(r_1) = f(t)f(s_1)f(r_2)\]
    But every element in the above equality are in \(B^ \times\), and so we are done. It is easy to check that \(h\) is a ring homomorphism.
\end{proof}

\begin{proposition}
    If \((A, j)\) satisfies the same universal property of \((S^{-1}R, \iota)\), where \(\iota(r) = r/1\), then there exists an isomorphism \(S^{-1}R \to A\), sending

\end{proposition}

\begin{enumerate}
    \item Take \(r/s \in S^{-1}R\), then
    \[\frac{r}{s} = \frac{0}{1} \iff \text{ there exists } u \in S \text{ with } ur = 0\]
    \item \(S^{-1}R = 0\) if and only if \(0 \in S\).

    \item In particular, \(\iota\) is injective if and only if \(S\) does not contain any zero divisors.
    \item \(\iota\) is always an epimorphism\footnote{A morphism \(f : X \to Y\) (in some category) is called an \emph{epimorphism}\index{epimorphism} if for all \(g_1, g_2 : Y \to Z\), with \(g_1 \circ f = g_2 \circ f\), we have \(g_1 = g_2\).}, but usually not surjective. For example, \(\iota : \Z \to \Q\) is an epimorphism. If we have \(f, g : \Q \to A\) ring homomorphisms, with \(f \circ \iota = g \circ \iota\), then \(f = g\).
\end{enumerate}

\begin{proposition}
    If \(M\) is an \(R\)-module, \(S \sse R\) a multiplicative subset, then we have an isomorphism:
    \begin{align*}
        \frac{S^{-1}R} \otimes_R M &\to S^{-1}M \\
        \frac{r}{s} \otimes m &\mapsto \frac{rm}{s}
    \end{align*}
\end{proposition}

\begin{proof}
    We can define a bilinear map
    \begin{align*}
        S^{-1}R \times M &\to S^{-1}M \\
        \left(\frac{r}{s}, m\right) &\mapsto \frac{rm}{s}
    \end{align*}
    and thus, by the universal property we haev \(\varphi : S^{-1}R \otimes_R M \to S^{-1}M\). This is \(R\)-linear, and it is easy to see that \(\varphi\) is also \(S^{-1}R\)-linear. It is clear that \(\varphi\) is surjective, since
    \[\varphi\left(\frac{1}{s} \otimes m\right) = \frac{m}{s}\]

    We want to show that every tensor
    \[t = \sum_i \frac{r_i}{s_i} \otimes m_i \in S^{-1}R \otimes_R M\]
    is prime. Define \(s = \prod_i s_i\), and \(t_j = \prod_{i \ne j}s_i\). In this case,
    \begin{align*}
        \sum \frac{r_i}{s_i} \otimes m_i &= \sum \frac{1}{s_i} \otimes (r_i m_i) \\
        &= \sum \frac{t_i}{s} \otimes (r_i m_i) \\
        &= \frac{1}{s} \otimes \left(\sum_i r_it_im_i\right)
    \end{align*}
    Using this, if
    \[\varphi\left(\frac{1}{s} \otimes m\right) = \frac{m}{s} = 0 = \frac{0}{1}\]
    That is, there exists \(u \in S\), such that \(um = 0\). In this case,
    \[\frac{1}{s} \otimes m = \frac{u}{us} \otimes m = \frac{1}{us} \otimes (um) = 0\]
\end{proof}

With this, \(S^{-1}R \otimes (\cdots)\) acts on \(R\)-modules. But in fact, it also acts on \(R\)-linear maps.

\begin{lemma}
    If \(M\) is an \(S^{-1}R\)-module, then, we can restrict scalars on \(M\) from \(S^{-1}R\) to \(R\), then apply \(S^{-1}(\cdot)\). Then
    \[S^{-1}M \cong M\]
    as \(S^{-1}R\)-modules. Equivalently,
    \[M \cong S^{-1}R \otimes M\]
    as \(S^{-1}R\)-modules.
\end{lemma}

\begin{proof}

    \begin{align}
        M &\to S^{-1}M \\
        m &\mapsto \frac{m}{1}

\end{proof}

\begin{proposition}
    Let \(M\) be an \(R\)-module, \(L\) an \(S^{-1}R\)-module, \(f : M \to L\) is \(R\)-linear. Then there exists a unique \(h : S^{-1}M \to L\) which is \(S^{-1}R\)-linear, such that
    \[f(m) = h\left(\frac{m}{1}\right)\]
\end{proposition}

\begin{proof}
    We know that \(S^{-1}(\cdot) \otimes S^{-1}R \otimes (\cdot)\), and so it suffices to prove the result for the tensor product. With this, the localisation map is
    \begin{align*}
        \iota : M &\to S^{-1}R \otimes M \\
        m &\mapsto \frac{1}{1} \otimes m
    \end{align*}
    Let \(f : M \to L\) be \(R\)-linear. We then have that
    \[h : \id_{S^{-1}R} \otimes f : S^{-1}R \otimes_R M \to S^{-1}R \otimes_R L\]

    \[h\left(\frac{r}{s} \otimes m\right) = \frac{r}{s}f(m)\]
    For the uniqueness of \(h\), it follows from the fact that elements of the form \(\frac{1}{1} \otimes m\) generate \(S^{-1}R \otimes_R M\) as an \(S^{-1}R\)-module.
\end{proof}

\subsection{Extension and contraction of ideals}

Recall if \(f : A \to B\) is a ring homomorphism, we define the \emph{contraction of \(\mfb \idealof B\)}\index{contraction of an ideal} as
\[\mfb^c = f^{-1}(\mfb) \idealof A\]
and the \emph{extension of \(\mfa \idealof A\)}\index{extension of an ideal} as
\[\mfa^e = \inner{f(\mfa)} \idealof B\]
In examples sheet 1, we have a bijection
\[\{\text{contracted ideals of }A\} \bij \{\text{extended ideals of }B\}\]
To see this, we have that an ideal \(\mfa\) is contracted if and only if \(\mfa = \mfa^{ec}\), and an ideal \(\mfb\) is extended if and only if \(\mfb = \mfb^{ce}\), and so the bijection is given by extension/contraction.

Let \(S\) be a multiplicative subset of \(R\), and we will consider the ring homomorphism \(R \to S^{-1}R\), given by \(r \mapsto r/1\). For an ideal \(\mfa\) of \(R\), we have the \emph{extension}
\[\mfa^e = S^{-1}\mfa \idealof S^{-1}R\]
and for an ideal \(\mfb\) of \(S^{-1}R\), we have the contraction \(\mfb^c \idealof R\).

\begin{proposition}
    \[\mfa^e = S^{-1}\mfa = \left\{\frac{a}{s} \ \bigg\vert\ a \in \mfa, s \in S\right\}\]
\end{proposition}

\lec

\begin{proof}
    \(\mfa^e\) is the ideal generated by \(a/1\) for \(a \in \mfa\), and so \(\supseteq\) holds. But the right hand side is already an ideal, and so by minimality, equality holds.
\end{proof}

\begin{proposition}
    \(\mfa^{ec} = \bigcup_{s \in S} (\mfa : s)\) where \((\mfa : s) = \{r \in R \mid rs \in \mfa\}\).
\end{proposition}

\begin{proof}
    Take \(r \in \bigcup_{s \in S}(\mfa : s)\). That is, \(rs = a \in \mfa\), and so in \(S^{-1}R\),
    \[\frac{rs}{1} = \frac{a}{1} \implies \frac{r}{1} = \frac{a}{s} \in \mfa^e\]
    and so \(r \in \mfa^{ec}\). Conversly, if \(r \in \mfa^{ec}\), then
    \[\frac{r}{1} = \frac{a}{s}\]
    for some \(a \in \mfa, s \in S\). But this means that there exists \(u \in S\), such that \(urs = ua\). With this, \(r \in (\mfa : us)\), \(us \in S\) as \(S\) is multiplicative.
\end{proof}

Now suppose \(\mfb\) is an ideal of \(S^{-1}R\). Then
\[\mfb^c = \left\{r \in \R \ \bigg\vert\ \frac{r}{1} \in \mfb\right\}\]

\begin{proposition}
    \label{prop:contract-extend}
    \(\mfb^{ce} = \mfb\).
\end{proposition}

\begin{proof}
    \(\sse\) always holds. Take \(r/s \in \mfb\), then \(r/1 \in \mfb\). Thus, \(r \in \mfb^c\), and so \(r/1 \in \mfb^{ce}\), which means that \(r/s \in \mfb^{ce}\).
\end{proof}

\begin{proposition}
    Consider the loclisation map \(R \to S^{-1}R\), then
    \begin{enumerate}[(i)]
        \item Every ideal of \(S^{-1}R\) is extended.
        \item An ideal \(\mfa\) of \(R\) is contracted if and only if the image of \(S\) in \(R/\mfa\) contains no zero divisors of \(R/\mfa\).
        \item \(\mfa^e = S^{-1}R\) if and only if \(\mfa \cap S \ne \emptyset\).
        \item We have a bijection:
        \begin{align*}
            \{\mfp \in \Spec(R) \mid \mfp \cap S = \emptyset\} &\bij \Spec(S^{-1}R) \\
            \mfp &\mapsto \mfp^e \\
            \mfq^c &\mapsfrom \mfq
        \end{align*}
    \end{enumerate}
\end{proposition}

\begin{proof}
    (i) Follows from \cref{prop:contract-extend}. For (ii), \(\mfa\) is contracted if and only if \(\mfa^{ec} \sse \mfa\). But
    \[\mfa^{ec} = \bigcup_{s \in S}(\mfa : s)\]
    Thus, \(\mfa^{ec} \sse \mfa\) if and only if: for all \(r \in R\), if \(Sr \cap \mfa \ne \emptyset\), then \(r \in \mfa\). But \(Sr \cap \mfa \ne \empty\) is true if and only if \(0 + \mfa\) is in the image of \(S\), and \(r \in \mfa\) is the same as \(r + \mfa = 0\). Thus, \(\mfa\) is contracted if and only if the image of \(S\) in \(R/\mfa\) contains no zero divisors.

    For (iii), suppose \(\mfa \cap S \ne \emptyset\). Choose \(x \in \mfa \cap S\), then
    \[1 = \frac{x}{x} \in \mfa^e\]
    Conversely, if \(\mfa^e = S^{-1}R\). Then \(1 \in \mfa^e\), and so
    \[\frac{1}{1} = \frac{a}{s}\]
    for some \(a \in \mfa, s \in S\), and so there exists \(u \in S\) such that \(us = ua\). But \(us \in S\) as it is multiplicative, \(ua \in \mfa\) as it is an ideal.
    
    For (iv), first consider the contraction map \(\Spec(S^{-1}R) \to \{\mfp \in \Spec(R) \mid \mfp \cap S = \emptyset\}\). This makes sense as the contraction of a prime ideal is prime, and if \(\mfp \in \Spec(R)\) is contracted, by (ii), we see that \(S \cap \mfp\) is empty, since \(R/\mfp\) is an integral domain, and so the only zero divisor is zero.

    Moreover, this map is injective, since it has a left inverse, as all ideals in \(S^{-1}R\) are extended ideals, and so \(\mfq^{ce} = \mfq\). In the other direction, for a prime ideal \(\mfp \in \Spec(R)\), with \(\mfp \cap S = \emptyset\), we have seen that \(\mfp\) is contracted, and so \(\mfp^{ec} = \mfp\). With this, all we need to show is that \(\mfp^e\) is prime.
    
    We would like to show that \((S^{-1}R)/\mfp^e\) is an integral domain. We know that \(\mfp^e\) is not all of \(S^{-1}R\), and so \((S^{-1}R)/\mfp^e\) is not the zero ring. So we need to show that \((S^{-1}R)/\mfp^e\) has no zero divisors. We will do this by embedding \((S^{-1}R)/\mfp^e\) into \(\Frac(R/\mfp)\).

    Now consider the composition map

\subsubsection*{An application}

If \(I \idealof R\) is an ideal, then the \emph{radical of \(I\)} is
\[\sqrt{I} = \left\{r \in R \mid \exists m \ge 1 \text{ such that }r^m \in I\right\}\]
\begin{proposition}
    \[\sqrt{I} = \bigcap_{I \le \mfp \in \Spec(R)}\mfp\]
\end{proposition}
\begin{proof}
    Take \(x \in \sqrt{I}\), then \(x^n \in I\), and so for every \(\mfp \in \Spec(R)\), if \(I \sse \mfp\), then \(x^n \in \mfp\), and so \(x \in \mfp\). That is, \(\sse\) holds. For the other inclusion, take \(x \in R\), \(x \notin \sqrt{I}\). We know that \(I \ne R\), and \(R/I\) is not the zero ring. Let \(\overline{x} \in R/I\) be the image of \(x\). Consider
    \[(R/I)_{\overline x} = \{\overline{x}^n\}^{-1}(R/I)\]
    This is not the zero ring, since we did not invert zero. Therefore, \((R/I)_{\overline x}\) has a prime ideal, which corresponds to a prime ideal of \(R/I\) which avoids \(\overline x\), which in turn, corresponds to a prime ideal of \(R\), which contains \(I\), and avoids \(x\).
\end{proof}

\subsection{Local properties}

\begin{definition}
    [local ring]
    A ring \(R\) is \emph{local}\index{local ring} if it has a unique maximal ideal. We write \((R, \mfm)\) for the local ring \(R\) with maximal ideal \(\mfm\).
\end{definition}

\begin{example}
    Let \(\mfp \in \Spec(R)\). Then recall that we have a bijection
    \[\{\mfq \in \Spec(R) \mid \mfq \sse \mfp\} \bij \Spec(R_\mfp)\]
    given by extension and contraction. With this, all prime ideals of \(R_\mfp\) are contained in \(\mfp R_\mfp\). Thus, \((R_\mfp, \mfp R_\mfp)\) is a local ring.

    In particular, \(\Z_{\inner2}\) is a local ring, and the unique maximal ideal is
    \[\inner{2}\Z_{\inner{2}} = \left\{\frac{2a}{b} \ \bigg\vert\ a, b \in \Z, 2 \nmid b\right\}\]
\end{example}

\begin{proposition}
    \label{prop:zero-local}
    Let \(M\) be an \(R\)-module. Then the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(M = 0\),
        \item \(M_\mfp = 0\) for all \(\mfp \in \Spec(R)\),
        \item \(M_\mfm = 0\) for all \(\mfm \in \maxSpec(R)\).
    \end{enumerate}
    That is, being zero is a local property (i.e. it is localisable and local to global).
\end{proposition}

\begin{proof}
    The implications \((i) \implies (ii) \implies (iii)\) is clear. Suppose (iii) holds, and suppose for contradiction there exists \(m \in M\) non-zero. Consider
    \[\Ann_R(m) = \left\{r \in R \mid rm = 0\right\} \idealof R\]
    Since \(m \ne 0\), \(1 \notin \Ann_R(m)\). Take a maximal ideal \(\mfm\) containing \(\Ann_R(m)\). In this case,
    \[\frac{m}{1} = 0 \in M_\mfm\]
    That is, \(um = 0\) for some \(u \in R \setminus \mfm\). But in this case, \(u \notin \Ann_R(m)\). Contradiction.
\end{proof}

\section{Nakayama's lemma}

\begin{proposition}
    [Cayley-Hamilton]
    \label{prop:Cayley-Hamilton}
    \index{Cayley-Hamilton}
    Let \(M\) be a finitely generated \(R\)-module, \(f : M \to M\) an \(R\)-linear map, \(mfa \idealof R\) an ideal, with \(f(M) \sse \mfa M\). Then
    \[f^n + a_1f^{n-1} + a_n \id = 0\]
    where \(a_i \in \mfa\).
\end{proposition}

\begin{proof}
    Say \(M = \Span_R\{m_1, \dots, m_n\}\), then \(\mfa M = \Span_{\mfa}\{m_1, \dots, m_n\}\). Therefore,
    \[\begin{pmatrix}
        f(m_1) \\
        \vdots \\
        f(m_n)
    \end{pmatrix} = P\begin{pmatrix}
        m_1 \\
        \vdots \\
        m_n
    \end{pmatrix}\]
    where \(P \in \Mat_n(\mfa)\). Take \(\rho : R \to \End(M)\) to be the structure ring homomorphism of \(M\) as an \(R\)-module, then we can define
    \begin{align*}
        R[t] &\to \End_R(M) \\
        t &\mapsto f
    \end{align*}
    which makes \(M\) into an \(R[t]\)-module. Using this,
    \[t \cdot \begin{pmatrix}
        m_1 \\
        \vdots \\
        m_n
    \end{pmatrix} = P \begin{pmatrix}
        m_1 \\
        \vdots \\
        m_n
    \end{pmatrix}\]
    and so
    \[Q \begin{pmatrix}
        m_1 \\ \vdots \\ m_n = 0
    \end{pmatrix}\]
    where \(Q = t\cdot I_n - P = 0\). Multiplying by \(\adj(Q)\), we get that
    \[\det(Q) \begin{pmatrix}
        m_1 \\ \vdots \\ m_n
    \end{pmatrix} = 0\]
    Hence \(\det(Q) m = 0\) for all \(m \in M\), and so \(m \mapsto \det(Q)m\) is the zero map. But then \(\det(Q)\) gives the polynomial as required.
\end{proof}

\begin{corollary}
    \label{cor:am=m}
    Let \(M\) be a finitely generated \(R\)-module. \(\mfa \idealof R\) an ideal, if \(\mfa M = M\), then there exists \(a \in \mfa\) such that \(am = m\) for every \(m \in M\).
\end{corollary}

\begin{proof}
    Apply Cayley-Hamilton with \(f = \id_M\), we get that
    \[(1 + a_1 + \cdots + a_n)\id_M = 0\]
    and so we can take \(a = -(a_1 + \dots + a_n)\).
\end{proof}

\begin{definition}
    [Jacobson radical] The \emph{Jacobson radical}\index{Jacobson radical} of a ring \(R\) is
    \[J(R) = \bigcap_{\mfm \idealof R \text{ maximal}}\mfm\]
\end{definition}

\begin{example}
    If \((R, \mfm)\) is a local ring, then \(J(R) = \mfm\). On the other hand, \(J(\Z) = 0\).
\end{example}

\begin{proposition}
    \label{prop:mem-jacobson-iff}
    For \(x \in R\), \(x \in J(R)\) if and only if \(1 - xy\) is a unit in \(R\) for every \(y \in R\).
\end{proposition}

\begin{proof}
    Suppose that \(x \in J(R)\), and suppose for contradiction that \(1 - xy\) is not a unit, for some \(y \in R\). With this, \(1 - xy\) is contained in a maximal ideal \(\mfm\). Since \(x \in J(R)\), \(x \in \mfm\). Thus,
    \[1 = (1 - xy) + xy \in \mfm\]
    Contradiction. On the other hand, if \(x \notin J(R)\), then there exists a maximal ideal \(\mfm\) such that \(x \notin \mfm\). Then \(\mfm + \inner{x} = R\). In particular, there exists \(t \in \mfm, y \in R\) such that \(t + xy = 1\). In this case, \(1 - xy = t \in \mfm\), and so it is not a unit.
\end{proof}

\begin{proposition}
    [Nakayama's lemma] 
    \label{prop:nakayama}
    \index{Nakayama's lemma}
    Let \(M\) be a finitely generated \(R\)-module, \(\mfa \le J(R)\) is an ideal of \(R\), with \(\mfa M = M\). Then \(M = 0\).
\end{proposition}

\begin{proof}
    By \cref{cor:am=m}, there exists \(a \in \mfa\) such that \(am = m\) for all \(m \in M\). By \cref{prop:mem-jacobson-iff}, \(1 = a\) is a unit, and so we can multiply by \((1 -a)^{-1}\), to get that
    \[m = (1-a)^{-1}(1-a)m = (1-a)^{-1}\cdot0 = 0\]
\end{proof}

\begin{corollary}
    Let \(M\) be a finitely generated \(R\)-module, \(N \le M\) an \(R\)-submodule, \(\mfa \le J(R)\) an ideal, such that
    \[N + \mfa M = M\]
    then \(N = M\).
\end{corollary}

\begin{proof}
    \[\mfa \cdot \left(\frac{M}{N}\right) = \frac{\mfa M + N}{N} = \frac{M}{N}\]
    Therefore, by Nakayama, \(M/N = 0\), and so \(N = M\).
\end{proof}

\begin{definition}
    [integral]
    Let \(A\) be an \(R\)-algebra, \(x \in A\) is \emph{integral over \(R\)}\index{integral!element} if there exists \(f \in R[t]\) monic, such that \(f(x) = 0\).
\end{definition}

\begin{example}
    If \(K\) is a field, \(A\) is a \(K\)-algebra, \(x \in A\), then \(x\) is integral over \(K\) if and only if it is algebraic over \(K\).
\end{example}

\begin{example}

        \item the \(\Z\) integral elements of \(\Q(\sqrt 2)\) is \(\Z[\sqrt 2]\),
        \item the \(\Z\) integral elements of \(\Q(\sqrt 5)\) is \(\Z\left[\frac{1+\sqrt{5}}{2}\right]\)
    \end{enumerate}
    To see this, we can also recall Part II Number Fields and the ring of integers of a number field.
\end{example}

\begin{definition}
    [faithful]
    An \(R\)-module \(M\) is \emph{faithful}\index{faithful} if the structure ring homomorphism \(R \to \End_R(M)\) is injective.

    That is, for every non-zero \(r \in R\), there exists \(m \in M\) such that \(rm \ne 0\).
\end{definition}

\begin{example}
    Let \(R \sse A\) be rings, and so \(A\) is an \(R\)-module in a natural way. It must be faithful, since we have \(r1 = r\).
\end{example}

\begin{proposition}
    \label{prop:integral-iff-submod}
    Let \(R \sse A\) be rings, \(x \in A\). Then \(R[x] \sse A\) is a subring, which makes \(A\) into an \(R[x]\)-algebra (and thus an \(R[x]\)-module). Then \(x\) is \(R[x]\)-integral if and only if there exists \(M \sse A\) such that
    \begin{enumerate}
        \item \(M\) is a faithful \(R[x]\)-module, that is, \(M\) is an \(R\)-submodule of \(A\), \(xM \sse M\), and \(R[x] \to \End_{R[x]}(M)\) is injective,
        \item \(M\) is finitely generated as an \(R\)-module.
    \end{enumerate} 
\end{proposition}

\begin{proof}
    Suppose such an \(M\) exists. With this, we have an \(R\)-linear map \(f : M \to M\),
    \[f(m) = xm\]
    Since \(M\) is a finitely generated \(R\)-module, we can apply Cayley-Hamilton (\cref{prop:Cayley-Hamilton}), to get
    \[f^n + r_1 f^{n-1} + \dots + r_n = 0\]
    where \(r_i \in R\). Evaluating at \(m \in M\), we get that
    \[(x^n + r_1x^{n-1} + \cdots + r_n)(m) = 0\]
    Since \(M\) is a faithful \(R[x]\)-module,
    \(x^n + f_1x^{n-1} + \cdots + r_n = 0\)
    That is, \(x\) is integral over \(R\). Now suppose \(x\) is integral over \(R\). Then
    \[x^n + r_1x^{n-1} + \cdots + r_n = 0\]
    for some \(r_i \in R\). Take
    \[M = \Span_R\{1, x, \cdots, x^{n-1}\}\]
    satisfies \(xM = M\), and as \(1 \in M\), it is faithful. The fact that it is finitely generated is clear by definition.
\end{proof}

\lec

\begin{definition}
    [integral]
    Let \(A\) be an \(R\)-algebra. Then \(A\) is \emph{integral over \(R\)}\index{integral!extension} if every \(x \in A\) is integral over \(R\).
\end{definition}

\begin{definition}
    [finite over]
    Let \(A\) be an \(R\)-algebra, then \(A\) is \emph{finite over \(A\)}\index{finite!extension of algebras} if it is finitely generated as an \(R\)-module.
\end{definition}

\begin{proposition}
    \label{prop:finite-iff-fg-integral}
    Let \(A\) be an \(R\)-algebra. Then the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(A\) is a finitely generated integral \(R\)-algebra,
        \item \(A\) is generated as an \(R\)-algebra by a finite set of integral elements,
        \item \(A\) is finite over \(R\),
    \end{enumerate}
\end{proposition}

\begin{proof}
    (i) \(\implies\) (ii) is trivial. Suppose (ii) holds. Then \(A\) is generated by \(\alpha_1, \dots, \alpha_m\) as an \(R\)-algebra. But \(\alpha_i\) being integral implies that
    \[\alpha_i^{n_i} + r_{i, 1}\alpha_i^{n_i - 1} + \cdots + r_{i, n_i} = 0\]
    That is,
    \[\alpha_i^{n_i} \in \Span_R\{1, \alpha_i, \dots, \alpha_i^{n_i - 1}\}\]
    But this means that for all \(e_1, \dots e_n \ge 0\),
    \[\alpha_1^{e_1}\cdots \alpha_m^{e_m} \in \Span_R\{\alpha_1^{f_1}\cdots \alpha_m^{f_m} \mid 0 \le f_i \le n_i - 1\}\]
    Hence \(A\) is a finitely generated \(R\)-module.

    Finally, suppose (iii) holds. If \(A\) is finitely generated as an \(R\)-module, then it is necessarily finitely generated as an \(R\)-algebra. Choose \(\alpha \in A\), we would like to show that \(\alpha\) is integral over \(R\). Let \(\rho ; R \to A\) be the structure ring homomorphism of \(A\) as an \(R\)-algebra. Then \(\rho(R)\) is a subring of \(A\). With this, it then makes sense to consider \(\rho(R)[\alpha]\) as a subring of \(A\).

    Next, \(A\) is a \(\rho(R)[\alpha]\)-module, and it must be faithful as \(1 \in A\). Using this, and the fact that \(A\) is a finitely generated \(\rho(R)[\alpha]\)-module, so by \cref{prop:integral-iff-submod}, \(\alpha\) is integral over \(\rho(R)\). Equivalently, \(\alpha\) is integral over \(R\).
\end{proof}

\begin{proposition}
    If \(A\) is an \(R\)-algebra, \(\mcO\) is the integral elements of \(A\), then \(\mcO\) is an \(R\)-subalgebra of \(A\).
\end{proposition}

\begin{proof}
    Take \(x, y \in \mcO\). Then this is a finite set of \(R\)-integral elements, and so must generate an integral \(R\)-subalgebra of \(A\). But this contains \(x \pm y, xy\), which must then be integral. Hence \(\mcO\) is a ring. The fact that it is an \(R\)-subalgebra is clear.
\end{proof}

\begin{proposition}
    If \(A \sse B \sse C\) are rings, 
    \begin{enumerate}[(i)]
        \item if \(C\) is finite over \(B\), and \(B\) is finite over \(A\), then \(C\) is finite over \(A\).
        \item if \(C\) is integral over \(B\), \(B\) is integral over \(A\), then \(C\) is integral over \(A\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), if \(C = \Span_B\{\gamma_1, \dots, \gamma_n\}\), \(B = \Span_A\{\beta_1, \dots, \beta_\ell\}\), then \(C = \Span_A\{\beta_i\gamma_j\}\).

    For (ii), let \(c \in C\). We would like to show that \(c\) is \(A\)-integral. We know that \(c\) is \(B\)-integral, and so \(f(c) = 0\) for some 
    \[f(T) = T^n + b_1T^{n-1} + \cdots + b_n \in B[T]\]
    Hence \(f \in A[b_1, \dots, b_n][T]\). Set \(A' = A[b_1, \dots, b_n]\). Then we have inclusions
    \[A \sse A' \sse A'[c]\]
    Both inclusions are integral, as they are generated by finitely many integral elements. But this tells us that both extensions are finite by \cref{prop:finite-iff-fg-integral}. By (i), \(A \sse A'[c]\) is finite, and so \(A \sse A'[c]\) is integral, and so \(c\) is integral over \(A\).
\end{proof}

\begin{definition}
    Let \(A \sse B\) be rings. The \emph{integral closure}\index{integral closure} of \(A\) in \(B\) is
    \[\overline{A} = \{b \in B \mid b \text{ integral over }A\}\]
    We say that \(A\) is \emph{integrally closed}\index{integrally closed} if \(A = \overline{A}\).

    If \(A\) is an integral domain, then its \emph{integral closure} is its integral closure in \(\Frac(A)\), and it is \emph{integrally closed} if it is integrally closed in \(\Frac(A)\).
\end{definition}

\begin{example}
    Consider \(A = \Z[\sqrt{5}]\). This is not integrally closed, since \(\Frac(A) = \Q(\sqrt 5)\). In this case,
    \[\alpha = \frac{1 + \sqrt 5}{2} \in \Frac(A) \setminus A\]
    But \(\alpha\) is integral over \(A\), since \(\alpha^2 - \alpha - 1 = 0\).
\end{example}

\begin{example}
    \(\Z\) and \(k[t_1, \cdots, t_n]\) are integrally closed.
\end{example}

\begin{proposition}
    If \(A\) is a UFD, then \(A\) is integrally closed.
\end{proposition}

\begin{proof}
    Take \(x \in \Frac(A) \setminus A\), say \(x = a/b\), \(a, b \in A\), with some \(p \in A\) prime, \(p \mid b\) but \(p \nmid a\). If \(x\) is \(A\)-integral, then
    \[\left(\frac a b\right)^n + a_1\left(\frac a b\right)^{n-1} + \cdots + a_0 = 0\]
    Multiply through by \(b^n\), we get
    \[a^n = -b(a_1 + a_2b + \cdots + a_n b^{n-1})\]
    Since \(p \mid b\), \(p\) divides the right hand side, and so \(p \in a^n\). Thus, \(p \mid a\).
\end{proof}

\begin{lemma}
    If \(A \sse B\) are rings, \(\overline{A}\) the integral closure of \(A\) in \(B\), then \(\overline{A}\) is integrally closed over \(A\).
\end{lemma}

\begin{proof}
    If \(x \in B\) is integral over \(\overline A\), then we have integral extensions
    \[A \sse \overline A \sse \overline A[x]\]
    By transitivity, \(A \sse \overline A[x]\) is integral, and so \(x\) is integral over \(A\), that is, \(x \in \overline{A}\).
\end{proof}

\begin{proposition}
    Let \(A \sse B\) be rings,
    \begin{enumerate}[(i)]
        \item If \(B\) is integral over \(A\), 
        \begin{enumerate}[(a)]
            \item for every ideal \(\mfb\) of \(B\),
            \[\frac{B}{\mfb} \text{ is integral over }\frac{A}{\mfb \cap A}\]
            \item if \(S \sse A\) is a multiplicative set, then \(S^{-1}B\) is integral over \(S^{-1}A\),
        \end{enumerate}
        \item If \(\overline A\) is the integral closure of \(A\) in \(B\), then then \(S^{-1}\overline A\) is the integral closure of \(S^{-1}A\) in \(S^{-1}B\). That is, \(\overline{S^{-1}A} = S^{-1}\overline A\)
    \end{enumerate}
\end{proposition}

\begin{proof}
    See notes.
\end{proof}

\begin{lemma}
    Suppose \(A \sse B\) is an integral extension of rings,
    \begin{enumerate}[(i)]
        \item \(A \cap B^\times = A^\times\),
        \item if \(A, B\) are domains, then \(A\) is a field if and only if \(B\) is a field.
    \end{enumerate}
\end{lemma}

\begin{proof}
    For (i), \(\supseteq\) is clear. Conversely, take \(a \in A \cap B^\times\). Then there exists \(b \in B\) such that \(ab = 1\). We need to show that \(b \in A\). We know that \(b\) is integral over \(A\), that is,
    \[b^n + a_1b^{n-1} + \cdots + a_n = 0\]
    Multiply this by \(a^{n-1}\), we get
    \[b + a_1 + a_2a + \cdots + a_na^{n-1} = 0\]
    But \(a_1 + a_2a + \cdots + a_na^{n-1} \in A\), and so \(b \in A\).

    For (ii), suppose that \(B\) is a field. Then
    \[A^\times = A \cap B^\times = A \cap (B \setminus \{0\}) = A \setminus \{0\}\]
    and so \(A\) is a field. Now suppose \(A\) is a field. Let \(b \in B\) be non-zero. Since \(b\) is integral over \(A\),
    \[b^n + a_1 b^{n-1} + \cdots + a_n = 0\]
    where \(n\) is \emph{minimal}. With this,
    \[b(\underbrace{b^{n-1} + a_1b^{n-2} + \cdots + a_{n-1}}_{=\delta}) = -a_n\]
\lec
    By minimality, \(\delta \ne 0\). Therefore, \(a_n \ne 0\) as it is a domain. But \(a_n \in A\) is a unit, so
    \[b(a_n^{-1}\delta) = 1\]
    and so \(b\) is a unit.
\end{proof}

\begin{corollary}
    Let \(A \sse B\) be an integral extension of rings, \(\mfq\) a prime ideal of \(B\). Then \(\mfq\) is a maximal ideal of \(B\) if and only if \(\mfq \cap A\) is a maximal ideal of \(A\).
\end{corollary}

\begin{proof}
    We have a ring embedding
    \[\frac{A}{\mfq \cap A} \inj \frac{B}{\mfq}\]
    and these are integral domains as \(\mfq\) is prime. Moreover, this is an integral extension, and so we are done.
\end{proof}

\section{Noether normalisation and Hilbert's Nullstellensatz}

\subsection{Noether normalisation}

Throughout, let \(k\) be a field.

\begin{definition}
    [algebraically independent]
    If \(A\) is a \(k\)-algebra, and \(x_1, \dots, x_n \in A\), then \(x_1, \dots, x_n\) are \emph{\(k\)-algebraically independent}\index{algebraically independent} if for every \(p \in k[T_1, \dots, T_n]\) non-zero, \(p(x_1, \dots, x_n) \ne 0\). That is, the \(k\)-algebra homomorphism \(k[T_1, \dots, T_n] \to A\) given by sending \(T_i\) to \(x_i\) is injective.
\end{definition}

\begin{theorem}
    [Noether normalisation] 
    \label{thm:noether-normalisation}
    \index{Noether normalisation}
    If \(A \ne 0\) is a finitely generated \(k\)-algebra, then there exists \(x_1, \dots, x_n \in A\), which are \(k\)-algebraically independent, such that \(A\) is finite over \[A' = k[x_1, \dots, x_n]\]
\end{theorem}

\begin{example}
    [of the method of proof] Let \(A = k[t, t^{-1}]\). First of all, note that \(k[t] \sse k[t, t^{-1}]\) is not a finite extension. To see this, suppose it was, then \(t^{-1}\) is integral over \(k[t]\). That is,
    \[t^{-n} \in \Span_{k[t]}\{1, t^{-1}, \dots, t^{-(n-1)}\}\]
    Multiply through by \(t^n\), we get
    \[1 \in \Span_{k[t]}\{t^n, t^{n-1}, \dots, t\}\]
    which is a contradiction. However, let \(c \in k\) (which we will choose later). Then
    \[A = k[t, t^{-1}] = k[t, t^{-1} - ct]\]
    \begin{claim}
        \(k[T^{-1} - cT] \sse A\) is a finite extension for ``most'' c.
    \end{claim}

    \begin{proof}
        Since \(tt^{-1} - 1 =0\), we have that
        \[((t^{-1} - ct) + ct)t -1 = 0\]
        Expanding,
        \[ct^2 + (t^{-1}-ct)t - 1 = 0\]
        Thus, if \(c \ne 0\), then we can divide by \(c\) to show that \(t\) is integral over \(k[t - c t^{-1}]\).
    \end{proof}
\end{example}

\begin{proof}
    [Proof of \cref{thm:noether-normalisation} assuming \(k\) is infinite] We will induct on the minimal number \(m\) of generators of \(A\) as an \(k\)-algebra.

    \textbf{Base case: \(m = 0\)} is trivial since \(A = k\). We can take \(A' = A\).

    \textbf{Inducive step:} Suppose \(A\) is generated by \(x_1, \dots, x_m \in A\) as an \(k\)-algebra. If \(x_1, \dots, x_m\) are algebraically independent, then we can take \(A = A'\). Otherwise,
    \begin{claim}
        \label{claim:noether-normalisation}
        There exists \(c_1, \dots, c_{m-1} \in k\) such that \(x_m\) is integral over
        \[B = k[x_1 - c_1 x_m, \dots, x_{m-1} - c_{m-1}x_m]\]
    \end{claim}
    Assuming the claim, then \(A = B[x_n]\), and so \(A\) is finite over \(B\). But \(B\) is generated by \(m-1\) elements, and so by induction, \(B\) contains \(z_1, \dots, z_n \in B\), with \(B\) finite over \(A' = k[z_1, \dots, z_n]\). Then \(A\) is finite over \(A'\) by transitivity. 

    \begin{proof}
        [Proof of \cref{claim:noether-normalisation}] Since \(x_1, \dots, x_n\) are not algebraically independent over \(k\), there exists a non-zero \(f \in k[t_1, \dots, t_m]\), with
        \[f(x_1, \dots, x_m) = 0\]
        We would like to prove that \(x_m\) is integral over \(B\), where \(c_i \in k\) we will choose later. Write
        \[f = \sum_{i=0}^r f_{[i]}\]
        as a sum of homogeneous parts. Set \(F = f_{[r]}\) for the highest order part. For \(c_1, \dots, c_{m-1} \in k\), set
        \[g(t_1, \dots, t_m) = f(t_1 + c_1 t_m, \dots, t_{m-1} + c_{m-1}t_m, t_m) = F(c_1, \dots, c_{m-1}, 1)t_m^r + h(t_1, \dots, t_m)\]
        where each term in \(h\) has degree of \(t_m\) less than \(r\). Note
        \[g(x_1 - c_1x_m, \cdots, x_{m-1} - c_{m-1}x_m, x_m) = f(x_1, \dots, x_m) = 0\]
        and that \(g\) as a polynomial in \(t_m\) over \(k[t_1, \dots, t_{m-1}]\) has degree at most \(r\), and the coefficient of \(t_m^r\) is \(F(c_1, \dots, c_{m-1}, 1)\), Since \(F(t_1, \dots, t_m)\) is a non-zero homogeneous polynomial, and so \(F(t_1, \dots, t_{m-1}, 1)\) is not zero. Therefore, there are \(c_1, \dots, c_{m-1}\), with
        \[F(c_1, \dots, c_{m-1}) \ne 0\]
        since we are working over an infinite field (Schwartz-Zippel).
    \end{proof}
\end{proof}

\begin{remark}
    Noether normalisation is true for any field.
\end{remark}

From the example
\[k[t, t^{-1}] \cong \frac{k[x, y]}{\inner{xy - 1}}\]
Geometrically, \(xy - 1\) is a hyperbola. The projection onto the \(x\)-axis is not surjective, but the projection onto \(y = cx\) is surjective for \(c \ne 0\).

\subsection{Hilbert Nullstellensatz}

\begin{proposition}
    [Zariski's lemma] Let \(k \sse L\) be fields, with \(L\) finitely generated as a \(k\)-algebra. Then \(\dim_k(L) < \infty\).
\end{proposition}

\begin{proof}
    By Noether normalisation, we have a finite extension \(k[x_1, \dots, x_\ell] \le L\) where the \(x_i\) are algebraically independent. Moreover, this is an integral extension, and so \(k[x_1, \dots, x_\ell]\) is a field. So \(\ell = 0\). Hence \(k \le L\) is a finite extension.
\end{proof}

\lec

From now on, fix a field extension \(\Omega/k\), where \(\Omega\) is algebraically closed.

\begin{definition}
    [vanishing locus, algebraic set] For \(S \sse k[T_1, \dots, T_n]\), define
    \[\van(S) = \left\{x \in \Omega^n \mid f(x) = 0 \text{ for all }f \in S\right\}\]
    we call such sets \emph{\(k\)-algebraic sets}\index{k-algebraic set@\(k\)-algebraic set}
\end{definition}

\begin{definition}
    [ideal of a subset] \index{ideal of a subset} For \(X \sse \Omega^n\), define
    \[I(X) = \left\{f \in k[T_1, \dots, T_n] \mid f(x) = 0 \text{ for all }x \in X\right\}\idealof k[T_1, \dots, T_n]\]
\end{definition}

\begin{remark}
    Note \(\van(S) = \van(\inner{S})\).
\end{remark}

Recall from field theory that if \(L/k\) is a finite field extension, then there exists a \(k\)-homomorphism \(L \to \Omega\).

\begin{theorem}
    \index{Nullstellensatz}
    Let \(\mfa \idealof k[T_1, \dots, T_n]\) be an ideal. Then
    \begin{enumerate}[(i)]
        \item (Weak Nullstellensatz) \(\van(\mfa) = \emptyset\) if and only if \(1 \in \mfa\),
        \item (Strong Nullstellensatz) \(I(\van(\mfa)) = \sqrt{\mfa}\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    For (i), \(\impliedby\) is clear. Now suppose \(1 \notin \mfa\). Hence there exists a maximal ideal \(\mfm\) of \(k[T_1, \dots, T_n]\) containing \(\mfa\), and so \(L = k[T_1, \dots, T_n]/\mfm\) is a field, and it is also finitely generated as a \(k\)-algebra. By Zariski's lemma, \(\dim_k(L) < \infty\). Hence there exists a \(k\)-homomorphism \(L \to \Omega\).

    Consider the composition \(\varphi : k[T_1, \dots, T_n] \to L \to \Omega\). In this case, \(\ker(\varphi) = \mfm\). Define 
    \[x = (\varphi(T_1), \dots, \varphi(T_n)) \in \Omega^n\]
    Then for \(f \in k[T_1, \dots, T_n]\),
    \[\varphi(f) = f(\overline x)\]
    Hence for all \(f \in \mfa \sse \mfm\), 
    \[f(\overline{x}) = \varphi(f) = 0\]

    For (ii), let \(f \in \sqrt{\mfa}\). Then then \(f^\ell \in \mfa\) for some \(\ell\), and thus \(f^\ell(x) = 0\) for all \(x \in \van(\mfa)\). But we are working in a field, and so \(f(x) = 0\) for all \(x \in \van(\mfa)\), i.e. \(f \in I(\van(\mfa))\).

    Conversely, take \(f \in I(\van(\mfa))\). We want to show that \(f \in \sqrt{\mfa}\). Equivalently, \(\overline{f}\) is nilpotent in \(R = k[T_1, \dots, T_n]/\mfa\). In turn, this is equivalent to
    \[R_{\overline f} = 0\]
    But recall that
    \[R_{\overline f} = \frac{R[T_1, \dots, T_n, U]}{\mfa^e + \inner{Uf - 1}}\]
    Let \(\mfb = \mfa^e + \inner{UF - 1}\). Hence we need to show that \(1 \in \mfb\). By the Weak Nullstellensatz, it suffices to show \(\van{\mfb} = \emptyset\).

    Take \(x = (x_1, \dots, x_n, u) \in \van(\mfb) \sse \Omega^{n+1}\). Let \(x' = (x_1, \dots, x_n)\), then
    \[x' \in \van(\mfa)\]
    Hence \(f(x')\), since \(f \in I(\van(\mfa))\). Considering the canonical embedding \(k[T_1, \dots, T_n] \inj k[T_1, \dots, T_n, U]\), \(f(x') = 0\). Now \((Uf - 1)(x) = -1 \ne 0\), contradiction, as \(Uf -1 \in \mfb\).
\end{proof}

Recall \(\sqrt{\sqrt{I}} = \sqrt{I}\), and we have that
\begin{enumerate}
    \item if \(X \sse Y \sse \Omega^n\), then \(I(Y) \sse I(X)\),
    \item if \(S \sse T \sse k[T_1, \dots, T_n]\), then \(\van(T) \sse \van(S)\),
    \item if \(S \sse k[T_1, \dots, T_n]\), then \(S = I(\van(S))\),
    \item if \(X \sse \Omega^n\), then \(X \sse \van(I(X))\).
    \item if \(X \sse \Omega^n\) is an algebraic set, then \(X = \van(I(X))\). This follows from writing \(X = \van(\mfa)\).
    \item if \(X \sse \Omega^n\), then \(I(X)\) is a radical ideal.
\end{enumerate}

\begin{proposition}
    We have a bijection
    \begin{align*}
        \{k\text{-alg. subsets of }\Omega^n\} &\bij \left\{\text{radical ideals in }k[T_1, \dots, T_n]\right\} \\
        X &\mapsto I(X) \\
        \van(\mfa) &\mapsfrom \mfa
    \end{align*}
\end{proposition}

\begin{proof}
    We know \(I(X)\) is radical, and \(X = \van(I(X))\). Now take \(\mfa \in k[T_1, \dots, T_n]\) a radical ideal, then by the strong Nullstellensatz
    \[I(\van(\mfa)) = \sqrt{\mfa} = \mfa\] 
\end{proof}

\begin{remark}
    Note that we defined algebraic subsets with respect to \(k \sse \Omega\).
\end{remark}

\begin{corollary}
    Under the above correspondence, maximal ideals correspond to minimal non-empty algebraic sets. In particular, let \(k = \Omega\) be an algebraically closed field. Then we have a bijection
    \begin{align*}
        \Omega^n &\bij \{\text{maximal ideals of }\Omega[T_1, \dots, T_n]\} \\
        x = (x_1, \dots, x_n) &\mapsto \mfm_x = (T_1 - x_1, \dots, T_n - x_n)
    \end{align*}
\end{corollary}

\begin{proof}
    The first part is just the fact that \(\van\) and \(I\) are order reversing.

    Since \(\Omega[T_1, \dots, T_n]/\mfm_x = \Omega\), \(\mfm_x\) is a maximal ideal. Moreover, \(\mfm_x\) is the ideal of polynomials which vanish on \(x\). To see this,
    \[\mfm_x \sse I(\{x\})\]
    But \(\mfm_x\) is maximal, and \(I(\{x\})\) is a proper ideal, and so equality holds. Moreover, \(\van(\mfm_x) = \{x\}\). The claim follows from the inclusion reversing bijection from before.
\end{proof}

Note that the requirement that \(k = \Omega\) above is necessary. Consider the field extension \(\C/\R\). In \(\R[t]\), \(\inner{t^2 + 1}\) is a maximal ideal, but it corresponds to the points \(\{i, -i\} \sse \C\). In general, for \(\Omega/k\) as above, each point \(x \in k^n\) is a minimal \(k\)-algebraic subsets of \(\Omega^n\), but there can be more. If \(\Char(k) = 0\), then \(x \in \Omega^n\) is \(k\)-algebraic if and only if the coordinates are in \(k\). More generally, if \(\Omega/k\) is separable.

On the other hand, if \(k = \bb F_p(x)\) is the field of rational functions over \(\bb F_p\), \(\Omega = \overline k\), \(n = 1\). Consider the polynomial
\[T^p - x \in k[T]\]
By Frobenius and that \(k\) is algebraically closed, \(T^p - x = (T - x^{1/p})^p\) over \(\Omega\). Hence
\[\van(T^p - x) = \{x^{1/p}\}\]

Finally, note that every prime ideal is radical.

\begin{definition}

\section{Integral and finite extensions again}

\begin{definition}
    [integral over an ideal]
    If \(A \sse B\), \(\mfa \idealof A\), \(x \in B\) is \emph{integral over \(\mfa\)}\index{integral!element over an ideal} if
    \[x^n + a_1x^{n-1} + \cdots + a_n = 0\]
    where \(a_i \in \mfa\).
\end{definition}

\lec

\begin{definition}
    [integral closure over an ideal]

    If \(A \sse B\) rings, \(\mfa \idealof A\), then the \emph{integral closure of \(\mfa\) in \(B\)}\index{integral closure!over an ideal} is
    \[\{x \in B \mid x \text{ is }\mfa\text{-integral}\}\]
\end{definition}

\begin{proposition}
    If \(A \sse B\) are rings, \(\overline{A}\) the integral closure of \(A \sse B\), \(\mfa \sse A\) is an ideal. Then the integral closure of \(\mfa\) in \(B\) is
    \[\sqrt{\mfa \overline A}\]
    where we take the radical in \(\overline A\).
\end{proposition}

\begin{proof}
    Suppose \(b \in B\) is \(\mfa\)-integral, then
    \[b^n + a_1b^{n-1} + a_n = 0\]
    with \(a_i \in \mfa\). In particular, \(\mfb\) is integral over \(A\), and therefore, \(b_0, \dots, b_{n-1} \in \overline A\). Using the above,
    \[b^n \in \mfa \overline A\]
    and so \(b \in \sqrt{\mfa\overline A}\).

    Now suppose \(b \in \sqrt{\mfa \overline A}\). Then \(b^n \in \mfa \overline A\) for some \(n\), and so
    \[b^n = \sum_{i=1}^m a_i x_i \tag{\(*\)}\]
    where \(a_i \in \mfa\), \(x_i \in \overline A\). Define the algebra
    \[M := A[x_1, \dots, x_m]\]
    Since each \(x_i\) is integral over \(A\), \(M\) is a finite \(A\)-algebra. Moreover, from \((*)\), \(b^nM \sse \mfa M\). Now define \(f : M \to M\),
    \[f(m) = b^nm\]
    This satisfies \(f(M) \sse \mfa M\), and \(f\) is \(A\)-linear. Therefore, by Cayley-Hamilton,
    \[f^\ell + \alpha_1f^{\ell - 1} + \cdots + \alpha_\ell = 0 \in \End_R(M)\]
    where each \(\alpha_i \in \mfa\). Evaluating this at \(1 \in A\), we get that
    \[b^{n\ell} + \alpha_1 b^{n(\ell - 1)} + \cdots + \alpha_\ell = 0 \in B\]
    and so \(b\) is \(\mfa\)-integral.
\end{proof}

\begin{corollary}
    Suppose \(A \sse B\) are rings, \(\mfa \idealof A\), \(b \in B\), then \(b\) is \(\mfa\)-integral if and only if \(b\) is \(\sqrt{\mfa}\)-integral.
\end{corollary}

\begin{proof}
    By the proposition, it suffices to show
    \[\sqrt{\mfa \overline A} = \sqrt{\sqrt{\mfa}\,\overline A}\]
    \(\sse\) is clear. For \(\supseteq\), note that in general, \(\sqrt{I}^e \sse \sqrt{I^e}\). Applying this to the above, we have that
    \[\sqrt\mfa\,\overline A \sse \sqrt{\mfa\overline A}\]
    and so
    \[\sqrt{\sqrt{\mfa}\,\overline A} \sse \sqrt{\mfa\overline A}\]
\end{proof}

\begin{proposition}
    \label{prop:integral-min-poly}
    Let \(A\) be an integrally closed\footnote{in \(\Frac(A)\)} integral domain, and \(A \sse B\) rings, \(B\) is an integral domain, and an ideal \(\mfa \idealof A\). Let \(b \in B\), We have a field extension \(\Frac(B)/\Frac(A)\), and the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(b\) is integral over \(\mfa\)
        \item \(b\) is algebraic over \(\Frac(A)\), with minimal polynomial over \(\Frac(A)\) of the form
        \[T^n + a_1T^{n-1} + \cdots + a_0\]
        where \(a_i \in \sqrt{\mfa}\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    Suppose (ii) holds, then \(b\) is integral over \(\sqrt{\mfa}\) by definition. By the corollary, \(b\) is integral over \(\mfa\).

    Now suppose (i) holds. Let \(F = \Frac(A)\). Then we have that
    \[b^n + a_1b^{n-1} + \cdots + a_n = 0\]
    where \(a_i \in \mfa\). Set
    \[h(T) = T^n + a_1T^{n-1} + \cdots + a_n \in F[T]\]
    Then \(h(b) = 0\), and so \(b\) is algebraic over \(\Frac(A)\). Now let \(f\) be the minimial polynomial of \(b\) over \(F\). Let \(\Omega/F\) be an algebraically closed field. In this case,
    \[f = \prod_{i=1}^\ell(T - \alpha_i) \tag{\(*\)}\]
    where each \(\alpha_i \in \Omega\). We would like to show that the coefficient of \(f\) are in \(\sqrt{\mfa}\). Since \(A\) is integrally closed, the integral closure of \(\mfa\) in \(F\) is \(\sqrt{\mfa} \idealof A\). Thus, it suffices to show that the coefficients of \(f\) are \(\mfa\)-integral. Note that by definition, the coefficient of \(f\) are in \(F\).

    Expanding \((*)\), we see the coefficients of \(f\) are sums of products of the \(\alpha_i\). By the proposition, the integral closure of \(\mfa\) in \(\Omega\) is closed under sums and products (as it is an ideal). Therefore, we need to show that each \(\alpha_i\) is integral over \(A\).

    In this case, \(\alpha_i\) and \(b\) have the same minimal polynomial over \(\Frac(A)\), and therefore, there exists \(\varphi_i : F(b) \to F(\alpha_i)\), which is a \(F\)-homomorphism, with \(\varphi_i(b) = \alpha_i\). Since \(h\) has coefficients in \(F\),
    \[h(\alpha_i) = h(\varphi_i(b)) = \varphi(h_i(b)) = 0\]
\end{proof}

\subsection{Cohen-Seidenberg theorems}

Let \(\iota : A \inj B\) be the inclusion map. Then we have a pullback
\begin{align*}
    i^* : \Spec(B) &\to \Spec(A) \\
    \mfq *\mapsto \mfq \cap A
\end{align*}
We are interested in studying \(i^*\), in particular its fibres.

\begin{proposition}
    [incomparability] If \(A \sse B\) is an integral extension, \(\mfq, \mfq' \in \Spec(B)\), \(\mfq \sse \mfa'\), and \(\mfq \cap A = \mfq' \cap A\). Then \(\mfq = \mfq'\).
\end{proposition}

That is, the elements of the fibres are pairwise incomparable.

\begin{proof}
    Let \(\mfp = \mfq \cap A = \mfq' \cap A\), and \(S = A\setminus \mfp\). \(\mfq\) and \(\mfq'\) are prime ideals of \(B\) not intersecting \(S\), So
    \[\mfq = (S^{-1}\mfq)^c\]
    where by \(S^{-1}\mfq\), we mean the extension of \(\mfq\) to \(S^{-1}B\). Note this is not the localisation of \(B\) at \(\mfp\), since \(\mfp\) need not be a prime in \(B\). Similarly, \(\mfq' = (S^{-1}\mfq')^c\). We would like to show that
    \[S^{-1}\mfq = S^{-1}\mfq'\]
    To see this,
    \[S^{-1}\mfq \cap A_\mfp = S^{-1}\mfq \cap S^{-1}A = S^{-1}(\mfq \cap A) = S^{-1}\mfp = \mfp A_\mfp\]
    Similarly, \(S^{-1}\mfq' \cap A_{\mfp} = \mfp A_\mfp\), which is the unique maximal ideal of \(A_\mfp\).

    Since \(A \sse B\) is an integral extension, so is \(A_\mfp \sse S^{-1}B\). Therefore, the contractions \(S^{-1}\mfq, S^{-1}\mfq'\) are maximal ideals of \(S^{-1}B\). But \(\mfq \sse \mfq'\), and so they are equal.
\end{proof}

\section{Primary decomposition}

\begin{definition}
    [primary ideal]
    Let \(I\) be an ideal of \(R\), then \(I\) is \emph{primary}\index{primary ideal} if \(R/I\) is non-zero, and every zero divisor in \(R/I\) is nilpotent.
\end{definition}

\begin{remark}
    Contrast this with \(I\) being prime if \(R/I\) is an integral domain, and \(I\) is radical if \(R/I\) has no non-zero nilpotent elements.

    In particular, any prime ideal is radical and primary. Note \(R\) is radical, but not prime nor primary.
\end{remark}

\lec

\begin{example}
    In \(\Z\), \(\inner{6}\) is radical, but not primary, since in \(\R/6\), there are no non-zero nilpotent elements, but \(2 \times 3 = 6\). But \(\inner{9}\) is primary, but not radical.

    More generally, for \(x \ne 0\),
    \begin{itemize}
        \item \(\inner{x}\) if and only if \(x\) is prime,
        \item \(\inner{x}\) is radical if and only if \(x\) is square free,
        \item \(\inner{x}\) is primary if and only if \(x = p^n\) for some prime \(p\).
    \end{itemize}
\end{example}

\begin{proposition}
    Let \(I \idealof R\) be a proper ideal.
    \begin{enumerate}[(i)]
        \item if \(I\) is primary, then \(\mfp = \inner{I}\) is prime, and we say that \(I\) is \(\mfp\)-primary,
        \item if \(\sqrt{I}\) is maximal, then \(I\) is primary,
        \item if \(\mfq_1, \dots, \mfq_n\) are all \(\mfp\)-primary, then so is \(\mfq_1 \cap \cdots \cap \mfq_n\),
        \item if \(I\) has a \emph{primary decomposition}\index{primary decomposition}, i.e.
        \[I = \mfq_1 \cap \cdots \cap \mfq_n \tag{\(*\)}\]
        where \(\mfq_i\) is primary, then \(I\) has a \emph{minimal primary decomposition}\index{minimal prime decomposition}, i.e. like \((*)\), but \(\sqrt{\mfq_1}, \dots, \sqrt{\mfq_n}\) are distinct, and none of the \(\mfq_i\) can be dropped, 
        \item if \(R\) is Noetherian, then every ideal \(I\) has a primary decomposition
    \end{enumerate}
\end{proposition}

\begin{proof}
    Examples sheet.
\end{proof}

\begin{example}
    In \(\Z\),
    \[\inner{90} = \inner{2} \cap \inner{3^2} \cap \inner{5}\]
\end{example}

\begin{example}
    For a prime ideal \(\mfp\) of \(R\), if \(\mfp^n\) is primary, then \(\mfp^n\) is \(\mfp\)-primary, as \(\sqrt{\mfp^n} = \mfp\).
    \begin{enumerate}
        \item \textbf{Not every primary ideal is a power of a prime}. Let \(R = k[x, y]\), \(\mfq = \inner{x, y^2}\). To see that \(\mfq\) is primary, \(\sqrt{\mfq} = \inner{x, y}\), which is a maximal ideal, and so \(\mfq\) is \(\inner{x, y}\)-primary. Alternatively, \(k[x, y]/\mfq = k[y]/\inner{y^2}\). If \(f \in k[y]\) and \(f + \inner{y^2}\) is a zero divisor, then \(y\) divides \(f\), and so \(f + \inner{y^2}\) is nilpotent.
        
        On the other hand, if \(\mfq = \mfp^n\), then \(\sqrt{\mfq} = \mfp\), but \(\sqrt{\mfq} = \inner{x, y}\). But we have that
        \[\inner{x, y}^2 \subset \inner{x, y^2} \subset \inner{x, y}\]

        \item \textbf{Power of a prime does not have to be primary.} Let \(R = k[x, y, z]/\inner{xy - z^2} = k[\overline{x}, \overline y, \overline z]\), Let \(\mfp = \inner{\overline x, \overline z}\). We will show that \(\mfp\) is prime, but \(\mfp^2\) is not primary. In this case,
        \[R/\mfp = k[y]\]
        which is an integral domain, and so \(\mfp\) is prime. On the other hand,
        \[\mfp^2 = \inner{\overline{x}^2, \overline{x}\overline{z}, \overline{z}^2}\]
        With this,
        \[\overline{x}\overline{y} = \overline{z}^2 \in \mfp^2\]
        so the image of \(\overline{x}\overline{y}\) in \(R/\mfp^2\) is zero. But \(\overline{x} + \mfp^2 \ne 0\), and so \(\overline{y} + \mfp^2\) is a zero divisor in \(R/\mfp^2\). But
        \[R/\mfp^2 = k[x, y, z]/\inner{xy - z^2, x^2, xz, z^2}\]
        and no power of \(y\) is in \(\inner{xy - z^2, x^2, xz, z^2}\).
    \end{enumerate}
\end{example}

\begin{theorem}
    Let \(I = \mfq_1 \cap \cdots \cap \mfq_n\) be a minimal primary decomposition. Let \(\mfp_i = \sqrt{\mfq_i}\), then
    \begin{enumerate}[(i)]
        \item (\emph{associated primes of \(I\)}\index{associated prime}) \(\mfp_1, \dots, \mfp_n\) are determined only by \(I\),
        \item (\emph{isolated primes of \(I\)}\index{isolated prime}) the minimal elements amongst the \(\mfp_1, \dots, \mfp_n\) are exactly the minimal primes of \(R\) containing \(I\),
        \item if \(\mfp_1, \dots, \mfp_t\) are the isolated primes of \(I\), then \(\mfq_1, \dots, \mfq_t\) are determined only by \(I\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    Examples sheet.
\end{proof}

\begin{definition}
    [embedded primes] The \emph{embedded primes}\index{embedded primes} of \(I\) are the associated primes which are not isolated.
\end{definition}

\begin{example}
    Let \(R = k[x, y]\), \(I = \inner{x^2, xy}\). Then we have primary decompositions
    \[I = \inner{x} \cap \inner{x, y}^2 = \inner{x} \cap \inner{x^2, y}\]
    In this case, \(\sqrt{\inner{x}} = \inner{x}\), \(\sqrt{\inner{x, y}^2} = \inner{x, y}\), and \(\sqrt{\inner{x^2, y}} = \inner{x, y}\).

    In this case, the associated primes are \(\inner{x}, \inner{x, y}\), which don't depend on the decomposition. In particular, \(\inner{x}\) is isolated and \(\inner{x, y}\) is embedded.

    Thining about this geometrically, \(\van(\inner{x, y}) \sse \van{\inner{x}}\), which is why we call them \emph{embedded}.
\end{example}

If \(I = \mfq_1 \cap \cdots \cap \mfq_n\) is a minimal primary decomposition, \(\mfp_i = \sqrt{\mfq_i}\). Say \(\mfp_1, \dots, \mfp_t\) are the isolated primes. Then
\[\sqrt{I} = \sqrt{\mfq_1} \cap \cdots \cap \sqrt{\mfq_t} = \mfp_1 \cap \cdots \cap \mfp_t\]
which is a (minimal) primary decomposition of \(\sqrt{I}\), and all associated primes are isolated. Thus, going from \(I\) ot \(\sqrt{I}\) is the same as forgetting the embedded primes of \(I\). 

Geometrically, in \(k[t_1, \dots, t_n]\), where \(k \sse \C\) is a subfield, then
\[\van(I) = \van(\sqrt{I})\]
and \(I(\van(I)) = \sqrt{I}\), thus \(\van(I)\) only sees \(\sqrt{I}\), or equivalently, it forgets about the embedded primes.

\section{Direct and inverse limits}

\lec

Let \(\msC\) be a category.

\begin{definition}
    [directed set]
    A \emph{directed set}\index{directed set} \((I, \le)\) is a poset, such that for all \(a, b \in I\), there exists \(c \in I\) such that \(a \le c, b \le c\).
\end{definition}

\begin{definition}
    [directed system] A \emph{direct system}\index{direct system} on \(I\) is objects \((X_i)_{i \in I}\) of \(\msC\), and for every \(i \le j\), a morphism \(f_{ij} : X_i \to X_j\), such that
    \begin{enumerate}
        \item \(f_{ii} = \id_{X_i}\) for all \(i\),
        \item \(f_{ik} = f_{jk}f_{ij}\) for all \(i \le j \le k\).
    \end{enumerate}
\end{definition}

\begin{definition}
    [inverse system]
    An \emph{inverse system}\index{inverse system} on \(I\) is objects \((Y_i)_{i \in I}\) of \(\msC\), and for every \(i \le j\), a morphism \(h_{ij} : X_j \to X_i\), such that
    \begin{enumerate}
        \item \(h_{ii} = \id_{Y_i}\) for all \(i\),
        \item \(f_{ik} = f_{ij}f_{jk}\) for all \(i \le j \le k\).
    \end{enumerate}
\end{definition}

\begin{example}
    Let \(I = (\N, \le)\), fix a prime \(\mfp\), consider the direct system
    \[X_i = \bb F_{p^{i!}}\]
    and \(f_{ij}\) being field embeddings. Recall if \(a \mid b\), then there exists an embedding \(\bb F_{p^a} \inj \bb F_{p^b}\), and that the set of all embeddings are given by
    \[x \mapsto \varphi(x)^{p^c}\]
    for \(0 \le c \le a - 1\). But we can just define \(f_{i, i+1}\), and the other maps are defined by composition.
\end{example}

\begin{example}
    Let \(I = (\N, \le)\), fix a prime \(p\), and consider
    \[Y_i = \Z/p^i\]
    and
    \begin{align*}
        h_{ij} : \Z/p^j &\to \Z/p^i \\
        x &\mapsto p^{i-j}x
    \end{align*}
    the natural projection map.
\end{example}

\begin{definition}
    [direct limit]
    Let \((I, \le)\) be a directed set. If \(D = ((X_i),(f_{ij}))\) forms a direct system, then the \emph{direct limit}\index{direct limit} of \(D\) is
    \[\dirlim X_i = \frac{\bigsqcup_i X_i}{\sim}\]
    where for \(x_i \in X_i, x_j \in X_j\), \(x_i \sim x_j\) if and only if there exists \(k\) such that \(f_{ik}(x_i) = f_{jk}(x_j)\). Equivalently, take the equivalence relation generated by \(x_i \sim f_{ij}(x_i)\) for all \(i \le j\).
\end{definition}

\begin{remark}
    If \(\mcD\) is a direct system in \(\msC\), then the direct limit is in \(\msC\) as well.
\end{remark}

\begin{definition}
    [inverse limit]
    Let \((I, \le)\) be a direct set. If \(E = ((Y_i), (h_{ij}))\) forms an inverse system, then the \emph{inverse limit}\index{inverse limit} of \(E\) is
    \[\invlim Y_i = \left\{y \in \prod_i Y_i \mid y_i = f_{ij}(y_j) \text{ for all }i \le j\right\}\]
\end{definition}

\begin{example}
    We claim that \(\bb F_p^{\text{alg}} = \dirlim \bb F_{p^{i!}}\) is an algebraic closure of \(\bb F_p\). 
    
    First we check that \(\bb F_p^{\text{alg}}\) is algebraic over \(\bb F_p\). Choose \([x] \in \bb F_p^{\text{alg}}\), say \(x \in \bb F_p^{i!}\), then \(x^{p^{i!}} - x = 0\), and so \([x]^{p^{i!}} - [x] = 0\).

    Next we check that it is algebraically closed. Let \([h] \in \bb F_p^{\text{alg}}[t]\). Since \([h]\) has finitely many coefficients, we have that \(h \in \bb F_{p^{i!}}[t]\). Considering a splitting field for \(h\), which is \(\bb F_{p^\ell}\), which in turn embeds into \(\bb F_{p^{\ell !}}\). Hence \(h\) splits over \(\bb F_{p^{\ell!}}\), and so \(h\) splits under the embdedding \(f_{i\ell} : \bb F_{p^{i!}} \to \bb F_{p^{\ell!}}\). This means that \([h]\) splits over the direct limit.
\end{example}

\begin{example}
    Let
    \[\Z_p = \invlim \frac{\Z}{p^i}\]
    be the ring of \(p\)-adic integers. For example,
    \(1 = (1, 1, 1, \dots)\)
    and
    \[-1 = (p-1, p^2-1, p^3 - 1, \dots)\]
\end{example}

\begin{definition}
    [\(\mfa\)-adic completion]
    Let \(R\) be a ring, \(\mfa \idealof R\) an ideal, then the \emph{\(\mfa\)-adic completion}\index{adic completion@\(\mfa\)-adic completion!of a ring} of \(R\) is
    \[\widehat R = \invlim \frac{R}{\mfa^i}\]
\end{definition}

\begin{example}
    If \(R = \Z\), \(\mfa = \inner{p}\), then \(\widehat R = \Z_p\).
\end{example}

\begin{example}
    If \(R = k[T]\), \(\mfa = \inner{T}\), then
    \[\widehat R = \invlim \frac{R}{\inner{T^i}} = k\powers T\]
\end{example}

\begin{definition}
    [\(\mfa\)-adic completion of a module]
    Let \(R\) be a ring, \(\mfa \idealof R\) be an ideal, \(M\) an \(R\)-module, then \emph{\(\mfa\)-adic completion}\index{adic completion@\(\mfa\)-adic completion!of a module} of \(M\) is
    \[\widehat{M} = \invlim\frac{M}{\mfa^i M}\]
    which is naturally a \(\widehat{M}\)-module. 
\end{definition}

\begin{definition}
    [filtration, completion with respect to a filtration] A \emph{filtration}\index{filtration} of an \(R\)-module \(M\) is a sequence \((M_n)\) of submodules of \(M\), with \(M_n \supseteq M_{n+1} \supseteq \cdots\), and \(M_0 = M\).
    
    The \emph{completion of \(M\) with respect to the filtration}\index{completion with respect to a filtration} is the inverse limit
    \[\invlim \frac{M}{M_n}\]
\end{definition}

\begin{theorem}
    Let \(R\) be a Noetherian ring, and let \(\mfa \idealof R\) be an ideal. Let \(\widehat R\) denote the \(\mfa\)-adic completion of \(R\).
    \begin{enumerate}[(i)]
        \item \(\widehat R\) is Noetherian,
        \item the functor \(\widehat R \otimes_R (\cdot)\) is exact.
        \item if \(M\) is a finitely generated \(R\)-module, then the natural map
        \[\widehat R \otimes M \to \widehat M\]
        is an \(\widehat R\)-linear isomorphism.
    \end{enumerate}
\end{theorem}

\begin{corollary}
    If \(R\) is a Noetherian ring, \(R\powers {T_1, \dots, T_n}\) is Noetherian.
\end{corollary}

\begin{proof}
    It is the \(\mfm\)-adic completion of \(R[T_1, \dots, T_n]\) at \(\mfm = \inner{T_1, \dots, T_n}\).
\end{proof}

\section{Filtration and graded rings}

\subsection{Graded rings and modules}

\begin{definition}
    [graded ring]
    A \emph{graded ring}\index{graded ring} \(A\) is a ring
    \[A = \bigoplus_{n = 0}^\infty A_n\]
    where each \(A_i\) is an additive subgroup of \(A\), and \(A_nA_m \sse A_{n+m}\).
\end{definition}

\begin{lemma}
    \(A_0\) is a subring of \(A\).
\end{lemma}

\begin{proof}
    The only thing we need to show is that \(1 \in A_0\). If \(A = A_0\) then we are done. Otherwise, choose \(z \in A_n\), and say
    \[1 = \sum_i y_i\]
    where \(y_i \in A_i\). Then \(y_iz \in A_{n + i}\). But \(z = 1z\), and so we must have that \(y_0 = 1, y_i = 0\) for \(i > 0\).
\end{proof}

\begin{example}
    \(A_d = k[T_1, \dots, T_n]\) is a graded ring, and in this case \(A_d\) is the degree \(d\) homogeneous polynomials.
\end{example}

\begin{definition}
    [irrelevant ideal] We call
    \[A_+ = \bigoplus_{n \ge 1}A_n\]
    the \emph{irrelevant ideal}\index{irrelevant ideal}.
\end{definition}

\(A_+\) is the kernel of the projection map \(A \to A_0\), and so \(A/A_+ \cong A_0\).

\begin{definition}
    [graded module]
    Let \(A\) be a graded ring. A \emph{graded \(A\)-module}\index{graded module} is an \(A\)-module \(M\), with
    \[M = \bigoplus_n M_n\]
    each \(M_i\) an additive subgroup, and \(A_nM_m \sse M_{n+m}\).
\end{definition}

\begin{proposition}
    Let \(A\) be a graded ring. Then \(A\) is Noetherian if and only if \(A_0\) is Noetherian and \(A\) is a finitely generated \(A_0\)-algebra.
\end{proposition}

\begin{proof}
    From Hilbert's basis theorem, if \(A_0\) is Noetherian and \(A\) is a finitely generated \(A_0\)-algebra, then \(A\) is Noetherian.

    Now suppose \(A\) is Noetherian. Then \(A_0 = A/A_+\) is the quotient of a Noetherian ring, and so Noetherian. Next, \(A_+\) is generated by the set of homogeneous elements of positive degree. Now \(A_+\) is finitely generated, as \(A\) is Noetherian. That is,
    \[A_+ = \inner{x_1, \dots, x_s}\]
    where \(x_i \in A_{k_i}, k_i > 0\). Let \(A'\) be the \(A_0\)-subalgebra of \(A\), defined by
    \[A' = A_0[x_1, \dots, x_s]\]
    We would like to show \(A = A'\). It suffices to show that \(A_n \sse A'\) for every \(A\). We will prove this by induction on \(n\). \(n = 0\) is clear.
    
    Now take \(y \in A_n\), \(n > 0\). Now \(y \in A_+\), and so we can write
    \[y = \sum_{i=1}^sr_ix_i\]
    where \(r_i \in A\). Apply the projection \(A \to A_n\), we get
    \[y = \sum_{i=1}^s a_i x_i\]
    where \(a_i \in A_{n - k_i}\). But as \(k_i > 0\), the induction hypothesis implies that each \(a_i\) is in \(A'\), and so \(y \in A'\).
\end{proof}

\subsection{Associated graded ring}

\begin{definition}
    [\(\mfa\)-filtration]
    Let \(\mfa \idealof R\) be an ideal, \(M\) an \(R\)-module. A filtration \((M_n)\) is an \emph{\(\mfa\)-filtration}\index{filtration@\(\mfa\)-filtration} if \(\mfa M_n \sse M_{n+1}\) for all \(n\).

    An \(\mfa\)-filtration is \emph{stable}\index{stable} if \(\mfa M_n = M_{n+1}\) for all sufficiently large \(n\).
\end{definition}

\begin{example}
    \((\mfa^nM)_{n \ge 0}\) is a stable \(\mfa\)-filtration of \(M\).
\end{example}

\begin{definition}
    [associated graded ring]
    If \(\mfa \idealof R\) is an ideal, then we have an \emph{associated graded ring}\index{associated graded ring}
    \[\mathrm{G}_\mfa(R) = \bigoplus_{n \ge 0}\frac{\mfa^n}{\mfa^{n+1}}\]
    We make this into a ring, by
    \[(x + \mfa^{n+1})(y + \mfa^{\ell + 1}) = xy + \mfa^{n+\ell+1}\]
    for \(x \in \mfa^n, y \in \mfa^\ell\).
\end{definition}

\begin{definition}
    [associated graded module]
    If \(\mfa \idealof R\) an ideal, \(M\) an \(R\)-module, \((M_n)_{n \ge 0}\) an \(\mfa\)-filtration of \(M\), then we have an \emph{associated graded module}\index{associated graded module}
    \[\mathrm{G}(M) = \bigoplus_{n \ge 0}\frac{M_n}{M_{n+1}}\]
    which is an \(\mathrm G_\mfa(R)\)-module, with module structure given by
    \[(x + \mfa^{n+1})(m + M_{\ell +1}) = xm + M_{n+\ell + 1}\]
\end{definition}

\begin{proposition}
    Let \(R\) be a Noetherian ring, \(\mfa \idealof R\) an ideal. Then
    \begin{enumerate}[(i)]
        \item \(\mathrm G_\mfa(R)\) is Noetherian,
        \item if \(M\) is a finitely generated \(R\)-module, \((M_n)\) is a stable \(\mfa\)-filtration of \(M\), then \(\mathrm G(M)\) is a finitely generated \(\mathrm G_\mfa(R)\)-module.
    \end{enumerate}
\end{proposition}

\begin{proof}
    For (i), since \(R\) is Noetherian, \(\mfa\) is finitely generated, say
    \[\mfa = \inner{x_1, \dots, x_s}\]
    Set \(\overline{x_i} = x_i + \mfa^2 \in \mfa/\mfa^2\). Then \(\mathrm G_\mfa(R)\) is generated as an \(R/\mfa\)-algebra by \(\overline{x_1}, \dots, \overline{x_n}\). But \(R/\mfa\) is a Noetherian ring, and so \(\mathrm G_\mfa(R)\) by the Hilbert Basis Theorem.

    For (ii), since \((M_n)\) is stable, so there exists \(N\) such that
    \[M_{N+r} = \mfa^rM_N\]
    Then \(\mathrm G(M)\) is generated by
    \[\bigoplus_{n \le N}\frac{M_n}{M_{n+1}}\]
    as a \(\mathrm G_\mfa(R)\)-module. But each \(M_n/M_{n+1}\) is a Noetherian \(R\)-module, annihilated by \(\mfa\). In particular, each \(M_n/M_{n+1}\) is a finitely generated \(R/\mfa\)-module. So
    \[\bigoplus_{n \le N}\frac{M_n}{M_{n+1}}\]
    is a finitely genertaed \(R/\mfa\)-module, and so it is a finitely generated \(\mathrm G_\mfa(R)\)-module.
\end{proof}

\subsection{Filtrations}

\begin{definition}
    [equivalent]
    Let \(M\) be an \(R\)-module. Then filtrations \((M_n), (M_n')\) of \(M\) are \emph{equivalent}\index{equivalent filtrations} if there exists \(n_0\) such that
    \[M_{n + n_0} \sse M'_n \qqtext{and} M'_{n + n_0} \sse M_n\]
    for all \(n \ge 0\).
\end{definition}

\begin{lemma}
    Let \(\mfa \idealof R\) be an ideal, \(M\) an \(R\)-module, \((M_n)\) is a stable \(\mfa\)-filtration on \(M\). Then \((M_n)\) is equivalent to \((\mfa^nM)\).
\end{lemma}

\begin{proof}
    We have that
    \[M_n \supseteq \mfa M_{n-1} \supseteq \cdots \supseteq \mfa^n M \supseteq \mfa^{n+n_0}M\]
    for all \(n_0 \ge 0\). In the other direction, there exist \(n_0 \ge 0\) such that \(\mfa M_n = M_{n+1}\) for all \(n \ge n_0\). Hence
    \[M_{n + n_0} = \mfa^nM_{n_0} \sse \mfa^n M\]
\end{proof}

Let \(\mfa \idealof R\) be an ideal, \(M\) an \(R\)-moddule, \((M_n)\) an \(\mfa\)-filtration of \(M\). Let
\[R^* = \bigoplus_{n=0}^\infty \mfa^n\]
and
\[M^* = \bigoplus_{n=0}^\infty M_n\]
Then \(R^*\) is a graded ring, and \(M^*\) is a graded \(R^*\)-module with the natural actions.

If \(R\) is Noetherian, then \(\mfa = \inner{x_1, \dots, x_r}\), and \(R^*\) is generated as an \(R\)-algebra by
\[x_1, \dots, x_n \in \mfa\]
Hence by the Hilbert basis theorem, \(R^*\) is Noetherian.

\begin{lemma}
    Let \(R\) be a Noetherian ring, \(M\) a finitely generated \(R\)-module, \((M_n)\) an \(\mfa\)-filtration. Then \(M^*\) is a finitely generated \(R^*\)-module if and only if the \(\mfa\)-filtration \((M_n)\) is stable.
\end{lemma}

\begin{proof}
    First of all, note that
    \begin{enumerate}
        \item Each \((M_n)\) is a finitely generated \(R\)-module. Since \(R\) is Noetherian, and \(M\) is finitely generated, \(M\) is Noetherian, and so every submodule is finitely generated.
        \item Consider the submodule
        \[M_n^* = M_0 \oplus \cdots \oplus M_n \oplus \mfa M_n \oplus \mfa^2 M_n \oplus \cdots\]
        of \(M^*\), then the ascending chain \((M_n^*)\) stabilises, if and only if \((M_n)\) is a stable \(\mfa\)-filtration.
    \end{enumerate}

    Suppose \(M^*\) is finitely generated. We know that \(R\) is Noetherian, and so \(R^*\) is Noetherian, and therefore, \(M^*\) is Noetherian. But then the ascending chain \((M_n^*)\) stabilises, and so \((M_n)\) is a stable \(\mfa\)-filtration by 2.

    Now suppose the filtration \((M_n)\) is stable. Then the sequence \((M_n^*)\) stabilises at some \(n_0\). Now note that
    \[M^* = \bigcup_n M_n^*\]
    Hence \(M^* = M_{n_0}^*\). But we know that
    \[M_0 \oplus \cdots \oplus M_{n_0}\]
    generates \(M_n^*\) as an \(R^*\)-module. But each \(M_n\) is a finitely generated \(R\)-module, and so \(M_0 \oplus \cdots \oplus M_{n_0}\) is a finitely generated \(R\)-module. Thus, \(M_n^*\) is a finitely generated \(R^*\)-module.
\end{proof}

\begin{proposition}
    [Artin-Rees] \index{Artin-Rees} Let \(R\) be a Noetherian ring, \(\mfa \idealof R\) an ideal, \(M\) a finitely generated \(R\)-module, \((M_\ell)\) a stable \(\mfa\)-filtration of \(M\), and \(N \sse M\) a submodule.

    Then \((N \cap M_\ell)\) is a stable \(\mfa\)-filtration of \(N\).
\end{proposition}

\begin{proof}
    First of all,
    \[\mfa(N \cap M_\ell) \sse N \cap \mfa M_\ell \sse N \cap M_{\ell + 1}\]
    and so \((N \cap M_\ell)\) is an \(\mfa\)-filtration. Define
    \[N^* = \bigoplus_{\ell=0}^\infty (N \cap M_\ell)\]
    This is an \(R^*\)-submodule of \(M^*\). Recall \(R\) is Noetherian, and so \(R^*\) is Noetherian. Since \((M_\ell)\) is stable, \(M^*\) is finitely generated, and so \(M^*\) is a Noetherian \(R^*\)-module. Hence \(N^*\) is a finitely generated \(R^*\)-module, and so \((N \cap M_\ell)\) is stable.
\end{proof}

\section{Dimension theory}

\begin{definition}
    [height]
    Let \(\mfp \in \Spec(R)\) be a prime. Then the \emph{height}\index{height} of \(\mfp\) is
    \[\idealht(\mfp) = \sup \{d \mid \mfp_0 \subsetneq \cdots \subsetneq \mfp_d = \mfp\}\]
\end{definition}

\begin{definition}
    [(Krull) dimension]

    The \emph{(Krull) dimension of a ring}\index{dimension@(Krull) dimension} is
    \[\dim(R) = \sup\{\idealht(\mfp) \mid \mfp \in \Spec(R)\} = \sup\{\idealht(\mfm) \mid \mfm \in \maxSpec(R)\}\]
\end{definition}

We can see that \(\dim(R_\mfp) = \idealht(\mfp)\), and so
\[\dim(R) = \sup\{\dim(R_\mfm) \mid \mfm \in \maxSpec(R)\}\]

\begin{definition}
    For an ideal \(I\) of \(R\),
    \[\idealht(I) = \inf\left\{\idealht(\mfp) \mid I \sse \mfp \in \Spec(R)\right\}\]
\end{definition}

\begin{proposition}
    If \(A \sse B\) is an integral extension of rings, then
    \begin{enumerate}[(i)]
        \item \(\dim(A) = \dim(B)\),
        \item if \(A, B\) are integral domains and \(k\)-algebras, where \(k\) is a field, then \(\trdeg_k(A) = \trdeg_k(B)\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    First, we show that \(\dim(A) \le \dim(B)\). Given a chain
    \[\mfp_0 \subsetneq \mfp_1 \subsetneq \cdots \subsetneq \mfp_d\]
    By lying over and going up, we have
    \[\mfq_0 \subseteq \mfq_1 \subseteq \cdots \subseteq \mfq_d\]
    with \(\mfq_i \cap A = \mfp_i\), and so \(\mfq_i \ne \mfq_{i+1}\). Thus, \(\dim(A) \le \dim(B)\).

    Next, we show \(\dim(A) \ge \dim(B)\). Let
    \[\mfq_0 \subsetneq \cdots \subsetneq \mfq_d\]
    be a chain in \(\Spec(B)\), then
    \[\mfq_0 \cap A \subsetneq \cdots \subsetneq \mfq_d \cap A\]
    is a chain in \(\Spec(A)\). By incomparability, \(\mfq_i \cap A \ne \mfq_{i+1} \cap A\), and so \(\dim(A) \ge \dim(B)\).

    (ii) is left as an exercise.
\end{proof}

Now if \(k\) is a field, \(A\) a finitely generated \(k\)-algebra, then by the Noether normalisation theorem, we had a \(k\)-algebra embedding
\[k[T_!, \dots, T_d] \inj A\]
which is an integral extension. Hence by the proposition,
\[\dim(A) = \dim(k[T_1, \dots, T_d]) = d\]
by examples sheet 3 question 10.

\subsection{Hilbert polynomials and functions}

Let \(A\) be a Noetherian graded ring. That is, \(A_0\) is Noetherian and \(A\) is a finitely generated \(A_0\)-algebra. Let \(M\) be a finitely generated graded \(A\)-module. Then each \(M_n\) is an \(A_0\)-module.

\begin{claim}
    \(M_n\) is a finitely generated \(A_0\)-module.
\end{claim}

\begin{proof}
    Say \(M = \Span_A\{m_1, \dots, m_t\}\), each \(m_i \in M_{r_i}\) homogeneous. Therefore,
    \[M_n = \left\{a_1m_1 + \dots + a_tm_t \mid a_i \in A_{n-r_i}\right\}\]
    We have that \(A = A_0[x_1, \dots, x_s]\), each \(x_i \in A_{k_i}\), \(k_i > 0\). Then
    \[M_n = \Span_{A_0} \left\{x_1^{e_1}\cdots x_s^{e_s}m_i \ \bigg\vert\ e_i \ge 0, \sum k_ie_i = n - r_i\right\}\]
\end{proof}

Now we will assume in addition that \(A_0\) is also Artinian. Therefore, each \(M_n\) is an Artinian and Noetherian module. Hence \(\ell(M_n) < \infty\)\footnote{That is, it has finite length. Equivalently, it has a composition series of finite length.}.

\begin{definition}
    [Poincar\'e series]
    Let \(A, M\) be as above. The \emph{Poincar\'e series}\index{Poincar\'e series} of \(M\) is
    \[P(M, T) = \sum_{n=0}^\infty \ell(M_n)T^n \in \Z\powers T\]
\end{definition}

    [Hilbert-Serre]
    \(P(M, T)\) is a rational function of the form

    \[(1-T^{k_s})P(M, T) = P(M, T) - T^{k_s}P(M, T) = P(L, T) - T^{k_s}P(K, T)\]
    But we can write the right hand side as
    \[\frac{f_1}{\prod_{i=1}^{s-1}(1-T^{k_i})} - \frac{T^{k_s}f_2}{\prod_{i=1}^{s-1}(1-T^{k_i})}\]
    by induction. Rearranging gives the result.
\end{proof}

Let \(d(M)\) be the order of the pole of \(P(M, T)\) at \(t = 1\). Then if \(M \ne 0\), \(d \ge 0\). See notes for details.

\begin{example}
    Let \(A = k[T_1, \dots, T_s]\), \(A_n\) the homogeneous parts. Then
    \begin{enumerate}
        \item \(A\) is generated as an \(A_0 = k\)-algebra by \(T_1, \dots, T_s\). In each case, \(k_i = 1\).
        \item \(\ell(A_n) = \dim_k(A_n) = {n+s-1 \choose s}\), which is a polynomial of degree \(s-1\) in \(n\) over \(\Q\). In this case,
        \item \[P(A, T) = \sum {n + s - 1 \choose n}T^n = \frac{1}{(1-T)^s}\] 
    \end{enumerate}
\end{example}

\begin{proposition}
    If \(k_1 = \dots = k_s = 1\), then there exists a polynomial \(\HP_M \in \Q[T]\), and \(n_0 \ge 1\), such that
    \[\ell(M_n) = \HP_M(n)\]
    for all \(n \ge N_0\). Moreover,
    \[\deg(\HP_M) = d(M) - 1\]
    This is called the \emph{Hilbert polynomial}\index{Hilbert polynomial}.
\end{proposition}

\begin{proof}
    Let \(d = d(M) \ge 0\). Then we can write
    \[\sum_{n\ge 0}\ell(M_n)T^n = \frac{f(T)}{(1-T)^d}\]
    where \(f \in \Z[T]\), with \(f(1) \ne 0\). Write
    \[f = \sum_{k=0}^{\deg(f)}a_kT^k\]
    for \(a_k \in \Z\). Next,
    \[\frac{1}{(1-T)^d} = \sum_{j=0}^\infty b_jT^j\]
    where \(b_j = {j + d - 1 \choose j}\). Then
    \[\ell(M_n) = \sum_{i=0}^{\deg(f)}a_{n-i}b_i\]
    for \(n \ge \deg(f)\). Since \(a_i \in \Z\), \(b_j\) is a polynomial in \(j\) over \(\Q\) of degree \(d-1\). Moreover, the leading coefficient of \(b_i\) is
    \[\frac{1}{(d-1)!}\]
    Hence \(\ell(M_n) = p(n)\), where \(p \in \Q[T]\). All we need to show is that \(\deg(p) = d - 1\). The coefficient of \(T^{d-1}\) in \(p\) is
    \[\sum_{i=0}^{\deg(f)}a_i \frac{1}{(d-1)!} = \frac{f(1)}{(d-1)!}\]
    which is non-zero, as \(f(1) \ne 0\) by assumption.
\end{proof}

\begin{lemma}

        \item If \(\mfq\) is \(\mfm\)-primary, then \(A/\mfq\) is Artinian.

\end{lemma}

\begin{theorem}
    [dimension] If \((A, \mfm)\) is a Noetherian local ring, then
    \[\dim(A) = \delta(A) = d(\mathrm{G}_\mfm(A))\]
    where
    \begin{align*}
        \delta(A) &= \min\{\delta(\mfq) \mid \mfq \sse A \text{ }\mfm\text{-primary}\} \\
        \delta(\mfq) &= \text{minimal number of generators for }\mfq 
    \end{align*}
    and \(d(\mathrm{G}_\mfm(A))\) is the order of the pole at \(T = 1\) of the rational function associated to the Poincar\'e series of \(\mathrm{G}_\mfm(A)\). That is, the order of the pole at \(1\) of
    \[\sum_{n \ge 0}\ell\left(\frac{\mfm^n}{\mfm^{n+1}}\right)T^n\]
\end{theorem}

\begin{corollary}
    [Krull's height theorem] Let \(A\) be a Noetherian ring, \(\mfa = (x_1, \dots, x_r) \sse A\) an ideal. Let \(\mfa \le \mfp\) be a minimal prime of \(\mfa\). Then
    \[\idealht(\mfp) \le r\]
\end{corollary}

\begin{proof}
    First of all, we claim that
    \[\sqrt{\mfa A_\mfp} = \mfp A_\mfp\]
    To see this, let \(\mfn \in \Spec(A)\) be such that \(\mfa A_\mfp \sse \mfn\), then
    \[\mfa \sse (\mfa A_\mfp)^c \sse \mfn^c \sse \mfp\]
    Then by minimality, \(\mfn^c = \mfp\). Hence \(\mfn^{ce} = \mfp^e\), and the result follows. Thus, \(\mfa A_\mfp\) is \(\mfp A_\mfp\)-primary. On the other hand,
    \[\mfa A_\mfp = \inner{\frac{x_1}{1}, \dots, \frac{x_r}{1}}\]
    Then
    \[\idealht(\mfp) = \dim(A_\mfp) = \delta(A_\mfp) \le \delta(\mfa A_\mfp) \le r\]
\end{proof}

Let \((A, \mfm)\) be a Noetherian local ring, \(\mfq \idealof A\) an \(\mfm\)-primary ideal. Say \(\delta(\mfq) = s\), and \(\mfq = \inner{x_1, \dots, x_s}\). Then
\[\mathrm{G}_\mfq(A) = \frac{A}{\mfq} \oplus \frac{\mfq}{\mfq^2} \oplus \bigoplus_{n \ge 2}\frac{\mfq^n}{\mfq^{n+1}}\]
In this case, \(A/\mfq\) is Artinian, and the images of \(x_1, \dots, x_s\) generate \(\mfq/\mfq^2\) as an \(A/\mfq\) algebra, the \(x_i\) are of degree \(1\). Here, we have that
\[\ell\left(\frac{\mfq^n}{\mfq^{n+1}}\right) < \infty\]
From the Hilbert polynomial, \(\ell\left(\frac{\mfq^n}{\mfq^{n+1}}\right)\) is eventually a polynomial, of degree \(\le s - 1 = \delta(\mfq) - 1\).

Fix \(\mfq_0 \sse A\) \(\mfm\)-primary, with \(\delta(\mfq_0) = \delta(A)\). With this, we have two special cases. We will write \(\deg(\ell(\mfq^n/\mfq^{n+1}))\) for the degree of the corresponding Hilbert polynomial.

First of all,
\[\deg(\ell(\mfq_0^n/\mfq_0^{n+1})) \le \delta(A) - 1\]
and
\[\deg(\ell(A/\mfq_0^n)) = \sum_{i=0}^{n-1}\ell(\mfq_0^i/\mfq_0^{i+1}) \le \delta(A)\]
Next,
\[\deg(\ell(\mfm^n/\mfm^{n+1})) = d(\mathrm{G}_\mfm(A)) - 1\]
and
\[\deg(\ell(A/\mfm^n)) = d(\mathrm{G}_\mfm(A))\]
Moreover, there exists \(t \ge 1\) such that
\[\mfm^t \sse \mfq \sse \mfm\]
and so
\[\ell(A/\mfm^n) \le \ell(A/\mfq_0^n) \le \ell(A/\mfm^{tn})\]
Thus, we must have that \(\deg(\ell(A/\mfm^n)) = \deg(\ell(A/\mfq_0^n))\).

\begin{proposition}
    \(\delta(A) \ge d(\mathrm{G}_\mfm(A))\)
\end{proposition}

\begin{proof}
    \begin{align*}
        \delta(A) &= \delta(\mfq_0) \\
        &\ge \deg(\ell(A/\mfq_0^n)) \\
        &= \deg(\ell(A/\mfm^n)) \\
        &= d(\mathrm{G}_\mfm(A))
    \end{align*}
\end{proof}

\begin{proposition}
    If \(x \in \mfm\) is not a zero divisor, then
    \[d\left(\mathrm{G}_{\mfm/xA}(A/xA)\right) \le d(\mathrm{G}_\mfm(A)) - 1\]
\end{proposition}

\begin{proof}
    We know that \((A/xA, \mfm/xA)\) is still a local ring. In this case,
    \[d(\mathrm{G}_\mfm(A)) = \deg(\ell(A/\mfm^n))\]
    and
    \[d(\mathrm{G}_{\mfm/xA}(A/xA)) = \deg(\ell((\mfm^n + xA)/xA))\]
    We want to show that
    \[\deg(\ell(A/(\mfm + xA))) \le \deg(\ell(A/\mfm^n)) - 1\]

\begin{proposition}
    \(\dim(A) \ge \delta(A)\). That is, there exists \(\mfq \idealof A\) \(\mfm\)-primary, generated by \(\dim(A)\) elements.
\end{proposition}

\begin{proof}
    The height of \(\mfm\) is exactly \(\dim(A)\). Thus, for any other prime \(\mfp \in \Spec(A)\), \(\idealht(\mfp) < \dim(A)\). So what we want is to form an ideal \(\mfq = \inner{x_1, \dots, x_d}\), with \(\idealht(\mfq) = \dim(A)\), since then for any minimal prime containing \(\mfq\), we must have that the height of the prime is \(\dim(A)\), and so \(\sqrt{\mfq} = \mfm\), and so \(\mfq\) is \(\mfm\)-primary.
    
    We construct \(\inner{x_1, \dots, x_d}\) inductively, such that if
    \[\mfq_i = \inner{x_1, \dots, x_i}\]
    then
    \[\idealht(\mfq_i) \ge i\]
    For the base case \(i = 0\), we can just use \(\mfq_0 = 0\). For the inductive step, assume \(\mfq_{i-1}\) has \(\idealht(\mfq_i) \ge i - 1\). We claim that there are only finitely many \(\mfp_1, \dots, \mfp_t\) prime ideals, such that \(\mfq_{i-1} \sse \mfp_j\), and \(\idealht(\mfp_j) = i - 1\). If not, since \(\idealht{\mfq_{i-1}} \ge i-1\), each \(\mfp_j\) is a minimal prime of \(\mfq_i\). But in a Noetherian ring, every ideal has finitely many minimal primes.

    Now \(i - 1 < \dim(A) = \idealht(\mfm)\), and so \(\mfm\) is not contained in \(\mfp_j\) for all \(j\), and so \(\mfm\) is not contained in their union, by prime avoidance. So we can take \(x_i \in \mfm\), with \(x_i \notin \mfp_j\) for any \(j\). Define
    \[\mfq_i = \inner{x_1, \dots, x_i}\]
    Then if \(\mfp\) is prime, which contains \(\mfq_i\), then it contains \(\mfq_{i-1}\) and \(x_i\). Hence it cannot be any of the \(\mfp_j\) above. Thus, \(\height(\mfp) \ge i\) as required.
\end{proof}
