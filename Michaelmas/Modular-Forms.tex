\section{Introduction}

\begin{definition}
    Define the upper half plane\index{upper half plane}
    \[\mfH = \left\{\tau \in \C \mid \im(\tau) > 0\right\}\]
    and the positive determinant group
    \[\GL_2(\R)^+ = \left\{g \in \GL_2(\R) \mid \det(g) > 0\right\}\]
    and\index{\(\Gamma(1)\)}
    \[\Gamma(1) = \SL_2(\Z)\]
\end{definition}

\begin{lemma}
    \(\GL_2(\R)^+\) acts transitively on \(\mfH\) via M\"obius transformations.
\end{lemma}

\begin{proof}
    Suppose \(g = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix} \in \GL_2(\R)^+\), and \(\tau \in \mfH\). Then
    \[\Im(g\tau) = \frac{1}{2i}\left(\frac{a\tau + b}{c\tau + d} - \frac{a\overline\tau + b}{c\overline\tau + d}\right) = \frac{1}{2i}\frac{(ad - bc)(\tau - \overline \tau)}{\abs{c\tau + d}^2} = \frac{\deg(g)\Im(\tau)}{\abs{c\tau + d}^2} > 0\]
    For transitivity, for \(x + iy \in \mfH\),
    \[\begin{pmatrix}
        y & x \\ 0 & 1
    \end{pmatrix} \vdot i = x + iy\]
\end{proof}

\begin{definition}
    Let \(g \in \GL_2(\R)^+\), \(\tau \in \mfH\), define \(j(g, \tau) = c\tau + d\), where \(g = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix}\). This is called the \emph{modular cocycle}\index{modular cocycle}. If \(k\) is an integer, \(f : \mfH \to \C\) is any function, then
    \begin{align*}
        f\vert_k[g] : \mfh &\to \C \\
        \tau &\mapsto \det(g)^{k-1}f(g\tau)j(g, \tau)
    \end{align*}
    is called the \emph{weight \(k\) action of \(g\) on \(f\)}\index{weight \(k\) action}.
\end{definition}

\begin{lemma}
    This is a \emph{right} action of \(\GL_2(\R)^+\). That is,
    \[f\vert_k[gh] = \left(f\vert_k[g]\right)\vert_k[h]\]
\end{lemma}

\begin{proof}
    \begin{align*}
        f\vert_k[g]\vert_k[h](\tau) &= \det(h)^{k-1}f\vert_k[g](h\tau)j(h, \tau)^{-k} = \det(h)^{k-1}\det(g)^{k-1}f(gh\tau)j(g, h\tau)^{-k}j(h, \tau)^{-k} \\
        &\stackrel{?}{=} \det(gh)^{k-1}f(gh\tau)j(gh, \tau)^{-k}
    \end{align*}
    Therefore, suffices to show that
    \[j(gh, \tau) = j(g, h\tau)j(h, \tau)\]
    Note that if \(g = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix}\), then \(g \begin{pmatrix}
        \tau \\ 1
    \end{pmatrix} = \begin{pmatrix}
        a\tau + b \\ c\tau + d
    \end{pmatrix} = j(g, \tau)\begin{pmatrix}
        g\tau \\ 1
    \end{pmatrix}\), This means that,
    \[j(gh, \tau)\begin{pmatrix}
        gh\tau \\ 1
    \end{pmatrix} = gh\begin{pmatrix}
        \tau \\ 1
    \end{pmatrix} = g\left(j(h, \tau)\begin{pmatrix}
        h\tau \\ 1
    \end{pmatrix}\right) = j(g, h\tau)j(h, \tau)\begin{pmatrix}
        gh\tau \\ 1
    \end{pmatrix}\]
\end{proof}

\underline{Formulae:}

For \(g, h \in \GL_2(\R)^+\), \(\tau \in \mfh\),
\[\Im(g\tau) = \frac{\det(g)\Im(\tau)}{\abs{j(g, \tau)}^2}\]
and
\[j(gh, \tau) = j(g, h\tau)j(h, \tau)\]
\begin{definition}
    Let \(k \in \Z\), \(\Gamma \le \Gamma(1)\) a finite index subgroup. A \emph{weakly modular function of weight \(k\) and level \(\Gamma\)}\index{weakly modular function of weight \(j\) and level \(\Gamma\)} is a meromorphic function \(f : \mfH \to \C\), which is invariant under the weight \(k\) action of \(\Gamma\). That is, such that for all \(\gamma \in \Gamma\),
    \[f\vert_k[\gamma] = f\] 
\end{definition}
We will define a modular form (when \(\Gamma = \Gamma(1)\)) next time, but they are weakly modular functions, which are holomorphic in \(\mfH\) and at \(\infty\).

In fact, modular forms of fixed weight and level live in finite dimensional \(\C\)-vector spaces \(M_k(\Gamma)\), which are the main objects of study in this course.

\underline{Why do we study modular forms?}

\begin{enumerate}
    \item They are related to the theory of elliptic functions. Let \(E/\C\) be an elliptic curve. Let \(\omega\) be a non-zero holomorphic \(1\)-form. Then there exists a unique lattice \(\Lambda \le \C\), and an isomorphism of Riemann surfaces, \(\phi : \C/\Lambda \to E\), such that \(\phi^*(\omega) = \dd z\). We can show that \(E\) is isomorphic to the elliptic curve \(y^2 = 4x^3 - 60 G_4(\Lambda)x - 140 G_6(\Lambda)\), where for \(k \in \Z\),
    \[G_k(\Lambda) = \sum_{\lambda \in \Lambda \setminus 0}\lambda^{-k}\]
    which converges for \(\lambda > 2\). If \(\tau \in \mfH\), then we have an associated lattice \(\Lambda_\tau = \Z \oplus \Z_\tau\), and the function
    \[G_k(\tau) = G_k(\Lambda_\tau)\]
    is a modular form of weight \(k\) and level \(\Gamma(1)\). This is called the \emph{Eisenstein series}. Moreover, \(\mfH/\SL_2(\Z)\) can be identified (as a set) with the set of isomorphism classes of elliptic curves over \(\C\).
    \item Modular forms \(f\) have Fourier expansions
    \[\sum_{n \in \Z}a_n q^n\]
    where \(a_n \in \C\), and often serve as generating functions for arithmetically interesting sequences \(a_n\). One example is
    \[\vartheta(q) = \sum_n e^{\pi i n^2\tau}\]
    If \(k \in 2\N\), then \(\vartheta^{2k}\) is a modular form, and on the other hand,
    \[\vartheta^{2k} = \sum_{n \in \Z}r_k(n)e^{\pi i n\tau}\]
    where \(r_k(n)\) is the number of ways to write \(n\) as a sum of \(k\) squares. By expressing \(\vartheta^{2k}\) in terms of ther modular forms, we can prove formulae such as
    \[r_4(n) = 8\sum_{d \mid n, 4 \not\mid d}d\]
    \item The Riemann \(\zeta\) function is an important object in number theory. Properties include
    \begin{enumerate}
        \item The Euler product
        \[\zeta(s) = \prod_p(1 - p^{-s})^{-1}\]
        \item A meromorphic extension to \(\C\),
        \item A functional equation relating \(\zeta(s)\) and \(\zeta(1-s)\).
    \end{enumerate}
    A \emph{Dirichlet \(L\)-series} is a function of the form
    \[\sum_{n \ge 1}a_n n^{-s}\]
    which has similar properties is called an \(L\)-function. Modular forms can be used to construct interesting examples of \(L\)-functions. Take \(M_k(\Gamma)\) and decompose them under the action of Hecke operators. In particular, if \(\Gamma = \Gamma(1)\), we get a decomposition into lines, called Hecke eigenforms.
    \item Connections to the Langland programme, which predicts (among other things) a relation between modular forms and other objects in arithmetic geometry. A special case of this is the \emph{Modularity conjecture}, which says that there is a bijection between elliptic curves over \(\Q\) (up to isogeny) and the set of Hecke eigenforms of weight \(2\). This implies Fermat's last theorem. The bijection is formulated in the language of Hecke operators and \(L\)-functions.
    
    \underline{Homework:} Handout on Moodle called ``Reminder of Complex Analysis''.
\end{enumerate}

\section{Modular forms on \(\Gamma(1)\)}

Recall a \emph{meromorphic function} on \(U \sse \C\) is a closed subset \(A \sse U\) and a holomorphic function \(f : U \setminus A \to \C\), such that for every \(a \in A\), there exists \(\delta > 0\), such that
\[D_*(a, \delta) \sse U \setminus A\]
and there exists an integer \(n \ge \Z\) such that
\[(z-a)^nf(z)\]
defines a holomorphic function on \(D(a, \delta)\). Such an \(a \in A\) is called a \emph{pole} of \(f\). \(f\) then has a \emph{Laurent expansion}
\[\sum_{m \in \Z}a_m(z-a)^m\]
which is absolutely convergent on \(D(a, \delta)\).

\begin{lemma}
    Let \(f\) be a weakly modular function of weight \(k\) and level \(\Gamma(1)\). Then there exists a unique meromorphic function \(\tilde f\) on \(D_*(0, 1)\) such that
    \[f(\tau) \tilde f(e^{2\pi i \tau})\]
\end{lemma}

\begin{proof}
    By assumption, \(f\) is meromorphic on \(\mfH\), let \(\gamma = \begin{pmatrix}
        1 & 1 \\ 0 & 1
    \end{pmatrix} \in \Gamma(1)\). Then
    \[f\vert_k[\gamma](\tau) = f(\gamma\tau) = f(\tau)\]
    as \(f\) is invariant under the weight \(k\) action of \(\gamma\). But
    \[f(\gamma\tau) = f(\tau + 1)\]
    \underline{Existence:} Locally, let \(a \in D_*(0, 1), \delta > 0\) be such that \(D(a, \delta) \sse D_*(0, 1)\). Define \(\tilde f\) in the disc by
    \[\tilde f(q) = f\left(\frac{1}{2\pi i}\log(q)\right)\]
    where \(\log\) is any branch of logarithm defined in \(D(a, \delta)\). This is independent of the choice of branch of logarithm, since any two branches differ by \(2\pi i\), and \(f(\tau) = f(\tau + 1)\). Therefore, this defines \(\tilde f\) on \(D_*(0, 1)\).

    \underline{Uniqueness:} Since the map \(\tau \mapsto e^{2\pi i \tau}\) is surjective, \(\tilde f\) is unique. 
\end{proof}

Suppose \(\tilde f\) extends to a meromorphic function on \(D(0, 1)\), then there exists \(\delta > 0\) such that \(\tilde f\) has a Laurent expansion
\[\tilde f(q) = \sum_{n \in \Z}a_nq^n\]
which is absolutely convergent on \(D_*(0, \delta)\). In particular, in the region where \(\Im(\tau) > \frac{1}{2\pi}\log(\delta)\), we have
\[f(\tau) = \sum_n a_n q^n\]
where \(q = e^{2\pi i \tau}\). We call this the \emph{\(q\)-expansion of a weakly modular function \(f\)}.

\begin{definition}
    Let \(f\) be a weakly modular function of weight \(k\) and level \(\Gamma(1)\). We say that \(f\) is \emph{meromorphic at \(\infty\)}\index{meromorphic at \(\infty\)} if \(\tilde f\) extends to a meromorphic function on \(D(0, 1)\).

    We say that \(f\) is \emph{holomorphic at \(\infty\)}\index{holomorphic at \(\infty\)} if \(\tilde f\) extends, and is holomorphic at \(q = 0\). In this case, we define
    \[f(\infty) = \tilde f(0) = \lim_{\Im(\tau) \to \infty}f(\tau)\]

    We say that \(f\) \emph{vanishes at \(\infty\)}\index{vanishes at \(\infty\)} if \(f\) is holomorphic at \(\infty\), and \(f(\infty) = 0\).
\end{definition}

\begin{definition}
    [modular form] A \emph{modular function}\index{modular function!of weight \(k\) and level \(\Gamma(1)\)} (of weight \(k\) and level \(\Gamma(1)\)) is a weakly modular function, which is meromorphic at \(\infty\). A \emph{modular form}\index{modular form!of weight \(k\) and level \(\Gamma(1)\)} is a weakly modular function which is holomorphic in \(\mfH\), and holomorphic at \(\infty\).

    A \emph{cuspidal modular form}\index{cuspoidal modular form} is a modular form which vanishes at \(\infty\).
\end{definition}

\begin{remark}
    We let
    \[M_k(\Gamma(1))\]
    be the set of \emph{modular forms of weight \(k\) and level \(\Gamma(1)\)}\index{modular form!of weight \(k\) and level \(\Gamma(1)\)!\(M_k(\Gamma(1))\)}, and
    \[S_k(\Gamma(1)) \sse M_k(\Gamma(1))\]
    be the set of \emph{cuspidal modular forms of weight \(k\) and level \(\Gamma(1)\)}\index{cuspidal modular form!\(S_k(\Gamma(1))\)}. These are \(\C\)-vector spaces. If \(k\) is odd, then they are zero. To see this, consider the matrix
    \[\gamma = \begin{pmatrix}
        -1 & 0 \\ 0 & -1
    \end{pmatrix} \in \Gamma(1)\]
    and the weight \(k\)-action is
    \[f\vert_k[\gamma](\tau) = (-1)^kf(\tau) = f(\tau)\]
    We will now consider even weights only.
\end{remark}

If \(k \in \Z\) is even, let
\[G_k(\tau) = \sum_{\lambda \in \Lambda_\tau \setminus 0}\lambda^{-k} = \sum_{(m, n) \in \Z^2 \setminus 0} (m \tau + n)^{-k}\]
where \(\Lambda_\tau = \Z_\tau \oplus \Z \le \C\).

If \(\gamma \in \Gamma(1)\), then we formally have
\[G_k\vert_k[\gamma](\tau) = G_k(\gamma\tau)j(\gamma, \tau)^{-k} = \sum_{\lambda \in \Lambda_{\gamma\tau}\setminus 0}\lambda^{-k}j(\gamma, \tau)^{-k}\]
But if \(\gamma = \begin{pmatrix}
    a & b \\ c & d
\end{pmatrix}\), then
\[\Lambda_{\gamma\tau} = \Z\frac{a\tau + b}{c\tau + d} \oplus \Z = (c\tau + d)^{-1}\left(\Z(a \tau + b) \oplus \Z(c \tau + d)\right) = (c\tau + d)^{-1}\Lambda_\tau\]
Therefore,
\[G_k\vert_k[\gamma](\tau) = \sum_{\lambda \in (c\tau + d)^{-1}\Lambda_\tau \setminus 0}\lambda^{-k}(c\tau + d)^{-k} = \sum_{\lambda \in \Lambda_\tau \setminus 0}\lambda^{-k} = G_k(\tau)\]
This is justified only when the series defining \(G_k(\tau)\) converges absolutely.

\begin{proposition}
    Let \(k > 2\) be an even integer. Then \(G_k(\tau)\) converges absolutely, and defines a modular form of weight \(k\) and level \(\Gamma(1)\), with
    \[G_k(\infty) = 2\zeta(k)\]
    where \(\zeta\) is the Riemann \(\zeta\)-function. We call \(G_k\) the \emph{weight \(k\) Eisenstein series}\index{weight \(k\) Eisenstein series, \(G_k\)}.
\end{proposition}

\begin{remark}
    We will see later that \(M_2(\Gamma(1)) = 0\), so this is optimal.
\end{remark}

\begin{proof}
    We want to show absolute and local uniform convergence in \(\mfH\), since this shows that \(G_k\) is holomorphic.

    Let \(A \ge 2\), and define
    \[\Omega_A = \left\{\tau \in \mfH \ \bigg\vert\ \Im(\tau) \ge \frac{1}{A} \text{ and }\Re(\tau) \in [-A, A]\right\}\]

    We will show uniform convergence in \(\Omega_A\). If \(\tau \in \Omega_A\), \(x \in \R\), then
    \[\abs{\tau + x} \ge \begin{cases}
        \frac{1}{A} & \abs{x} \le 2A \\
        \frac{\abs{x}}{2} & \abs{x} \ge 2A
    \end{cases}\]
    Therefore, 
    \[\tau + x \ge \max\left\{\frac{1}{A}, \frac{\abs{x}}{2A^2}\right\} \ge \max\left\{\frac{1}{A^2}, \frac{\abs{x}}{2A^2}\right\} \ge \frac{1}{2A^2}\max\{1, \abs{x}\}\]

    If \((m, n) \in \Z^2\), then we get that
    \[\abs{m \tau + n}\ge \frac{1}{2A^2}\max\{\abs{m}, \abs{n}\}\]
    If \(\tau \in \Omega_A\), then

    % \begin{align*}
        
    %     \sum_{(m, n) \in \Z^2\setminus 0}\abs{m\tau + n}^{-k} \le \left(\frac{1}{2A^2}\right)^{-k}\sum_{(m, n) \in \Z^2 \setminus 0}\max\{\abs{m}, \abs{n}\}^{-k}
    % %  \\
    % %     &= (2A^2)^{-k}\sum_{d \in \N}d^{-k} \cdot \abs{\left\{(m, n) \in \Z^2 \mid \max\{\abs{m}, \abs{n}\} = d\right\}} \\
    % %     &< \infty
    % \end{align*}
    \begin{align*}
        \sum_{(m, n) \in \Z^2 \setminus 0}\abs{m\tau + n}^{-k} &\le (2A^2)^k \sum_{(m, n)\in \Z^2 \setminus 0} \max\{\abs{m}, \abs{n}\}^{-k} \\
        &= (2A^2)^k\sum_d d^{-k} \cdot \#\{(m, n) \in \Z^2 \setminus 0 \mid \max\{\abs{m}, \abs{n}\} = d\} \\
        &= (2A^2)^{-k} \sum_d d^{1-k} \\
        &= 8(2A^2)^k\zeta(k-1) < \infty
    \end{align*}

    as \(k > 2\). This shows uniform convergence in \(\Omega_A\) by the Weierstrass \(M\)-test.

    We now know that \(G_k\) is holomorphic in \(\mfH\), and invariant under the weight \(k\) action of \(\Gamma(1)\). It remains to show to show \(G_k\) is holomorphic at \(\infty\), with \(G_k(\infty) = 2\zeta(k)\). It suffices to show that
    \[\lim_{\Im(\tau) \to \infty}G_k(\tau) = 2\zeta(k)\]
    By uniform convergence, we can exchange the limit and sum. But
    \[\lim_{\Im(\tau) \to \infty}(m \tau + n)^{-k} = \begin{cases}
        0 & m \ne 0 \\
        n^{-k} & m = 0
    \end{cases}\]
\end{proof}

If we consider the \emph{\(q\)-expansion}\index{q-expansion@\(q\)-expansion}
\[f(\tau) = \sum_{n \ge 0}a_n q^n\]
where \(q = e^{2\pi i\tau}\), then we have that
\[f(\infty) = \tilde f(0) = a_0\]

We define
\[E_k(\tau) = \frac{G_k(\tau)}{2\zeta(k)} = 1 + \sum_{n \ge 1}a_n q^n\]\index{Ek@\(E_k\)}
and we will see that \(a_n \in \Q\) for all \(n \ge 1\). We can construct more modular forms from these: if
\[f \in M_k(\Gamma(1)) \qqtext{and}g \in M_\ell(\Gamma(1))\]
then
\[fg \in M_{k+\ell}(\Gamma(1))\]
\underline{Exercise:} Check. For \(fg\) holomorphic at \(\infty\), we can use that the \(q\)-expansions multiply.

For example,
\[E_4^3, E_6^2 \in M_{12}(\Gamma(1))\]
and in fact,
\[\Delta = E_4^2 - E_6^2 \in S_{12}(\Gamma(1))\]
and \(\Delta\) is called the \emph{Ramanujan \(\Delta\) function}\index{Ramanujan \(\Delta\)-function}.

Next, we want to show \(M_k(\Gamma(1))\) is finite dimensional. We will study the space \(\Gamma(1)\backslash \mfH\). To do this, we will introduce a \emph{fundamental set}\index{fundamental set} \(\mcF' \sse \mfH\) for the \(\Gamma(1)\)-action, which contains exactly one element from each \(\Gamma(1)\)-orbit.
\begin{align*}
    \mcF &= \left\{\tau \in \mcH \ \big\vert\  \abs{\Re(\tau)} \le \frac{1}{2}, \abs{\tau} \ge 1\right\} \\
    \mcF' &= \left\{\tau \in \mcF \ \big\vert\  \Re(\tau) < \frac{1}{2} \text{ and if }\abs{\tau} = 1\text{ then } -\frac{1}{2} \le \Re(\tau) \le 0\right\}
\end{align*}

Define
\[T = \begin{pmatrix}
    1 & 1 \\ 0 & 1
\end{pmatrix} \qquad S = \begin{pmatrix}
    0 & 1 \\ -1 & 0
\end{pmatrix}\]

Then every element of \(\mcF\) is conjugate under \(S, T\) to an element of \(\mcF'\). In particular, \(T(\tau) = \tau + 1\), \(S(\tau) = -\overline{\tau}\).

\begin{proposition}
    Let \(G = \Gamma(1)/\{\pm I\}\). Then
    \begin{enumerate}[(i)]
        \item For every \(\tau \in \mfH\), \(\tau\) is \(\Gamma(1)\)-conjugate to an element of \(\mcF'\),
        \item if \(\tau, \tau' \in \mcF'\) are \(\Gamma(1)\)-conjugate, then \(\tau = \tau'\), 
        \item if \(\tau \in \mcF'\), then \(\Stab_G(\tau) = 1\), except for
        \[\Stab_G(i) = \inner{S} \qqtext{and}\Stab_G(\rho) = \inner{ST}\]
        where \(\rho = e^{2\pi i/3}\).
        \item \(\Gamma(1)\) is generated by \(S, T\).
    \end{enumerate}
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/mf3 (2).jpg}
    \end{center}
\end{proposition}

\begin{proof}
    Define \(H = \inner{S, T} \le G\).
    \begin{claim}
        Every \(\tau \in \mfH\) is \(H\)-conjugate to an element of \(\mcF'\).
    \end{claim}
    \begin{proof}
        [Proof of claim] By an easy observation, and as \(S, T \in \mfH\), it suffices to show every \(\tau \in \mfH\) is \(H\)-conjugate to an element of \(\mcF\).

        Let \(\tau \in \mfH\), and recall that if \(\gamma = \begin{pmatrix}
            a & b \\ c & d
        \end{pmatrix}\), then
        \[\Im(\gamma\tau) = \frac{\Im(\tau)}{\abs{c\tau + d}^2}\]
        In particular, for all \(R > 0\), the intersection \(H\tau \cap \{\Im(\tau') > R\}\) is finite, since \(\Im(\tau) > R\) if and only if
        \[\abs{c \tau + d}^2 < \frac{\Im(\tau)}{R}\]
        But \(\Lambda_\tau = \Z \oplus \Z\tau\) is a lattice, and so its intersection with any compact subset is finite.

        In particular, there exists \(h \in H\) such that \(\Im(h\tau) \ge \Im(h'\tau)\) for all \(h' \in H\). That is, the maximum value of the imaginary part is attained. By replacing \(\tau\) with \(h\tau\), we can assume
        \[\Im(\tau) \ge \Im(h\tau)\]
        for all \(h \in H\). Moreover, \(\Im(T\tau) = \Im(\tau)\), and so we can assume that
        \[-\frac{1}{2} \le \Re(\tau) \le \frac{1}{2}\]
        Setting \(h = S\), we have that
        \[\Im(\tau)\ge \Im(S\tau) = \frac{\Im(\tau)}{\abs{\tau}^2}\]
        Hence \(\abs{\tau} \ge 1\).
    \end{proof}
    Note that this also proves (i). Now given \(\tau, \tau' \in \mcF'\), and suppose \(\gamma\tau = \tau'\) for some
    \[\gamma = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix} \in \Gamma(1)\]
    We want to show that \(\gamma = \pm I\) or \(\tau = i\) or \(\tau = \rho\). Without loss of generality, \(\Im(\tau') = \Im(\gamma\tau) \ge \Im(\tau)\). This means that
    \[\Im(\gamma(\tau)) = \frac{\Im(\tau)}{\abs{c\tau + d}^2} \ge \Im(\tau)\]
    and so \(\abs{c\tau + d} \le 1\).

    If \(\tau \in \mcF'\) then \(\Im(\tau) \ge \sqrt{3}{2}\), with equality if and only if \(\tau = \rho\). Hence
    \[\abs{c\tau + d} \ge \abs{c}\im(\tau) \ge \abs{c}\frac{\sqrt{3}}{2}\]
    Hence \(\abs{c} = 0\) or \(1\). Therefore, \(c = 0, 1\) or \(-1\).

    If \(c = 0\), then
    \[\gamma = \begin{pmatrix}
        a & b \\
        0 & d
    \end{pmatrix}\]
    In this case, \(ad = 1\), and so \(\gamma = \pm T^n\) for some \(T\). But as the real part of \(\tau\) and \(\tau'\) lie in \(\Ico{-1/2, 1/2}\), we must have \(n=0\), and so \(\gamma = \pm I\).

    If \(c = 1\), then
    \[\gamma = \begin{pmatrix}
        a & b \\
        1 & d
    \end{pmatrix}\]
    and \(\abs{\tau + d} \le 1\). The only circles centered at integers, and with radius \(1\) which intersects \(\mcF'\) are centred at \(-d=-1\) and \(d=0\). 
    
    \begin{center}
        \includegraphics[scale=0.1]{diagrams/mf3 (3).jpg}
    \end{center}
    
    The only possibilities are \(d = 0\), and so \(\abs{\tau} = 1\), or \(d = 1\), and so \(\tau = \rho\).

    If \(c = 1, d= 0\), \(\abs{\tau} = 1\), then
    \[\gamma = \begin{pmatrix}
        a & -1 \\
        1 & 0
    \end{pmatrix}\]
    and \(\gamma(\tau) = a - \tau^{-1} = a - \overline{\tau}\), and
    \[\Re(\gamma\tau) = a - \Re(\tau) \in \Re(\mcF' \cap \{\abs{\tau} = 1\}) = \left[-1/2, 0\right]\]
    But we also have that \(\Re(\gamma(\tau)) \in a - [-1/2, 0] = a + [0, 1/2]\). But the intersection between
    \[[-1/2, 0] \cap (a + [0, 1/2])\]
    is non-empty only if \(a = 0\), and the unique point in the intersection is \(0\), or \(a = - 1\), and the unique point of intersection is \(\Re(\tau) = \Re(\gamma\tau) = -1/2\). The first case is \(\gamma = i\) and the second is \(\gamma = \rho\).

    If \(a = 0\), then
    \[\gamma = \begin{pmatrix}
        0 & -1 \\
        1 & 0
    \end{pmatrix} = -S\]
    and if \(a = -1\), then
    \[\gamma = \begin{pmatrix}
        -1 & -1 \\
        1 & 0
    \end{pmatrix}\]

    Computing,
    \[ST = \begin{pmatrix}
        0 & 1 \\
        -1 & -1
    \end{pmatrix}\]
    and
    \[(ST)^2 = \begin{pmatrix}
        -1 & -1 \\
        0 & 1
    \end{pmatrix}\]
    and
    \[(ST)^3 = \begin{pmatrix}
        1 & 0\\
        0 & 1
    \end{pmatrix} = I\]

    Now if \(c = 1, d = 1, \tau = \rho\), then
    \[\gamma = \begin{pmatrix}
        a & b \\
        1 & 1
    \end{pmatrix}\]
    Then
    \[\gamma\rho = \frac{a\rho+ b}{\rho + 1} = \rho\]
    We have that \(\rho^2 + \rho + 1 = 0\). In particular, \(\rho + 1 = -\rho^2\). Hence
    \[a\rho + b = \rho^2 + \rho = -1\]
    But since \(a, b \in \Z\), and \(\rho \in \C\), which is linearly independent of \(\R\), and so \(a\rho + b = -1\) implies \(a = 0\) and \(b = -1\). Therefore,
    \[\gamma =\begin{pmatrix}
        0 & -1 \\
        1 & 1
    \end{pmatrix} = -ST\]
    The case \(c = -1\) can be reduced to the case \(c = 1\) by replacing \(\gamma = -\overline{\gamma}\).

    We have now shown (ii) and (iii). That is, \(\Gamma(1)\) is generated by \(S, T\). As \(S^2 = -I\), it suffices to show \(G = H\). Choose \(\tau \in \Int(\mcF)\), then \(\Stab_G(\tau) = \{I\}\). Let \(g \in G\), there exists \(h \in H\) such that \(hg\tau \in \mcF'\). But then by (ii) and (iii), we must therefore have \(hg\tau = \tau\). This then implies \(hg \in \Stab_G(\tau) = 1\), and so \(g = h^{-1} \in H\).
\end{proof}
