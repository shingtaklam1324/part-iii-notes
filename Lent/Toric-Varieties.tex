\section{What is a toric variety?}

\begin{definition}
    [affine variety] An \emph{affine variety}\index{affine variety} is \(V = \Spec(R)\), where \(R\) is a finitely generated integral\footnote{\(R\) is an integral domain.} \(\C\)-algebra.
\end{definition}

\underline{Fact:} Given such an \(R\), we can choose generators and relations, and write
\[R = \frac{\C[x_1, \dots, x_n]}{I}\]
where \(I\) is a prime ideal. This gives an affine embedding. i.e. a closed embedding
\[V = \van(I) = \A^n\]
Note that the affine embedding is not unique.

\begin{definition}
    [(complex) variety] A \emph{(complex) variety}\index{variety} is a scheme \(X\) which is integral (reduced and irreducible), separated, and of finite type.
\end{definition}

We can avoid schemes, and glue varieties out of affine varieties. We need 
\begin{enumerate}
    \item a finite collection \(\{V_\alpha\}_{\alpha \in A}\) of affine varieties, say \(V_\alpha = \Spec(R_\alpha)\), 
    \item for \(\alpha, \beta \in A\), we need Zariski opens \(V_{\alpha\beta} \sse V_\alpha\),
    \item algebraic transition maps \(g_{\alpha\beta} : V_{\alpha\beta} \to V_{\beta\alpha}\), satisfying the usual cocycle conditions,
    \item check that the resulting
    \[X = \frac{\bigsqcup_\alpha V_\alpha}{\sim}\]
    is irreducible and separated (Hausdorff).
\end{enumerate}

\begin{definition}
    [torus] An \emph{(algebraic complex) torus of dimension \(n\)}\index{torus} is 
    \[\T^n = (\C^*)^n = \Spec(\C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]) = (\A^1 \setminus \{0\})^n\]
\end{definition}

Note that this is not the same as the usual torus \((S^1)^n\). In particular, we have an affine embedding
\[\T^n = \Spec\left(\frac{\C[z_0, \dots, z_n]}{\inner{z_0\cdots z_n - 1}}\right) = \van(z_0\cdots z_n - 1) \sse \A^{n+1}\]
\underline{Exercise:} Check that we have an isomorphism of rings
\[\C[z_1^{\pm 1}, \dots, z_n^{\pm 1}] \cong \frac{\C[z_0, \dots, z_n]}{\inner{z_0\cdots z_n - 1}}\]

The torus is also a \emph{commutative group scheme}\index{commutative group scheme}, that is, we have algebraic morphisms
\begin{align*}
    e : \bullet &\to \T^n \\
    m : \T^n \times \T^n &\to \T^n \\
    i : \T^n &\to \T^n
\end{align*}
making \(\T^n\) into an abelian group with identity \(e\), multiplication \(m\) and inverse \(i\). On the level of coordinate rings, we have the \emph{identity section},
\begin{align*}
    e^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C \\
    t_i^{\pm 1} &\mapsto 1
\end{align*}
The multiplication map is given by pointwise operations, i.e.
\[m((x_1, \dots, x_n), (y_1, \dots, y_n)) = (x_1y_1, \dots, x_ny_n)\]
With this, we have \emph{comultiplication}
\begin{align*}
    m^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}] \otimes \C[y_1^{\pm 1}, \dots, y_n^{\pm 1}] \\
    t_i &\mapsto x_i \otimes y_i
\end{align*}
and the inverse map is given by
\begin{align*}
    i^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] \\
    t_i &\mapsto t_i^{-1}
\end{align*}

\begin{definition}
    [toric variety] A \emph{toric variety}\index{toric variety} is a variety \(X\) with a dense open \(\T^n \sse X\), and an action (i.e. algebraic map)
    \[a : \T^n \times X \to X\]
    which extends the multiplication on \(X\).
\end{definition}

\begin{example}
    \(\T^n\) is itself a toric variety.
\end{example}
Cohomology
\begin{example}
    \(\A^n = \Spec(\C[x_1, \dots, x_n])\), with the torus
    \[\Spec(\C[x_1^{\pm 1}, \dots, x_n^{\pm 1}])\]
    with
    \[a((t_1, \dots, t_n), (x_1, \dots, x_n)) = (t_1x_1, \dots, t_nx_n)\]
\end{example}

\begin{example}
    Projective space
    \[\P^n = \Proj(\C[z_0, \dots, z_n]) = \{(z_0 : \dots : z_n) \mid (z_0, \dots, z_n) \ne (0, \dots, 0)\}\]
    is also a toric variety, with torus
    \[\T^n = \{(z_0 : \dots : z_n) \mid z_0\cdots z_n \ne 0\}\]
    with action given coordinate-wise again, i.e.
    \[a((t_0: \dots: t_n), (z_0:\dots:z_n)) = (t_0z_0:\dots:t_nz_n)\]
    in non-homogeneous coordinates (i.e. in the affine \(z_0 \ne 0\)) of the torus, we can define the action by
    \[(t_1, \dots, t_n) \cdot (z_0:\dots : z_n) = (z_0 : t_1z_1: \dots : t_nz_n)\]
    We can check that \(\T^n\) is contained in all of the standard opens \(U_i = D(z_i)\), and if we choose the usual affine coordinates, then \(\T^n\) embeds into affine space as above.
\end{example}

Toric varieties are determined by the dimension \(n\) of the torus, and the \emph{toric boundary}\index{toric boundary} \(X \setminus \T^n\). For example, \(\P^2 \setminus \T^2\) has three components,
\begin{enumerate}
    \item \(V_0 = \van(z_0)\),
    \item \(V_1 = \van(z_1)\),
    \item \(V_2 = \van(z_2)\)
\end{enumerate}
The \(V_i\) intersect pairwise, but not all three at the same time. Each \(V_i\) is invariant under the action of the torus, so \(\P^2\) can be represented as

\section{Characters, cocharacters and lattices}

Recall from the last lecture that \(\T^n = (\C^*)^n\) is an algebraic group.

\begin{definition}
    [character]
    A \emph{character} \(\chi\) of \(\T^n\) is a morphism \(\chi : \T^n \to \C^*\) of algebraic groups. That is, \(\chi : \T^n \to \C^*\) is a morphisms of schemes, such that
    \[\chi(e) = 1 \qqtext{and}(\chi^\sharp \otimes \chi^\sharp) \circ m^\sharp = m^\sharp \circ \chi^\sharp\]
\end{definition}

We will write \(\Hom_{\algp}(\cdot, \cdot)\) for the set of morphisms of algebraic groups.

\begin{theorem}
    The characters of \(\T^n\) are
    \[M = \Hom_\algp(\T^n, \C^*) = \Z^n\]
\end{theorem}

\begin{proof}
    First of all, note that
    \[\Hom_\algp(\T^n, \C^*) = \left(\Hom(\C^*, \C^*)\right)^n\]
    Thus, without loss of generality \(n = 1\). Now we need a \(\C\)-algebra map
    \[\chi^\sharp : \C[x^{\pm 1}] \to \C[y^{\pm 1}]\]
    This is determined by \(\chi^\sharp(x) = p(y, y^{-1})\). Note that \(x\) is a unit in \(\C[x^{\pm 1}]\), and so \(\chi^\sharp(x)\) is a unit. Thus, \(p(y, y^{-1}) = ay^k\), for some \(a \in \C, k \in \Z\).

    Next, we need that \(e^\sharp \circ \chi^\sharp = \chi^\sharp \circ e^\sharp\). That is,
    \[a = e^\sharp(\chi^\sharp(x)) = \chi^\sharp(e^\sharp(x)) = \chi^\sharp(1) = 1\]
    Hence \(\chi^\sharp(x) = x^k\), for some \(k \in \Z\). Note that \(\chi^\sharp(x \otimes x) = \chi^\sharp(x) \otimes \chi^\sharp(x)\) automatically, so we are done.
\end{proof}

The characters of \(\T^n\) are of the form
\[\chi(t_1, \dots, t_n) = t_1^{a_1}\cdots t_n^{a_n}\]
for some \((a_1, \dots, a_n) \in \Z^n\). Expressions of the form
\[t_1^{a_1}\cdots t_n^{a_n}\]
are called \emph{Laurent monomials}. We will write
\[\chi^{\mathbf a}(\mathbf t) = \mathbf t^{\mathbf a} = t_1^{a_1}\cdots t_n^{a_n}\]
for this character. The correspondence is then
\begin{align*}
    \Hom_\algp(\T^n, \C^*) &\to \Z^n \\
    \chi^{\mathbf a} \mapsto \mathbf a
\end{align*}

The characters form a group. Given characters \(\chi, \chi' : \T^n \to \C^*\), 
\[\chi\cdot\chi'(\mathbf t) = \chi(\bf t)\cdot \chi'(\bf t)\]
Thus, we have taht
\[\chi^{\mathbf a}\cdot \chi^{\mathbf a'} = \chi^{\mathbf a + \mathbf a'}\]
Hence the identification \(\Hom_\algp(\T^n, \C^*) \cong \Z^n\) is also a group isomorphism.

\begin{definition}
    [cochacter]
    A \emph{cocharacter} of \(\T^n\) is
    \[\lambda \in \Hom_\algp(\C^*, \T^n)\]
    We also call them \emph{one-parameter subgroup}.
\end{definition}

We can check that
\[N = \Hom_\algp(\C^*, \T^n) \cong \Z^n\]
by showing that any \(\lambda \in N\) is of the form
\[\lambda(t) = (t^{u_1}, \dots, t^{u_n})\]
for \(\mathbf u = (u_1, \dots, u_n) \in \Z\). So we can write the above cocharacter as \(\lambda_{\mathbf u}\) as before.  \(N\) is a group, and the identification is a group isomorphism. That is,
\[\lambda_{\mathbf u} \cdot \lambda_{\mathbf u'} = \lambda_{\mathbf u + \mathbf u'}\]

\begin{example}
    For \(n = 2\), we can identify
    \[\T^2 = \left\{\begin{pmatrix}
        t_1 & 0 \\ 0 & t_2
    \end{pmatrix}\ \bigg\vert\ t_1t_2 \ne 0\right\}\]
    Under the identification \(M \cong N \cong \Z^2\),
    \[\chi^{(a, b)}\begin{pmatrix}
        t_1 & 0 \\
        0 & t_2
    \end{pmatrix} = t_1^a\cdot t_2^b\]
    and
    \[\lambda_{(u, v)}(t) = \begin{pmatrix}
        t^u & 0 \\
        0 & t^v
    \end{pmatrix}\]
\end{example}

\begin{definition}
    [lattice]
    A \emph{lattice} is a free abelian group of finite rank.
\end{definition}

Note that with this definition, any lattice is isomorphic to \(\Z^n\) when we choose a basis.

\begin{example}
    \(\Z, 2\Z, \Z \oplus \Z, \Z \oplus 3\Z, \dots\) are all lattices.
\end{example}

There is a \emph{perfect pairing}\footnote{That is, it induces isomorphism \(\Hom_\Z(M, \Z) = M^\vee \cong N\) and vice versa.}, given by
\begin{align*}
    \inner{\cdot, \cdot} : M \times N &\to \Z \\
    \inner{\mathbf a, \mathbf u} &= \mathbf a \vdot \mathbf u = a_1 u_1 + \cdots + a_nu_n
\end{align*}
In a coordinate-free manner,
\begin{align*}
    M \times N = \Hom(\C^*, \T^n) \times \Hom(\T^n, \C^*) &\to \Hom(\C^*, \C^*) \\
    (\lambda, \chi) &\mapsto \chi \circ \lambda
\end{align*}
Using the identification \(\Hom(\C^*, \C^*) \cong \Z\) as before gives the pairing.

We can describe a torus \(\T_N\) starting from a lattice \(N\), through
\[\T_N = N \otimes_\Z \C^*\]
with this,
\[\Hom(\C^*, \T_N) = N\]
Formally,
\[\T_N = \Spec(\Z[N] \otimes_\Z \C[x, x^{-1}])\]

With this, give a torus \(\T\), we get two dual lattices \(N, M\). Conversely, given a lattice \(N\), we get a torus \(\T_N\) and a dual lattice \(M\). Choosing a basis on one gives a basis on the other two.

\section{Cocharacters and limit points - Part I}

Back to toric varieties. Let \(X\) be a toric variety, with torus \(\T\). We can think of \(X\) as a ``partial compactification'' of \(T\). That is, we can construct \(X\) by adding limit points\footnote{In the Euclidean topology.} to \(\T\).

Given a one-parameter subgroup \(\mathbf u \in N\), we have
\[\lambda_{\mathbf u} : \C^* \to \T \sse X\]
We can ask when does
\[\lim_{t \to 0}\lambda_{\mathbf u}(t)\]
exist in \(X\). The idea will be that we can determine \(X\) by saying which \(\lambda_{\mathbf u}\) have a limit. See later.

\begin{example}
    Consider \(\T^2 \sse \P^2_{z_0:z_1:z_2}\). The embedding is given by \((t_1, t_2) \mapsto (1 : t_1 : t_2)\). Choose \(\mathbf u \in \Z^2\), so \(\mathbf u = (u_1, u_2)\).
    \[\lim_{t \to 0}\lambda_{\mathbf u}(t) = \lim_{t \to 0}(1 : t^{u_1} : t^{u_2})\]

    All these limit points exist, since \(\P^2\) is complete\footnote{Or in algebraic geometry, it satisfies the valuative criterion for properness for some choice of DVR.}. In particular, if \(u_1, u_2 > 0\), the limit is \((1 : 0 : 0)\). If \(u_1 < 0, u_2 > u_1\), the limit is \((0 : 1: 0)\). Similarly if \(u_2 < 0, u_1 > u_2\) then the limit is \((0 : 0 : 1)\).

    The upshot is that we have partitioned \(N_\R = N \otimes \R = \R^2\) into cones, so that if \(\mathbf u_1, \mathbf u_2\) are in the same cone if and only if they have the same limit. In fact, we can construct \(\P^2\) from this.
\end{example}

and the set \(\{u_i\}_{i=1}^r\) is the set of generators of the cone.

\begin{definition}

    The \emph{dimension} of a cone \(\sigma\) is \(\dim(\sigma) = \dim(V)\), where \(V = \Span\{\sigma\} \sse N_\R\). A cone is \emph{top-dimensional} if \(V = N_\R\).
\end{definition}

\begin{definition}
    [strongly convex] \(\sigma\) is \emph{strongly convex} if it does not contain any lines through the origin. Equivalently, \(\sigma \cap -\sigma = \{0\}\).

\underline{Fact:} \(\check{\check{\sigma}} = \sigma\).

\begin{lemma}
    The dual \(\check\sigma\) of a cone \(\sigma\) is a cone.
\end{lemma}

\begin{proof}

    \[\check\sigma = \Cone(m_1, \dots, m_s)\]

    \[m^\perp = H_m = \{u \in N_\R \mid \inner{u, m} = 0\}\]

        [supporting hyperplane] A \emph{supporting hyperplane} of \(\sigma\) is \(H_m\), such that \(\sigma \sse H^+_m\). Equivalently, \(m \in \check\sigma\).
    \end{definition}

    Now note that
    \[\check\sigma = \Cone(m_1, \dots, m_s) \iff \sigma = \bigcap_{i=1}^s H_{m_i}^+\]
    So to construct \(\check\sigma\), we need to express \(\sigma\) as an intersection of half spaces.

    For example, \(\check\sigma_3 = \Cone((1, 0), (0, 1))\)
    
    and similarly, \(\check\sigma_4 = \Cone((2, -1), (0, 1))\).

    In general, we can do this by identifying the \emph{facets} \(\tau \faceof \sigma\), and taking the corresponding dual element \(m_\tau \in M\), and
    \[\check\sigma = \Cone(m_\tau \mid \tau \faceof \sigma)\]
    when \(\sigma\) is top dimensional.

        A \emph{face} \(\tau \faceof \sigma\) is the intersection \(\sigma \cap H_m\), for a supporting hyperplane \(H_m\). Note that the faces are all cones, and we call \(\tau\) a \emph{facet} if \(\dim(\tau) = \dim(\sigma) - 1\), and \(\tau\) an \emph{edge} if \(\dim(\tau) = 1\).

\end{proof}

\subsection{Commutative semigroup}

\begin{definition}
    [commutative semigroup]
    A \emph{commutative semigroup} is \((S, +)\), such that \(+\) is an associative commutative binary operation on \(S\).
\end{definition}

In particular, a semigroup with identity is called a \emph{monoid}. \(S\) is \emph{finitely generated} if there exists \(A \sse S\) finite, such that
\[S = \left\{\sum_n \lambda_n a_n \ \bigg\vert\ \lambda_n \in \Z_{> 0}, a_n \in A\right\}\]
\(S\) is \emph{affine} if it is finitely generated, and \((S, +)\) is a subsemigroup of a lattice \((\Z^n, +)\).

\begin{example}
    Any group is a semigroup. Given a (not necessarily unital) ring \(R\), \(R\) under multiplication is a semigroup. For example, \(\Z_{> 0}\) is a semigroup under multiplication, but this is not finitely generated.
\end{example}

\begin{definition}
    Let \(\sigma\) be a cone. Define
    \[S_\sigma = \check\sigma \cap M \sse M_\R\]
    for the semigroup associated to \(\sigma\).
\end{definition}

This is a semigroup as cones are closed under addition. Now we want to show that this is affine. This does not follow immediately from being a subgroup of \((\Z^n, +)\). For example, consider
\[S = \{(x, y) \mid x \le \sqrt2 y\}\]
This is a semigroup, but it is not finitely generated.

\begin{lemma}
    [Gordan's lemma] If \(\sigma\) is a cone, then \(S_\sigma\) is a finitely generated semigroup.
\end{lemma}

\begin{proof}
    We know that \(\check\sigma = \Cone(m_1, \dots, m_r)\), for some \(m_i \in M\). Say \(x \in S_\sigma\). By definition,
    \[x = \sum_{i=1}^r \alpha_i m_i\]
    where \(\alpha_i \in \R_{\ge 0}\). Write
    \[\alpha_i = a_i + b_i\]
    where \(a_i = \lfloor\alpha_i\rfloor\), \(b_i = \{\alpha_i\} \in \Ico{0, 1}\). Then
    \[x = \sum_i a_i m_i + \sum_i b_i m_i\]
    Now \(x \in M\), and \(\sum_i a_i m_i \in M\), and \(M\) is a group, so
    \[\sum_i b_im_i \in M\]
    On the other hand, \(\sum b_i m_i\) is contained in a compact set
    \[K = \left\{\sum c_i m_i \mid 0 \le c_i \le 1\right\}\]
    Since \(M\) is discrete, \(K\) is compact, \(K \cap M\) is finitely many points, say \(K \cap M = \{m_{r+1}, \dots, m_\ell\}\). Then \(S_\sigma\) is generated by
    \[m_1, \dots, m_r, m_{r+1}, \dots, m_\ell\]
    so it is finitely generated.
\end{proof}

\begin{example}
    When \(\sigma = \Cone(1)\), then \(\check\sigma = \Cone(1)\), and so \(S_\sigma\) is generated by \(1\).

    Now if \(\sigma = \Cone(e_1) \sse \R^2\), then \(\check\sigma = \Cone(e_1, e_2, -e_2)\), and these are the generators of \(S_\sigma\).

    If \(\sigma = \Cone(e_1, e_2) \sse \R^2\), then \(\check\sigma = \Cone(e_1, e_2)\), which are the generators of \(S_\sigma\).

    Finally, if \(\sigma = \Cone(e_1, e_1 + 2e_2)\), then \(\check\sigma = \Cone(e_2, 2e_1 - e_2)\). Here, Gordan's lemma shows that \(S_\sigma\) is generated by
    \[e_2, 2e_1 - e_2, e_1\]
\end{example}

\subsection{Coordinate ring}

\begin{definition}
    The algebra \(\C[S]\) of a semigroup \((S, +)\) is the \(\C\)-algebra of finite sums
    \[\C[S] = \left\{\sum_i a_i x^{s_i} \mid a_i \in \C, s_i \in S\right\}\]
    with \(x^s x^{s'} = x^{s + s'}\).
\end{definition}

\begin{example}
    The semigroup \(\Z^n\) is generated by \(e_1, \dots, e_n, -e_1, \dots, -e_n\), so
    \[\C[\Z^n] = \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]\]
    where \(x_i^{\pm 1} = x^{\pm e_i}\).
\end{example}

\begin{lemma}
    If \((S, +)\) is a finitely generated semigroup, \(\C[S]\) is a finitely generated \(\C\)-algebra. If \((S, +)\) is affine, then \(\C[S]\) is the algebra of an affine variety.
\end{lemma}

\begin{proof}
    If \(S\) is generated by \(A = \{s_1, \dots, s_\ell\}\), then \(\{x^{s_1}, \dots, x^{s_\ell}\}\) generates \(\C[S]\) as a \(\C\)-algebra.

    Now suppose \(S\) is affine. Then \(\C[S]\) is finitely generated by the above, and we have that \((S, +) \sse (\Z^n, +)\), so we have
    \[\C[S] \sse \C[\Z^n] = \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]\]
    So \(\C[S]\) is a subring of an integral domain, hence it is an integral domain. To conclude, every finitely generated integral domain is the coordinate ring of an affine variety.
\end{proof}

Define \(R_\sigma = \C[S_\sigma]\). From this, we know that \(U_\sigma = \Spec(R_\sigma)\) is an affine variety.

\begin{example}
    For \(\sigma = \Cone(1)\), let \(x = x^{e_1}\), then \(\C[S_\sigma] = \C[x]\), and so \(U_\sigma = \A^1\). The inclusion \(\C[x] \sse \C[x^{\pm 1}]\) defines the torus.

\section{Opens in \(U_\sigma\)}

We will see that \(\tau \faceof \sigma\), correspond to principal opens \(U_\tau \sse U_\sigma\). Recall that a face is \(\tau = \sigma \cap H_m\) for some \(m \in \check\sigma \cap M = S_\sigma\).

\begin{theorem}
    If \(\tau = \sigma \cap H_m\), then
    \[\C[S_\tau] = \C[S_\sigma]_{\chi^{\mathbf m}} = \C[S_\sigma][\chi^{\pm \mathbf m}]\]
\end{theorem}

\begin{proof}
    First, since \(\tau \in H_m\), \(\inner{u, m} = 0\) for all \(u \in \tau\). So \(m\) and \(-m\) are in \(\check\tau \cap M = S_\tau\). Also, if \(m' \in S_\sigma\), then \(\inner{m', u} \ge 0\) for all \(u \in \tau\). Putting these together,
    \[\C[S_\sigma]_{\chi^{\mathbf m}} \sse \C[S_\tau]\]
    For the reverse inclusion, take \(m' \in S_\tau \setminus S_\sigma\). We want to show \(m' + km \in S_\sigma\) for some \(k \in \Z\). For \(u \in \sigma\setminus \tau\), \(\inner{m', u} < 0\). But we also have that \(\inner{m, u} > 0\), so we can pick \(k\) large enough such that \(\inner{m' + km, u} > 0\). This tells us that \(m' + km \in S_\sigma\)\footnote{Since we only need to show positivity on the finitely many generators of the cone.}.
\end{proof}

\begin{example}
    Let \(\sigma = \Cone(e_1, e_2)\). Let \(\star = (0, 0)\). This is a face, and \(S_\star = \C[x^{\pm 1}, y^{\pm 1}]\). Hence \(U_\star = \T \sse \A^2\).

    Now if \(\tau_1\) is the face \(y = 0\), then we get
    \[\C[S_{\tau_1}] = \C[x, y]_{\chi^{e_2} = y} = \C[x, y^{\pm 1}]\]
    So
    \[U_{\tau_1} = \A^1 \times \C^*\]
\end{example}

\section{Fans and toric varieties}

\begin{definition}
    [fan] A \emph{fan} \(\Sigma\) is a finite collection of strongly convex cones \(\sigma \sse N_\R\), such that
    \begin{enumerate}
        \item for all \(\sigma \in \Sigma\), \(\tau \faceof \sigma\), then \(\tau \in \Sigma\).
        \item for any \(\sigma_1, \sigma_2 \in \Sigma\), \(\tau = \sigma_1 \cap \sigma_2\) is a face of both.
    \end{enumerate}
\end{definition}

\begin{notation}
    The \emph{support} of \(\Sigma\) is
    \[\abs{\Sigma} = \bigcup_{\sigma \in \Sigma}\sigma \sse N_\R\]
    and we will write
    \[\Sigma(r) \sse \Sigma\]
    for the subset of \(r\)-dimensional cones.
\end{notation}

\begin{example}
    First, let \(\sigma = \Cone(e) \sse \R\), then \(\Sigma_1 = \{\sigma, -\sigma, \star\}\) is a fan.

    Finally, consider the rays passing through \((0, 1), (1, 0), (0, -1), (-1, a)\), for soem \(a \in \Z_{> 0}\).
\end{example}

By the above construction, we can associate to each \(\sigma \in \Sigma\) an affine toric variety, and if \(\tau \faceof \sigma\) is a face, then \(U_\tau \sse U_\sigma\) is an open. We'll see two alternative ways of describing toric varieties:
\begin{enumerate}
    \item toric ideals, \(V = \van(I_L) \sse \A^s\), where \(I_L\) is a toric ideal,
    \item torus embedding, \(V = \overline{\Phi_\msA(T)}\), where \(\Phi_{\msA} : T \to \A^s\) given by monomials.
\end{enumerate}

Recall for \(\sigma = \Cone((1, 0), (1, 2))\), we have \(\sigma^\vee = \Cone((0, 1), (2, -1))\), and
\[S_\sigma = \Z_{\ge 0}\msA\]
where \(\msA = \{A, B, C\}\), where \(A = (0, 1), B = (1, 0), C = (2, -1)\). We have a map
\begin{align*}
    \pi : \Z^3_{A, B, C} &\to M \cong \Z^2 \\
    A &\mapsto (0, 1) \\
    B &\mapsto (1, 0) \\
    C &\mapsto (2, -1)
\end{align*}
As a matrix,
\[\begin{pmatrix}
    0 & 1 & 2 \\
    1 & 0 & -1
\end{pmatrix}\]
Then
\[\ker(\pi) = \Z\begin{pmatrix}
    1 \\ -2 \\ 1
\end{pmatrix}\]
If we restrict to \(\N_{A, B, C}^3\) and \(S_\sigma\), then we have a surjection of semigroups. In particular, \(S_\sigma = \N_{A, B, C}^3/\sim\), where \(\sim\) is the relation \(A + C = 2B\). On algebras,
\[\C[S_\sigma] = \frac{\C[\N_{A, B, C}]}{I_L}\]
Let \(x = \chi^A, y = \chi^B, z = \chi^C\). Then \(A + C = 2B\) corresponds to \(xz = y^2\). Then \(I_L = \inner{xz - y^2}\). In particular, this shows that we have an affine embedding \(U_\sigma = \van(I_L) \sse \A^3\).

On the other hand, \(\chi^A, \chi^B, \chi^C\) are maps \(T \to \C^*\). We can define a map
\begin{align*}
    \Phi_\msA : T &\to \A^3 \\
    (s, t) &\mapsto \left(\chi^A(s, t), \chi^B(s, t), \chi^C(s, t)\right) = (t, s, s^2t^{-1})
\end{align*}
In this case,
\[U_\sigma = \overline{\Phi_\msA(T)}\]
Moreover, \(\Phi_\msA\) with coordinates \(s, t\), acts on \(U_\sigma\) by
\[(s, t)(x,y,z) = (tx, sy, s^2t^{-1}z)\]

More generally, if \(S\) is an affine semigroup, generated by \(\msA = \{m_1, \dots, m_s\}\), where \(m_i \in M\), a lattice. So \(S = \Z_{\ge 0}\msA\). We can associate an ideal and a torus embedding as above.

\begin{definition}
    A \emph{lattice ideal} is an ideal \(I_L\) in \(\C[x_1, \dots, x_s]\), of the form
    \[I_L = \inner{x^\alpha - x^\beta \mid \alpha - \beta \in L}\]
    for some lattice \(L\). Note here we use the multi-index notation.

    \(I_L\) is \emph{toric} if it is prime.
\end{definition}

Now given \(S\) as above, consider the map
\[\pi : \bigoplus_{i=1}^s \Z \cdot m_i \to M\]
and \(L = \ker(\pi)\) is a sublattice. These relations give an ideal
\[I_L = \inner{x^{\ell^+} - x^{\ell^-} \ \big\vert\ \ell^+ - \ell^- \in L; \ell^+, \ell^- \in \bigoplus\Z_{\ge 0}\cdot m_i}\]
This is an ideal of
\[\C[x_1, \dots, x_s] = \C\left[\bigoplus_{i=1}^s\Z_{\ge 0}\cdot m_i\right]\]
where \(x_i = \chi^{m_i}\). We can define a variety \(V = \van(I_L) \sse \A^s\).

Now for \(\Phi_{\msA}\), we recall that \(M = \Z S\), and \(T_N = \Spec(\C[M])\). We can define
\begin{align*}
    \Phi_{\msA} : T_N &\to \A^s \\
    \mathbf t &\mapsto \left(\chi^{m_1}(\mathbf t), \dots, \chi^{m_s}(\mathbf t)\right)
\end{align*}

\begin{theorem}
    If \(V\) is an affine variety, then the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(V\) is an affine toric variety,
        \item \(V = \Spec(\C[S])\) for an affine semigroup \(S\),
        \item \(V = \van(I_L)\) for a toric ideal \(I_L\),
        \item \(V = \overline{\Phi_{\msA}(T)}\) for some finite collection \(\msA\) generating an affine semigroup.
    \end{enumerate}
\end{theorem}

\begin{proof}
    We have seen (ii) implies (iii) and that (iii) implies (iv). To see that (iv) implies (i), note that \(\Phi_\msA\) gives an action of \(T\) on \(\A^s\), and if \(V = \overline{\Phi_\msA(T)}\), then \(\Phi_{\msA}(T) \sse V\) is a dense open torus, and we just need to check that for \(t \in T\), \(t \vdot V \sse V\). Now \(t\cdot V\) contains \(\Phi_{\msA}(T)\), so it is contained in the closure of \(\Phi_{\msA}(T)\), which is \(V\).

    For (iv) implies (i), we need a lemma from representation theory:

    \begin{lemma}
        If \(A \sse \C[M]\) is a subrepresentation of \(T_N\), then \(A\) splits into
        \[A = \bigoplus_{m \in S}\C\cdot \chi^m\]
        for some subset\footnote{Since the multiplicity in \(\C[M]\) is one for each eigenspace, each eigenspace appears either zero or one times, so we can consider a \emph{subset} without thinking about multiplicity.} \(S \sse M\).
    \end{lemma}

    Note that here, multiplication by \(t\) defines a map \(\T_N \to \T_N\), which in turn is a map on coordinate rings \(\C[M] \to \C[M]\).

    Applying the lemma to \(\C[V] \to \C[M]\). The collection \(S\) of monomial appearing in \(\C[V]\) needs to form a semigroup \(S\), as \(\C[V]\) is a ring. This is finitely generated as \(\C[V]\) is finitely generated, as it is the coordinate ring of an affine variety.
\end{proof}

\[\{\text{affine toric variety}\} \bij \{\text{affine semigroups}\}\]
Restricting, we get

\begin{definition}
    [normal]
    An integral domain \(R\) with \(K = \Frac(R)\) is \emph{normal} (i.e. integrally closed) if for all \(x \in K\), with \(x\) being a root of a monic polynomial with coefficients in \(R\), \(x \in R\).
    
    Equivalently, if \(x^n + a_{n-1}x^{n-1} + \dots + a_0\), with \(a_i \in R\), then \(x \in R\).
\end{definition}

For an affine variety, it is normal if and only if its coordinate ring is normal. In general, we need to consider stalks, or an affine cover.

\begin{example}
    We'll show that \(R = \C[x, y]/\inner{y^2 - x^3}\) is not normal. Note that
    \[x = \frac{y^2}{x^2} = \left(\frac{y}{x}\right)^2 \in \Frac(R)\]
    Now
    \[z = \frac{y}{x} \notin R\]
    but \(z^2 - x^2 = 0\).
\end{example}

\begin{definition}
    [saturated]
    An affine semigroup \(S \sse M\) is \emph{saturated} if for every \(m \in M, k \in \N\) with \(km \in S\), then \(m \in S\).
\end{definition}

\begin{lemma}
    The semigroup \(S_\sigma\) of a (strongly convex) cone is saturated.
\end{lemma}

\begin{proof}
    We defined \(S_\sigma\) as the intersection of a cone (which is convex) and \(M\).
\end{proof}

\begin{theorem}
    If \(V\) is an affine toric variety, with torus \(T_N\), then the following are equivalent:
    \begin{enumerate}[(i)]
        \item \(V\) is normal,
        \item \(V = \Spec(\C[S_\sigma])\), for some cone \(\sigma\),
        \item \(V = \Spec(\C[S])\), for some saturated affine semigroup \(S\),
    \end{enumerate}
\end{theorem}

\begin{proof}
    For (i) implies (iii), if \(V\) is affine toric, then \(V = \Spec(\C[S])\), for some affine semigroup \(S\). Say \(m \in M\), and \(km \in S\), for some \(k \in \Z_{> 0}\). This means that \(\chi^{km} \in \C[S]\), and \(\chi^m \in \C(S) = \Frac(\C[S])\). But \(\chi^{m}\) is the root of the monic integral polynomial
    \[y^k - \chi^km = 0\]
    so \(\chi^m \in \C[S]\) as \(\C[S]\) is normal.

    For (iii) implies (ii), let \(S\) be saturated and affine, and let \(\msA\) be a finite generating set. Define \(\sigma^\vee = \Cone(\msA)\). Then \(S = M \cap \sigma^\vee\). Taking the dual gives \(\sigma\).

    For (ii) implies (i), take \(\sigma = \Cone(v_1, \dots, v_r)\). Let \(\rho_i\) be the ray generated by \(v_i\). Without loss of generality, we can assume that \(v_i\) is a minimal generator of \(\rho_i\). Now we have that
    \[\sigma^\vee = \bigcap_{i=1}^r H_{v_i}^+ = \bigcap_{i=1}^r \rho_i^\vee\]
    Intersecting with the lattice,
    \[S_\sigma = \bigcap_{i=1}^r S_{\rho_i}\]
    and so
    \[\C[S_\sigma] = \bigcap_{i=1}^r \C[\rho_i]\]
    We just have to check the varieties \(U_i = \Spec(\C[S_{\rho_i}])\) are normal. But if we only have one ray \(\rho\), say generated by \(x_1\), then
    \[\C[S_\rho] = \C[x_1, x_2^{\pm 1}, \dots, x_n^{\pm 1}]\]
    So \(U_\rho = \A^1 \times (\C^*)^{n-1}\) which is normal. 
\end{proof}

With this, we have a correspondence
\[\{\text{normal affine toric varieties}\} \bij \{\text{strongly convex cones}\}\]

\[\begin{tikzcd}
	{\C^*} & T & {\C^*}
	\arrow["{\lambda_u}", from=1-1, to=1-2]
	\arrow["{\chi^m}", from=1-2, to=1-3]
\end{tikzcd}\]
    is \(x \mapsto x^d\). Moreover,
    \[u \in \sigma \iff \inner{u, m} \ge 0 \text{ for all }m \in S_\sigma \iff \lim_{t \to 0}(\chi^m \circ \lambda_u)(t) \text{ exists for all }m \in S_\sigma\]
    So we need to show that
    \[\lim_{t \to 0}(\chi^m \circ \lambda_u)(t) \text{ exists for all }m \in S_\sigma \iff \lim_{t \to 0}\lambda_u(t) \text{ exists in }U_\sigma\]
    \(\impliedby\) is clear. Conversely, recall (a closed point) \(x \in U_\sigma\) is a \(\C\)-algebra homomorphism
    \[x^\sharp : \C[S_\sigma] \to \C\]
    Define
    \[x_0(\chi^m) = \lim_{t \to 0}\chi^m(\lambda_u(t)) = \begin{cases}
        0 & \text{if } \inner{u, m} > 0 \\
        1 & \text{if } \inner{u, m} = 0 \\
    \end{cases}\]
    Then
    \[\lim_{t \to 0}\lambda_u(t) = x_0\]
\end{proof}

\begin{remark}
    \[\lim_{t \to 0}\chi^m(\lambda_u(t)) = \begin{cases}
        0 & \text{if } \inner{u, m} > 0 \\
        1 & \text{if } \inner{u, m} = 0 \\
    \end{cases}\]
    and the limit does not exist if \(\inner{u, m} < 0\).
\end{remark}

\begin{definition}
    [distinguished point]
    The \emph{distinguished point of \(\sigma\)} is
    \begin{align*}
        x_\sigma : \C[S_\sigma] &\to \C \\
        \chi^m &\mapsto \begin{cases}
            0 & \text{if }m \in S_\sigma \setminus \sigma^\perp \\
            1 & \text{otherwise}
        \end{cases}
    \end{align*}
\end{definition}
Note \(m \in S_\sigma \setminus \sigma^\perp\) if and only if \(\inner{u, m} > 0\) for all \(m\).

\begin{lemma}
    We have a one-to-one correspondence between points \(x \in U_\sigma\) and semigroup homomorphisms
    \[\gamma_x : (S_\sigma, +) \to (\C, \cdot)\]
\end{lemma}

\begin{proof}
    For \(x \in U_\sigma\), we have a corresponding map
    \begin{align*}
        x^\sharp : \C[S_\sigma] &\to \C \\
        x^\sharp(\chi^m) &= \chi^m(x)
    \end{align*}
    and we can take \(\gamma_x(m) = \chi^m(x)\). Conversely, given \(\gamma : S_\sigma \to \C\), define
    \begin{align*}
        x_\gamma^\sharp : \C[S_\sigma] &\to \C \\
        x_\gamma^\sharp(\chi^m) &= \gamma(m)
    \end{align*}
\end{proof}

\begin{example}
    The distinguished point \(x_\sigma \in U_\sigma\) corresponds to the semigroup homomorphism
    \begin{align*}
        \gamma_\sigma : S_\sigma &\to \C \\
        \gamma_\sigma(m) &= \begin{cases}
            1 & m \in \sigma^\perp \cap S_\sigma \\
            0 & m \in S_\sigma \setminus \sigma^\perp
        \end{cases}
    \end{align*}
\end{example}

\begin{corollary}
    For all \(u \in \Int(\sigma) \cap N\),
    \[\lim_{t \to 0}\lambda_u(t) = x_\sigma\]
\end{corollary}

\begin{example}
    Consider the cone \(\sigma = \Cone(e_1, e_2) \sse \R^2\). In this case, \(\sigma^\perp = \{0\}\). Now \(U_\sigma = \A^2\), and we would like to find \(x_\sigma, x_{\tau_1}\), where \(\tau_1 = \Cone(e_1)\).

    First of all, in this case, we have that \(\gamma_\sigma(0) = 1\), and \(\gamma_\sigma(m) = 0\) otherwise. The corresponding point \(x_\sigma\) is given by
    \begin{align*}
        x_\sigma : \C[S_\sigma] &\to \C \\
        \chi^m &\mapsto \begin{cases}
            1 & m = 0 \\
            0 & \text{otherwise}
        \end{cases}
    \end{align*}
    This is just evaluation at zero of the polynomial ring. So \(x_\sigma = (0, 0) \in \A^2\). Next, we want to show that this is the limit point for all one-parameter subgroups \(\lambda_u\), for \(u \in \Int(\sigma) \cap N\). That is, \(u = (a, b)\), with \(a, b > 0\). Now
    \[\lambda_{(a, b)}(t) = (t^a, t^b)\]
    Since \(a, b > 0\),
    \[\lim_{t \to 0}\lambda_{(a, b)}(t) = (0, 0) = x_\sigma\]

    For \(\tau_1\), we have that \(\tau_1^\vee = \Cone(e_1, e_2, e_2^{-1})\). In this case,
    \[U_{\tau_1} = \A^1 \times \C^*\]
    and \(S_{\tau_1} = \Z_{\ge 0} e_1 \oplus \Z e_2\). First of all, \((a, b) \in \Int(\tau_1) \cap N\) if and only if \(a > 0, b = 0\). So we have
    \[\lambda_{(a, 0)}(t) = (t^a, 1)\]
    and so
    \[\lim_{t \to 0}\lambda_{(a, 0)}(t) = (0, 1)\]
    On the other hand,
    \begin{align*}
        \gamma_{\tau_1} : S_{\tau_1} &\to \C \\
        e_1 &\mapsto 0 \\
        e_2 &\mapsto 1
    \end{align*}
    and so the map is
    \begin{align*}
        \C[x, y^{\pm 1}] &\to \C \\
        x &\mapsto 0 \\
        y &\mapsto 1
    \end{align*}
    and so \(x_{\tau_1} = (0, 1)\).
\end{example}

\begin{example}
    For \(\sigma = \Cone(\emptyset) = \{0\} \sse N_\R\). Then \(\sigma^\vee = M_\R\), and \(S_\sigma = \Z e_1 \oplus \cdots \oplus \Z e_n\). Then
    \[\C[S_\sigma] = \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]\]
    The distinguished point is given by \(\gamma_\sigma(m) = 1\) for all \(m\). So we have the point \((1, \dots, 1)\), which is the identity of the group \(T_N = U_\sigma\). This is the limit point for all \(u \in \Int(\sigma)\). 
\end{example}

\section{Smoothness}

\begin{lemma}
    If \(\sigma = \Cone(v_1, \dots, v_k)\) for \(v_i \in N\), and suppose \(v_1, \dots, v_k\) can be completed to a \(\Z\)-basis \(v_1, \dots, v_n\) of \(N\). Then \(U_\sigma \cong \A^k \times (\C^*)^{n-k}\).
\end{lemma}

Moreover, we claim that these are the only non-singular affine toric varieties.

\begin{proof}
    Complete \(v_1, \dots, v_k\) to a basis \(v_1, \dots, v_n\). Then \(S_\sigma\) is generated by \(w_1^*, \dots, w_k^*, \pm w^*_{k+1}, \dots, \pm w^*_n\). So
    \[\C[S_\sigma] = \C[x_1, \dots, x_k] \otimes \C[x_{k+1}^{\pm 1}, \dots, x_n^{\pm 1}]\]
\end{proof}

Recall if \(v_1, \dots, v_n \in \Z^n\) forms a basis of \(\R^n\), it does not have to form a basis of \(\Z^n\).

\begin{example}
    Take \(v_1 = (1, 0), v_2 = (1, 2)\). These form a \(\R\)-basis, but not a \(\Z\)-basis. In this case, \(U_\sigma\) is a singular variety.
\end{example}

In general, if \(v_1, \dots, v_n \in \Z^n\) are \(\R\)-linearly independent, then they form a basis of \(\Z^n\) if and only if
\[V = \begin{pmatrix}
    \vdots & & \vdots \\
    v_1 & \cdots & v_n \\
    \vdots & & \vdots
\end{pmatrix} \in \GL(n, \Z)\]
Equivalently, \(\det(V) = \pm 1\).

\begin{definition}
    [regular point]
    Let \(X\) be a variety of dimension \(n\). Then \(x \in X\) is called a \emph{regular} or \emph{nonsingular} point if \(\dim_\C(\Omega_{X, x}) = n\).

    Otherwise, we say that \(x\) is a singular point.
\end{definition}

Here, \(\Omega_{X,x}\) is the Zariski cotangent space to \(x\) at \(X\). Recall that a closed point \(x \in \Spec(R)\) corresponds to \(x^\sharp : R \to \C\), and defines a maximal ideal \(\mfm_x = \ker(x^\sharp)\).

Then
\[\Omega_{X, x} = \frac{\mfm_x}{\mfm_x^2}\]
This is a \(\C\)-vector space.

\begin{definition}
    We say that \(X\) is \emph{smooth} or \emph{nonsingular}, if \(X\) has no singular points.
\end{definition}

Note that over some fields, smooth implies non-singular, but the converse is false. An alternative definition of smoothness:

\begin{proposition}
    [Jacobian criterion] Let \(V = \van(f_1, \dots, f_r) \sse \A^s\) be an affine variety of dimension \(d\). Then \(V\) is \emph{smooth} at \(x \in V\) if and only if the matrix
    \[J_{(f_1, \dots, f_r)}(x) = \left(\pdv{f_i}{x_j}\right)_{i = 1, \dots, r; j = 1, \dots, s}\]
    has rank \(s - d\) for all \(x \in V\).
\end{proposition}

We can identify \(\ker(J(x)) = (\mfm_x/\mfm_x^2)^\vee\). In terms of differential geometry, we have a map \(F : \A^s \to \A^r\), and \(\ker(\dd F) = \T V\).
