\section{What is a toric variety?}

\begin{definition}
    [affine variety] An \emph{affine variety}\index{affine variety} is \(V = \Spec(R)\), where \(R\) is a finitely generated integral\footnote{\(R\) is an integral domain.} \(\C\)-algebra.
\end{definition}

\underline{Fact:} Given such an \(R\), we can choose generators and relations, and write
\[R = \frac{\C[x_1, \dots, x_n]}{I}\]
where \(I\) is a prime ideal. This gives an affine embedding. i.e. a closed embedding
\[V = \van(I) = \A^n\]
Note that the affine embedding is not unique.

\begin{definition}
    [(complex) variety] A \emph{(complex) variety}\index{variety} is a scheme \(X\) which is integral (reduced and irreducible), separated, and of finite type.
\end{definition}

We can avoid schemes, and glue varieties out of affine varieties. We need 
\begin{enumerate}
    \item a finite collection \(\{V_\alpha\}_{\alpha \in A}\) of affine varieties, say \(V_\alpha = \Spec(R_\alpha)\), 
    \item for \(\alpha, \beta \in A\), we need Zariski opens \(V_{\alpha\beta} \sse V_\alpha\),
    \item algebraic transition maps \(g_{\alpha\beta} : V_{\alpha\beta} \to V_{\beta\alpha}\), satisfying the usual cocycle conditions,
    \item check that the resulting
    \[X = \frac{\bigsqcup_\alpha V_\alpha}{\sim}\]
    is irreducible and separated (Hausdorff).
\end{enumerate}

\begin{definition}
    [torus] An \emph{(algebraic complex) torus of dimension \(n\)}\index{torus} is 
    \[\T^n = (\C^*)^n = \Spec(\C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]) = (\A^1 \setminus \{0\})^n\]
\end{definition}

Note that this is not the same as the usual torus \((S^1)^n\). In particular, we have an affine embedding
\[\T^n = \Spec\left(\frac{\C[z_0, \dots, z_n]}{\inner{z_0\cdots z_n - 1}}\right) = \van(z_0\cdots z_n - 1) \sse \A^{n+1}\]
\underline{Exercise:} Check that we have an isomorphism of rings
\[\C[z_1^{\pm 1}, \dots, z_n^{\pm 1}] \cong \frac{\C[z_0, \dots, z_n]}{\inner{z_0\cdots z_n - 1}}\]

The torus is also a \emph{commutative group scheme}\index{commutative group scheme}, that is, we have algebraic morphisms
\begin{align*}
    e : \bullet &\to \T^n \\
    m : \T^n \times \T^n &\to \T^n \\
    i : \T^n &\to \T^n
\end{align*}
making \(\T^n\) into an abelian group with identity \(e\), multiplication \(m\) and inverse \(i\). On the level of coordinate rings, we have the \emph{identity section},
\begin{align*}
    e^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C \\
    t_i^{\pm 1} &\mapsto 1
\end{align*}
The multiplication map is given by pointwise operations, i.e.
\[m((x_1, \dots, x_n), (y_1, \dots, y_n)) = (x_1y_1, \dots, x_ny_n)\]
With this, we have \emph{comultiplication}
\begin{align*}
    m^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}] \otimes \C[y_1^{\pm 1}, \dots, y_n^{\pm 1}] \\
    t_i &\mapsto x_i \otimes y_i
\end{align*}
and the inverse map is given by
\begin{align*}
    i^\sharp : \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] &\to \C[t_1^{\pm 1}, \dots, t_n^{\pm 1}] \\
    t_i &\mapsto t_i^{-1}
\end{align*}

\begin{definition}
    [toric variety] A \emph{toric variety}\index{toric variety} is a variety \(X\) with a dense open \(\T^n \sse X\), and an action (i.e. algebraic map)
    \[a : \T^n \times X \to X\]
    which extends the multiplication on \(X\).
\end{definition}

\begin{example}
    \(\T^n\) is itself a toric variety.
\end{example}
Cohomology
\begin{example}
    \(\A^n = \Spec(\C[x_1, \dots, x_n])\), with the torus
    \[\Spec(\C[x_1^{\pm 1}, \dots, x_n^{\pm 1}])\]
    with
    \[a((t_1, \dots, t_n), (x_1, \dots, x_n)) = (t_1x_1, \dots, t_nx_n)\]
\end{example}

\begin{example}
    Projective space
    \[\P^n = \Proj(\C[z_0, \dots, z_n]) = \{(z_0 : \dots : z_n) \mid (z_0, \dots, z_n) \ne (0, \dots, 0)\}\]
    is also a toric variety, with torus
    \[\T^n = \{(z_0 : \dots : z_n) \mid z_0\cdots z_n \ne 0\}\]
    with action given coordinate-wise again, i.e.
    \[a((t_0: \dots: t_n), (z_0:\dots:z_n)) = (t_0z_0:\dots:t_nz_n)\]
    in non-homogeneous coordinates (i.e. in the affine \(z_0 \ne 0\)) of the torus, we can define the action by
    \[(t_1, \dots, t_n) \cdot (z_0:\dots : z_n) = (z_0 : t_1z_1: \dots : t_nz_n)\]
    We can check that \(\T^n\) is contained in all of the standard opens \(U_i = D(z_i)\), and if we choose the usual affine coordinates, then \(\T^n\) embeds into affine space as above.
\end{example}

Toric varieties are determined by the dimension \(n\) of the torus, and the \emph{toric boundary}\index{toric boundary} \(X \setminus \T^n\). For example, \(\P^2 \setminus \T^2\) has three components,
\begin{enumerate}
    \item \(V_0 = \van(z_0)\),
    \item \(V_1 = \van(z_1)\),
    \item \(V_2 = \van(z_2)\)
\end{enumerate}
The \(V_i\) intersect pairwise, but not all three at the same time. Each \(V_i\) is invariant under the action of the torus, so \(\P^2\) can be represented as

\section{Characters, cocharacters and lattices}

Recall from the last lecture that \(\T^n = (\C^*)^n\) is an algebraic group.

\begin{definition}
    [character]
    A \emph{character} \(\chi\) of \(\T^n\) is a morphism \(\chi : \T^n \to \C^*\) of algebraic groups. That is, \(\chi : \T^n \to \C^*\) is a morphisms of schemes, such that
    \[\chi(e) = 1 \qqtext{and}(\chi^\sharp \otimes \chi^\sharp) \circ m^\sharp = m^\sharp \circ \chi^\sharp\]
\end{definition}

We will write \(\Hom_{\algp}(\cdot, \cdot)\) for the set of morphisms of algebraic groups.

\begin{theorem}
    The characters of \(\T^n\) are
    \[M = \Hom_\algp(\T^n, \C^*) = \Z^n\]
\end{theorem}

\begin{proof}
    First of all, note that
    \[\Hom_\algp(\T^n, \C^*) = \left(\Hom(\C^*, \C^*)\right)^n\]
    Thus, without loss of generality \(n = 1\). Now we need a \(\C\)-algebra map
    \[\chi^\sharp : \C[x^{\pm 1}] \to \C[y^{\pm 1}]\]
    This is determined by \(\chi^\sharp(x) = p(y, y^{-1})\). Note that \(x\) is a unit in \(\C[x^{\pm 1}]\), and so \(\chi^\sharp(x)\) is a unit. Thus, \(p(y, y^{-1}) = ay^k\), for some \(a \in \C, k \in \Z\).

    Next, we need that \(e^\sharp \circ \chi^\sharp = \chi^\sharp \circ e^\sharp\). That is,
    \[a = e^\sharp(\chi^\sharp(x)) = \chi^\sharp(e^\sharp(x)) = \chi^\sharp(1) = 1\]
    Hence \(\chi^\sharp(x) = x^k\), for some \(k \in \Z\). Note that \(\chi^\sharp(x \otimes x) = \chi^\sharp(x) \otimes \chi^\sharp(x)\) automatically, so we are done.
\end{proof}

The characters of \(\T^n\) are of the form
\[\chi(t_1, \dots, t_n) = t_1^{a_1}\cdots t_n^{a_n}\]
for some \((a_1, \dots, a_n) \in \Z^n\). Expressions of the form
\[t_1^{a_1}\cdots t_n^{a_n}\]
are called \emph{Laurent monomials}. We will write
\[\chi^{\mathbf a}(\mathbf t) = \mathbf t^{\mathbf a} = t_1^{a_1}\cdots t_n^{a_n}\]
for this character. The correspondence is then
\begin{align*}
    \Hom_\algp(\T^n, \C^*) &\to \Z^n \\
    \chi^{\mathbf a} \mapsto \mathbf a
\end{align*}

The characters form a group. Given characters \(\chi, \chi' : \T^n \to \C^*\), 
\[\chi\cdot\chi'(\mathbf t) = \chi(\bf t)\cdot \chi'(\bf t)\]
Thus, we have taht
\[\chi^{\mathbf a}\cdot \chi^{\mathbf a'} = \chi^{\mathbf a + \mathbf a'}\]
Hence the identification \(\Hom_\algp(\T^n, \C^*) \cong \Z^n\) is also a group isomorphism.

\begin{definition}
    [cochacter]
    A \emph{cocharacter} of \(\T^n\) is
    \[\lambda \in \Hom_\algp(\C^*, \T^n)\]
    We also call them \emph{one-parameter subgroup}.
\end{definition}

We can check that
\[N = \Hom_\algp(\C^*, \T^n) \cong \Z^n\]
by showing that any \(\lambda \in N\) is of the form
\[\lambda(t) = (t^{u_1}, \dots, t^{u_n})\]
for \(\mathbf u = (u_1, \dots, u_n) \in \Z\). So we can write the above cocharacter as \(\lambda_{\mathbf u}\) as before.  \(N\) is a group, and the identification is a group isomorphism. That is,
\[\lambda_{\mathbf u} \cdot \lambda_{\mathbf u'} = \lambda_{\mathbf u + \mathbf u'}\]

\begin{example}
    For \(n = 2\), we can identify
    \[\T^2 = \left\{\begin{pmatrix}
        t_1 & 0 \\ 0 & t_2
    \end{pmatrix}\ \bigg\vert\ t_1t_2 \ne 0\right\}\]
    Under the identification \(M \cong N \cong \Z^2\),
    \[\chi^{(a, b)}\begin{pmatrix}
        t_1 & 0 \\
        0 & t_2
    \end{pmatrix} = t_1^a\cdot t_2^b\]
    and
    \[\lambda_{(u, v)}(t) = \begin{pmatrix}
        t^u & 0 \\
        0 & t^v
    \end{pmatrix}\]
\end{example}

\begin{definition}
    [lattice]
    A \emph{lattice} is a free abelian group of finite rank.
\end{definition}

Note that with this definition, any lattice is isomorphic to \(\Z^n\) when we choose a basis.

\begin{example}
    \(\Z, 2\Z, \Z \oplus \Z, \Z \oplus 3\Z, \dots\) are all lattices.
\end{example}

There is a \emph{perfect pairing}\footnote{That is, it induces isomorphism \(\Hom_\Z(M, \Z) = M^\vee \cong N\) and vice versa.}, given by
\begin{align*}
    \inner{\cdot, \cdot} : M \times N &\to \Z \\
    \inner{\mathbf a, \mathbf u} &= \mathbf a \vdot \mathbf u = a_1 u_1 + \cdots + a_nu_n
\end{align*}
In a coordinate-free manner,
\begin{align*}
    M \times N = \Hom(\C^*, \T^n) \times \Hom(\T^n, \C^*) &\to \Hom(\C^*, \C^*) \\
    (\lambda, \chi) &\mapsto \chi \circ \lambda
\end{align*}
Using the identification \(\Hom(\C^*, \C^*) \cong \Z\) as before gives the pairing.

We can describe a torus \(\T_N\) starting from a lattice \(N\), through
\[\T_N = N \otimes_\Z \C^*\]
with this,
\[\Hom(\C^*, \T_N) = N\]
Formally,
\[\T_N = \Spec(\Z[N] \otimes_\Z \C[x, x^{-1}])\]

With this, give a torus \(\T\), we get two dual lattices \(N, M\). Conversely, given a lattice \(N\), we get a torus \(\T_N\) and a dual lattice \(M\). Choosing a basis on one gives a basis on the other two.

\section{Cocharacters and limit points - Part I}

Back to toric varieties. Let \(X\) be a toric variety, with torus \(\T\). We can think of \(X\) as a ``partial compactification'' of \(T\). That is, we can construct \(X\) by adding limit points\footnote{In the Euclidean topology.} to \(\T\).

Given a one-parameter subgroup \(\mathbf u \in N\), we have
\[\lambda_{\mathbf u} : \C^* \to \T \sse X\]
We can ask when does
\[\lim_{t \to 0}\lambda_{\mathbf u}(t)\]
exist in \(X\). The idea will be that we can determine \(X\) by saying which \(\lambda_{\mathbf u}\) have a limit. See later.

\begin{example}
    Consider \(\T^2 \sse \P^2_{z_0:z_1:z_2}\). The embedding is given by \((t_1, t_2) \mapsto (1 : t_1 : t_2)\). Choose \(\mathbf u \in \Z^2\), so \(\mathbf u = (u_1, u_2)\).
    \[\lim_{t \to 0}\lambda_{\mathbf u}(t) = \lim_{t \to 0}(1 : t^{u_1} : t^{u_2})\]

    All these limit points exist, since \(\P^2\) is complete\footnote{Or in algebraic geometry, it satisfies the valuative criterion for properness for some choice of DVR.}. In particular, if \(u_1, u_2 > 0\), the limit is \((1 : 0 : 0)\). If \(u_1 < 0, u_2 > u_1\), the limit is \((0 : 1: 0)\). Similarly if \(u_2 < 0, u_1 > u_2\) then the limit is \((0 : 0 : 1)\).

    The upshot is that we have partitioned \(N_\R = N \otimes \R = \R^2\) into cones, so that if \(\mathbf u_1, \mathbf u_2\) are in the same cone if and only if they have the same limit. In fact, we can construct \(\P^2\) from this.
\end{example}

We will also write
\[\sigma = \Cone(u_1, \dots, u_r)\]
and the set \(\{u_i\}_{i=1}^r\) is the set of generators of the cone.

\begin{definition}
    [dimension]
    The \emph{dimension} of a cone \(\sigma\) is \(\dim(\sigma) = \dim(V)\), where \(V = \Span\{\sigma\} \sse N_\R\). A cone is \emph{top-dimensional} if \(V = N_\R\).
\end{definition}

\begin{definition}
    [strongly convex] \(\sigma\) is \emph{strongly convex} if it does not contain any lines through the origin. Equivalently, \(\sigma \cap -\sigma = \{0\}\).
\end{definition}

    \item Then we will generate a commutative semigroup \(S_\sigma\),
    \item from this, we will define \(R_\sigma\).
\end{enumerate}

\underline{Fact:} \(\check{\check{\sigma}} = \sigma\).

\begin{lemma}
    The dual \(\check\sigma\) of a cone \(\sigma\) is a cone.
\end{lemma}

\begin{proof}
    We'll define an algorithm to find \(m_1, \dots, m_s \in M\), such that
    \[\check\sigma = \Cone(m_1, \dots, m_s)\]
    From linear algebra, for \(m \in M = N^\vee\), we have a hypersurface
    \[m^\perp = H_m = \{u \in N_\R \mid \inner{u, m} = 0\}\]
    We also have half spaces
    \[H_m^+ = \{u \in N_\R \mid \inner{m, u} \ge 0\}\]

    \begin{definition}
        [supporting hyperplane] A \emph{supporting hyperplane} of \(\sigma\) is \(H_m\), such that \(\sigma \sse H^+_m\). Equivalently, \(m \in \check\sigma\).
    \end{definition}

    Now note that
    \[\check\sigma = \Cone(m_1, \dots, m_s) \iff \sigma = \bigcap_{i=1}^s H_{m_i}^+\]
    So to construct \(\check\sigma\), we need to express \(\sigma\) as an intersection of half spaces.

    For example, \(\check\sigma_3 = \Cone((1, 0), (0, 1))\)
    
    and similarly, \(\check\sigma_4 = \Cone((2, -1), (0, 1))\).

    In general, we can do this by identifying the \emph{facets} \(\tau \faceof \sigma\), and taking the corresponding dual element \(m_\tau \in M\), and
    \[\check\sigma = \Cone(m_\tau \mid \tau \faceof \sigma)\]
    when \(\sigma\) is top dimensional.

    \begin{definition}
        A \emph{face} \(\tau \faceof \sigma\) is the intersection \(\sigma \cap H_m\), for a supporting hyperplane \(H_m\). Note that the faces are all cones, and we call \(\tau\) a \emph{facet} if \(\dim(\tau) = \dim(\sigma) - 1\), and \(\tau\) an \emph{edge} if \(\dim(\tau) = 1\).
    \end{definition}

    Note by taking \(m = 0\), \(\sigma \faceof \sigma\).

    When \(\sigma\) is not top dimensional, then \(\Span_\R(\sigma) = W \ne N_\R\). Write \(\overline{\sigma}\) for \(\sigma\) considered as a subspace of \(W\). Now \(\overline\sigma\) is top-dimensional, and so we can the dual
    \[\check{\overline{\sigma}} \sse \frac{M_\R}{W^\perp}\]

    \[\check\sigma = \Cone(m_1, \dots, m_s, \pm m_{s+1}, \dots, \pm m_k)\]
\end{proof}

\subsection{Commutative semigroup}

\begin{definition}
    [commutative semigroup]
    A \emph{commutative semigroup} is \((S, +)\), such that \(+\) is an associative commutative binary operation on \(S\).
\end{definition}

In particular, a semigroup with identity is called a \emph{monoid}. \(S\) is \emph{finitely generated} if there exists \(A \sse S\) finite, such that
\[S = \left\{\sum_n \lambda_n a_n \ \bigg\vert\ \lambda_n \in \Z_{> 0}, a_n \in A\right\}\]
\(S\) is \emph{affine} if it is finitely generated, and \((S, +)\) is a subsemigroup of a lattice \((\Z^n, +)\).

\begin{example}
    Any group is a semigroup. Given a (not necessarily unital) ring \(R\), \(R\) under multiplication is a semigroup. For example, \(\Z_{> 0}\) is a semigroup under multiplication, but this is not finitely generated.
\end{example}

\begin{definition}
    Let \(\sigma\) be a cone. Define
    \[S_\sigma = \check\sigma \cap M \sse M_\R\]
    for the semigroup associated to \(\sigma\).
\end{definition}

This is a semigroup as cones are closed under addition. Now we want to show that this is affine. This does not follow immediately from being a subgroup of \((\Z^n, +)\). For example, consider
\[S = \{(x, y) \mid x \le \sqrt2 y\}\]
This is a semigroup, but it is not finitely generated.

\begin{lemma}
    [Gordan's lemma] If \(\sigma\) is a cone, then \(S_\sigma\) is a finitely generated semigroup.
\end{lemma}

\begin{proof}
    We know that \(\check\sigma = \Cone(m_1, \dots, m_r)\), for some \(m_i \in M\). Say \(x \in S_\sigma\). By definition,
    \[x = \sum_{i=1}^r \alpha_i m_i\]
    where \(\alpha_i \in \R_{\ge 0}\). Write
    \[\alpha_i = a_i + b_i\]
    where \(a_i = \lfloor\alpha_i\rfloor\), \(b_i = \{\alpha_i\} \in \Ico{0, 1}\). Then
    \[x = \sum_i a_i m_i + \sum_i b_i m_i\]
    Now \(x \in M\), and \(\sum_i a_i m_i \in M\), and \(M\) is a group, so
    \[\sum_i b_im_i \in M\]
    On the other hand, \(\sum b_i m_i\) is contained in a compact set
    \[K = \left\{\sum c_i m_i \mid 0 \le c_i \le 1\right\}\]
    Since \(M\) is discrete, \(K\) is compact, \(K \cap M\) is finitely many points, say \(K \cap M = \{m_{r+1}, \dots, m_\ell\}\). Then \(S_\sigma\) is generated by
    \[m_1, \dots, m_r, m_{r+1}, \dots, m_\ell\]
    so it is finitely generated.
\end{proof}

\begin{example}
    When \(\sigma = \Cone(1)\), then \(\check\sigma = \Cone(1)\), and so \(S_\sigma\) is generated by \(1\).

    Now if \(\sigma = \Cone(e_1) \sse \R^2\), then \(\check\sigma = \Cone(e_1, e_2, -e_2)\), and these are the generators of \(S_\sigma\).

    If \(\sigma = \Cone(e_1, e_2) \sse \R^2\), then \(\check\sigma = \Cone(e_1, e_2)\), which are the generators of \(S_\sigma\).

    Finally, if \(\sigma = \Cone(e_1, e_1 + 2e_2)\), then \(\check\sigma = \Cone(e_2, 2e_1 - e_2)\). Here, Gordan's lemma shows that \(S_\sigma\) is generated by
    \[e_2, 2e_1 - e_2, e_1\]
\end{example}

\subsection{Coordinate ring}

\begin{definition}
    The algebra \(\C[S]\) of a semigroup \((S, +)\) is the \(\C\)-algebra of finite sums
    \[\C[S] = \left\{\sum_i a_i x^{s_i} \mid a_i \in \C, s_i \in S\right\}\]
    with \(x^s x^{s'} = x^{s + s'}\).
\end{definition}

\begin{example}
    The semigroup \(\Z^n\) is generated by \(e_1, \dots, e_n, -e_1, \dots, -e_n\), so
    \[\C[\Z^n] = \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]\]
    where \(x_i^{\pm 1} = x^{\pm e_i}\).
\end{example}

\begin{lemma}
    If \((S, +)\) is a finitely generated semigroup, \(\C[S]\) is a finitely generated \(\C\)-algebra. If \((S, +)\) is affine, then \(\C[S]\) is the algebra of an affine variety.
\end{lemma}

\begin{proof}
    If \(S\) is generated by \(A = \{s_1, \dots, s_\ell\}\), then \(\{x^{s_1}, \dots, x^{s_\ell}\}\) generates \(\C[S]\) as a \(\C\)-algebra.

    Now suppose \(S\) is affine. Then \(\C[S]\) is finitely generated by the above, and we have that \((S, +) \sse (\Z^n, +)\), so we have
    \[\C[S] \sse \C[\Z^n] = \C[x_1^{\pm 1}, \dots, x_n^{\pm 1}]\]
    So \(\C[S]\) is a subring of an integral domain, hence it is an integral domain. To conclude, every finitely generated integral domain is the coordinate ring of an affine variety.
\end{proof}

Define \(R_\sigma = \C[S_\sigma]\). From this, we know that \(U_\sigma = \Spec(R_\sigma)\) is an affine variety.

\begin{example}
    For \(\sigma = \Cone(1)\), let \(x = x^{e_1}\), then \(\C[S_\sigma] = \C[x]\), and so \(U_\sigma = \A^1\). The inclusion \(\C[x] \sse \C[x^{\pm 1}]\) defines the torus.

    When \(\sigma = \Cone(e_1) \sse \R^2\), let \(x = x^{e_1}, y = x^{e_2}\), then \(\C[S_\sigma] = \C[x, y^{\pm 1}]\), and \(U_\sigma = \A^1 \times \C^*\). The torus is given by the inclusion
    \[\C[x, y^{\pm 1}] \sse \C[x^{\pm 1}, y^{\pm 1}]\]
    
    When \(\sigma = \Cone(e_1, e_2) \sse \R^2\), set \(x = x^{e_1}, y = x^{e_2}\), then \(U_\sigma = \Spec(\C[x, y]) = \A^2\).
    
    Finally, when \(\sigma = \Cone(e_1, e_1 + 2e_2)\). Let \(A = e_1, B = 2e_1 - e_2, C = e_2\). Then \(S_\sigma\) is generated by \(A, B, C\). We have a relation \(B + A = 2C\). Define \(x = x^A, y = x^B, z = x^C\), then
    \[\C[S_\sigma] = \frac{\C[x, y, z]}{\inner{xy = z^2}}\]
    With this, \(U_\sigma = \Spec(\C[S_\sigma]) = \van(xy - z^2) \sse \A^3\). In particular, this is a \emph{singular} toric variety. In this case, the torus is given by
    \[\frac{\C[x, y, z]}{\inner{xy - z^2}} \sse \C[x^{\pm 1}, z^{\pm 1}] = \C[M]\]
    To see this, we have an inclusion
    \[\frac{\C[x, y, z]}{\inner{xy - z^2}} \sse \frac{\C[x^{\pm 1}, y, z^{\pm 1}]}{\inner{xy - z^2}}\]
\end{example}

\begin{remark}
    We know that \(S_\sigma \sse M\), and so we have an inclusion of algebras \(\C[S_\sigma] \sse \C[M]\). A fact from commutative algebra is that, the image of the map
    \[\T^n = \Spec(\C[M]) \to U_\sigma = \Spec(\C[S_\sigma])\]
    is dense, as these are integral domains.
\end{remark}

\begin{remark}
    \(\T^n\) may map to some smaller torus \(\T^{n'} \sse U_\sigma\), with \(n' \le n\). We claim that if \(\sigma\) is strongly convex, then \(\T^n \sse U_\sigma\) as a dense open subset. See examples sheet 1, questions 1, 3. For this, note that \(\sigma\) is strongly convex if and only if \(\check\sigma\) is top dimensional.
\end{remark}

From now on, restrict to strongly convex \(\sigma\).
